     1→/**
     2→ * The deterministic fake editor (task W1-d): an in-memory implementation of the
     3→ * `window.effectcraft` surface for unit tests and E2E fixtures. Also the basis
     4→ * for the later Playwright mock mode: install the returned object as
     5→ * `window.effectcraft` in a test page and drive it with `script()`.
     6→ *
     7→ * It mirrors the verified upstream semantics, not just the shape:
     8→ * - `request` rejects with plain strings (editor/apps/effectcraft-web/src/api.rs).
     9→ * - `addFile` keeps only the last path segment of the name
    10→ *   (`editor/apps/effectcraft-web/src/files.rs` `path_for`).
    11→ * - `engine.batch` resolves `$N.key` references, rolls back a failing atomic
    12→ *   batch, and folds the run into ONE undo step whose label is the batch label
    13→ *   (editor/crates/engine/src/commands/batch.rs).
    14→ * - `edit.undo`/`edit.redo` return `{undone}`/`{redone}` labels;
    15→ *   `edit.history.list` returns `{states, current, branches}` — the fake's
    16→ *   history is linear (branches: 0) and ids are positional (`s0`, `s1`, …)
    17→ *   instead of upstream's content-keyed hex ids; labels are the step that made
    18→ *   each state, the oldest state being "Original".
    19→ * - Error strings use the verified upstream formats: "unknown command `x`",
    20→ *   "invalid parameters for `cmd`: …", "no active composition",
    21→ *   "no composition x", "no composition is open", "select a layer first",
    22→ *   "nothing to undo", "nothing to redo".
    23→ * - Command registry ids/labels/menus/shortcuts are copied from the upstream
    24→ *   spec registrations (see comments for sources).
    25→ *
    26→ * Approximations (documented divergences): `renderFrame` returns a constant
    27→ * 1×1 transparent PNG stub; `editor.state.times` is comp seconds (upstream
    28→ * serializes ticks); `project.summary` items are the comps only (the fake has
    29→ * no separate footage/solid items); no UI element registry (`ui.elements`,
    30→ * `ui.click`, `ui.key` are rejected as unknown methods).
    31→ */

   479→export function createFakeEditor(overrides: FakeEditorOverrides = {}): FakeEditor {
   480→  const version = overrides.version ?? "0.13.0-fake";
   481→  const latencyMs = overrides.latencyMs ?? 0;
   482→
   483→  let project: FakeProject = { comps: [], layers: [], keyframes: [] };
   484→  let windowSize: [number, number] = [...(overrides.windowSize ?? [1280, 800])];
   485→  let activeComp: number | null = null;
   486→  let selectedLayers: number[] = [];
   487→  let playhead = overrides.playhead ?? 0;
   488→  let playing = overrides.playing ?? false;
   489→  let focusedPanel = overrides.focusedPanel ?? "Viewer";
   490→  let windowFocused = true;
   491→  let projectPath: string | null = null;
   492→  let dirty = false;
   493→  let nextCompId = 1;
   494→  let nextLayerId = 1;
   495→  let nextUid = 1;
   496→
   497→  const files = new Map<string, Uint8Array>();
   498→  const undoStack: FakeHistoryEntry[] = [];
   499→  const redoStack: FakeHistoryEntry[] = [];
   500→
   501→  // False while `engine.batch` runs its steps: individual steps must not leave undo
   502→  // entries — the whole batch folds into ONE (batch.rs `fold`).
   503→  let journaling = true;
   504→  let suppressedJournals = 0;
   505→
   506→  function init(): void {
   507→    for (const seed of overrides.comps ?? []) {
   508→      const comp = makeComp(seed.name, seed.width, seed.height, seed.duration);
   509→      project.comps.push(comp);
   510→    }
   511→    const first = project.comps[0];
   512→    if (first) activeComp = first.id;
   513→  }
   514→
   515→  function makeComp(
   516→    name: string,
   517→    width?: number,
   518→    height?: number,
   519→    duration?: number,
   520→    frameRate?: number,
   521→    background?: string,
   522→  ): FakeComp {
   523→    const comp: FakeComp = {
   524→      id: nextCompId++,
   525→      name: uniqueCompName(name),
   526→      width: num(width, 1920),
   527→      height: num(height, 1080),
   528→      duration: num(duration, 10),
   529→      frameRate: num(frameRate, 29.97),
   530→      background: background ?? "#000000",
   531→    };
   532→    return comp;
   533→  }
   534→
   535→  function uniqueCompName(name: string): string {
   536→    if (!project.comps.some((c) => c.name === name)) return name;
   537→    let n = 2;
   538→    while (project.comps.some((c) => c.name === `${name} ${n}`)) n++;
   539→    return `${name} ${n}`;
   540→  }
   541→
   542→  function uniqueLayerName(base: string): string {
   543→    if (!project.layers.some((l) => l.name === base)) return base;
   544→    let n = 2;
   545→    while (project.layers.some((l) => l.name === `${base} ${n}`)) n++;
   546→    return `${base} ${n}`;
   547→  }

   577→  /** One journaled edit (upstream `Session::edit`): push the PRE state, clear redo. */
   578→  function journal(label: string): void {
   579→    if (!journaling) {
   580→      suppressedJournals++; // would have journaled — the batch folds it
   581→      return;
   582→    }
   583→    undoStack.push({ label, snapshot: snapshot() });
   584→    redoStack.length = 0;
   585→    dirty = true;
   586→  }

   595→  function resolveComp(ref: unknown): FakeComp {
   596→    const active = activeCompOrNull();
   597→    if (ref === undefined || ref === null) {
   598→      if (!active) throw "no active composition";
   599→      return active;
   600→    }
   601→    if (typeof ref === "number") {
   602→      const comp = project.comps.find((c) => c.id === ref);
   603→      if (!comp) throw `no composition ${ref}`;
   604→      return comp;
   605→    }
   606→    if (typeof ref === "string") {
   607→      const comp = project.comps.find((c) => c.name === ref);
   608→      if (!comp) throw `no composition ${ref}`;
   609→      return comp;
   610→    }
   611→    throw `invalid parameters for \`comp\`: ${JSON.stringify(ref)}`;
   612→  }

   614→  function resolveLayer(ref: unknown): FakeLayer {
   615→    // upstream has_layers: a selection or an explicit layer is required
   616→    let id: number | null = null;
   617→    if (typeof ref === "number") id = ref;
   618→    else if (typeof ref === "string" && /^\d+$/.test(ref)) id = Number(ref);
   619→    else if (ref === undefined && selectedLayers.length > 0) id = selectedLayers[0] ?? null;
   620→    const layer = id === null ? undefined : project.layers.find((l) => l.id === id);
   621→    if (!layer) {
   622→      throw activeCompOrNull() ? "select a layer first" : "no composition is open";
   623→    }
   624→    return layer;
   625→  }

   655→  // -- $N reference substitution (batch.rs `reference`/`substitute`) ----------
   656→
   657→  function reference(ref: string, results: unknown[]): unknown | undefined {
   658→    const rest = ref.slice(1); // strip '$'
   659→    const parts = rest.split(".");
   660→    const n = Number(parts[0]);
   661→    if (!Number.isInteger(n) || n < 1) return undefined; // not a $N reference
   662→    const path = parts.slice(1);
   663→    if (path.some((p) => p.length === 0 || !/^[A-Za-z0-9_]+$/.test(p))) {
   664→      throw `invalid parameters for \`$ref\`: ${ref}`;
   665→    }
   666→    const prior = results[n - 1];
   667→    if (prior === undefined) {
   668→      throw `\`${ref}\` refers to step ${n}, but only ${results.length} step(s) ran before`;
   669→    }
   670→    let v: unknown = prior;
   671→    for (const p of path) {
   672→      if (typeof v === "object" && v !== null && !Array.isArray(v)) {
   673→        v = (v as Record<string, unknown>)[p];
   674→      } else if (Array.isArray(v) && /^\d+$/.test(p)) {
   675→        v = v[Number(p)];
   676→      } else {
   677→        throw `\`${ref}\`: step ${n} returned ${JSON.stringify(prior)}, which has no \`${p}\``;
   678→      }
   679→      if (v === undefined) {
   680→        throw `\`${ref}\`: step ${n} returned ${JSON.stringify(prior)}, which has no \`${p}\``;
   681→      }
   682→    }
   683→    return v;
   684→  }
   685→
   686→  function substitute(value: unknown, results: unknown[]): unknown {
   687→    if (typeof value === "string") {
   688→      if (value.startsWith("$")) {
   689→        const resolved = reference(value, results);
   690→        if (resolved !== undefined) return resolved;
   691→      }
   692→      return value;
   693→    }
   694→    if (Array.isArray(value)) return value.map((v) => substitute(v, results));
   695→    if (isRecord(value)) {
   696→      const out: Record<string, unknown> = {};
   697→      for (const [k, v] of Object.entries(value)) out[k] = substitute(v, results);
   698→      return out;
   699→    }
   700→    return value;
   701→  }

   800→  function cmdBatch(p: Record<string, unknown>): unknown {
   801→    const steps = p.steps;
   802→    if (!Array.isArray(steps)) {
   803→      throw "invalid parameters for `engine.batch`: missing `steps`: [{command, params?}, …]";
   804→    }
   805→    const label = str(p.label) ?? "Batch";
   806→    const atomic = p.atomic === undefined ? true : p.atomic === true;
   807→    const pre = snapshot();
   808→    const results: unknown[] = [];
   809→    // batch.rs `fold`: applied steps become ONE undo entry named `label` —
   810→    // pushed only when any step actually journaled.
   811→    const fold = (): void => {
   812→      if (suppressedJournals > 0) {
   813→        undoStack.push({ label, snapshot: pre });
   814→        redoStack.length = 0;
   815→        dirty = true;
   816→      }
   817→      suppressedJournals = 0;
   818→    };
   819→    journaling = false;
   820→    try {
   821→      for (let i = 0; i < steps.length; i++) {
   822→        const step = steps[i];
   823→        const id = isRecord(step) ? (str(step.command) ?? str(step.id)) : undefined;
   824→        if (id === undefined) {
   825→          throw "invalid parameters for `engine.batch`: each step needs `command` (a command id)";
   826→        }
   827→        if ((NOT_IN_BATCH as readonly string[]).includes(id)) {
   828→          throw `invalid parameters for \`engine.batch\`: \`${id}\` can't run inside a batch`;
   829→        }
   830→        let params: Record<string, unknown>;
   831→        try {
   832→          params = (substitute(isRecord(step) ? (step.params ?? {}) : {}, results) ?? {}) as Record<
   833→            string,
   834→            unknown
   835→          >;
   836→        } catch (err) {
   837→          if (atomic) restore(pre);
   838→          else fold();
   839→          const note = atomic
   840→            ? "; the batch was rolled back (nothing changed)"
   841→            : `; steps 1–${i} were applied`;
   842→          throw `step ${i + 1} (${id}): invalid parameters for \`engine.batch\`: ${String(err)}${note}`;
   843→        }
   844→        try {
   845→          results.push(runCommand(id, params));
   846→        } catch (err) {
   847→          if (atomic) restore(pre);
   848→          else fold();
   849→          const note = atomic
   850→            ? "; the batch was rolled back (nothing changed)"
   851→            : `; steps 1–${i} were applied`;
   852→          throw `step ${i + 1} (${id}): ${String(err)}${note}`;
   853→        }
   854→      }
   855→    } finally {
   856→      journaling = true;
   857→    }
   858→    fold();
   859→    return { steps: results.length, results };
   860→  }

   862→  function cmdUndo(): unknown {
   863→    const label = undoStack.length > 0 ? (undoStack[undoStack.length - 1]?.label ?? null) : null;
   864→    const entry = undoStack.pop();
   865→    if (entry) {
   866→      redoStack.push({ label: entry.label, snapshot: snapshot() });
   867→      restore(entry.snapshot);
   868→    }
   869→    return { undone: label };
   870→  }

   872→  function cmdRedo(): unknown {

   877→      restore(entry.snapshot);
   878→      dirty = true;
   879→    }
   880→    return { redone: label };
   881→  }

  1042→  /** The run/list-time guards of JOURNALED commands (queries are `always`, query! macro). */
  1043→  function enabledCheck(spec: FakeCommandSpec): { ok: boolean; why: string | null } {
  1044→    const active = activeCompOrNull();
  1045→    switch (spec.id) {
  1046→      case "layer.newSolid":
  1047→      case "layer.newText":
  1048→        // has_comp — commands/mod.rs:393 "no composition is open"
  1049→        return active ? { ok: true, why: null } : { ok: false, why: "no composition is open" };
  1050→      case "layer.rename":
  1051→      case "prop.set":
  1052→        // has_layers — commands/mod.rs:397 "select a layer first"
  1053→        if (!active) return { ok: false, why: "no composition is open" };
  1054→        return selectedLayers.length > 0
  1055→          ? { ok: true, why: null }
  1056→          : { ok: false, why: "select a layer first" };
  1057→      case "edit.undo":
  1058→        // can_undo — edit.rs:12 "nothing to undo"
  1059→        return undoStack.length > 0
  1060→          ? { ok: true, why: null }
  1061→          : { ok: false, why: "nothing to undo" };
  1062→      case "edit.redo":
  1063→        // can_redo — edit.rs:15 "nothing to redo"
  1064→        return redoStack.length > 0
  1065→          ? { ok: true, why: null }
  1066→          : { ok: false, why: "nothing to redo" };
  1067→      default:
  1068→        // `always` (comp.new, engine.batch, every query!)
  1069→        return { ok: true, why: null };
  1070→    }
  1071→  }

  1139→  function queryHistoryList(): Record<string, unknown> {
  1140→    // Linear approximation of the upstream history tree (engine/src/history.rs):
  1141→    // every state's label is the step that MADE it (the oldest is `ROOT_LABEL`
  1142→    // "Original"), `parent` is the state index it was made from, ids are
  1143→    // positional (`s0`, `s1`, …) instead of upstream's content-keyed hex ids,
  1144→    // and there are no branches (`branches: 0`).
  1145→    const states: Array<Record<string, unknown>> = [];
  1146→    const n = undoStack.length;
  1147→    for (let i = 0; i <= n; i++) {
  1148→      const label = i === 0 ? ROOT_LABEL : (undoStack[i - 1]?.label ?? ROOT_LABEL);
  1149→      states.push({
  1150→        index: i,
  1151→        id: `s${i}`,
  1152→        label,
  1153→        parent: i === 0 ? null : i - 1,
  1154→        depth: 0,
  1155→        current: i === n,
  1156→        line: true,
  1157→        future: false,
  1158→      });
  1159→    }
  1160→    for (let j = 0; j < redoStack.length; j++) {
  1161→      const idx = n + 1 + j;
  1162→      states.push({
  1163→        index: idx,
  1164→        id: `s${idx}`,
  1165→        label: redoStack[j]?.label ?? ROOT_LABEL,
  1166→        parent: idx - 1,
  1167→        depth: 0,
  1168→        current: false,
  1169→        line: true,
  1170→        future: true,
  1171→      });
  1172→    }
  1173→    return { states, current: n, branches: 0 };
  1174→  }

  1309→  function runCommand(command: string, params: Record<string, unknown>): unknown {
  1310→    const spec = REGISTRY.find((s) => s.id === command);
  1311→    if (!spec) throw `unknown command \`${command}\``;
  1312→    const failWith = overrides.failOn?.[command];
  1313→    if (failWith !== undefined) throw failWith;
  1314→    const run = COMMANDS[command] ?? QUERIES[command];
  1315→    if (run === undefined) throw `unknown command \`${command}\``;
  1316→    // `execute_checked`: edit commands run their enabled guard (queries use `always`).
  1317→    if (COMMANDS[command] !== undefined) {
  1318→      const en = enabledCheck(spec);
  1319→      if (!en.ok) throw `command \`${command}\` is not available right now: ${en.why}`;
  1320→    }
  1321→    return run(params);
  1322→  }
