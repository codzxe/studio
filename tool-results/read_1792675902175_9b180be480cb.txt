   186→function defaultSleep(ms: number): Promise<void> {
   187→  if (ms <= 0) return Promise.resolve();
   188→  return new Promise((resolve) => {
   189→    setTimeout(resolve, ms);
   190→  });
   191→}

   193→/** Serialized query string; `undefined`/`null` values are skipped. */
   194→export function buildQuery(
   195→  params: Record<string, string | number | boolean | null | undefined>,
   196→): string {
   197→  const search = new URLSearchParams();
   198→  for (const [key, value] of Object.entries(params)) {
   199→    if (value === undefined || value === null) continue;
   200→    search.set(key, String(value));
   201→  }
   202→  const text = search.toString();
   203→  return text === "" ? "" : `?${text}`;
   204→}

   221→interface ClientInternals {
   222→  readonly baseUrl: string;
   223→  readonly fetchImpl: FetchLike;
   224→  readonly extraHeaders: Record<string, string>;

   228→}

   233→
   234→class ApiClientImpl {
   235→  constructor(readonly core: ClientInternals) {}
   236→
   237→  /**
   238→   * One HTTP call: envelope unwrap, response validation, GET retry with
   239→   * backoff. Never surfaces raw response text — errors are typed ApiErrors.
   240→   */
   241→  async request<T>(plan: RequestPlan<T>): Promise<T> {
   242→    const url = `${this.core.baseUrl}${plan.path}${plan.query ? buildQuery(plan.query) : ""}`;
   243→
   244→    let attempt = 0;
   245→    for (;;) {
   246→      let status = 0;
   247→      let statusText = "";
   248→      let text = "";
   249→      try {
   250→        const init = this.buildInit(plan);
   251→        const response = await this.core.fetchImpl(url, init);
   252→        status = response.status;
   253→        statusText = response.statusText;
   254→        text = await response.text();
   255→      } catch (error) {
   256→        if (isAbortError(error) || plan.signal?.aborted) throw error;
   257→        const netError = networkError(error);
   258→        if (!this.shouldRetry(plan, netError, attempt)) throw netError;
   259→        await this.backoff(plan, attempt);
   260→        attempt += 1;
   261→        continue;
   262→      }
   263→
   264→      if (status < 200 || status > 299) {
   265→        const apiError = await fromResponse({
   266→          status,
   267→          statusText,
   268→          text: async () => text,
   269→        });
   270→        if (!this.shouldRetry(plan, apiError, attempt)) throw apiError;
   271→        await this.backoff(plan, attempt);
   272→        attempt += 1;
   273→        continue;
   274→      }
   275→
   276→      return this.unwrapOk<T>(status, text, plan);
   277→    }
   278→  }
   279→
   280→  private buildInit(plan: RequestPlan<unknown>): RequestInit {
   281→    const headers = new Headers(this.core.extraHeaders);
   282→    headers.set("Accept", "application/json");
   283→    if (plan.body !== undefined) headers.set("Content-Type", "application/json");
   284→    if (plan.idempotencyKey !== undefined) headers.set("Idempotency-Key", plan.idempotencyKey);
   285→    const init: RequestInit = {
   286→      method: plan.method,
   287→      headers,
   288→      signal: plan.signal,
   289→    };
   290→    if (plan.body !== undefined) init.body = JSON.stringify(plan.body);
   291→    return init;
   292→  }
   293→
   294→  /** Only idempotent-safe GETs retry, and only when the failure is retryable. */
   295→  private shouldRetry(plan: RequestPlan<unknown>, error: ApiError, attempt: number): boolean {
   296→    if (plan.method !== "GET") return false;
   297→    if (attempt >= this.core.retryDelays.length) return false;
   298→    if (error.status === 0) return true; // network error before any response
   299→    if (error.status === 408 || error.status === 429 || error.status >= 500) {
   300→      return error.retryable;
   301→    }
   302→    return false;
   303→  }
   304→
   305→  private backoff(plan: RequestPlan<unknown>, attempt: number): Promise<void> {
   306→    const delay = this.core.retryDelays[attempt] ?? 0;
   307→    return sleepWithAbort(this.core.sleep, delay, plan.signal);
   308→  }
   309→
   310→  /** 2xx handling: empty body → `null`; otherwise a §7.3 Ok envelope is required. */
   311→  private unwrapOk<T>(status: number, text: string, plan: RequestPlan<T>): T {
   312→    if (text === "") {
   313→      if (plan.responseSchema === undefined) return null as T;
   314→      throw malformedResponseError(status);
   315→    }
   316→    let json: unknown;
   317→    try {
   318→      json = JSON.parse(text);
   319→    } catch {
   320→      throw malformedResponseError(status);
   321→    }
   322→    if (isErr(json)) {
   323→      const error: Err["error"] = json.error;
   324→      // A 2xx status with an Err body is still the §7.3 failure arm.
   325→      throw new ApiError({
   326→        code: error.code,
   327→        message: error.message,
   328→        hint: error.hint,
   329→        retryable: error.retryable,
   330→        requestId: error.requestId,
   331→        status,
   332→        cause: json,
   333→      });
   334→    }
   335→    if (!isOk(json)) throw malformedResponseError(status);
   336→    const data = json.data;
   337→    if (plan.responseSchema === undefined) return data as T;
   338→    try {
   339→      return plan.responseSchema.parse(data);
   340→    } catch (error) {
   341→      throw malformedResponseError(status, error);
   342→    }
   343→  }
   344→
   345→  /** Client-side §7.2 request validation — fails fast, never sends. */
   346→  validateBody<T>(schema: z.ZodType<T>, body: unknown, path: string): T {
   347→    try {
   348→      return schema.parse(body);
   349→    } catch (error) {
   350→      if (isApiError(error)) throw error;
   351→      throw invalidRequestError(
   352→        Object.assign(new Error(`invalid body for ${path}`), { cause: error }),
   353→      );
   354→    }
   355→  }
   356→
   357→  autoKey(): string {
   358→    return this.core.newKey();
   359→  }
   360→
   361→  // -- GETs don't send Idempotency-Key; mutations always do (auto when absent).
   362→  async get<T>(
   363→    path: string,
   364→    query: Record<string, string | number | boolean | undefined> | undefined,
   365→    schema: z.ZodType<T>,
   366→    signal?: AbortSignal,
   367→  ): Promise<T> {
   368→    return this.request<T>({ method: "GET", path, query, responseSchema: schema, signal });
   369→  }
   370→
   371→  async mutate<T>(
   372→    method: "POST" | "PATCH" | "DELETE",
   373→    path: string,
   374→    plan: {
   375→      body?: unknown;
   376→      bodySchema?: z.ZodType<unknown>;
   377→      responseSchema?: z.ZodType<T>;
   378→      idempotencyKey?: string;
   379→      signal?: AbortSignal;
   380→    },
   381→  ): Promise<T> {
   382→    const body =
   383→      plan.bodySchema !== undefined
   384→        ? this.validateBody(plan.bodySchema, plan.body, path)
   385→        : plan.body;
   386→    return this.request<T>({
   387→      method,
   388→      path,
   389→      body,
   390→      responseSchema: plan.responseSchema,
   391→      idempotencyKey: plan.idempotencyKey ?? this.autoKey(),
   392→      signal: plan.signal,
   393→    });
   394→  }
   395→}

   534→/**
   535→ * Builds the typed API client. `baseUrl` includes the version segment
   536→ * (default `"/v1"` for same-origin browser use). All mutation methods send
   537→ * `Idempotency-Key` (§7.2) — pass your own per logical operation, or let the
   538→ * client derive one.
   539→ */
   540→export function createApiClient(options: ApiClientOptions = {}): ApiClient {
   541→  const core: ClientInternals = {
   542→    baseUrl: (options.baseUrl ?? DEFAULT_BASE_URL).replace(/\/+$/, ""),
   543→    fetchImpl: options.fetch ?? ((input, init) => fetch(input, init)),
   544→    extraHeaders: { ...(options.headers ?? {}) },
   545→    newKey: options.idempotencyKey ?? (() => crypto.randomUUID()),
   546→    sleep: options.sleep ?? defaultSleep,
   547→    retryDelays: options.retry?.delaysMs ?? RETRY_DELAYS_MS,
   548→  };
   549→  const impl = new ApiClientImpl(core);
   550→
   551→  const mutationKey = (options: CallOptions | undefined): string | undefined => {
   552→    const explicit = options?.idempotencyKey;
   553→    if (explicit !== undefined) {
   554→      // §7.2: client-generated, ≥8 chars (contracts IDEMPOTENCY_KEY_SCHEMA).
   555→      return impl.validateBody(IDEMPOTENCY_KEY_SCHEMA, explicit, "Idempotency-Key");
   556→    }
   557→    return undefined;
   558→  };
   559→
   560→  const queryOf = (
   561→    schema: z.ZodType<unknown>,
   562→    query: unknown,
   563→  ): Record<string, string | number | boolean | undefined> => {
   564→    const parsed = impl.validateBody(schema, query ?? {}, "list query");
   565→    const out: Record<string, string | number | boolean | undefined> = {};
   566→    for (const [k, v] of Object.entries(parsed as Record<string, unknown>)) {
   567→      out[k] = v as string | number | boolean | undefined;
   568→    }

   841→/** Two-phase version upload (§9.2/§9.3): descriptor POST + signed-URL byte PUT. */
   842→async function uploadVersion(
   843→  impl: ApiClientImpl,
   844→  projectId: string,
   845→  input: {
   846→    readonly baseVersionId: string;
   847→    readonly sha256: string;
   848→    readonly label?: string;
   849→    readonly agentRunId?: string;
   850→    readonly bytes: Blob | Uint8Array;
   851→  },
   852→  options: CallOptions | undefined,
   853→): Promise<ProjectVersion> {
   854→  const { bytes } = input;
   855→  if (!(bytes instanceof Blob) && !(bytes instanceof Uint8Array)) {
   856→    throw invalidRequestError(new Error("bytes must be a Blob or Uint8Array"));
   857→  }
   858→  const sizeBytes = bytes instanceof Blob ? bytes.size : bytes.byteLength;
   859→  const descriptor = impl.validateBody(
   860→    UPLOAD_VERSION_SCHEMA,
   861→    {
   862→      baseVersionId: input.baseVersionId,
   863→      sha256: input.sha256,
   864→      label: input.label,
   865→      agentRunId: input.agentRunId,
   866→      sizeBytes,
   867→    },
   868→    "upload version descriptor",
   869→  );
   870→
   871→  const response = await impl.mutate<UploadVersionResponse>(
   872→    "POST",
   873→    `/projects/${encodeId(projectId)}/versions`,
   874→    {
   875→      body: descriptor,
   876→      responseSchema: UPLOAD_VERSION_RESPONSE_SCHEMA,
   877→      idempotencyKey: options?.idempotencyKey ?? impl.autoKey(),
   878→      signal: options?.signal,
   879→    },
   880→  );
   881→
   882→  // Phase two: bytes go straight to object storage via the signed URL — the
   883→  // storage response is NOT a §7.3 envelope, so only the status is checked.
   884→  const { url, headers } = response.upload;
   885→  const put = await impl.core.fetchImpl(url, {
   886→    method: "PUT",
   887→    headers: { ...headers },
   888→    // TS 5.7's DOM BodyInit doesn't accept Uint8Array<ArrayBufferLike>
   889→    // (SharedArrayBuffer edge); runtime fetch (browser + undici) does.
   890→    body: bytes instanceof Blob ? bytes : (bytes as unknown as BodyInit),
   891→    signal: options?.signal,
   892→  });
   893→  if (!put.ok) {
   894→    throw uploadFailedError(put.status);
   895→  }
   896→  return response.version;
   897→}
