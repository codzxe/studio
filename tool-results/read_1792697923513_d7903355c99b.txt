   571→/** Raw chat POST (the api() helper has no abort/signal seam). */
   572→async function chatStream(
   573→  fx: Fixture,
   574→  body: ChatBody,
   575→  options: ChatCallOptions = {},
   576→): Promise<Response> {
   577→  const headers: Record<string, string> = {
   578→    origin: TEST_APP_ORIGIN,
   579→    "content-type": "application/json",
   580→    accept: "text/event-stream",
   581→    cookie: fx.jar.header,
   582→    "x-workspace-id": fx.workspaceId,
   583→  };
   584→  if (options.lastEventId !== undefined) headers["last-event-id"] = options.lastEventId;
   585→  return fetch(`${server.base}/v1/threads/${fx.threadId}/chat`, {
   586→    method: "POST",
   587→    headers,
   588→    body: JSON.stringify(body),
   589→    signal: options.signal,
   590→    redirect: "manual",
   591→  });
   592→}

   646→/** Poll until probe() is true (the route emits run.completed BEFORE its DB writes). */
   647→async function waitFor(
   648→  description: string,
   649→  probe: () => Promise<boolean>,
   650→  timeoutMs = 4000,
   651→): Promise<void> {
   652→  const deadline = Date.now() + timeoutMs;
   653→  for (;;) {
   654→    if (await probe()) return;
   655→    if (Date.now() > deadline) throw new Error(`timed out waiting for ${description}`);
   656→    await new Promise((resolve) => {
   657→      setTimeout(resolve, 25);
   658→    });
   659→  }
   660→}
