   185→export interface GoldenExpectation {
   186→  /**
   187→   * Today's REAL-pipeline outcome. "compiled" ⇒ the assertions below hold on
   188→   * the compiled result now. "blocked" ⇒ the compile answers a TYPED FATAL
   189→   * (fail-closed: no batch is produced at all) because of FINDING G-1/G-6 —
   190→   * `commands` then carries the pinned POST-FIX golden for the fix.
   191→   */
   192→  readonly compile: "compiled" | "blocked";
   193→  /** The exact fatal issue code asserted when `compile` is "blocked". */
   194→  readonly fatalCode?: "empty_batch";
   195→  /**
   196→   * Registry command ids of the compiled batch, in order (the §M5.2 golden).
   197→   * For blocked fixtures: the post-reconciliation golden per the compiler's
   198→   * own translation table — every id is in AGENT_COMMAND_IDS (self-checked).
   199→   */
   200→  readonly commands: readonly string[];
   201→  /** Inclusive [min, max] range on the compiled batch step count. */
   202→  readonly batchStepRange: readonly [number, number];
   203→  /** Server-RECOMPUTED risk carried on the plan event (never the model's). */
   204→  readonly risk: PlanRisk;
   205→  /** estimateCredits({ risk, steps: plan.steps }) — never the model's number. */
   206→  readonly estimatedCredits: number;
   207→  /** Layer NAMES needing client-side id resolution, in request order (§8.7). */
   208→  readonly resolutionRequests: readonly string[];
   209→  /** Expected WARN issue codes on a compiled result (subset match). */
   210→  readonly warningCodes: readonly string[];
   211→  /** Present when the plan carries a save_version annotation. */
   212→  readonly saveVersionLabel?: string;
   213→  /** Finding references explaining a non-golden expectation. */
   214→  readonly findings: readonly string[];
   215→}

   233→function baseContext(overrides: Partial<ContextSelection> = {}): ContextSelection {
   234→  return {
   235→    projectSummary: true,
   236→    activeComp: true,
   237→    selection: false,
   238→    playhead: true,
   239→    focusedPanel: true,
   240→    selectedProperties: false,
   241→    currentFrame: false,
   242→    recentCommands: false,
   243→    comments: false,
   244→    ...overrides,
   245→  };
   246→}

   248→function selectionContext(): ContextSelection {
   249→  return baseContext({ selection: true, selectedProperties: true });
   250→}

   252→/** A §8.6 plan step: curated write tool id + args + the model's ADVISORY
   253→ * batch proposal (pseudo command ids = the bare tool id, per the M5-C write
   254→ * tool carrier convention — never a real registry id, so the wire proofs can
   255→ * assert the compiled registry ids NEVER enter the model conversation). */
   256→function step(
   257→  title: string,
   258→  tool: string,
   259→  args: unknown,
   260→  proposedParams: Record<string, unknown>,
   261→): EditPlan["steps"][number] {
   262→  return {
   263→    title,
   264→    tool,
   265→    args,
   266→    batch: {
   267→      label: title,
   268→      steps: [{ command: tool.replace(/^studio\./, ""), params: proposedParams }],
   269→    },
   270→  };
   271→}
