   157→async function uploadVersion(
   158→  ws: Workspace,
   159→  projectId: string,
   160→  fixture: { bytes: Buffer; sha256: string },
   161→): Promise<string> {
   162→  const query = new URLSearchParams({
   163→    sha256: fixture.sha256,
   164→    sizeBytes: String(fixture.bytes.byteLength),
   165→  });
   166→  const response = await api(
   167→    server.base,
   168→    `/v1/projects/${projectId}/versions?${query.toString()}`,
   169→    {
   170→      method: "POST",
   171→      jar: ws.jar,
   172→      headers: {
   173→        "x-workspace-id": ws.id,
   174→        "idempotency-key": unique("upload"),
   175→        "content-type": "application/octet-stream",
   176→      },
   177→      rawBody: new Uint8Array(fixture.bytes),
   178→    },
   179→  );
   180→  expect(response.status).toBe(201);
   181→  const body = (await response.json()) as { data: { id: string } };
   182→  return body.data.id;
   183→}

   215→interface Envelope<T> {
   216→  readonly status: number;
   217→  readonly data?: T;
   218→  readonly error?: { code: string };
   219→}
   220→
   221→async function call<T>(
   222→  ws: Workspace,
   223→  path: string,
   224→  init: { method: string; body?: unknown; headers?: Record<string, string> } = { method: "GET" },
   225→): Promise<Envelope<T>> {
   226→  const response = await api(server.base, path, {
   227→    method: init.method,
   228→    jar: ws.jar,
   229→    headers: { "x-workspace-id": ws.id, ...init.headers },
   230→    body: init.body,
   231→  });
   232→  const text = await response.text();
   233→  const parsed = JSON.parse(text) as { data?: T; error?: { code: string } };
   234→  return { status: response.status, data: parsed.data, error: parsed.error };
   235→}

   287→/** Seed COMPLETED render minutes this month (frames_total / frameRate). */
   288→async function seedCompletedSeconds(
   289→  ws: Workspace,
   290→  projectId: string,
   291→  seconds: number,
   292→): Promise<void> {
   293→  const version = await server.deps.repos.projects.get(ws.id, projectId);
   294→  const render = await server.deps.repos.renders.create(ws.id, {
   295→    projectId,
   296→    versionId: version.currentVersionId ?? "ver_missing",
   297→    compId: "seed",
   298→    format: "mp4",
   299→    settingsJson: { frameRate: 30 },
   300→    createdBy: ws.userId,
   301→  });
   302→  await server.deps.repos.renders.markRunning(ws.id, render.id);
   303→  await server.deps.repos.renders.heartbeat(ws.id, render.id, {
   304→    framesTotal: seconds * 30,
   305→  });
   306→  await server.deps.repos.renders.markCompleted(ws.id, render.id, { objectKey: "seed/out.mp4" });
   307→}
