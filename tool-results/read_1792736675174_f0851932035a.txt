    84→/**
    85→ * The billing surface `@studio/api-client` exposes. Production builds it from
    86→ * `createApiClient` (same-origin `/v1` + `x-workspace-id`); tests fake it.
    87→ */
    88→export interface BillingPort {
    89→  /** POST /v1/billing/checkout — subscribe/switch to a paid plan (D-4). */
    90→  readonly checkout: (body: CheckoutRequest, options?: CallOptions) => Promise<CheckoutResponse>;
    91→  /** POST /v1/billing/topup — one-time-payment credit pack. */
    92→  readonly topup: (body: TopUpRequest, options?: CallOptions) => Promise<TopUpResponse>;
    93→  /** POST /v1/billing/portal — Stripe customer portal session. */
    94→  readonly portal: (body?: PortalRequest, options?: CallOptions) => Promise<PortalResponse>;
    95→  /** GET /v1/billing/state — the ONE billing read (refreshes after switches). */
    96→  readonly state: (options?: CallOptions) => Promise<BillingState>;
    97→  /** GET /v1/credits/ledger — the append-only credit log (cursor pages). */
    98→  readonly ledger: LedgerPageFetcher;
    99→}

   101→/**
   102→ * Compile-time pin (and tiny adapter): the production `ApiClient["billing"]`
   103→ * plus `credits.ledger` satisfies {@link BillingPort} — the page never
   104→ * hand-rolls a call.
   105→ */
   106→export function asBillingPort(client: ApiClient): BillingPort {
   107→  return {
   108→    checkout: client.billing.checkout,
   109→    topup: client.billing.topup,
   110→    portal: client.billing.portal,
   111→    state: client.billing.state,
   112→    ledger: (query) => client.credits.ledger(query),
   113→  };
   114→}

   116→/** Same-origin API root (Caddy fronts `/v1` in production — README §Env). */
   117→const API_BASE_URL = "/v1";

   119→/** Builds the production port for one workspace page load. */
   120→function createDefaultBillingPort(workspaceId: string): BillingPort {
   121→  return asBillingPort(
   122→    createApiClient({ baseUrl: API_BASE_URL, headers: { "x-workspace-id": workspaceId } }),
   123→  );

   166→// ---------------------------------------------------------------------------
   167→// Typed action errors (§7.3 surface: message + hint + request id)
   168→// ---------------------------------------------------------------------------
   169→
   170→interface BillingActionError {
   171→  readonly message: string;
   172→  readonly hint?: string;
   173→  readonly requestId?: string;
   174→}
   175→
   176→function toActionError(error: unknown): BillingActionError {
   177→  if (isApiError(error)) {
   178→    return { message: error.message, hint: error.hint, requestId: error.requestId };
   179→  }
   180→  return { message: "Something went wrong on our side.", hint: "Try again in a moment." };
   181→}

   183→// ---------------------------------------------------------------------------
   184→// Status presentation (§8.8 plan card; days remaining for trialing/grace)
   185→// ---------------------------------------------------------------------------
   186→
   187→const LIVE_STATUSES = new Set(["trialing", "active", "past_due", "grace"]);
   188→
   189→type BadgeVariant = "neutral" | "success" | "warning" | "danger" | "ice";
   190→
   191→interface StatusView {
   192→  readonly badge: BadgeVariant;
   193→  readonly label: string;
   194→  readonly line: string;
   195→}
   196→
   197→function dayWord(count: number): string {
   198→  return `${formatCount(count)} ${count === 1 ? "day" : "days"} left`;
   199→}
   200→
   201→function statusView(state: BillingState, now: Date): StatusView {
   202→  switch (state.status) {
   203→    case "trialing":
   204→      return {
   205→        badge: "ice",
   206→        label: "Trial",
   207→        line:
   208→          state.trialEnd === null
   209→            ? "Your trial is running."
   210→            : `Ends ${formatDate(state.trialEnd)} — ${dayWord(daysUntil(state.trialEnd, now))}.`,
   211→      };
   212→    case "active":
   213→      return state.cancelAtPeriodEnd
   214→        ? {
   215→            badge: "warning",
   216→            label: "Active",
   217→            line: `Renewal is cancelled — your plan ends ${formatDate(state.currentPeriodEnd)}.`,
   218→          }
   219→        : {
   220→            badge: "success",
   221→            label: "Active",
   222→            line: `Renews ${formatDate(state.currentPeriodEnd)}.`,
   223→          };
   224→    case "past_due":
   225→      return {
   226→        badge: "danger",
   227→        label: "Past due",
   228→        line: "The card on file was declined — update your payment to keep creating.",
   229→      };
   230→    case "grace":
   231→      return {
   232→        badge: "warning",
   233→        label: "Grace period",
   234→        line:
   235→          state.graceEnd === null
   236→            ? "Update your payment to keep creating."
   237→            : `Grace period ends ${formatDate(state.graceEnd)} — ${dayWord(daysUntil(state.graceEnd, now))} to update payment.`,
   238→      };
   239→    case "canceled":
   240→      return {
   241→        badge: "danger",
   242→        label: "Cancelled",
   243→        line: "Choose a plan below to subscribe again. Your work is never deleted.",
   244→      };
   245→    case "incomplete":
   246→      return {
   247→        badge: "warning",
   248→        label: "Payment incomplete",
   249→        line: "Finish payment to activate your subscription.",
   250→      };
   251→    case "none":
   252→      return {
   253→        badge: "neutral",
   254→        label: "No subscription",
   255→        line: "Choose a plan below to start creating.",
   256→      };
   257→    default:
   258→      // Wire drift: an unknown status never breaks the page.
   259→      return {
   260→        badge: "neutral",
   261→        label: "Status unavailable",
   262→        line: "Refresh to see the latest billing status.",
   263→      };
   264→  }
   265→}

   282→  const [state, setState] = useState<BillingState>(initialState);
   283→  const [busy, setBusy] = useState<string | null>(null);
   284→  const [actionError, setActionError] = useState<BillingActionError | null>(null);
   285→  const [cancelOpen, setCancelOpen] = useState(false);
   286→  const now = useMemo(() => new Date(nowIso), [nowIso]);
   287→
   288→  const isLive = LIVE_STATUSES.has(state.status);
   289→  const current = CURRENT_PLAN_DISPLAY[state.plan] ?? { name: state.plan, priceCents: null };
   290→  const priceLine =
   291→    current.priceCents === null
   292→      ? "Included during the trial"
   293→      : formatMonthlyPrice(current.priceCents);
   294→
   295→  const ledger = useLedgerPages(port.ledger);
   296→
   297→  async function refresh(): Promise<void> {
   298→    if (busy !== null) return;
   299→    setBusy("refresh");
   300→    setActionError(null);
   301→    try {
   302→      setState(await port.state());
   303→    } catch (error) {
   304→      setActionError(toActionError(error));
   305→    } finally {
   306→      setBusy(null);
   307→    }
   308→  }
   309→

   315→    try {
   316→      const session = await port.portal({
   317→        returnUrl: `${window.location.origin}/w/billing`,
   318→      });
   319→      window.location.assign(session.url);
   320→    } catch (error) {
   321→      setActionError(toActionError(error));
   322→    } finally {
   323→      setBusy(null);
   324→    }
   325→  }
   326→
   327→  const status = statusView(state, now);
   328→  const seatsLine =
   329→    seatsUsed === undefined
   330→      ? `Up to ${formatCount(state.entitlements.seats)} seats`
   331→      : `${formatCount(seatsUsed)} of ${formatCount(state.entitlements.seats)} seats used`;

   333→  return (
   334→    <main id="main" data-testid="billing-view" className="min-h-svh bg-app px-5 py-10">
   335→      <div className="mx-auto flex w-full max-w-[1100px] flex-col gap-6">
   336→        <header className="flex flex-wrap items-end justify-between gap-4">
   337→          <div className="flex flex-col gap-1">
   338→            <h1 className="text-title-l font-medium text-primary">Billing</h1>
   339→            <p className="text-body-s text-muted">
   340→              Plan, credits and invoices for {workspaceName}.
   341→            </p>
   342→          </div>
   343→          <div className="flex items-center gap-2">
   344→            <Button
   345→              variant="ghost"
   346→              size="sm"
   347→              loading={busy === "refresh"}
   348→              disabled={busy !== null && busy !== "refresh"}
   349→              onClick={() => {
   350→                void refresh();
   351→              }}
   352→              data-testid="billing-refresh"
   353→            >
   354→              Refresh
   355→            </Button>
   356→            <a
   357→              href="/w"
   358→              className="text-body-s text-secondary underline underline-offset-3 hover:text-primary"
   359→            >
   360→              Back to workspace
   361→            </a>
   362→          </div>
   363→        </header>
   364→
   365→        {/* §8.8 low-credit banner (RAW µc; never an auto-charge). */}
   366→        <LowCreditBanner
   367→          balanceMicrocredits={state.credits.balance}
   368→          allowanceMicrocredits={state.entitlements.monthlyCreditAllowance}
   369→          onTopUp={
   370→            canManage
   371→              ? () => {
   372→                  document
   373→                    .querySelector<HTMLElement>('[data-testid="topup-flow"]')
   374→                    ?.scrollIntoView();
   375→                }
   376→              : undefined
   377→          }
   378→        />
   379→
   380→        {actionError !== null && (
   381→          <div
   382→            role="alert"
   383→            data-testid="billing-error"
   384→            className="flex flex-col gap-1 rounded-sm border border-danger/30 bg-danger/10 p-4 text-body-s"

   386→            <span className="font-medium text-primary">{actionError.message}</span>
   387→            {actionError.hint !== undefined && (
   388→              <span className="text-secondary">{actionError.hint}</span>
   389→            )}

   557→        {/* ── Credit ledger (M6-G kit; the append-only log) ────────────── */}

   575→        {/* ── Cancel ──────────────────────────────────────────────────── */}

   592→              data-testid="billing-cancel-button"

   606→              }

   616→        {canManage && isLive && state.cancelAtPeriodEnd && (

   631→              }}

   635→            </Button>

   638→      </div>

   647→const INVOICE_BADGE: Readonly<Record<BillingInvoiceRow["status"], BadgeVariant>> = {
   648→  draft: "neutral",
   649→  open: "ice",
   650→  paid: "success",
   651→  void: "neutral",
   652→  uncollectible: "danger",
   653→};
   654→
   655→function formatInvoiceAmount(amountCents: number, currency: string): string {
   656→  if (!Number.isFinite(amountCents) || amountCents < 0 || currency === "") return "—";
   657→  try {
   658→    return new Intl.NumberFormat("en-US", { style: "currency", currency }).format(
   659→      amountCents / 100,
   660→    );
   661→  } catch {
   662→    return "—";
   663→  }
   664→}
   665→
   666→function InvoiceTable({ invoices }: { invoices: readonly BillingInvoiceRow[] }) {
   667→  if (invoices.length === 0) {
   668→    return (
   669→      <p className="text-body-s text-muted" data-testid="billing-invoices-empty">
   670→        Invoices appear here after your first payment.
   671→      </p>
   672→    );
   673→  }
   674→  return (
   675→    <ul className="flex flex-col divide-y divide-border" data-testid="billing-invoice-list">
   676→      {invoices.map((invoice) => (
   677→        <li
   678→          key={invoice.id}
   679→          data-testid="billing-invoice-row"
   680→          className="flex flex-wrap items-center justify-between gap-3 py-3"
   681→        >
   682→          <span className="text-body-s text-secondary tabular-nums">
   683→            {formatDate(invoice.date)}
   684→          </span>
   685→          <span className="text-body-s font-medium text-primary tabular-nums">
   686→            {formatInvoiceAmount(invoice.amountCents, invoice.currency)}
   687→          </span>
   688→          <Badge variant={INVOICE_BADGE[invoice.status]} data-testid="billing-invoice-status">
   689→            {invoice.status}
   690→          </Badge>
   691→          <span className="flex items-center gap-3">
   692→            {invoice.hostedUrl !== null && (
   693→              <a
   694→                href={invoice.hostedUrl}
   695→                className="text-body-s text-secondary underline underline-offset-3 hover:text-primary"
   696→                data-testid="billing-invoice-view"
   697→              >
   698→                View
   699→              </a>
   700→            )}
   701→            {invoice.pdfUrl !== null && (
   702→              <a
   703→                href={invoice.pdfUrl}
   704→                className="text-body-s text-secondary underline underline-offset-3 hover:text-primary"
   705→                data-testid="billing-invoice-pdf"
   706→              >
   707→                PDF
   708→              </a>
   709→            )}
   710→          </span>
   711→        </li>
   712→      ))}
   713→    </ul>
   714→  );
   715→}
