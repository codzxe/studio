    13→async function rejects(promise: Promise<unknown>): Promise<unknown> {
    14→  return promise.then(
    15→    () => {
    16→      throw new Error("expected a rejection");
    17→    },
    18→    (e: unknown) => e,
    19→  );

   183→describe("fake editor: engine.batch", () => {
   184→  it("runs $N-referenced steps and folds into ONE undo step", async () => {
   185→    const fake = createFakeEditor();
   186→    await fake.execute("comp.new", { name: "B" });
   187→    const undoBefore = fake.state.history.undo.length;
   188→
   189→    const result = (await fake.execute("engine.batch", {
   190→      label: "Build",
   191→      steps: [
   192→        { command: "layer.newSolid", params: { name: "Box", color: "#ff0000" } },
   193→        {
   194→          command: "prop.set",
   195→          params: { layer: "$1.layer", path: "transform/opacity", value: 40 },
   196→        },
   197→        { command: "layer.rename", params: { layer: "$1.layer", name: "Red Box" } },
   198→      ],
   199→    })) as { steps: number; results: Array<{ layer?: number }> };
   200→
   201→    expect(result.steps).toBe(3);
   202→    const layerId = result.results[0]?.layer;
   203→    const layer = fake.state.project.layers.find((l) => l.id === layerId);
   204→    expect(layer?.name).toBe("Red Box");
   205→    expect(layer?.props["transform/opacity"]?.value).toBe(40);
   206→    expect(fake.state.history.undo.length).toBe(undoBefore + 1);
   207→    expect(fake.state.history.undo.at(-1)?.label).toBe("Build");
   208→
   209→    // ONE edit.undo reverts the whole batch
   210→    const undone = (await fake.execute("edit.undo", {})) as { undone: string | null };
   211→    expect(undone.undone).toBe("Build");
   212→    expect(fake.state.project.layers).toEqual([]);
   213→    expect(fake.state.history.undo.length).toBe(undoBefore);
   214→
   215→    const redone = (await fake.execute("edit.redo", {})) as { redone: string | null };
   216→    expect(redone.redone).toBe("Build");
   217→    expect(fake.state.project.layers.length).toBe(1);
   218→  });

   220→  it("a failing atomic step rolls everything back and names the step", async () => {
   221→    const fake = createFakeEditor();
   222→    await fake.execute("comp.new", {});
   223→    await fake.execute("layer.newSolid", {});
   224→    const before = fake.state.project.layers.length;
   225→    const undoBefore = fake.state.history.undo.length;
   226→
   227→    const err = await rejects(
   228→      fake.execute("engine.batch", {
   229→        steps: [
   230→          { command: "layer.newText", params: { text: "x" } },
   231→          { command: "prop.set", params: { layer: "$1.layer", path: "transform/nope", value: 1 } },
   232→        ],
   233→      }),
   234→    );
   235→    expect(String(err).startsWith("step 2 (prop.set):")).toBe(true);
   236→    expect(String(err)).toContain("rolled back (nothing changed)");

   238→    expect(fake.state.history.undo.length).toBe(undoBefore); // no undo entry for the batch
   239→
   240→    // the rolled-back batch left no undo entry: undoing still reverts the
   241→    // PREVIOUS step, not the batch
   242→    const undone = (await fake.execute("edit.undo", {})) as { undone: string | null };
   243→    expect(undone.undone).toBe("New Solid");
