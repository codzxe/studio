    57→type ThreadMode = "auto" | "manual" | "reject-once";
    58→
    59→interface HarnessOptions {
    60→  /** "auto" resolves creation on the next microtask; "manual" waits for resolveCreate(); "reject-once" fails the first create. */
    61→  readonly threadMode?: ThreadMode;
    62→  /** "fail" makes every models.list attempt reject (entitlement failure → []). */
    63→  readonly models?: "ok" | "fail";
    64→}
    65→
    66→interface RecordedStreamCall {
    67→  readonly threadId: string;
    68→  readonly body: ChatRequest;
    69→  readonly handlers: {
    70→    onEvent?: (event: unknown) => void;
    71→    onError?: (error: unknown) => void;
    72→    onDone?: (summary: unknown) => void;
    73→  };
    74→  readonly cancel: ReturnType<typeof vi.fn>;
    75→}
    76→
    77→interface Harness {
    78→  readonly api: () => ChatContextApi;
    79→  readonly createThreadCalls: () => number;
    80→  readonly resolveCreateThread: () => void;
    81→  readonly modelsListCalls: () => number;
    82→  readonly streamCalls: RecordedStreamCall[];
    83→  readonly postEvents: Array<{ runId: string; body: Record<string, unknown> }>;
    84→  readonly bridgeCalls: { inspect: number; queries: string[] };
    85→  readonly bus: ChatShortcutBus;
    86→}
    87→
    88→const THREAD_IDS = ["thr_1", "thr_2"];
    89→
    90→function makeThread(id: string): Thread {
    91→  return {
    92→    id,
    93→    projectId: "prj_test",
    94→    title: null,
    95→    createdBy: "usr_1",
    96→    createdAt: "2026-10-22T00:00:00.000Z",
    97→    updatedAt: "2026-10-22T00:00:00.000Z",
    98→  };
    99→}
   100→
   101→const FAST_MODEL = {
   102→  id: "studio-fast",
   103→  tier: "fast" as const,
   104→  use: "Cheap planning, classification, quick edits",
   105→  creditsPer1kInput: 0.01,
   106→  creditsPer1kOutput: 0.03,
   107→};
   108→
   109→const MODELS_OK = { plan: "trial", models: [FAST_MODEL] };
   110→
   111→function makeHarness(options: HarnessOptions = {}): Harness {
   112→  const threadMode: ThreadMode = options.threadMode ?? "auto";
   113→  const modelsFail = options.models === "fail";
   114→
   115→  let createCalls = 0;
   116→  let rejectedOnce = false;
   117→  const manualResolvers: Array<(thread: Thread) => void> = [];
   118→  const streamCalls: RecordedStreamCall[] = [];
   119→  const postEvents: Array<{ runId: string; body: Record<string, unknown> }> = [];
   120→  let modelsCalls = 0;
   121→
   122→  const client = {
   123→    threads: {
   124→      create: (_projectId: string) => {
   125→        createCalls += 1;
   126→        const thread = makeThread(THREAD_IDS[createCalls - 1] ?? "thr_x");
   127→        if (threadMode === "manual") {
   128→          return new Promise<Thread>((resolve) => {
   129→            manualResolvers.push(resolve);
   130→          });
   131→        }
   132→        if (threadMode === "reject-once" && !rejectedOnce) {
   133→          rejectedOnce = true;
   134→          return Promise.reject(new Error("threads are unavailable right now"));
   135→        }
   136→        return Promise.resolve(thread);
   137→      },
   138→    },
   139→    models: {
   140→      list: () => {
   141→        modelsCalls += 1;
   142→        if (modelsFail) return Promise.reject(new Error("models endpoint down"));
   143→        return Promise.resolve(MODELS_OK);
   144→      },
   145→    },
   146→    chat: {
   147→      stream: (threadId: string, body: ChatRequest, handlers: RecordedStreamCall["handlers"]) => {
   148→        const call: RecordedStreamCall = {
   149→          threadId,
   150→          body,
   151→          handlers,
   152→          cancel: vi.fn(async () => undefined),
   153→        };
   154→        streamCalls.push(call);
   155→        return {
   156→          paused: false,
   157→          cancel: call.cancel,
   158→          abort: () => undefined,
   159→          reconnect: vi.fn(async () => undefined),
   160→        };
   161→      },
   162→    },
   163→    runs: {
   164→      postEvent: async (runId: string, body: Record<string, unknown>) => {
   165→        postEvents.push({ runId, body });
   166→        return null;
   167→      },
   168→    },
   169→  } as unknown as ApiClient;
   170→
   171→  // Recorded fake bridge (M4-C2's recordBridge pattern, trimmed to the calls
   172→  // the provider makes: inspect + query).
   173→  const fake: FakeEditor = createFakeEditor({ version: "0.13.0" });
   174→  const bridgeCalls = { inspect: 0, queries: [] as string[] };
   175→  const inner = attachFakeBridge(fake);
   176→  const bridge: EditorBridge = {
   177→    ...inner,
   178→    inspect: async () => {
   179→      bridgeCalls.inspect += 1;
   180→      return inner.inspect();
   181→    },
   182→    query: ((command: string, params?: unknown) => {
   183→      bridgeCalls.queries.push(command);
   184→      return inner.query(command, params);
   185→    }) as EditorBridge["query"],
   186→  };
   187→
   188→  const bus: ChatShortcutBus = createChatShortcutBus();
   189→
   190→  let captured: ChatContextApi | null = null;
   191→  render(
   192→    <ChatProvider
   193→      client={client}
   194→      bridge={bridge}
   195→      workspaceId="ws_test"
   196→      projectId="prj_test"
   197→      shortcutBus={bus}
   198→    >
   199→      {(api) => {
   200→        captured = api;
   201→        return <div data-testid="provider-probe" data-chat-phase={api.snapshot.phase} />;
   202→      }}
   203→    </ChatProvider>,
   204→  );
   205→
   206→  return {
   207→    api: () => {
   208→      if (captured === null) throw new Error("provider did not render");
   209→      return captured;
   210→    },
   211→    createThreadCalls: () => createCalls,
   212→    resolveCreateThread: () => {
   213→      const resolve = manualResolvers.shift();
   214→      if (resolve === undefined) throw new Error("no pending threads.create to resolve");
   215→      resolve(makeThread(THREAD_IDS[createCalls - 1] ?? "thr_x"));
   216→    },
   217→    modelsListCalls: () => modelsCalls,
   218→    streamCalls,
   219→    postEvents,
   220→    bridgeCalls,
   221→    bus,
   222→  };
   223→}
   224→
   225→beforeEach(() => {
   226→  executorRecorder.deps.length = 0;
   227→  executorRecorder.calls.length = 0;
   228→});
   229→
   230→afterEach(() => {
   231→  cleanup();
   232→  vi.useRealTimers();
   233→});
