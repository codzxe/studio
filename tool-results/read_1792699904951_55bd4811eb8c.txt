     1→// @studio/config — fail-fast environment schema (PRODUCT_ARCHITECTURE §14).
     2→//
     3→// Every Studio env var is declared exactly once here: name, purpose, group,
     4→// secret flag and a Zod schema. `parseEnv` refuses to start the process with a
     5→// single fatal error listing EVERY missing/invalid variable together with its
     6→// purpose. Secret values are never echoed — reasons are redacted against the
     7→// raw input before they can reach a log line.
     8→import { z } from "zod";
     9→import { Brand } from "./brand.js";
    10→
    11→/** Structurally compatible with `process.env`. */
    12→export type EnvSource = Record<string, string | undefined>;
    13→
    14→/** Deployment environments (§14.3). */
    15→export type AppEnv = "local" | "staging" | "production";
    16→
    17→/** Structured log levels. */
    18→export type LogLevel = "debug" | "info" | "warn" | "error";
    19→
    20→/** Sections of the typed config object and of the generated .env.example. */
    21→export type EnvGroupName =
    22→  | "app"
    23→  | "db"
    24→  | "queue"

    26→  | "stripe"

    34→/** Metadata for one env var — drives error messages and the .env.example generator. */
    35→export interface EnvVarMeta {
    36→  readonly name: string;
    37→  readonly purpose: string;
    38→  readonly group: EnvGroupName;
    39→  /** Required unconditionally. */
    40→  readonly required: boolean;
    41→  /** Value must never appear in errors, logs or examples with real data. */
    42→  readonly secret: boolean;
    43→  /** Optional with a schema default; the .env.example shows the default. */
    44→  readonly hasDefault: boolean;
    45→  /** Placeholder/example emitted by writeEnvExample (never a real secret). */
    46→  readonly example: string;
    47→  /** Extra cross-field rule worth restating next to the example. */
    48→  readonly condition?: string;
    49→}

   416→}

   418→// ───────────────────────────────────────────────────────────────────────────
   419→// Zod building blocks (zod v4). Env vars are strings; numerics are coerced with
   420→// explicit, human-readable reasons. Custom messages avoid echoing received
   421→// values, and `redact()` is the belt-and-braces net.
   422→// ───────────────────────────────────────────────────────────────────────────
   423→
   424→const httpUrl = z
   425→  .url({ error: "must be a valid absolute URL" })
   426→  .refine(
   427→    (value: string) => value.startsWith("http://") || value.startsWith("https://"),
   428→    "must be an http(s) URL",
   429→  );
   430→
   431→const postgresUrl = z
   432→  .string()
   433→  .min(1)
   434→  .refine(
   435→    (value: string) => /^postgres(ql)?:\/\//.test(value),
   436→    "must be a PostgreSQL URL starting with postgres:// or postgresql://",
   437→  );
   438→
   439→const valkeyUrl = z
   440→  .string()
   441→  .min(1)
   442→  .refine(
   443→    (value: string) => /^rediss?:\/\//.test(value),
   444→    "must be a Valkey URL starting with redis:// or rediss://",
   445→  );
   446→
   447→const bucketName = z
   448→  .string()
   449→  .min(1)
   450→  .refine(
   451→    (value: string) => /^[a-z0-9][a-z0-9.-]{1,61}[a-z0-9]$/.test(value),
   452→    "must be a valid S3 bucket name (lowercase letters, digits, dots, hyphens)",
   453→  );
   454→
   455→/** Whole positive integer, in plain digits, ≥ min. */
   456→function wholeNumber(min: number) {
   457→  return z.coerce
   458→    .number({ error: "must be a whole number" })
   459→    .int("must be a whole number")
   460→    .min(min, `must be at least ${min}`);
   461→}
   462→
   463→const appEnvSchema = z.enum(["local", "staging", "production"], {
   464→  error: 'must be one of: "local", "staging", "production"',
   465→});
   466→
   467→const logLevelSchema = z.enum(["debug", "info", "warn", "error"], {
   468→  error: 'must be one of: "debug", "info", "warn", "error"',
   469→});
   470→
   471→/** Strict boolean; accepts true/false, 1/0, yes/no, on/off in any case. */
   472→const envBool = z.string().transform((raw: string, ctx: z.RefinementCtx): boolean => {
   473→  const value = raw.trim().toLowerCase();
   474→  if (["true", "1", "yes", "on"].includes(value)) {
   475→    return true;
   476→  }
   477→  if (["false", "0", "no", "off"].includes(value)) {
   478→    return false;
   479→  }
   480→  ctx.addIssue({
   481→    code: "custom",
   482→    message: "must be a boolean (true/false; 1/0, yes/no, on/off also accepted)",
   483→  });
   484→  return z.NEVER;
   485→});
   486→
   487→const editorPublicPath = z
   488→  .string()
   489→  .min(1)
   490→  .refine(
   491→    (value: string) => value.startsWith("/") && value.length > 1 && !value.endsWith("/"),
   492→    "must be an absolute path like /editor with no trailing slash",
   493→  );
   494→
   495→/** JSON object mapping plan names to Stripe price ids. */
   496→const stripePriceIds = z
   497→  .string()
   498→  .transform((raw: string, ctx: z.RefinementCtx): unknown => {
   499→    try {
   500→      return JSON.parse(raw) as unknown;
   501→    } catch {
   502→      ctx.addIssue({
   503→        code: "custom",
   504→        message: "must be valid JSON mapping plan names to Stripe price ids",
   505→      });
   506→      return z.NEVER;
   507→    }
   508→  })
   509→  .pipe(z.record(z.string(), z.string()));
   510→
   511→// ───────────────────────────────────────────────────────────────────────────
   512→// The flat schema — one key per declared var. Defaults are OUTPUT-typed under
   513→// zod v4 (`.default(n)` returns `n` unparsed), so numeric/boolean defaults are
   514→// literal numbers/booleans, never strings.
   515→// ───────────────────────────────────────────────────────────────────────────
   516→
   517→export const envSchema = z.object({
   518→  // app
   519→  APP_URL: httpUrl,
   520→  APP_ENV: appEnvSchema,
   521→  LOG_LEVEL: logLevelSchema.default("info"),
   522→  MOCK_MODEL: envBool.default(false),
   523→  // db
   524→  DATABASE_URL: postgresUrl,
   525→  // queue
   526→  VALKEY_URL: valkeyUrl.optional(),
   527→  // storage — optional as a group: local dev runs the gateway's filesystem
   528→  // driver (.local/storage) when the endpoint is unset; staging/production
   529→  // must set the whole group (cross-checked below).
   530→  S3_ENDPOINT: httpUrl.optional(),
   531→  S3_REGION: z.string().min(1).optional(),
   532→  S3_ACCESS_KEY_ID: z.string().min(1).optional(),
   533→  S3_SECRET_ACCESS_KEY: z.string().min(1).optional(),
   534→  S3_BUCKET: bucketName.optional(),
   535→  ASSET_MAX_BYTES: wholeNumber(1),
   536→  PROJECT_MAX_BYTES: wholeNumber(1),
   537→  // stripe
   538→  STRIPE_SECRET_KEY: z.string().min(1).optional(),
   539→  STRIPE_WEBHOOK_SECRET: z.string().min(1),
   540→  STRIPE_PRICE_IDS: stripePriceIds,
   541→  // auth
   542→  AUTH_SECRET: z.string().min(32, "must be at least 32 characters (openssl rand -base64 32)"),
   543→  AUTH_GOOGLE_ID: z.string().min(1),
   544→  AUTH_GOOGLE_SECRET: z.string().min(1),
   545→  AUTH_GITHUB_ID: z.string().min(1),
   546→  AUTH_GITHUB_SECRET: z.string().min(1),
   547→  // litellm
   548→  LITELLM_BASE_URL: httpUrl,
   549→  LITELLM_API_KEY: z.string().min(1),
   550→  LITELLM_VIRTUAL_KEY: z.string().min(1).optional(),
   551→  // editor
   552→  EDITOR_PUBLIC_PATH: editorPublicPath.default("/editor"),
   553→  EDITOR_DEV_SERVER_URL: httpUrl.optional(),
   554→  EDITOR_WASM_MAX_MB: wholeNumber(1).default(18),
   555→  // agent
   556→  CHAT_MAX_STEPS: wholeNumber(1),
   557→  AGENT_BATCH_MAX_STEPS: wholeNumber(1).default(100),
   558→  AGENT_BATCH_MAX_BYTES: wholeNumber(1).default(2 * 1024 * 1024),
   559→  AGENT_BATCH_TIMEOUT_MS: wholeNumber(1).default(60_000),
   560→  // lease
   561→  LEASE_HEARTBEAT_MS: wholeNumber(1).default(120_000),
   562→  LEASE_TTL_MS: wholeNumber(1).default(300_000),
   563→  // obs
   564→  OTEL_EXPORTER_OTLP_ENDPOINT: httpUrl.optional(),
   565→});
   566→
   567→/** Flat, validated env values (internal; the public shape is {@link StudioEnv}). */
   568→type FlatEnv = z.output<typeof envSchema>;

   570→/** Typed, grouped configuration returned by parseEnv/getEnv. Frozen deeply. */
   571→export interface StudioEnv {
   572→  readonly app: {
   573→    readonly url: string;
   574→    readonly env: AppEnv;
   575→    readonly logLevel: LogLevel;
   576→    readonly mockModel: boolean;
   577→  };
   578→  readonly db: {
   579→    readonly url: string;
   580→  };
   581→  readonly queue: {
   582→    /** Valkey for queues/rate limits; undefined when unset (features degrade). */
   583→    readonly url: string | undefined;
   584→  };
   585→  readonly storage: {
   586→    /** Undefined only when APP_ENV is "local" and the filesystem driver is used. */
   587→    readonly endpoint: string | undefined;
   588→    readonly region: string | undefined;
   589→    readonly accessKeyId: string | undefined;
   590→    readonly secretAccessKey: string | undefined;

   636→export type EnvProblemKind = "missing" | "invalid" | "rejected";
   637→
   638→/** One failing variable: kind, name, purpose and a reason that never leaks values. */
   639→export interface EnvProblem {
   640→  readonly kind: EnvProblemKind;
   641→  readonly name: string;
   642→  readonly purpose: string;
   643→  readonly reason: string;
   644→}

   646→/**
   647→ * Typed fatal error thrown by {@link parseEnv} when the environment is invalid.
   648→ * The message lists every failing variable with its purpose; secret values are
   649→ * never included. Entry points should catch this, log it once, and exit.
   650→ */
   651→export class EnvFatalError extends Error {
   652→  readonly code = "env_invalid" as const;
   653→  readonly appName: string;
   654→  readonly problems: readonly EnvProblem[];
   655→
   656→  constructor(appName: string, problems: readonly EnvProblem[]) {
   657→    super(formatEnvFatalMessage(appName, problems));
   658→    this.name = "EnvFatalError";
   659→    this.appName = appName;
   660→    this.problems = problems;
   661→  }
   662→}
   663→
   664→/** Type guard for fail-fast handlers in app entrypoints. */
   665→export function isEnvFatalError(value: unknown): value is EnvFatalError {
   666→  return value instanceof EnvFatalError;
   667→}
   668→
   669→function formatEnvFatalMessage(appName: string, problems: readonly EnvProblem[]): string {
   670→  const lines = problems.map((problem) => {
   671→    return `  - ${problem.name}: ${problem.reason}. (${problem.purpose})`;
   672→  });
   673→  return [
   674→    `[${appName}] environment is invalid — refusing to start (${problems.length} problem(s)):`,
   675→    ...lines,
   676→    "Values are never included in this message; fix the listed variables in your shell or .env and restart.",
   677→  ].join("\n");
   678→}
   679→
   680→/** Masks the raw value if a schema message happened to echo it. */
   681→function redact(reason: string, raw: string | undefined): string {
   682→  if (raw === undefined || raw === "") {
   683→    return reason;
   684→  }
   685→  return reason.split(raw).join("«redacted»");
   686→}
   687→
   688→interface RawZodIssue {
   689→  readonly code: string;
   690→  readonly path: readonly PropertyKey[];
   691→  readonly message: string;
   692→  readonly input?: unknown;
   693→}
   694→
   695→function issueIsMissing(issue: RawZodIssue): boolean {
   696→  return issue.code === "invalid_type" && issue.input === undefined;
   697→}
   698→
   699→/** Blank/whitespace-only values count as unset (defaults then apply). */
   700→function normalizeSource(source: EnvSource): Record<string, string | undefined> {
   701→  const normalized: Record<string, string | undefined> = {};
   702→  for (const spec of ENV_VAR_SPECS) {
   703→    const raw = source[spec.name];
   704→    normalized[spec.name] = raw !== undefined && raw.trim() !== "" ? raw : undefined;
   705→  }
   706→  return normalized;
   707→}
   708→
   709→function issuesToProblems(
   710→  issues: readonly RawZodIssue[],
   711→  normalized: Record<string, string | undefined>,
   712→): EnvProblem[] {
   713→  const problems: EnvProblem[] = [];
   714→  const seen = new Set<string>();
   715→  for (const issue of issues) {
   716→    const name = typeof issue.path[0] === "string" ? issue.path[0] : "(unknown)";
   717→    // One problem per variable: a chained schema (url → http refine) can emit
   718→    // several issues for one bad value; the first carries the best reason.
   719→    if (seen.has(name)) {
   720→      continue;
   721→    }
   722→    seen.add(name);
   723→    const raw = normalized[name];
   724→    if (issueIsMissing(issue) && raw === undefined) {
   725→      problems.push({
   726→        kind: "missing",
   727→        name,
   728→        purpose: envPurpose(name),
   729→        reason: "missing",
   730→      });
   731→      continue;
   732→    }
   733→    problems.push({
   734→      kind: "invalid",
   735→      name,
   736→      purpose: envPurpose(name),
   737→      reason: redact(issue.message, raw),
   738→    });
   739→  }
   740→  return problems;
   741→}
   742→
   743→/** Cross-field rules the flat schema cannot express. */
   744→function crossCheckProblems(env: FlatEnv): EnvProblem[] {
   745→  const problems: EnvProblem[] = [];
   746→  const push = (name: string, reason: string): void => {
   747→    problems.push({ kind: "rejected", name, purpose: envPurpose(name), reason });
   748→  };
   749→
   750→  if (env.APP_ENV !== "local" && env.S3_ENDPOINT === undefined) {
   751→    push(
   752→      "S3_ENDPOINT",
   753→      `required when APP_ENV is "${env.APP_ENV}" — object storage must be configured outside local (S3_ENDPOINT, S3_REGION, S3_ACCESS_KEY_ID, S3_SECRET_ACCESS_KEY, S3_BUCKET)`,
   754→    );
   755→  }
   756→  if (env.APP_ENV !== "local" && env.S3_BUCKET === undefined) {
   757→    push(
   758→      "S3_BUCKET",
   759→      `required when APP_ENV is "${env.APP_ENV}" — the filesystem storage driver is local-only`,
   760→    );
   761→  }
   762→  if (env.APP_ENV !== "local" && env.STRIPE_SECRET_KEY === undefined) {
   763→    push(
   764→      "STRIPE_SECRET_KEY",
   765→      `required when APP_ENV is "${env.APP_ENV}" — Stripe checkout, portal and webhooks cannot run without it`,
   766→    );
   767→  }
   768→  if (env.APP_ENV === "production" && env.MOCK_MODEL) {
   769→    push(
   770→      "MOCK_MODEL",
   771→      'must not be enabled when APP_ENV is "production" — the offline mock bypasses real model routing and billing',
   772→    );
   773→  }
   774→  if (env.APP_ENV === "production" && env.EDITOR_DEV_SERVER_URL !== undefined) {
   775→    push(
   776→      "EDITOR_DEV_SERVER_URL",
   777→      "is dev-only — production serves a built static editor bundle, never a dev server; unset it",
   778→    );
   779→  }
   780→  if (env.LEASE_TTL_MS <= env.LEASE_HEARTBEAT_MS) {
   781→    push(
   782→      "LEASE_TTL_MS",
   783→      `must exceed LEASE_HEARTBEAT_MS (currently ${env.LEASE_HEARTBEAT_MS} ms) or leases would expire between heartbeats`,
   784→    );
   785→  }
   786→  return problems;
   787→}

   789→function deepFreeze<T>(value: T): T {
   790→  if (value !== null && typeof value === "object" && !Object.isFrozen(value)) {
   791→    for (const key of Object.keys(value as Record<string, unknown>)) {
   792→      deepFreeze((value as Record<string, unknown>)[key]);
   793→    }
   794→    Object.freeze(value);
   795→  }
   796→  return value;
   797→}

   852→/**
   853→ * Validates `source` against the Studio env schema and returns the typed,
   854→ * grouped, frozen config. Throws {@link EnvFatalError} naming EVERY missing or
   855→ * invalid variable with its purpose — never any value.
   856→ */
   857→export function parseEnv(source: EnvSource): StudioEnv {
   858→  const normalized = normalizeSource(source);
   859→  const result = envSchema.safeParse(normalized);
   860→  if (!result.success) {
   861→    throw new EnvFatalError(Brand.codename, issuesToProblems(result.error.issues, normalized));
   862→  }
   863→  const crossProblems = crossCheckProblems(result.data);
   864→  if (crossProblems.length > 0) {
   865→    throw new EnvFatalError(Brand.codename, crossProblems);
   866→  }
   867→  return groupEnv(result.data);
   868→}
   869→
   870→const envCache = new WeakMap<EnvSource, StudioEnv>();
   871→
   872→/**
   873→ * Process-wide accessor. Defaults to `process.env` and parses once per source
   874→ * object (WeakMap), so apps get a cheap singleton while tests can pass fresh
   875→ * fixtures. Throws {@link EnvFatalError} on the first invalid environment.
   876→ */
   877→export function getEnv(source: EnvSource = process.env): StudioEnv {
   878→  const cached = envCache.get(source);
   879→  if (cached !== undefined) {
   880→    return cached;
   881→  }
   882→  const parsed = parseEnv(source);
   883→  envCache.set(source, parsed);
   884→  return parsed;
   885→}

   889→// ───────────────────────────────────────────────────────────────────────────

   891→/**
   892→ * Renders the .env.example body from {@link ENV_VAR_SPECS}: every variable,
   893→ * its purpose and requirement rule, plus a placeholder value (never a secret).
   894→ */
   895→export function renderEnvExample(): string {
   896→  const lines: string[] = [
   897→    `# ${Brand.productName} environment template.`,
   898→    "# Generated by @studio/config (writeEnvExample). Copy to .env and fill in real values.",
   899→    "# Never commit real secrets; startup fails fast listing anything missing or invalid.",
   900→    "",
   901→  ];
   902→  for (const group of GROUP_ORDER) {
   903→    lines.push(`# ── ${group} ──`);
   904→    for (const spec of ENV_VAR_SPECS.filter((entry) => entry.group === group)) {
   905→      const requirements: string[] = [];
   906→      if (spec.required) {
   907→        requirements.push("Required.");
   908→      } else if (spec.hasDefault) {
   909→        requirements.push(`Default: ${spec.example}.`);
   910→      } else {
   911→        requirements.push("Optional.");
   912→      }
   913→      if (spec.secret) {
   914→        requirements.push("Secret — never commit.");
   915→      }
   916→      lines.push(`# ${spec.name} — ${spec.purpose}`);
   917→      lines.push(`# ${requirements.join(" ")}`);
   918→      if (spec.condition !== undefined) {
   919→        lines.push(`# ${spec.condition}`);
   920→      }
   921→      lines.push(`${spec.name}=${spec.example}`);
   922→      lines.push("");
   923→    }
   924→  }
   925→  return `${lines.join("\n")}\n`;
   926→}
   927→
   928→/**
   929→ * Writes the generated .env.example to `path` (uses a dynamic `node:fs/promises`
   930→ * import so importing `env.ts` from a browser bundle stays side-effect free).
   931→ */
   932→export async function writeEnvExample(path: string | URL): Promise<void> {
   933→  const { writeFile } = await import("node:fs/promises");
   934→  await writeFile(path, renderEnvExample(), { encoding: "utf8", mode: 0o644 });
   935→}
