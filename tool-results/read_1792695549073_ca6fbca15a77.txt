     1→/**
     2→ * `attachEditorBridge` — the typed wrapper around the frozen editor iframe's
     3→ * `window.effectcraft` global (PRODUCT_ARCHITECTURE §6.2).
     4→ *
     5→ * Verified against the shipped dist (READ-ONLY sources, v0.13.0):
     6→ * - `window.effectcraft` + `window.effectcraftLoad = {}` are defined synchronously
     7→ *   by the inline module script in `apps/web/public/editor/index.html` (from
     8→ *   `editor/apps/effectcraft-web/web/index.html`) before the wasm runs.
     9→ * - `effectcraftLoad.wasmMs` is set once the module compiles, `readyMs` once
    10→ *   `ec.start()` returns; on failure the page sets `effectcraftLoad.error =
    11→ *   String(e)` (and never `readyMs`). The upstream smoke test
    12→ *   (`editor/apps/effectcraft-web/tests/smoke.mjs`) waits for
    13→ *   `effectcraftLoad.readyMs || effectcraftLoad.error`.
    14→ * - `effectcraft.request`/`execute` reject with plain STRINGS
    15→ *   (`editor/apps/effectcraft-web/src/api.rs`): "Studio is not running yet"
    16→ *   (de-branded dist string) when the control channel is not up (transient,
    17→ *   retry), "request dropped" when it disconnects. Errors from the control
    18→ *   channel itself reject with that channel's `error` message.
    19→ * - Upstream documents a 60 s control-request timeout
    20→ *   (`editor/docs/control-protocol.md`); the web dist keeps requests pending
    21→ *   until they are answered or dropped, so the bridge enforces its own
    22→ *   `callTimeoutMs` (default 15 s) client-side and normalizes to `BridgeError`.

    63→/** Transient startup rejections from api.rs ("Studio is not running yet" in the shipped dist) — retried once. */
    64→const NOT_READY_RE = /not running yet|not ready|still (loading|starting)|starting up/i;

    67→export interface AttachOptions {
    68→  /** Per-call timeout in ms (default 15 s). */
    69→  readonly callTimeoutMs?: number;
    70→  /** Handshake timeout in ms (default 60 s). */
    71→  readonly readyTimeoutMs?: number;
    72→  /** Expected editor version prefix; a mismatch emits an "error" event (attach still resolves). */
    73→  readonly version?: string;
    74→  /**
    75→   * Pre-attach event handler: sees the early "ready" emission, the
    76→   * version-mismatch "error" and the handshake "error"s (`EDITOR_LOAD_FAILED`
    77→   * when the dist's wasm failed to boot, `EDITOR_NOT_READY` on timeout) that
    78→   * fire before `attachEditorBridge` resolves. Routed into the same bus as
    79→   * `on(…)`.
    80→   */
    81→  readonly onEvent?: (event: EditorEventName, payload: unknown) => void;
    82→}

    84→class EventBus {
    85→  private readonly listeners = new Map<EditorEventName, Set<EditorEventHandler>>();
    86→  /** "ready" is sticky: late subscribers get the (single) emission replayed. */
    87→  private stickyReady: { payload: unknown } | null = null;
    88→
    89→  on(event: EditorEventName, cb: EditorEventHandler): () => void {
    90→    let set = this.listeners.get(event);
    91→    if (!set) {
    92→      set = new Set();
    93→      this.listeners.set(event, set);
    94→    }
    95→    set.add(cb);
    96→    if (event === "ready" && this.stickyReady) cb(this.stickyReady.payload);
    97→    return () => {
    98→      set?.delete(cb);
    99→    };
   100→  }
   101→
   102→  emit(event: EditorEventName, payload: unknown): void {
   103→    if (event === "ready") this.stickyReady = { payload };
   104→    for (const cb of [...(this.listeners.get(event) ?? [])]) cb(payload);
   105→  }
   106→
   107→  clear(): void {
   108→    this.listeners.clear();
   109→    this.stickyReady = null;
   110→  }
   111→}

   125→  if (err instanceof z.ZodError) {

   146→function isNotReadyMessage(message: string): boolean {
   147→  return NOT_READY_RE.test(message);
   148→}
   149→
   150→function classify(label: string, err: unknown): BridgeError {
   151→  if (err instanceof BridgeError) return err;
   152→  const message = messageOf(err);
   153→  if (DISCONNECTED_RE.test(message)) {
   154→    return new BridgeError("EDITOR_DISCONNECTED", `${label}: ${message}`, true, err);
   155→  }
   156→  if (isNotReadyMessage(message)) {
   157→    return new BridgeError("EDITOR_NOT_READY", `${label}: ${message}`, true, err);
   158→  }
   159→  return new BridgeError("EDITOR_CALL_FAILED", `${label}: ${message}`, false, err);
   160→}

   162→function withTimeout<T>(label: string, promise: Promise<T>, timeoutMs: number): Promise<T> {
   163→  return new Promise<T>((resolve, reject) => {
   164→    const timer = setTimeout(() => {
   165→      reject(new BridgeError("EDITOR_TIMEOUT", `${label}: timed out after ${timeoutMs} ms`, true));
   166→    }, timeoutMs);
   167→    const settle = (fn: (v: T | PromiseLike<T>) => void, value: T) => {
   168→      clearTimeout(timer);
   169→      fn(value);
   170→    };
   171→    promise.then(
   172→      (value) => settle(resolve, value),
   173→      (err) => {
   174→        clearTimeout(timer);
   175→        reject(err);
   176→      },
   177→    );
   178→  });
   179→}

   181→/**
   182→ * Poll for `effectcraftLoad.readyMs` (the dist sets it once `ec.start()`
   183→ * returns) or fail fast on `effectcraftLoad.error` — the exact two signals
   184→ * the shipped host page writes (`apps/web/public/editor/index.html`), and the
   185→ * same pair the upstream smoke test waits for.
   186→ *
   187→ * The handshake is the one place the bridge polls: the dist exposes no ready
   188→ * event, and `window.effectcraft` exists before the app runs. The `on(…)`
   189→ * event bus below never polls.
   190→ */
   191→async function waitForReady(iframe: HTMLIFrameElement, readyTimeoutMs: number): Promise<void> {
   192→  const deadline = Date.now() + readyTimeoutMs;
   193→  for (;;) {
   194→    const w = contentWindowOf(iframe);
   195→    const ec = w?.effectcraft;
   196→    const load = w?.effectcraftLoad;
   197→    if (ec && load) {
   198→      // Order matters: the dist sets `error` on any boot failure and never
   199→      // `readyMs`; if both are somehow present, the failure wins.
   200→      if (load.error !== undefined) {
   201→        throw new BridgeError("EDITOR_LOAD_FAILED", `editor failed to start: ${load.error}`, false);
   202→      }
   203→      if (load.readyMs !== undefined) return;
   204→      // A ready() hook (not in upstream v0.13): awaited once, bounded by the deadline.
   205→      const ready = (ec as { ready?: unknown }).ready;
   206→      if (typeof ready === "function") {
   207→        const remaining = Math.max(deadline - Date.now(), 0);
   208→        try {
   209→          await withTimeout("ready()", Promise.resolve(ready.call(ec)), remaining);
   210→          return;
   211→        } catch (err) {
   212→          throw classify("ready()", err);
   213→        }
   214→      }
   215→    }
   216→    if (Date.now() >= deadline) {
   217→      throw new BridgeError(
   218→        "EDITOR_NOT_READY",
   219→        `editor did not become ready within ${readyTimeoutMs} ms`,
   220→        true,
   221→      );
   222→    }
   223→    await delay(READY_POLL_MS);
   224→  }
   225→}

   231→/** The editor wasm's boot phase, read from the dist's `effectcraftLoad`. */
   232→export type EditorLoadStatus = "loading" | "ready" | "failed";
   233→
   234→/**
   235→ * A snapshot of the editor's boot state, taken from the shipped dist's
   236→ * `window.effectcraftLoad` (`apps/web/public/editor/index.html`):
   237→ * `{}` while loading, `wasmMs` once the module compiled, `readyMs` once
   238→ * `ec.start()` returned, `error` when the wasm failed to boot.
   239→ */
   240→export interface EditorLoadState {
   241→  /** `loading` until `readyMs`/`error` appears (or the globals are not yet visible). */
   242→  readonly status: EditorLoadStatus;
   243→  /** Present when `status === "failed"` — the dist's `String(e)`. */
   244→  readonly error?: string;
   245→  /** Set once the wasm module compiled (present on both ready and failed boots). */
   246→  readonly wasmMs?: number;
   247→  /** Set once `ec.start()` returned (present only on a ready boot). */
   248→  readonly readyMs?: number;
   249→}
   250→
   251→/**
   252→ * Read the editor's boot state synchronously — the public seam for "the wasm
   253→ * failed to boot" (M3.7). Works before (and without) `attachEditorBridge`:
   254→ * the host page can poll this while it shows its loading screen, then render
   255→ * the failure + retry UI the moment `status === "failed"`, and call
   256→ * `attachEditorBridge` once `status === "ready"`.
   257→ *
   258→ * - Returns `{ status: "loading" }` while the iframe has not exposed the
   259→ *   dist globals yet (including cross-origin reads, which throw and are
   260→ *   swallowed — the state is genuinely unknowable from this frame).
   261→ * - Same-origin only, like the rest of the bridge (§6.1).
   262→ *
   263→ * The `attachEditorBridge` rejection is the async counterpart: it fails with
   264→ * `BridgeError` code `EDITOR_LOAD_FAILED` carrying the same `error` string.
   265→ */
   266→export function editorLoadState(iframe: HTMLIFrameElement): EditorLoadState {
   267→  let load: EffectcraftLoad | undefined;
   268→  try {
   269→    load = (iframe.contentWindow as EffectcraftWindow | null)?.effectcraftLoad;
   270→  } catch {
   271→    return { status: "loading" }; // cross-origin: the dist's globals are not readable
   272→  }
   273→  if (!load) return { status: "loading" };
   274→  if (load.error !== undefined) {
   275→    return { status: "failed", error: load.error, wasmMs: load.wasmMs, readyMs: load.readyMs };
   276→  }
   277→  if (load.readyMs !== undefined) {
   278→    return { status: "ready", wasmMs: load.wasmMs, readyMs: load.readyMs };
   279→  }
   280→  return { status: "loading", wasmMs: load.wasmMs };
   281→}

   283→/** Where the editor is loaded relative to the current origin. Same-origin only (§6.1). */
   284→export function sameOrigin(iframe: HTMLIFrameElement): boolean {
   285→  const src = iframe.getAttribute("src") ?? "";
   286→  try {
   287→    const url = new URL(src, globalThis.location.href);
   288→    return url.origin === globalThis.location.origin;
   289→  } catch {
   290→    return true; // no/relative src → same origin
   291→  }

   294→/**
   295→ * Attach to the editor iframe. Resolves once `window.effectcraft` exists and
   296→ * reports ready (`effectcraftLoad.readyMs`, or an optional `ready()` hook).
   297→ *
   298→ * Rejects with a `BridgeError` when the editor cannot boot:
   299→ * - `EDITOR_LOAD_FAILED` — the dist's `effectcraftLoad.error` is set (the wasm
   300→ *   failed to boot; the `error` string is in the message). Use
   301→ *   {@link editorLoadState} to observe this without awaiting the handshake.
   302→ * - `EDITOR_NOT_READY` — the handshake timed out (default 60 s).
   303→ *
   304→ * A load failure or handshake timeout is also emitted on the event bus as an
   305→ * `"error"` event (visible pre-attach via `AttachOptions.onEvent`).
   306→ *
   307→ * Every resolved method call is wrapped with a timeout (default 15 s), a
   308→ * single retry on transient "not ready" rejections, and `BridgeError`
   309→ * normalization (`EDITOR_TIMEOUT` | `EDITOR_CALL_FAILED` | `EDITOR_NOT_READY`
   310→ * | `EDITOR_LOAD_FAILED` | `EDITOR_DISCONNECTED`).
   311→ */
   312→export function attachEditorBridge(
   313→  iframe: HTMLIFrameElement,
   314→  opts: AttachOptions = {},
   315→): Promise<EditorBridge> {
   316→  const callTimeoutMs = opts.callTimeoutMs ?? DEFAULT_CALL_TIMEOUT_MS;
   317→  const readyTimeoutMs = opts.readyTimeoutMs ?? DEFAULT_READY_TIMEOUT_MS;
   318→  const bus = new EventBus();
   319→  if (opts.onEvent) {
   320→    for (const event of ["ready", "focus", "resize", "error"] as const) {
   321→      bus.on(event, (payload) => opts.onEvent?.(event, payload));
   322→    }
   323→  }
   324→  let disposed = false;
   325→
   326→  /** The handshake, shared by attach and `bridge.ready()`: emits bus "error"s. */
   327→  async function awaitReady(): Promise<void> {
   328→    try {
   329→      await waitForReady(iframe, readyTimeoutMs);
   330→    } catch (err) {
   331→      const bridgeErr = err instanceof BridgeError ? err : classify("ready", err);
   332→      bus.emit("error", bridgeErr);
   333→      throw bridgeErr;
   334→    }
   335→  }
   336→
   337→  async function call<T>(
   338→    label: string,
   339→    fn: (ec: EffectcraftGlobal, w: EffectcraftWindow) => T | Promise<T>,
   340→    timeoutMs: number = callTimeoutMs,
   341→  ): Promise<T> {
   342→    try {
   343→      if (disposed) {
   344→        throw new BridgeError("EDITOR_DISCONNECTED", `${label}: bridge is disposed`, false);
   345→      }
   346→      const w = contentWindowOf(iframe);
   347→      if (!w) {
   348→        throw new BridgeError(
   349→          "EDITOR_DISCONNECTED",
   350→          `${label}: iframe contentWindow is gone`,
   351→          false,
   352→        );
   353→      }
   354→      const ec = w.effectcraft;
   355→      if (!ec) {
   356→        throw new BridgeError(
   357→          "EDITOR_NOT_READY",
   358→          `${label}: window.effectcraft is missing (editor reloading?)`,
   359→          true,
   360→        );
   361→      }
   362→
   363→      let lastErr: unknown = undefined;
   364→      for (let attempt = 0; ; attempt++) {
   365→        if (attempt > 0) await delay(RETRY_DELAY_MS);
   366→        try {
   367→          // fn may throw synchronously (readFile/info are sync upstream) — Promise.resolve captures it.
   368→          return await withTimeout(label, Promise.resolve(fn(ec, w)), timeoutMs);
   369→        } catch (err) {
   370→          lastErr = err;
   371→          const retryable = !(err instanceof BridgeError) && isNotReadyMessage(messageOf(err));
   372→          if (attempt >= 1 || !retryable) break;
   373→        }
   374→      }
   375→      throw lastErr;
   376→    } catch (err) {
   377→      const bridgeErr = classify(label, err);
   378→      bus.emit("error", bridgeErr);
   379→      throw bridgeErr;
   380→    }
   381→  }
   382→
   383→  function rawInfo(): EditorInfo {
   384→    if (disposed) {
   385→      throw new BridgeError("EDITOR_DISCONNECTED", "info(): bridge is disposed", false);
   386→    }
   387→    const w = contentWindowOf(iframe);
   388→    const ec = w?.effectcraft;
   389→    if (!w || !ec) {
   390→      throw new BridgeError("EDITOR_NOT_READY", "info(): window.effectcraft is missing", true);
   391→    }
   392→    // info() is synchronous upstream; validate the bits the bridge relies on.
   393→    const parsed = parseOr(EditorInfoSchema, "info", ec.info());
   394→    const load = parseOr(EffectcraftLoadInfoSchema, "info.load", w.effectcraftLoad ?? {});
   395→    return { ...parsed, load };
   396→  }

   400→  function attachWindowEvents(w: EffectcraftWindow): void {
   401→    const onFocus = () => bus.emit("focus", { focused: true, window: w });
   402→    const onBlur = () => bus.emit("focus", { focused: false, window: w });
   403→    const onResize = () =>
   404→      bus.emit("resize", { width: w.innerWidth, height: w.innerHeight, window: w });
   405→    w.addEventListener("focus", onFocus);
   406→    w.addEventListener("blur", onBlur);
   407→    w.addEventListener("resize", onResize);
   408→    detachFns.push(() => {
   409→      w.removeEventListener("focus", onFocus);
   410→      w.removeEventListener("blur", onBlur);
   411→      w.removeEventListener("resize", onResize);
   412→    });
   413→  }
   414→
   415→  const detachFns: Array<() => void> = [];
   416→
   417→  return (async (): Promise<EditorBridge> => {
   418→    await awaitReady();
   419→
   420→    const w = contentWindowOf(iframe);
   421→    if (!w) {
   422→      throw new BridgeError("EDITOR_DISCONNECTED", "iframe contentWindow is gone", false);
   423→    }
   424→    attachWindowEvents(w);
   425→
   426→    const info = rawInfo();
   427→    bus.emit("ready", {
   428→      readyMs: w.effectcraftLoad?.readyMs,
   429→      wasmMs: w.effectcraftLoad?.wasmMs,
   430→      version: info.version,
   431→    });
   432→    if (opts.version !== undefined && !info.version.startsWith(opts.version)) {
   433→      bus.emit(
   434→        "error",
   435→        new Error(
   436→          `editor version ${JSON.stringify(info.version)} does not match expected ${JSON.stringify(opts.version)}`,
   437→        ),
   438→      );
   439→    }
   440→
   441→    const bridge: EditorBridge = {
   442→      async ready(): Promise<void> {
   443→        if (disposed) {
   444→          throw new BridgeError("EDITOR_DISCONNECTED", "bridge is disposed", false);
   445→        }
   446→        await awaitReady();
   447→      },

   456→        const params = filter === undefined ? {} : { filter };
   457→        return call("commands", (ec) => ec.request("engine.commands", params)).then((v) =>
   458→          parseOr(z.array(CommandInfoSchema), "commands", v),
   459→        );
   460→      },
   461→

   463→        return call("describe", (ec) => ec.execute("command.describe", { command })).then((v) =>
   464→          parseOr(CommandSchemaSchema, "describe", v),

   466→      },

   489→      async batch(input: Parameters<EditorBridge["batch"]>[0]): Promise<BatchResult> {
   490→        let parsed: AgentBatch;
   491→        try {
   492→          // §8.5 bounds, `$N.key` refs and the 2 MB cap — via @studio/contracts
   493→          parsed = normalizeAgentBatch(input);
   494→        } catch (err) {
   495→          const bridgeErr = new BridgeError(
   496→            "EDITOR_CALL_FAILED",
   497→            `invalid agent batch: ${zodMessage(err)}`,
   498→            false,
   499→            err,
   500→          );
   501→          bus.emit("error", bridgeErr);
   502→          throw bridgeErr;
   503→        }
   504→        const payload = { label: parsed.label, atomic: true, steps: parsed.steps };
   505→        const timeoutMs = Math.min(parsed.timeoutMs ?? callTimeoutMs, MAX_BATCH_TIMEOUT_MS);
   506→        return call(
   507→          `batch ${JSON.stringify(parsed.label)}`,
   508→          (ec) => ec.execute("engine.batch", payload),
   509→          timeoutMs,
   510→        ).then((v) => parseOr(BatchResultSchema, "batch", v));
   511→      },
   512→
   513→      renderFrame(params: RenderFrameParams): Promise<RenderFrameResult> {
   514→        return call("renderFrame", (ec) => ec.renderFrame(params)).then((v) =>
   515→          parseOr(RenderFrameResultSchema, "renderFrame", v),
   516→        );
   517→      },

   519→      addFile(file: Blob | File | string, name?: string): Promise<string> {
   520→        return call("addFile", async (ec) =>
   521→          parseOr(z.string(), "addFile", await ec.addFile(file, name)),
   522→        );
   523→      },
   524→
   525→      async addAsset(asset: CloudAsset): Promise<string> {
   526→        const virtualPath = `/assets/${encodeURIComponent(asset.assetId)}/${sanitizeAssetName(asset.name)}`;
   527→        let blob: Blob;
   528→        if (asset.bytes !== undefined) {
   529→          blob =
   530→            asset.bytes instanceof Blob
   531→              ? asset.bytes
   532→              : new Blob([new Uint8Array(asset.bytes)], { type: "application/octet-stream" });
   533→        } else if (asset.url !== undefined) {
   534→          try {
   535→            const res = await fetch(asset.url);
   536→            if (!res.ok) {
   537→              throw new BridgeError(
   538→                "EDITOR_CALL_FAILED",
   539→                `addAsset: fetching ${asset.url} failed with HTTP ${res.status}`,
   540→                false,
   541→              );
   542→            }
   543→            blob = await res.blob();
   544→          } catch (err) {
   545→            if (err instanceof BridgeError) throw err;
   546→            throw new BridgeError(
   547→              "EDITOR_CALL_FAILED",
   548→              `addAsset: fetching ${asset.url} failed (${messageOf(err)})`,

   561→        }

   624→  /**
   625→   * A stable, filesystem-safe asset name (§6.3). The path the editor actually
   626→   * assigns is what callers must keep — upstream `addFile` basename-es the
   627→   * request (`editor/apps/effectcraft-web/src/files.rs` `path_for`).
   628→   */
   629→  function sanitizeAssetName(name: string): string {
   630→    const cleaned = name
   631→      .replace(/[/\\]/g, "-")
   632→      .replace(/[^A-Za-z0-9._-]+/g, "-")
   633→      .replace(/-{2,}/g, "-")
   634→      .replace(/^[.-]+/, "")
   635→      .slice(0, 120);
   636→    return cleaned.length > 0 ? cleaned : "asset";
   637→  }

   641→// forwardShortcut — EDITOR_CHAT_UX §9.1/§9.2

   643→
   644→export type EditorShortcutName =
   645→  | "toggleChat" // Cmd/Ctrl+L
   646→  | "inlineEdit" // Cmd/Ctrl+K
   647→  | "agent" // Cmd/Ctrl+I
   648→  | "stop" // Cmd/Ctrl+.
   649→  | "newChat" // Cmd/Ctrl+Shift+L
   650→  | "undoAgentRun" // Cmd/Ctrl+Shift+U
   651→  | "escape" // Esc
   652→  | "send"; // Cmd/Ctrl+Enter
   653→
   654→export interface EditorShortcutEvent {
   655→  readonly shortcut: EditorShortcutName;
   656→  readonly key: string;
   657→  readonly window: Window;
   658→  readonly native: KeyboardEvent;
   659→  preventDefault(): void;
   660→  stopPropagation(): void;
   661→}
   662→
   663→// biome-ignore lint/suspicious/noConfusingVoidType: handlers may return nothing (handled) or `false` (declined)
   664→export type EditorShortcutHandler = (event: EditorShortcutEvent) => boolean | void;
   665→
   666→export interface ShortcutHandlers {
   667→  readonly toggleChat?: EditorShortcutHandler;
   668→  readonly inlineEdit?: EditorShortcutHandler;
   669→  readonly agent?: EditorShortcutHandler;
   670→  readonly stop?: EditorShortcutHandler;
   671→  readonly newChat?: EditorShortcutHandler;
   672→  readonly undoAgentRun?: EditorShortcutHandler;
   673→  readonly escape?: EditorShortcutHandler;
   674→  readonly send?: EditorShortcutHandler;
   675→}
   676→
   677→function matchShortcut(e: KeyboardEvent): EditorShortcutName | undefined {
   678→  const mod = e.metaKey || e.ctrlKey; // `command` is ⌘ on macOS, Ctrl elsewhere (control-protocol.md)
   679→  const key = e.key.toLowerCase();
   680→  if (mod && !e.shiftKey && !e.altKey) {
   681→    if (key === "l") return "toggleChat";
   682→    if (key === "k") return "inlineEdit";
   683→    if (key === "i") return "agent";
   684→    if (key === ".") return "stop";
   685→    if (key === "enter") return "send";
   686→  }
   687→  if (mod && e.shiftKey && !e.altKey) {
   688→    if (key === "l") return "newChat";
   689→    if (key === "u") return "undoAgentRun";
   690→  }
   691→  if (!mod && !e.shiftKey && !e.altKey && !e.ctrlKey && !e.metaKey && key === "escape") {
   692→    return "escape";
   693→  }
   694→  return undefined;
   695→}
   696→
   697→/**
   698→ * Attach a same-origin key listener on the iframe's `contentWindow` and
   699→ * forward the chat shortcuts (EDITOR_CHAT_UX §9.2): the parent handles
   700→ * L / K / I / `.` and their Shift combos while the editor keeps every other
   701→ * shortcut. `Cmd/Ctrl+Enter` (send/apply) and `Esc` (close popover) forward
   702→ * too — the editor is not asked to know about the chat.
   703→ *
   704→ * A handler that returns `false` explicitly declines the event (nothing is
   705→ * prevented, the editor sees it as usual); anything else — `undefined` or
   706→ * `true` — marks it handled for the parent and the native event gets
   707→ * `preventDefault()` + `stopPropagation()`.
   708→ *
   709→ * Returns a detach function.
   710→ */
   711→export function forwardShortcut(iframe: HTMLIFrameElement, handlers: ShortcutHandlers): () => void {
   712→  const w = contentWindowOf(iframe);
   713→  if (!w) return () => {};
   714→  const onKeyDown = (e: KeyboardEvent) => {
   715→    if (e.repeat) return;
   716→    const shortcut = matchShortcut(e);
   717→    const handler = shortcut !== undefined ? handlers[shortcut] : undefined;
   718→    if (shortcut === undefined || handler === undefined) return;
   719→    let used = false;
   720→    const event: EditorShortcutEvent = {
   721→      shortcut,
   722→      key: e.key,
   723→      window: w,
   724→      native: e,
   725→      preventDefault: () => {
   726→        used = true;
   727→        e.preventDefault();
   728→      },
   729→      stopPropagation: () => {
   730→        used = true;
   731→        e.stopPropagation();
   732→      },
   733→    };
   734→    const handled = handler(event);
   735→    if (handled !== false) {
   736→      used = true;
   737→      e.preventDefault();
   738→      e.stopPropagation();
   739→    } else if (used) {
   740→      e.preventDefault();
   741→      e.stopPropagation();
   742→    }
   743→  };
   744→  w.addEventListener("keydown", onKeyDown);
   745→  return () => {
   746→    w.removeEventListener("keydown", onKeyDown);
   747→  };
   748→}
