    38→interface RecordedBridge {
    39→  bridge: EditorBridge;
    40→  executeCalls: Array<{ command: string; params: unknown }>;
    41→  queryCalls: Array<{ command: string; params: unknown }>;
    42→}
    43→
    44→/** Wraps a bridge so every execute/query is recorded (execute mutates!). */
    45→function recordBridge(bridge: EditorBridge): RecordedBridge {
    46→  const executeCalls: RecordedBridge["executeCalls"] = [];
    47→  const queryCalls: RecordedBridge["queryCalls"] = [];
    48→  const wrapped: EditorBridge = {
    49→    ...bridge,
    50→    execute: async <T>(command: string, params?: unknown) => {
    51→      executeCalls.push({ command, params: structuredClone(params) });
    52→      return await bridge.execute<T>(command, params);
    53→    },
    54→    query: <T>(command: string, params?: unknown) => {
    55→      queryCalls.push({ command, params });
    56→      return bridge.query<T>(command, params);
    57→    },
    58→  };
    59→  return { bridge: wrapped, executeCalls, queryCalls };
    60→}

    62→interface HarnessOptions {
    63→  /** Injected failure for the fake's `engine.batch` (the fake's own `failOn`). */
    64→  failBatchWith?: string;
    65→  /** The autosave outcome (default: a fresh saved version). */
    66→  saveOutcome?: "saved" | "conflict" | "error" | "not-dirty";
    67→  /** Injected ApiError for the first applyRun (default: success). */
    68→  applyRunError?: ApiError;
    69→  /** Injected ApiError for the first undoRun (default: success). */
    70→  undoRunError?: ApiError;
    71→  /** The session's current version id (reapply gate). */
    72→  currentVersionId?: string | null;
    73→}

    75→interface Harness {
    76→  fake: FakeEditor;
    77→  recorded: RecordedBridge;
    78→  controller: ReturnType<typeof createApplyController>;
    79→  /** Ordered milestone log across every fake (bridge/autosave/api). */
    80→  timeline: string[];
    81→  autosaveReasons: string[];
    82→  applyCalls: Array<{ runId: string; body: Record<string, unknown> }>;
    83→  undoCalls: Array<{ runId: string; body: Record<string, unknown> }>;
    84→  events: ApplyEngineEvent[];
    85→  api: ApplyApiClient;
    86→  autosave: ApplyAutosavePort;
    87→}

    89→const FAKE_SHA = "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa";
    90→
    91→async function makeHarness(options: HarnessOptions = {}): Promise<Harness> {
    92→  const fake = createFakeEditor({
    93→    comps: [{ name: "Main" }],
    94→    ...(options.failBatchWith !== undefined
    95→      ? { failOn: { "engine.batch": options.failBatchWith } }
    96→      : {}),
    97→  });
    98→  // Layer 1 "Hello" (text) in comp 1 "Main"; journaled → selected, dirty.
    99→  await fake.execute("layer.newText", { text: "Hello" });
   100→
   101→  const recorded = recordBridge(attachFakeBridge(fake));
   102→  const timeline: string[] = [];
   103→  const autosaveReasons: string[] = [];
   104→  const applyCalls: Harness["applyCalls"] = [];
   105→  const undoCalls: Harness["undoCalls"] = [];
   106→  const events: ApplyEngineEvent[] = [];
   107→  let versionCounter = 0;
   108→
   109→  const autosave: ApplyAutosavePort = {
   110→    saveNow: async (reason: string) => {
   111→      timeline.push(`saveNow:${reason}`);
   112→      autosaveReasons.push(reason);
   113→      if (options.saveOutcome === "conflict") return { saved: false, conflict: true };
   114→      if (options.saveOutcome === "error")
   115→        return { saved: false, error: new Error("upload failed") };
   116→      if (options.saveOutcome === "not-dirty") return { saved: false };
   117→      versionCounter += 1;
   118→      return {
   119→        saved: true,
   120→        version: {
   121→          id: `ver_applied_${versionCounter}`,
   122→          label: null,
   123→          sha256: FAKE_SHA,
   124→          sizeBytes: 128,
   125→        },
   126→      };
   127→    },
   128→    markDirty: (reason?: string) => {
   129→      timeline.push(`markDirty:${reason ?? "unnamed"}`);
   130→    },
   131→  };
   132→
   133→  const api: ApplyApiClient = {
   134→    runs: {
   135→      applyRun: async (runId, body) => {
   136→        timeline.push("applyRun");
   137→        applyCalls.push({ runId, body: body as Record<string, unknown> });
   138→        if (options.applyRunError !== undefined && applyCalls.length === 1) {
   139→          throw options.applyRunError;
   140→        }
   141→        return { applied: true, versionId: body.versionId, creditsCommitted: 12 };
   142→      },
   143→      undoRun: async (runId, body) => {
   144→        timeline.push("undoRun");
   145→        undoCalls.push({ runId, body: body as Record<string, unknown> });
   146→        if (options.undoRunError !== undefined && undoCalls.length === 1) {
   147→          throw options.undoRunError;
   148→        }
   149→        return { undone: true };
   150→      },
   151→    },
   152→  };
   153→
   154→  const controllerOptions: CreateApplyControllerOptions = {
   155→    bridge: recorded.bridge,
   156→    autosave,
   157→    api,
   158→    ...(options.currentVersionId !== undefined
   159→      ? { session: { currentVersionId: () => options.currentVersionId ?? null } }
   160→      : {}),
   161→    onEvent: (event) => {
   162→      events.push(event);
   163→    },
   164→  };
   165→
   166→  return {
   167→    fake,
   168→    recorded,
   169→    controller: createApplyController(controllerOptions),
   170→    timeline,
   171→    autosaveReasons,
   172→    applyCalls,
   173→    undoCalls,
   174→    events,
   175→    api,
   176→    autosave,
   177→  };
   178→}
