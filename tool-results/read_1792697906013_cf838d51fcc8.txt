   233→type ValidatedToolCall =
   234→  | { readonly ok: true; readonly name: string; readonly args: unknown }
   235→  | { readonly ok: false; readonly message: string };

   237→function validateToolCall(call: AggregatedToolCall): ValidatedToolCall {
   238→  const tool = TOOL_BY_ID.get(call.name);
   239→  if (tool === undefined) {
   240→    return {
   241→      ok: false,
   242→      message: `Unknown tool "${call.name}". Available tools: ${RUNNER_TOOLS.map((t) => t.id).join(", ")}.`,
   243→    };
   244→  }
   245→  let parsedArgs: unknown;
   246→  try {
   247→    parsedArgs = JSON.parse(call.arguments === "" ? "{}" : call.arguments);
   248→  } catch {
   249→    return {
   250→      ok: false,
   251→      message: `Tool arguments for ${tool.id} must be a single valid JSON object.`,
   252→    };
   253→  }
   254→  // Args are validated against the tool's curated Zod schema from the
   255→  // contracts catalog (`tool.args` — complete for all 13 tools today).
   256→  // M5-A's WRITE_TOOL_ARGS_SCHEMAS registry is the pinned forward surface
   257→  // (M5-SEAMS); once it lands as the authoritative per-tool ZodObjects, the
   258→  // integrator flips this ONE lookup to consult it for write tools.
   259→  const checked = tool.args.safeParse(parsedArgs);
   260→  if (!checked.success) {
   261→    const detail = checked.error.issues
   262→      .slice(0, 3)
   263→      .map((issue) => `${issue.path.join(".") || "(root)"}: ${issue.message}`)
   264→      .join("; ");
   265→    return { ok: false, message: `Invalid arguments for ${tool.id}: ${detail}` };
   266→  }
   267→  return { ok: true, name: tool.id, args: checked.data };
   268→}

   274→interface UsageAccumulator {

   280→function applyUsage(usage: UsageAccumulator, event: ModelStreamEvent): void {
   281→  if (event.type === "usage") {

   283→    usage.outputTokens += event.usage.outputTokens;

   288→function messageChars(message: OpenAIChatMessage): number {
   289→  const serialized = JSON.stringify(message);
   290→  return serialized === undefined ? 0 : serialized.length;
   291→}
   292→
   293→/**
   294→ * §8.8 commit math: ceil(input/1000 × creditsPer1kInput + output/1000 ×
   295→ * creditsPer1kOutput), minimum 1, from MODEL_CATALOG for the run's alias.
   296→ * When the provider sent no usage chunk, estimate tokens ≈ chars/4 over the
   297→ * assembled conversation (assistant turns as output, the rest as input).
   298→ */
   299→export function committedCreditsFor(
   300→  model: string,
   301→  usage: UsageAccumulator,
   302→  messages: readonly OpenAIChatMessage[],
   303→): number {
   304→  const descriptor = MODEL_CATALOG[model as ModelId] ?? MODEL_CATALOG["studio-fast"];
   305→  let inputTokens = usage.inputTokens;
   306→  let outputTokens = usage.outputTokens;
   307→  if (!usage.seen) {
   308→    let inputChars = 0;
   309→    let outputChars = 0;
   310→    for (const message of messages) {
   311→      if (message.role === "assistant") outputChars += messageChars(message);
   312→      else inputChars += messageChars(message);
   313→    }
   314→    inputTokens = Math.ceil(inputChars / 4);
   315→    outputTokens = Math.ceil(outputChars / 4);
   316→  }
   317→  const cost =
   318→    (inputTokens / 1000) * descriptor.creditsPer1kInput +
   319→    (outputTokens / 1000) * descriptor.creditsPer1kOutput;
   320→  return Math.max(1, Math.ceil(cost));
   321→}

   327→/** Extract the JSON object from a model answer (fences/brace tolerance). */
   328→function extractJsonObject(text: string): unknown {
   329→  let candidate = text.trim();
   330→  if (candidate.startsWith("```")) {
   331→    candidate = candidate.replace(/^```[a-zA-Z]*\s*/, "").replace(/```\s*$/, "");
   332→  }
   333→  try {
   334→    return JSON.parse(candidate);
   335→  } catch {
   336→    const start = candidate.indexOf("{");
   337→    const end = candidate.lastIndexOf("}");
   338→    if (start !== -1 && end > start) {
   339→      try {
   340→        return JSON.parse(candidate.slice(start, end + 1));
   341→      } catch {
   342→        return undefined;
   343→      }
   344→    }
   345→    return undefined;
   346→  }
   347→}

   392→/** Deterministic HUMAN summary of one write-tool call (§8.6 display data;
   393→ * args shapes per contracts write-tools.ts — layers BY NAME, STRICT objects). */
   394→export function writeToolSummary(toolId: string, args: Record<string, unknown>): string {
   395→  const layerNames = array(args.layers)
   396→    .map((layer) =>
   397→      typeof layer === "object" && layer !== null
   398→        ? str((layer as Record<string, unknown>).name)
   399→        : "",
   400→    )
   401→    .filter((name) => name !== "");
   402→  switch (toolId) {
   403→    case "studio.create_composition":
   404→      return `Create composition "${str(args.name)}"`;
   405→    case "studio.create_text": {
   406→      const content = str(args.text);
   407→      const label = str(args.name) || (content.length > 40 ? `${content.slice(0, 39)}…` : content);
   408→      return `Add text layer "${label}"`;
   409→    }
   410→    case "studio.set_properties": {
   411→      const updates = array(args.properties).length;
   412→      const noun = updates === 1 ? "property" : "properties";
   413→      const layers = layerNames.length === 0 ? 1 : layerNames.length;
   414→      return `Set ${updates} ${noun} on ${layers} layer${layers > 1 ? "s" : ""}`;
   415→    }
   416→    case "studio.add_effect":
   417→      return `Add effect "${effectName(args.effect)}" to ${layerNames.length} layer${
   418→        layerNames.length === 1 ? "" : "s"
   419→      }`;
   420→    case "studio.set_keyframes":
   421→      return `Set ${array(args.keys).length} keyframes on ${str(args.propertyPath)} of ${
   422→        layerNames.length
   423→      } layer${layerNames.length === 1 ? "" : "s"}`;
   424→    case "studio.apply_preset": {
   425→      const preset =
   426→        typeof args.preset === "object" && args.preset !== null
   427→          ? str((args.preset as Record<string, unknown>).name) ||
   428→            str((args.preset as Record<string, unknown>).builtinId)
   429→          : "";
   430→      const targets = layerNames.length > 0 ? layerNames.length : 1; // omitted layers ⇒ current selection
   431→      return `Apply preset "${preset}" to ${targets} layer${targets === 1 ? "" : "s"}`;
   432→    }
   433→    case "studio.undo_run":
   434→      return "Undo the most recent applied agent run"; // undo_run takes NO args (one undo step, law #6)
   435→    case "studio.save_version":
   436→      return str(args.label) === ""
   437→        ? "Save a version snapshot of the project"
   438→        : `Save a version snapshot labeled "${str(args.label)}"`;
   439→    default:
   440→      return `Propose ${toolId.replace(/^studio\./, "")}`;
   441→  }
   442→}
   443→
   444→function effectName(effect: unknown): string {
   445→  if (typeof effect === "object" && effect !== null) {
   446→    return str((effect as Record<string, unknown>).name);
   447→  }
   448→  return "";
   449→}

   451→/** Affected-layer rows for the carrier plan (§8.6 card data; bounded ≤ 50). */
   452→function writeToolAffectedLayers(
   453→  toolId: string,
   454→  args: Record<string, unknown>,
   455→): Array<{ name: string; change: string }> {
   456→  const layers: Array<{ name: string; change: string }> = [];
   457→  const push = (name: string, change: string): void => {
   458→    if (name !== "" && layers.length < MAX_PLAN_AFFECTED_LAYERS) layers.push({ name, change });
   459→  };
   460→  const namedLayers = array(args.layers)
   461→    .map((layer) =>
   462→      typeof layer === "object" && layer !== null
   463→        ? str((layer as Record<string, unknown>).name)
   464→        : "",
   465→    )
   466→    .filter((name) => name !== "");
   467→  switch (toolId) {
   468→    case "studio.set_properties": {
   469→      const change = `Set ${array(args.properties).length} propert${
   470→        array(args.properties).length === 1 ? "y" : "ies"
   471→      }`;
   472→      for (const name of namedLayers) push(name, change);
   473→      break;
   474→    }
   475→    case "studio.add_effect":
   476→      for (const name of namedLayers) push(name, `Add effect "${effectName(args.effect)}"`);
   477→      break;
   478→    case "studio.set_keyframes":
   479→      for (const name of namedLayers) push(name, `Keyframes on ${str(args.propertyPath)}`);
   480→      break;
   481→    case "studio.apply_preset": {
   482→      const change = `Apply preset ${
   483→        typeof args.preset === "object" && args.preset !== null
   484→          ? str((args.preset as Record<string, unknown>).name) ||
   485→            str((args.preset as Record<string, unknown>).builtinId)
   486→          : ""
   487→      }`;
   488→      for (const name of namedLayers) push(name, change);
   489→      break;
   490→    }
   491→    default:
   492→      break;
   493→  }
   494→  return layers;
   495→}

   497→/**
   498→ * The one-step EditPlan CARRIER handed to the compile seam for a write-tool
   499→ * call. CARRIER CONVENTION (M5-C ⇄ M5-B seam): the single step is the
   500→ * curated tool id + validated args — `steps[].tool` is authoritative for
   501→ * the compiler's per-step translation (M5-SEAMS: "the plan-compiler maps
   502→ * tool+args → registry command steps"), and the pseudo `batch.step.command`
   503→ * carries the UNPREFIXED tool id purely to satisfy PLAN_STEP_SCHEMA's
   504→ * non-empty batch bound. The compiler MUST NOT execute `steps[].batch`
   505→ * (model-proposed plan batches are validation targets, §8.7 — F-3 class).
   506→ * INTEGRATOR: if M5-B's CompilePlanInput grows a dedicated write-tool arm
   507→ * (e.g. `{ tool, args }`), flip this carrier to it and delete the convention.
   508→ */
   509→export function writeToolCarrierPlan(tool: ToolDefinition, args: unknown): EditPlan {
   510→  const record = (typeof args === "object" && args !== null ? args : {}) as Record<string, unknown>;
   511→  const summary = clamp280(writeToolSummary(tool.id, record));
   512→  const carrier = {
   513→    intent: WRITE_TOOL_INTENTS[tool.id] ?? "edit",
   514→    summary,
   515→    risk: tool.risk,
   516→    requiresApproval: tool.risk !== "low",
   517→    affectedLayers: writeToolAffectedLayers(tool.id, record),
   518→    estimatedCredits: estimateCredits({ risk: tool.risk, steps: [{}] }),
   519→    steps: [
   520→      {
   521→        title: summary,
   522→        tool: tool.id,
   523→        args,
   524→        batch: {
   525→          label: `Agent: ${summary}`.slice(0, MAX_BATCH_LABEL_LENGTH),
   526→          steps: [{ command: tool.id.replace(/^studio\./, ""), params: record }],
   527→        },
   528→      },
   529→    ],
   530→  };
   531→  return EDIT_PLAN_SCHEMA.parse(carrier);
   532→}
   533→
   534→/** First compiler issue as a bounded model/tool message (mirrors PLAN_INVALID). */
   535→function compileIssueMessage(issues: readonly PlanIssue[]): string {
   536→  const first = issues[0]?.message ?? "The proposed change failed validation.";
   537→  const suffix = issues.length > 1 ? ` (+${issues.length - 1} more issues)` : "";
   538→  return `${first}${suffix}`.slice(0, MAX_PLAN_SUMMARY_LENGTH);
   539→}

   542→// The runner
   543→// ---------------------------------------------------------------------------
   544→
   545→interface ModelRound {
   546→  readonly text: string;
   547→  readonly calls: readonly AggregatedToolCall[];
   548→}
   549→
   550→interface PlanAttempt {
   551→  readonly plan: EditPlan | null;
   552→  readonly errorMessage?: string;
   553→  readonly cancelled?: true;
   554→}
   555→
   556→export function createLiteLLMRunner(deps: LiteLLMRunnerDeps): AgentRunner {
   557→  const logger = deps.logger ?? nullLogger;
   558→  const bearer =
   559→    deps.virtualKey !== undefined && deps.virtualKey !== "" ? deps.virtualKey : deps.apiKey;
   560→  const tools = openAiToolSpecs();
   561→  // M5-B seam: the injected compiler for tests, the real ../plan-compiler.ts
   562→  // default otherwise (see defaultCompilePlan above).
   563→  const runCompilePlan = deps.compilePlan ?? defaultCompilePlan;
   564→  const clientDeps = {
   565→    baseUrl: deps.baseUrl,
   566→    bearer,
   567→    fetch: deps.fetch,
   568→    sleep: deps.sleep,
   569→  };
   570→
   571→  /** Stream one model call; `onDelta` streams text out (main rounds only). */
   572→  async function readModelRound(
   573→    model: string,
   574→    messages: readonly OpenAIChatMessage[],
   575→    usage: UsageAccumulator,
   576→    options: { readonly withTools: boolean; readonly onDelta?: (delta: string) => void },
   577→    signal: AbortSignal,
   578→  ): Promise<ModelRound> {
   579→    const request: ChatCompletionRequest = {
   580→      model,
   581→      messages,
   582→      tools: options.withTools ? tools : undefined,
   583→      signal,
   584→    };
   585→    let text = "";
   586→    const calls: AggregatedToolCall[] = [];
   587→    for await (const event of streamChatCompletion(clientDeps, request)) {
   588→      applyUsage(usage, event);
   589→      if (event.type === "text") {
   590→        text += event.delta;
   591→        options.onDelta?.(event.delta);
   592→      } else if (event.type === "tool_call") {
   593→        calls.push(event.call);
   594→      }
   595→    }
   596→    return { text, calls };
   597→  }
   598→
   599→  /**
   600→   * studio.plan_edit (§8.4): server-side structured-output call to the SAME
   601→   * alias — a strict JSON prompt validated with EDIT_PLAN_SCHEMA, 2 attempts
   602→   * (the second appends the validation error). Never emits message deltas;
   603→   * the plan reaches the client via the {type:"plan"} event.
   604→   */
   605→  async function serverSidePlan(
   606→    model: string,
   607→    goal: string,
   608→    conversation: readonly OpenAIChatMessage[],
   609→    usage: UsageAccumulator,
   610→    signal: AbortSignal,
   611→  ): Promise<PlanAttempt> {
   612→    const attemptMessages: OpenAIChatMessage[] = [
   613→      ...conversation,
   614→      { role: "user", content: planInstruction(goal) },
   615→    ];
   616→    for (let attempt = 1; attempt <= 2; attempt += 1) {
   617→      if (signal.aborted) return { plan: null, cancelled: true };
   618→      let round: ModelRound;
   619→      try {
   620→        round = await readModelRound(model, attemptMessages, usage, { withTools: false }, signal);
   621→      } catch (error) {
   622→        if (signal.aborted) return { plan: null, cancelled: true };
   623→        if (error instanceof ModelUnavailableError) {
   624→          return { plan: null, errorMessage: "The model service is unavailable right now." };
   625→        }
   626→        return { plan: null, errorMessage: "The plan could not be generated." };
   627→      }
   628→      const parsed = extractJsonObject(round.text);
   629→      const validation = planValidationError(parsed);
   630→      if (parsed !== undefined && validation === "") {
   631→        return { plan: EDIT_PLAN_SCHEMA.parse(parsed) };
   632→      }
   633→      if (attempt === 2) {
   634→        return {
   635→          plan: null,
   636→          errorMessage:
   637→            validation === ""
   638→              ? "The plan was not valid JSON."
   639→              : `The proposed plan failed validation: ${validation}`,
   640→        };
   641→      }
   642→      attemptMessages.push({ role: "assistant", content: round.text });
   643→      attemptMessages.push({
   644→        role: "user",
   645→        content: `That was not a valid EditPlan: ${
   646→          validation || "not valid JSON"
   647→        }. Respond again with STRICT JSON only — no prose, no markdown fences — matching the EditPlan shape exactly.`,
   648→      });
   649→    }
   650→    return { plan: null, errorMessage: "The proposed plan failed validation twice." };
   651→  }
   652→
   653→  return {
   654→    async start(run: AgentRun, stream: AgentRunStream): Promise<AgentRunOutcome> {
   655→      const signal = stream.signal;
   656→      const usage: UsageAccumulator = { inputTokens: 0, outputTokens: 0, seen: false };
   657→      const conversation: OpenAIChatMessage[] = [...buildMessages(run)];
   658→
   659→      /** The pinned compile seam is "typed, never throws"; a violating
   660→       * compiler implementation degrades to a typed issue instead of
   661→       * failing the run (defense-in-depth for the M5-B handoff). */
   662→      const compilePlanSafe = async (input: CompilePlanInput): Promise<CompilePlanResult> => {
   663→        try {
   664→          return await runCompilePlan(input);
   665→        } catch (error) {
   666→          logger.error("plan_compiler_threw", {
   667→            runId: run.runId,
   668→            detail: error instanceof Error ? error.name : "unknown",
   669→          });
   670→          return { ok: false, issues: [{ message: "The plan compiler failed unexpectedly." }] };
   671→        }
   672→      };
   673→
   674→      const cancelled = (): AgentRunOutcome => ({ kind: "cancelled" });
   675→      const failed = (code: string, message: string, retryable: boolean): AgentRunOutcome => ({
   676→        kind: "failed",
   677→        code,
   678→        message,
   679→        retryable,
   680→      });
   681→
   682→      stream.emit({ type: "run.started", model: run.model });
   683→      await stream.yieldToStream();
   684→      if (signal.aborted) return cancelled();
   685→
   686→      const modelFailure = (error: unknown): AgentRunOutcome => {
   687→        if (signal.aborted || (error instanceof Error && error.name === "AbortError")) {
   688→          return cancelled();
   689→        }
   690→        if (error instanceof ModelUnavailableError) {
   691→          return failed(
   692→            "RUN_MODEL_UNAVAILABLE",
   693→            "The model service is unavailable right now.",
   694→            true,
   695→          );
   696→        }
   697→        logger.error("litellm_runner_error", {
   698→          runId: run.runId,
   699→          detail: error instanceof Error ? `${error.name}: ${error.message}` : "unknown",
   700→        });
   701→        return failed("INTERNAL", "Something went wrong on our side.", true);
   702→      };
   703→
   704→      let invalidToolCalls = 0;
   705→      let rounds = 0;
   706→
   707→      while (true) {
   708→        if (signal.aborted) return cancelled();
   709→        if (rounds >= deps.maxSteps) {
   710→          return failed("RUN_STEP_LIMIT", "The agent used too many steps for one message.", false);
   711→        }
   712→        rounds += 1;
   713→        // M4.3 redaction: length + sha256 prefix only — never the body.
   714→        const fingerprint = fingerprintMessages(conversation);
   715→        logger.debug("litellm_model_call", {
   716→          runId: run.runId,
   717→          step: rounds,
   718→          model: run.model,
   719→          promptChars: fingerprint.chars,
   720→          promptSha256: fingerprint.sha256.slice(0, 16),
   721→          promptMessages: fingerprint.messageCount,
   722→        });
   723→
   724→        let round: ModelRound;
   725→        try {
   726→          round = await readModelRound(
   727→            run.model,
   728→            conversation,
   729→            usage,
   730→            {
   731→              withTools: true,
   732→              onDelta: (delta) => {
   733→                stream.emit({ type: "message.delta", delta });
   734→              },
   735→            },
   736→            signal,
   737→          );
   738→        } catch (error) {
   739→          return modelFailure(error);
   740→        }
   741→        if (signal.aborted) return cancelled();
   742→
   743→        conversation.push(
   744→          round.calls.length > 0
   745→            ? {
   746→                role: "assistant",
   747→                content: round.text,
   748→                tool_calls: round.calls.map((call) => ({
   749→                  id: call.id,
   750→                  type: "function" as const,
   751→                  function: { name: call.name, arguments: call.arguments },
   752→                })),
   753→              }
   754→            : { role: "assistant", content: round.text },
   755→        );
   756→        if (round.calls.length === 0) break; // finish_reason stop — final answer
   757→
   758→        for (const call of round.calls) {
   759→          const validated = validateToolCall(call);
   760→          if (!validated.ok) {
   761→            invalidToolCalls += 1;
   762→            if (invalidToolCalls >= 2) {
   763→              return failed(
   764→                "RUN_INVALID_TOOL",
   765→                "The agent requested a tool call we can't validate.",
   766→                false,
   767→              );
   768→            }
   769→            // One repair turn (§8.9): the model sees the validation error as
   770→            // the tool result and may retry with valid args.
   771→            conversation.push({
   772→              role: "tool",
   773→              tool_call_id: call.id,
   774→              content: JSON.stringify({ error: "VALIDATION_FAILED", message: validated.message }),
   775→            });
   776→            continue;
   777→          }
   778→
   779→          stream.expectToolCall(call.id, validated.name);
   780→          stream.emit({
   781→            type: "tool.started",
   782→            toolCallId: call.id,
   783→            name: validated.name,
   784→            args: validated.args,
   785→          });
   786→          await stream.yieldToStream();
   787→          if (signal.aborted) return cancelled();
   788→
   789→          const writeTool = registeredWriteTool(validated.name);
   790→          if (validated.name === "studio.plan_edit") {
   791→            const goal = (validated.args as { goal?: unknown }).goal;
   792→            const plan = await serverSidePlan(
   793→              run.model,
   794→              typeof goal === "string" ? goal : "",
   795→              conversation,
   796→              usage,
   797→              signal,
   798→            );
   799→            if (plan.cancelled === true) return cancelled();
   800→            if (plan.plan !== null) {
   801→              // M5 (§M5.2): run the server-side compiler on the plan and
   802→              // attach its output to the {type:"plan"} event — compiledBatch
   803→              // + resolutionRequests + server-recomputed risk/credits. All
   804→              // optional (back-compat); a compile failure emits the plan
   805→              // UN-enriched (still valid display data; Apply is gated
   806→              // downstream on compiledBatch's absence). The tool result and
   807→              // the model-facing content stay the BARE plan (M4 behavior)
   808→              // — the compiled registry command ids never reach the model.
   809→              const compiled = await compilePlanSafe({ plan: plan.plan });
   810→              if (compiled.ok) {
   811→                stream.emit({
   812→                  type: "plan",
   813→                  plan: plan.plan,
   814→                  compiledBatch: compiled.batch,
   815→                  resolutionRequests: [...compiled.resolutionRequests],
   816→                  risk: compiled.risk,
   817→                  estimatedCredits: compiled.estimatedCredits,
   818→                });
   819→              } else {
   820→                logger.debug("plan_compile_rejected", {
   821→                  runId: run.runId,
   822→                  issues: compiled.issues.length,
   823→                });
   824→                stream.emit({ type: "plan", plan: plan.plan });
   825→              }
   826→              stream.emit({ type: "tool.result", toolCallId: call.id, result: plan.plan });
   827→              await stream.yieldToStream();
   828→              conversation.push({
   829→                role: "tool",
   830→                tool_call_id: call.id,
   831→                content: JSON.stringify(plan.plan),
   832→              });
   833→            } else {
   834→              stream.emit({
   835→                type: "tool.result",
   836→                toolCallId: call.id,
   837→                result: { error: "PLAN_INVALID", message: plan.errorMessage },
   838→              });
   839→              await stream.yieldToStream();
   840→              conversation.push({
   841→                role: "tool",
   842→                tool_call_id: call.id,
   843→                content: JSON.stringify({ error: "PLAN_INVALID", message: plan.errorMessage }),
   844→              });
   845→            }
   846→          } else if (writeTool !== null) {
   847→            // M5.1 PROPOSE-ONLY write tool: resolve SERVER-SIDE through the
   848→            // compile seam. NOTHING executes in the gateway — the proposed
   849→            // batch executes only when the CLIENT applies it (§8.7, law #6:
   850→            // one atomic engine.batch = one undo step = one version).
   851→            const carrier = writeToolCarrierPlan(writeTool, validated.args);
   852→            const compiled = await compilePlanSafe({ plan: carrier });
   853→            if (compiled.ok) {
   854→              // Client envelope = the pinned WRITE_TOOL_RESULT_SCHEMA
   855→              // (contracts write-tools.ts): ONLY the proposed batch + the
   856→              // canonical proposed-only note — the human summary rides the
   857→              // batch label ("Agent: <summary>"). Boundary re-check (law #2)
   858→              // before the result goes on the wire; a compiler result that
   859→              // fails the pinned shape degrades to PROPOSAL_INVALID below.
   860→              const checked = WRITE_TOOL_RESULT_SCHEMA.safeParse({
   861→                proposedBatch: {
   862→                  label: compiled.batch.label,
   863→                  steps: compiled.batch.steps,
   864→                },
   865→                note: PROPOSED_BATCH_NOTE,
   866→              });
   867→              if (checked.success) {
   868→                stream.emit({
   869→                  type: "tool.result",
   870→                  toolCallId: call.id,
   871→                  result: checked.data,
   872→                });
   873→                await stream.yieldToStream();
   874→                if (signal.aborted) return cancelled();
   875→                logger.debug("write_tool_proposed", {
   876→                  runId: run.runId,
   877→                  tool: writeTool.id,
   878→                  steps: compiled.batch.steps.length,
   879→                  risk: compiled.risk,
   880→                  estimatedCredits: compiled.estimatedCredits,
   881→                  resolutionRequests: compiled.resolutionRequests.length,
   882→                });
   883→                // LAW #5 / M4-SAFETY B2: the model is fed a REDACTED
   884→                // envelope — the human summary + step count + the
   885→                // proposed-only note ONLY. The compiled registry command ids
   886→                // inside proposedBatch never enter the model conversation.
   887→                conversation.push({
   888→                  role: "tool",
   889→                  tool_call_id: call.id,
   890→                  content: JSON.stringify({
   891→                    proposed: true,
   892→                    summary: carrier.summary,
   893→                    note: PROPOSED_BATCH_NOTE,
   894→                    stepCount: compiled.batch.steps.length,
   895→                  }),
   896→                });
   897→              } else {
   898→                // Pinned-envelope violation (compiler drift): degrade safely.
   899→                const rejected = {
   900→                  error: "PROPOSAL_INVALID",
   901→                  message: "The proposed change did not pass the proposal contract.",
   902→                };
   903→                stream.emit({ type: "tool.result", toolCallId: call.id, result: rejected });
   904→                await stream.yieldToStream();
   905→                if (signal.aborted) return cancelled();
   906→                conversation.push({
   907→                  role: "tool",
   908→                  tool_call_id: call.id,
   909→                  content: JSON.stringify(rejected),
   910→                });
   911→              }
   912→            } else {
   913→              // Compile rejected (typed issues — the run survives; the model
   914→              // may adapt within the step cap). Mirrors PLAN_INVALID.
   915→              logger.debug("write_tool_compile_rejected", {
   916→                runId: run.runId,
   917→                tool: writeTool.id,
   918→                issues: compiled.issues.length,
   919→              });
   920→              const rejected = {
   921→                error: "PROPOSAL_INVALID",
   922→                message: compileIssueMessage(compiled.issues),
   923→              };
   924→              stream.emit({ type: "tool.result", toolCallId: call.id, result: rejected });
   925→              await stream.yieldToStream();
   926→              if (signal.aborted) return cancelled();
   927→              conversation.push({
   928→                role: "tool",
   929→                tool_call_id: call.id,
   930→                content: JSON.stringify(rejected),
   931→              });
   932→            }
   933→          } else {
   934→            const waited = await stream.waitForToolResult(call.id);
   935→            if (!waited.ok) {
   936→              if (signal.aborted) return cancelled();
   937→              if (waited.reason === "timeout") {
   938→                return failed("EDITOR_DISCONNECTED", "The editor didn't respond in time.", true);
   939→              }
   940→              return cancelled();
   941→            }
   942→            // The runs route already emitted the log's tool.result event when
   943→            // the client POSTed it; do not duplicate it here.
   944→            conversation.push({
   945→              role: "tool",
   946→              tool_call_id: call.id,
   947→              content: JSON.stringify(waited.result),
   948→            });
   949→          }
   950→          if (signal.aborted) return cancelled();
   951→        }
   952→      }
   953→
   954→      return {
   955→        kind: "completed",
   956→        committedCredits: committedCreditsFor(run.model, usage, conversation),
   957→      };
   958→    },
   959→
   960→    cancel() {
   961→      // The registry's AbortSignal is the single cancel path: it aborts the
   962→      // in-flight provider fetch/stream and every tool-result wait.
   963→    },
   964→  };
