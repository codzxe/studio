     1→     1→// @studio/gateway — the SERVER-SIDE PLAN COMPILER (BUILD_PLAN §M5 tasks 1-2,
     2→     2→// PRODUCT_ARCHITECTURE.md §8.5–§8.7, docs/phases/M5-SEAMS.md pinned seam,
     3→     3→// DECISIONS.md D-1: model loop ≤ 8 steps, plan ≤ 20 steps, compiled batch ≤
     4→     4→// 100 commands / 2 MB / 60 s).
     5→     5→//
     6→     6→// Mission (laws #5 + #6): the model proposes an EditPlan built from CURATED
     7→     7→// write-tool args; this module is the only thing that turns a plan into an
     8→     8→// engine `AgentBatch`. The per-step `batch` fields the model proposes are
     9→     9→// validated and ADVISORY ONLY — the compiled batch always comes from tool
    10→    10→// args translated against the curated command registry (deps.registry). One
    11→    11→// plan ⇒ ONE atomic batch ⇒ ONE undo step ⇒ ONE version (§8.4/§8.5).
    12→    12→//
    13→    13→// Tool → command translation table (every id verified against the real
    14→    14→// EffectCraft source at editor/crates/engine/src/commands/*.rs):
    15→    15→//
    16→    16→//   create_composition → comp.new {name, width, height, frameRate,
    17→    17→//                                  duration: args.durationSeconds}
    18→    18→//   create_text        → layer.newText {text, name?, position?,
    19→    19→//                                  size: fontSize?, font: fontFamily?,
    20→    20→//                                  fill: color?, comp: compId?}
    21→    21→//   set_properties     → prop.set × update {layer, path, value}
    22→    22→//   set_keyframes      → prop.addKey × keyframe {layer, path: propertyPath,
    23→    23→//                                  time, value}
    24→    24→//   add_effect         → effect.apply {effect, layers: [layer]}
    25→    25→//   apply_preset       → anim.applyBuiltinPreset {preset, layers: targets}
    26→    26→//   undo_run           → edit.undo {}
    27→    27→//   save_version       → NO engine command — a compile-time annotation
    28→    28→//                        (result.saveVersion) consumed by the apply flow.
    29→    29→//
    30→    30→// Ambiguity rulings (documented for the integrator — see the M5-B worklog):
    31→    31→//   * apply_preset NEVER compiles to `anim.applyPreset`: that command takes a
    32→    32→//     filesystem `path` (or an inline preset OBJECT), and a model-supplied
    33→    33→//     path breaks law #5 ("the model never sees filesystem paths"). Path-like
    34→    34→//     preset strings emit a `warn` and compile as builtin preset names.
    35→    35→//   * add_effect `params` and set_keyframes `easing` are CLIENT STATE (the
    36→    36→//     effect instance index and keyframe interpolation live in the editor);
    37→    37→//     they cannot be compiled server-side without guessing ids. They emit
    38→    38→//     `warn` issues instead of being silently dropped or fabricated.
    39→    39→//   * Layer references stay NAME strings in the compiled batch (the engine
    40→    40→//     accepts id|name|#n); `placeholders` + `resolutionRequests` let the
    41→    41→//     client (M5-F) substitute resolved ids before Apply. The compiler NEVER
    42→    42→//     fabricates ids; an injected `layerResolver` may substitute real ones.
    43→    43→//   * SEAMS names the request field `disambiguation` (the brief said
    44→    44→//     `candidates` — SEAMS wins per its own precedence rule). It carries the
    45→    45→//     plan's own `change` strings for that layer name.
    46→    46→//   * Risk and credits are RECOMPUTED from the compiled commands (never the
    47→    47→//     model's numbers). "High" is currently unreachable via the curated
    48→    48→//     tools (they compile to additive/property commands only) — it becomes
    49→    49→//     reachable when the registry widens past the eight write tools.
    50→    50→//
    51→    51→// Purity: idempotent, deterministic, no IO, no clocks (law #1: the result is
    52→    52→// a typed envelope and compilePlan NEVER throws, even on hostile input).

   253→   253→/** Deterministic, depth-capped canonical JSON (sorted object keys) used to
   254→   254→ * compare the model's proposed batch with the compiled translation. */
   255→   255→function stableStringify(value: unknown, depth = 0): string {
   256→   256→  if (depth > 64) return '"<deep>"';
   257→   257→  if (value === null) return "null";
   258→   258→  switch (typeof value) {
   259→   259→    case "string":
   260→   260→      return JSON.stringify(value);
   261→   261→    case "number":
   262→   262→      return Number.isFinite(value) ? String(value) : '"<nan>"';
   263→   263→    case "boolean":
   264→   264→    case "bigint":
   265→   265→      return String(value);
   266→   266→    default: {
   267→   267→      if (Array.isArray(value)) {
   268→   268→        return `[${value.map((item) => stableStringify(item, depth + 1)).join(",")}]`;
   269→   269→      }
   270→   270→      if (typeof value !== "object") return '"<unserializable>"';
   271→   271→      const record = value as Record<string, unknown>;
   272→   272→      const body = Object.keys(record)
   273→   273→        .sort()
   274→   274→        .map((key) => `${JSON.stringify(key)}:${stableStringify(record[key], depth + 1)}`)
   275→   275→        .join(",");
   276→   276→      return `{${body}}`;
   277→   277→    }
   278→   278→  }
   279→   279→}

   281→   281→function riskMax(a: PlanRisk, b: PlanRisk): PlanRisk {
   282→   282→  return PLAN_RISK_ORDER[a] >= PLAN_RISK_ORDER[b] ? a : b;
   283→   283→}

   285→   285→// ---------------------------------------------------------------------------
   286→   286→// Risk recompute (brief: from compiled commands — never the model's number)
   287→   287→// ---------------------------------------------------------------------------
   288→   288→
   289→   289→/**
   290→   290→ * Recomputed plan risk from the COMPILED commands plus an optional tool-catalog
   291→   291→ * floor. delete-ish / effect.remove / render/export / edit.undo+redo ⇒ at
   292→   292→ * least medium; nothing of the sort ⇒ low. Exported for tests and future
   293→   293→ * registry widening.
   294→   294→ */
   295→   295→export function recomputePlanRisk(
   296→   296→  steps: ReadonlyArray<Pick<BatchStep, "command">>,
   297→   297→  toolFloor: PlanRisk = "low",
   298→   298→): PlanRisk {
   299→   299→  let risk: PlanRisk = "low";
   300→   300→  for (const step of steps) {
   301→   301→    const id = step.command;
   302→   302→    if (DELETE_ISH_COMMAND.test(id) || RENDER_ISH_COMMAND.test(id) || HISTORY_COMMANDS.has(id)) {
   303→   303→      risk = riskMax(risk, "medium");
   304→   304→    }
   305→   305→  }
   306→   306→  return riskMax(risk, toolFloor);
   307→   307→}

   346→   346→  private pushIssue(issue: PlanIssue): void {
   347→   347→    const target = issue.severity === "fatal" ? this.fatals : this.warnings;
   348→   348→    if (target.length >= MAX_ISSUES) return; // bounded, deterministic

   358→   358→  }

   367→   367→
   368→   368→  /** Translate one plan step's curated args into registry command steps. */
   369→   369→  translate(planStep: number, tool: string, args: unknown): StepTranslation {
   370→   370→    const steps: BatchStep[] = [];
   371→   371→    const sites: LocalLayerSite[] = [];
   372→   372→    const argsObj: Record<string, unknown> = isPlainObject(args) ? args : {};
   373→   373→    const add = (command: string, params: Record<string, unknown>): void => {
   374→   374→      steps.push({ command, params });
   375→   375→    };
   376→   376→    const addSite = (param: string, layerName: string): void => {
   377→   377→      sites.push({ localStep: steps.length - 1, param, layerName });
   378→   378→    };
   379→   379→    const unusable = (field: string, why: string): void => {
   380→   380→      this.warn("unusable_args", `field \`${field}\` ${why}`, { step: planStep, tool });
   381→   381→    };
   382→   382→    const compilesToNothing = (): void => {
   383→   383→      // G-3 completeness ruling: a step that translates to ZERO commands
   384→   384→      // makes the WHOLE plan fatally incomplete — the user must never approve
   385→   385→      // N edits and silently get fewer (M5-M gp-11 finding).
   386→   386→      this.fatal(
   387→   387→        "empty_step",
   388→   388→        "step compiled to no engine commands; the plan is rejected as incomplete",
   389→   389→        { step: planStep, tool },
   390→   390→      );
   391→   391→    };
   392→   392→    /** Read `[{name}]`-shaped layer refs (M5-A LAYER_REFS_SCHEMA) as names. */
   393→   393→    const readLayerNames = (value: unknown): string[] => {
   394→   394→      const names: string[] = [];
   395→   395→      if (!Array.isArray(value)) return names;
   396→   396→      for (const entry of value) {
   397→   397→        const name = isPlainObject(entry) ? readString(entry, "name") : undefined;
   398→   398→        if (name !== undefined) names.push(name);
   399→   399→        else unusable("layers", "entry has no usable name (skipped)");
   400→   400→      }
   401→   401→      return names;
   402→   402→    };
   403→   403→
   404→   404→    switch (tool) {
   405→   405→      case "create_composition": {
   406→   406→        const params: Record<string, unknown> = {};
   407→   407→        const name = readString(argsObj, "name");
   408→   408→        if (name !== undefined) params.name = name;
   409→   409→        for (const key of ["width", "height", "frameRate"] as const) {
   410→   410→          const value = argsObj[key];
   411→   411→          if (isFiniteNumber(value)) params[key] = value;
   412→   412→          else if (value !== undefined) unusable(key, "is not a finite number");
   413→   413→        }
   414→   414→        const duration = argsObj.durationSeconds;
   415→   415→        if (isFiniteNumber(duration)) params.duration = duration;
   416→   416→        else if (duration !== undefined) unusable("durationSeconds", "is not a finite number");
   417→   417→        if (params.name === undefined) {
   418→   418→          compilesToNothing();
   419→   419→          break;
   420→   420→        }
   421→   421→        add("comp.new", params);
   422→   422→        break;
   423→   423→      }
   424→   424→
   425→   425→      case "create_text": {
   426→   426→        const params: Record<string, unknown> = {};
   427→   427→        const text = readString(argsObj, "text");
   428→   428→        if (text !== undefined) params.text = text;
   429→   429→        else if ("text" in argsObj) unusable("text", "is not a non-empty string");
   430→   430→        const name = readString(argsObj, "name");
   431→   431→        if (name !== undefined) params.name = name;
   432→   432→        for (const key of ["font", "style", "fill", "stroke"] as const) {
   433→   433→          const value = readString(argsObj, key);
   434→   434→          if (value !== undefined) params[key] = value;
   435→   435→        }
   436→   436→        for (const key of ["size", "strokeWidth", "tracking", "leading"] as const) {
   437→   437→          const value = argsObj[key];
   438→   438→          if (isFiniteNumber(value)) params[key] = value;
   439→   439→          else if (value !== undefined) unusable(key, "is not a finite number");
   440→   440→        }
   441→   441→        const box = argsObj.box;
   442→   442→        if (box !== undefined) {
   443→   443→          const [bx, by, w, h] = Array.isArray(box) ? box : [];
   444→   444→          if (isFiniteNumber(bx) && isFiniteNumber(by) && isFiniteNumber(w) && isFiniteNumber(h)) {
   445→   445→            params.box = [bx, by, w, h];
   446→   446→          } else {
   447→   447→            unusable("box", "is not an [x, y, w, h] number tuple");
   448→   448→          }
   449→   449→        }
   450→   450→        if (params.text === undefined) {
   451→   451→          compilesToNothing();
   452→   452→          break;
   453→   453→        }
   454→   454→        const position = argsObj.position;
   455→   455→        if (position !== undefined) {
   456→   456→          const [x, y] = Array.isArray(position) ? position : [];
   457→   457→          if (isFiniteNumber(x) && isFiniteNumber(y)) {
   458→   458→            params.position = [x, y];
   459→   459→          } else {
   460→   460→            unusable("position", "is not a [x, y] number pair");
   461→   461→          }
   462→   462→        }
   463→   463→        if (params.text === undefined) {
   464→   464→          compilesToNothing();
   465→   465→          break;
   466→   466→        }
   467→   467→        add("layer.newText", params);
   468→   468→        break;
   469→   469→      }
   470→   470→
   471→   471→      case "set_properties": {
   472→   472→        // M5-A shape: {layers: [{name}], properties: [{path, value}]} ->
   473→   473→        // cross-product prop.set per layer x property; a "name" path maps to
   474→   474→        // layer.rename (G-7 -- rename was otherwise inexpressible).
   475→   475→        const layers = readLayerNames(argsObj.layers);
   476→   476→        const properties = argsObj.properties;
   477→   477→        if (layers.length === 0) {
   478→   478→          unusable("layers", "no usable named layers");
   479→   479→          compilesToNothing();
   480→   480→          break;
   481→   481→        }
   482→   482→        if (!Array.isArray(properties)) {
   483→   483→          unusable("properties", "is not an array");
   484→   484→          compilesToNothing();
   485→   485→          break;
   486→   486→        }
   487→   487→        for (const property of properties) {
   488→   488→          if (!isPlainObject(property)) {
   489→   489→            unusable("properties", "contains a non-object entry (skipped)");
   490→   490→            continue;
   491→   491→          }
   492→   492→          const path = readString(property, "path");
   493→   493→          if (path === undefined) {
   494→   494→            unusable("properties", "entry has no path (skipped)");
   495→   495→            continue;
   496→   496→          }
   497→   497→          for (const layer of layers) {
   498→   498→            if (path === "name") {
   499→   499→              const value = readString(property, "value");
   500→   500→              if (value === undefined) {
   501→   501→                unusable("properties", "a rename needs a string value (skipped)");
   502→   502→                continue;
   503→   503→              }
   504→   504→              add("layer.rename", { layer, name: value });
   505→   505→            } else {
   506→   506→              const params: Record<string, unknown> = { layer, path };
   507→   507→              if ("value" in property) params.value = property.value;
   508→   508→              add("prop.set", params);
   509→   509→            }
   510→   510→            addSite("layer", layer);
   511→   511→          }
   512→   512→        }
   513→   513→        if (steps.length === 0) compilesToNothing();
   514→   514→        break;
   515→   515→      }
   516→   516→
   517→   517→      case "set_keyframes": {
   518→   518→        // M5-A shape: {layers: [{name}], propertyPath, keys: [{timeSeconds,
   519→   519→        // value, interpolation?}]} -> cross-product prop.addKey per layer x key.
   520→   520→        const layers = readLayerNames(argsObj.layers);
   521→   521→        const keys = argsObj.keys;
   522→   522→        const propertyPath = readString(argsObj, "propertyPath");
   523→   523→        if (layers.length === 0) {
   524→   524→          unusable("layers", "no usable named layers");
   525→   525→          compilesToNothing();
   526→   526→          break;
   527→   527→        }
   528→   528→        if (!Array.isArray(keys)) {
   529→   529→          unusable("keys", "is not an array");
   530→   530→          compilesToNothing();
   531→   531→          break;
   532→   532→        }
   533→   533→        for (const key of keys) {
   534→   534→          if (!isPlainObject(key)) {
   535→   535→            unusable("keys", "contains a non-object entry (skipped)");
   536→   536→            continue;
   537→   537→          }
   538→   538→          const time = key.timeSeconds;
   539→   539→          const interpolation = readString(key, "interpolation");
   540→   540→          for (const layer of layers) {
   541→   541→            const params: Record<string, unknown> = { layer };
   542→   542→            if (propertyPath !== undefined) params.path = propertyPath;
   543→   543→            if (isFiniteNumber(time)) params.time = time;
   544→   544→            else if (time !== undefined) {
   545→   545→              unusable("keys", "entry timeSeconds is not a finite number");
   546→   546→            }
   547→   547→            if ("value" in key) params.value = key.value;
   548→   548→            if (interpolation !== undefined && interpolation !== "linear") {
   549→   549→              this.warn(
   550→   550→                "uncompilable_arg",
   551→   551→                `interpolation \`${clip(interpolation, 24)}\` is editor-side keyframe state; the key compiles with the engine default (set it via the timeline after apply)`,
   552→   552→                { step: planStep, tool, param: "interpolation" },
   553→   553→              );
   554→   554→            }
   555→   555→            add("prop.addKey", params);
   556→   556→            addSite("layer", layer);
   557→   557→          }
   558→   558→        }
   559→   559→        if (steps.length === 0) compilesToNothing();
   560→   560→        break;
   561→   561→      }
   562→   562→
   563→   563→      case "add_effect": {
   564→   564→        // M5-A shape: {layers: [{name}], effect: {name}, params?} ->
   565→   565→        // effect.apply {layers: [names], effect: name}. Effect instance
   566→   566→        // params are editor state (uncompilable server-side; warn + set via
   567→   567→        // set_properties after apply).
   568→   568→        const layers = readLayerNames(argsObj.layers);
   569→   569→        if (layers.length === 0) {
   570→   570→          unusable("layers", "no usable named layers");
   571→   571→          compilesToNothing();
   572→   572→          break;
   573→   573→        }
   574→   574→        const effectObj = argsObj.effect;
   575→   575→        const effectName = isPlainObject(effectObj) ? readString(effectObj, "name") : undefined;
   576→   576→        const params: Record<string, unknown> = { layers: [...layers] };
   577→   577→        if (effectName !== undefined) params.effect = effectName;
   578→   578→        else unusable("effect", "has no usable name");
   579→   579→        add("effect.apply", params);
   580→   580→        layers.forEach((layer, index) => addSite(`layers[${index}]`, layer));
   581→   581→        const effectParams = argsObj.params;
   582→   582→        if (effectParams !== undefined) {
   583→   583→          if (isPlainObject(effectParams)) {
   584→   584→            for (const key of Object.keys(effectParams).sort()) {
   585→   585→              this.warn(
   586→   586→                "uncompilable_arg",
   587→   587→                `effect parameter \`${clip(key, 40)}\` cannot be compiled server-side (the effect instance index is editor state); set it via set_properties after apply`,
   588→   588→                { step: planStep, tool, param: key },
   589→   589→              );
   590→   590→            }
   591→   591→          } else {
   592→   592→            unusable("params", "is not an object");
   593→   593→          }
   594→   594→        }
   595→   595→        break;
   596→   596→      }
   597→   597→
   598→   598→      case "apply_preset": {
   599→   599→        // M5-A shape: {preset: {name}|{builtinId}, layers? [{name}], args?}.
   600→   600→        // D-21 ruling: when `layers` is omitted the plan's affectedLayers
   601→   601→        // names are the targets (never the live editor selection -- a
   602→   602→        // selection-capture hazard the compiler refuses).
   603→   603→        const explicitLayers = readLayerNames(argsObj.layers);
   604→   604→        const layers =
   605→   605→          explicitLayers.length > 0 ? explicitLayers : [...this.planLayerNames];
   606→   606→        if (layers.length === 0) {
   607→   607→          unusable(
   608→   608→            "layers",
   609→   609→            "no target layers: neither step layers nor plan affectedLayers name any",
   610→   610→          );
   611→   611→          compilesToNothing();
   612→   612→          break;
   613→   613→        }
   614→   614→        const presetObj = argsObj.preset;
   615→   615→        let preset: string | undefined;
   616→   616→        if (isPlainObject(presetObj)) {
   617→   617→          preset = readString(presetObj, "name") ?? readString(presetObj, "builtinId");
   618→   618→        }
   619→   619→        if (preset !== undefined && PRESET_PATH_LIKE.test(preset)) {
   620→   620→          this.warn(
   621→   621→            "uncompilable_arg",
   622→   622→            "preset looks like a filesystem path; paths are never agent-compilable " +
   623→   623→              "(compiled as a builtin preset name)",
   624→   624→            { step: planStep, tool, param: "preset" },
   625→   625→          );
   626→   626→        }
   627→   627→        if (preset === undefined) {
   628→   628→          unusable("preset", "needs a name or builtinId");
   629→   629→          compilesToNothing();
   630→   630→          break;
   631→   631→        }
   632→   632→        const params: Record<string, unknown> = { layers: [...layers], preset };
   633→   633→        const presetArgs = argsObj.args;
   634→   634→        if (isPlainObject(presetArgs)) {
   635→   635→          for (const key of Object.keys(presetArgs).sort()) {
   636→   636→            this.warn(
   637→   637→              "uncompilable_arg",
   638→   638→              `preset arg \`${clip(key, 40)}\` is editor-side state; set it via set_properties after apply`,
   639→   639→              { step: planStep, tool, param: key },
   640→   640→            );
   641→   641→          }
   642→   642→        }
   643→   643→        add("anim.applyBuiltinPreset", params);
   644→   644→        layers.forEach((layer, index) => addSite(`layers[${index}]`, layer));
   645→   645→        break;
   646→   646→      }
   647→   647→
   648→   648→      case "undo_run": {
   649→   649→        add("edit.undo", {});
   650→   650→        if ("runId" in argsObj) {
   651→   651→          this.warn(
   652→   652→            "undo_run_target",
   653→   653→            "undo_run named a specific run; the engine command reverts the most recently " +
   654→   654→              "applied batch — the gateway checks run ordering at undo time",
   655→   655→            { step: planStep, tool },
   656→   656→          );
   657→   657→        }
   658→   658→        break;
   659→   659→      }
   660→   660→
   661→   661→      case "save_version": {
   662→   662→        const rawLabel = argsObj.label;
   663→   663→        const label = typeof rawLabel === "string" ? rawLabel : undefined;
   664→   664→        if (this.saveVersion !== undefined && this.saveVersionStep !== undefined) {
   665→   665→          this.warn(
   666→   666→            "duplicate_save_version",
   667→   667→            "an earlier save_version step was superseded (the last labels the version)",
   668→   668→            {
   669→   669→              step: this.saveVersionStep,
   670→   670→              tool,
   671→   671→            },
   672→   672→          );
   673→   673→        }
   674→   674→        this.saveVersion = label !== undefined ? { label } : {};
   675→   675→        this.saveVersionStep = planStep;
   676→   676→        break;
   677→   677→      }
   678→   678→
   679→   679→      default: {
   680→   680→        // Exhaustiveness guard — lookupWriteTool already rejected unknown ids.
   681→   681→        this.fatal("unknown_tool", `tool \`${clip(tool, 64)}\` is not a write tool`, {
   682→   682→          step: planStep,
   683→   683→          tool,
   684→   684→        });
   685→   685→        break;
   686→   686→      }
   687→   687→    }
   688→   688→
   689→   689→    return { steps, sites };
   690→   690→  }

   692→   692→  /** Validate the model's PROPOSED batch for one step (advisory — warn only:
   693→   693→   * the compiled translation wins, law #5). */
   694→   694→  validateProposed(
   695→   695→    planStep: number,
   696→   696→    tool: string,
   697→   697→    proposed: readonly BatchStep[],
   698→   698→    translated: readonly BatchStep[],
   699→   699→  ): void {
   700→   700→    // Own-property allowlist scan — a "__proto__" id must not resolve through
   701→   701→    // the prototype chain (SEAMS: "never invent a command id").
   702→   702→    for (const [index, step] of proposed.entries()) {
   703→   703→      if (!registryHas(this.depsRegistry, step.command)) {
   704→   704→        this.warn(
   705→   705→          "unknown_command",
   706→   706→          `proposed command \`${clip(step.command, 64)}\` is not in the agent registry`,
   707→   707→          {
   708→   708→            step: planStep,
   709→   709→            tool,
   710→   710→            command: step.command,
   711→   711→            batchStep: index + 1,
   712→   712→          },
   713→   713→        );
   714→   714→      }
   715→   715→    }
   716→   716→    // Required params for the KNOWN proposed commands (reuse the contracts
   717→   717→    // helper; unknown-command duplicates were handled above).
   718→   718→    for (const issue of checkBatchCommands({ steps: [...proposed] }, this.depsRegistry)) {
   719→   719→      if (issue.reason !== "missing_required_param") continue;
   720→   720→      this.warn(
   721→   721→        "missing_required_param",
   722→   722→        `proposed \`${issue.command}\` is missing required param \`${issue.param}\``,
   723→   723→        {
   724→   724→          step: planStep,
   725→   725→          tool,
   726→   726→          command: issue.command,
   727→   727→          batchStep: issue.step,
   728→   728→          param: issue.param,
   729→   729→        },
   730→   730→      );
   731→   731→    }
   732→   732→    // Backwards-only $N.key refs (§8.5) within the proposed batch's own
   733→   733→    // 1-based numbering.
   734→   734→    for (const issue of validateBatchRefs(proposed)) {
   735→   735→      this.warn(
   736→   736→        "unresolvable_ref",
   737→   737→        `proposed batch has $-reference \`${issue.ref.raw}\` (${issue.reason})`,
   738→   738→        {
   739→   739→          step: planStep,
   740→   740→          tool,
   741→   741→          batchStep: issue.step,
   742→   742→        },
   743→   743→      );
   744→   744→    }
   745→   745→    // Disagreement: the compiled translation wins — warn, never merge.
   746→   746→    const agrees =
   747→   747→      proposed.length === translated.length &&
   748→   748→      proposed.every(
   749→   749→        (step, index) =>
   750→   750→          step.command === translated[index]?.command &&
   751→   751→          stableStringify(step.params) === stableStringify(translated[index]?.params),
   752→   752→      );
   753→   753→    if (!agrees) {
   754→   754→      this.warn(
   755→   755→        "batch_mismatch",
   756→   756→        "the model's proposed batch disagrees with the compiled translation; the translation stands",
   757→   757→        {
   758→   758→          step: planStep,
   759→   759→          tool,
   760→   760→        },
   761→   761→      );
   762→   762→    }
   763→   763→  }

   766→   766→  merge(planStep: number, translation: StepTranslation, toolRisk: PlanRisk | undefined): void {
   767→   767→    const base = this.compiledSteps.length;

   771→   771→    }
   772→   772→    this.toolRisks.push(toolRisk ?? "low");

   796→   796→/** The M5-A seam may key tools by bare id (`create_text`) or with the
   797→   797→ * `studio.` prefix — and the plan may use either form too. Deterministic
   798→   798→ * candidate order, own-property lookup only. */
   799→   799→function lookupWriteTool(deps: CompileDeps, tool: string): NormalizedTool | undefined {
   800→   800→  const bare = tool.startsWith(STUDIO_TOOL_PREFIX) ? tool.slice(STUDIO_TOOL_PREFIX.length) : tool;
   801→   801→  for (const candidate of [bare, `${STUDIO_TOOL_PREFIX}${bare}`, tool]) {
   802→   802→    if (!Object.prototype.hasOwnProperty.call(deps.writeTools, candidate)) continue;
   803→   803→    const entry = deps.writeTools[candidate];
   804→   804→    if (
   805→   805→      entry !== null &&
   806→   806→      typeof entry === "object" &&
   807→   807→      "safeParse" in entry &&
   808→   808→      typeof entry.safeParse === "function"
   809→   809→    ) {
   810→   810→      return { bareId: bare, argsSchema: entry as ZodTypeAny };
   811→   811→    }
   812→   812→    if (isPlainObject(entry)) {
   813→   813→      const def = entry as Record<string, unknown>;
   814→   814→      const args = def.args;
   815→   815→      const risk = def.risk;
   816→   816→      return {
   817→   817→        bareId: bare,
   818→   818→        argsSchema:
   819→   819→          args !== null &&
   820→   820→          typeof args === "object" &&
   821→   821→          "safeParse" in args &&
   822→   822→          typeof (args as ZodTypeAny).safeParse === "function"
   823→   823→            ? (args as ZodTypeAny)
   824→   824→            : undefined,
   825→   825→        risk: risk === "low" || risk === "medium" || risk === "high" ? risk : undefined,
   826→   826→      };
   827→   827→    }
   828→   828→  }
   829→   829→  return undefined;
   830→   830→}

   832→   832→/** Write `value` at a `layer`/`layers[i]`-style path inside a params object. */
   833→   833→function applyParamPath(
   834→   834→  params: Record<string, unknown>,
   835→   835→  path: string,
   836→   836→  value: string | number,
   837→   837→): void {
   838→   838→  const match = /^([A-Za-z0-9_]+)(?:\[(\d+)\])?$/.exec(path);
   839→   839→  if (match === null) return;
   840→   840→  const key = match[1];
   841→   841→  if (key === undefined) return;
   842→   842→  const indexText = match[2];
   843→   843→  if (indexText === undefined) {
   844→   844→    params[key] = value;
   845→   845→    return;
   846→   846→  }
   847→   847→  const list = params[key];
   848→   848→  if (Array.isArray(list)) {
   849→   849→    const index = Number.parseInt(indexText, 10);
   850→   850→    if (index >= 0 && index < list.length) list[index] = value;
   851→   851→  }
   852→   852→}

   862→   862→function compilePlanInner(plan: unknown, deps: CompileDeps): CompilePlanResult {
   863→   863→  const compiler = new Compiler(deps.registry);
   864→   864→
   865→   865→  // -- 0. pre-flight: the plan must serialize (cycle-hostile) and be sane.
   866→   866→  let serializedPlan: string;
   867→   867→  try {
   868→   868→    serializedPlan = JSON.stringify(plan) ?? "null";
   869→   869→  } catch {
   870→   870→    compiler.fatal("plan_too_large", "the plan is not serializable JSON");
   871→   871→    return { ok: false, issues: compiler.fatals };
   872→   872→  }
   873→   873→  if (serializedPlan.length > MAX_PLAN_BYTES) {
   874→   874→    compiler.fatal(
   875→   875→      "plan_too_large",
   876→   876→      `the plan is too large to compile (${serializedPlan.length} bytes)`,
   877→   877→    );
   878→   878→    return { ok: false, issues: compiler.fatals };
   879→   879→  }
   880→   880→
   881→   881→  // -- 1. defense-in-depth re-validation against the §8.6 schema (the runner
   882→   882→  // validates first; hostile direct calls stop here).
   883→   883→  const parsed = EDIT_PLAN_SCHEMA.safeParse(plan);
   884→   884→  if (!parsed.success) {
   885→   885→    const messages = parsed.error.issues.slice(0, 5).map((issue) => issue.message);
   886→   886→    compiler.fatal(
   887→   887→      "plan_schema",
   888→   888→      `the plan failed its schema: ${messages.join("; ") || "invalid shape"}`,
   889→   889→    );
   890→   890→    return { ok: false, issues: compiler.fatals };
   891→   891→  }
   892→   892→  const editPlan = parsed.data;
   893→   893→  compiler.planLayerNames = editPlan.affectedLayers
   894→   894→    .map((layer) => (typeof layer.name === "string" && layer.name.length > 0 ? layer.name : ""))
   895→   895→    .filter((name) => name !== "");
   896→   896→
   897→   897→  // -- 2. per-step: allowlist the tool, validate args, translate, validate
   898→   898→  // the proposed batch (warns only — the translation wins, law #5).
   899→   899→  for (const [index, planStep] of editPlan.steps.entries()) {
   900→   900→    const position = index + 1;
   901→   901→    if (typeof planStep.tool !== "string" || planStep.tool.length === 0) {
   902→   902→      compiler.fatal("unknown_tool", `step ${position} has no tool`, { step: position });
   903→   903→      continue;
   904→   904→    }
   905→   905→    const tool = lookupWriteTool(deps, planStep.tool);
   906→   906→    if (tool === undefined) {
   907→   907→      compiler.fatal(
   908→   908→        "unknown_tool",
   909→   909→        `tool \`${clip(planStep.tool, 64)}\` is not one of the 8 write tools`,
   910→   910→        {
   911→   911→          step: position,
   912→   912→          tool: planStep.tool,
   913→   913→        },
   914→   914→      );
   915→   915→      continue;
   916→   916→    }
   917→   917→
   918→   918→    if (tool.argsSchema !== undefined) {
   919→   919→      let argsOk = true;
   920→   920→      let argsMessage = "";
   921→   921→      try {
   922→   922→        const safe = tool.argsSchema.safeParse(planStep.args);
   923→   923→        if (!safe.success) {
   924→   924→          argsOk = false;
   925→   925→          argsMessage = safe.error.issues
   926→   926→            .slice(0, 5)
   927→   927→            .map((issue) => issue.message)
   928→   928→            .join("; ");
   929→   929→        }
   930→   930→      } catch {
   931→   931→        argsOk = false;
   932→   932→        argsMessage = "args could not be validated";
   933→   933→      }
   934→   934→      if (!argsOk) {
   935→   935→        compiler.fatal(
   936→   936→          "invalid_args",
   937→   937→          `args for \`${tool.bareId}\` failed validation: ${argsMessage}`,
   938→   938→          {
   939→   939→            step: position,
   940→   940→            tool: tool.bareId,
   941→   941→          },
   942→   942→        );
   943→   943→        continue;
   944→   944→      }
   945→   945→    }
   946→   946→
   947→   947→    const translation = compiler.translate(position, tool.bareId, planStep.args);
   948→   948→    compiler.validateProposed(position, tool.bareId, planStep.batch.steps, translation.steps);
   949→   949→    compiler.merge(position, translation, tool.risk);
   950→   950→  }
   951→   951→  if (compiler.fatals.length > 0) {
   952→   952→    return { ok: false, issues: compiler.fatals };
   953→   953→  }
   954→   954→
   955→   955→  // -- 3. bounds + registry gate on the COMPILED batch (violations are fatal).
   956→   956→  const steps: BatchStep[] = [...compiler.steps()];
   957→   957→  if (steps.length === 0) {
   958→   958→    compiler.fatal(
   959→   959→      "empty_batch",
   960→   960→      "the plan compiles to an empty command batch (only annotation steps)",
   961→   961→    );
   962→   962→    return { ok: false, issues: compiler.fatals };
   963→   963→  }
   964→   964→  if (steps.length > MAX_BATCH_STEPS) {
   965→   965→    compiler.fatal(
   966→   966→      "too_many_commands",
   967→   967→      `the compiled batch has ${steps.length} commands (max ${MAX_BATCH_STEPS})`,
   968→   968→    );
   969→   969→    return { ok: false, issues: compiler.fatals };
   970→   970→  }
   971→   971→  for (const issue of checkBatchCommands({ steps }, deps.registry)) {
   972→   972→    if (issue.reason === "unknown_command") {
   973→   973→      compiler.fatal(
   974→   974→        "unknown_command",
   975→   975→        `compiled command \`${issue.command}\` is not in the agent registry`,
   976→   976→        {
   977→   977→          batchStep: issue.step,
   978→   978→          command: issue.command,
   979→   979→        },
   980→   980→      );
   981→   981→    } else {
   982→   982→      compiler.fatal(
   983→   983→        "missing_required_param",
   984→   984→        `\`${issue.command}\` is missing required param \`${issue.param}\``,
   985→   985→        {
   986→   986→          batchStep: issue.step,
   987→   987→          command: issue.command,
   988→   988→          param: issue.param,
   989→   989→        },
   990→   990→      );
   991→   991→    }
   992→   992→  }
   993→   993→  // The compiled batch is name-based by construction — refs are validated as
   994→   994→  // defense-in-depth anyway (§8.5 backwards-only $N.key).
   995→   995→  for (const issue of validateBatchRefs(steps)) {
   996→   996→    compiler.fatal(
   997→   997→      "unresolvable_ref",
   998→   998→      `compiled batch has $-reference \`${issue.ref.raw}\` (${issue.reason})`,
   999→   999→      {
  1000→  1000→        batchStep: issue.step,
  1001→  1001→      },
  1002→  1002→    );
  1003→  1003→  }
  1004→  1004→  if (compiler.fatals.length > 0) {
  1005→  1005→    return { ok: false, issues: compiler.fatals };
  1006→  1006→  }
  1007→  1007→
  1008→  1008→  // -- 4. layer-name resolution: requests + placeholders (names stay names in
  1009→  1009→  // the batch; the client substitutes ids via the bridge — §8.7).
  1010→  1010→  const declaredNames: string[] = [];
  1011→  1011→  const disambiguation = new Map<string, string[]>();
  1012→  1012→  for (const layer of editPlan.affectedLayers) {
  1013→  1013→    if (typeof layer.name !== "string" || layer.name.length === 0) continue;
  1014→  1014→    if (!declaredNames.includes(layer.name)) declaredNames.push(layer.name);
  1015→  1015→    const notes = disambiguation.get(layer.name) ?? [];
  1016→  1016→    if (!notes.includes(layer.change)) notes.push(layer.change);
  1017→  1017→    disambiguation.set(layer.name, notes);
  1018→  1018→  }
  1019→  1019→  const uniqueNames = [...declaredNames];
  1020→  1020→  for (const site of compiler.layerSites()) {
  1021→  1021→    if (!uniqueNames.includes(site.layerName)) uniqueNames.push(site.layerName);
  1022→  1022→  }
  1023→  1023→  if (uniqueNames.length > MAX_PLAN_AFFECTED_LAYERS) {
  1024→  1024→    compiler.fatal(
  1025→  1025→      "too_many_layer_names",
  1026→  1026→      `the plan references ${uniqueNames.length} distinct layer names (max ${MAX_PLAN_AFFECTED_LAYERS})`,
  1027→  1027→    );
  1028→  1028→    return { ok: false, issues: compiler.fatals };
  1029→  1029→  }
  1030→  1030→
  1031→  1031→  const warnedUndeclared = new Set<string>();
  1032→  1032→  for (const site of compiler.layerSites()) {
  1033→  1033→    if (declaredNames.includes(site.layerName) || warnedUndeclared.has(site.layerName)) continue;
  1034→  1034→    warnedUndeclared.add(site.layerName);
  1035→  1035→    compiler.warn(
  1036→  1036→      "undeclared_layer",
  1037→  1037→      `layer \`${clip(site.layerName, 64)}\` is used by a step but not declared in the plan`,
  1038→  1038→      {
  1039→  1039→        step: site.planStep,
  1040→  1040→      },
  1041→  1041→    );
  1042→  1042→  }
  1043→  1043→
  1044→  1044→  const resolved = new Map<string, string | number>();
  1045→  1045→  if (deps.layerResolver !== undefined) {
  1046→  1046→    for (const name of uniqueNames) {
  1047→  1047→      try {
  1048→  1048→        const id = deps.layerResolver(name);
  1049→  1049→        if (id !== undefined) resolved.set(name, id);
  1050→  1050→      } catch {
  1051→  1051→        compiler.warn(
  1052→  1052→          "resolver_error",
  1053→  1053→          "the layer resolver failed; the layer name stays in the batch",
  1054→  1054→        );
  1055→  1055→      }
  1056→  1056→    }
  1057→  1057→  }
  1058→  1058→
  1059→  1059→  const requests: LayerResolutionRequest[] = [];
  1060→  1060→  const requestIdByName = new Map<string, string>();
  1061→  1061→  let requestIndex = 0;
  1062→  1062→  for (const name of uniqueNames) {
  1063→  1063→    if (resolved.has(name)) continue; // substituted server-side — no request
  1064→  1064→    requestIndex += 1;
  1065→  1065→    const requestId = `lr-${requestIndex}`;
  1066→  1066→    requestIdByName.set(name, requestId);
  1067→  1067→    requests.push({ requestId, layerName: name, disambiguation: disambiguation.get(name) ?? [] });
  1068→  1068→  }
  1069→  1069→
  1070→  1070→  const placeholders: BatchPlaceholder[] = [];
  1071→  1071→  for (const site of compiler.layerSites()) {
  1072→  1072→    const resolvedId = resolved.get(site.layerName);
  1073→  1073→    if (resolvedId !== undefined) {
  1074→  1074→      // Substitute the authoritative id (resolver-provided, never fabricated).
  1075→  1075→      const step = steps[site.globalStep];
  1076→  1076→      if (step !== undefined) applyParamPath(step.params, site.param, resolvedId);
  1077→  1077→      placeholders.push({
  1078→  1078→        step: site.globalStep + 1,
  1079→  1079→        param: site.param,
  1080→  1080→        layerName: site.layerName,
  1081→  1081→        resolvedId,
  1082→  1082→      });
  1083→  1083→      continue;
  1084→  1084→    }
  1085→  1085→    placeholders.push({
  1086→  1086→      step: site.globalStep + 1,
  1087→  1087→      param: site.param,
  1088→  1088→      layerName: site.layerName,
  1089→  1089→      requestId: requestIdByName.get(site.layerName),
  1090→  1090→    });
  1091→  1091→  }
  1092→  1092→
  1093→  1093→  // -- 5. assemble the ONE atomic batch (§8.4/§8.5: "Agent: <summary>").
  1094→  1094→  const label = `${AGENT_LABEL_PREFIX}${clip(editPlan.summary, MAX_BATCH_LABEL_LENGTH - AGENT_LABEL_PREFIX.length)}`;
  1095→  1095→  const batch: AgentBatch = { label, atomic: true, steps };
  1096→  1096→
  1097→  1097→  const serializedBatch = JSON.stringify(batch);
  1098→  1098→  if (serializedBatch !== undefined && serializedBatch.length > MAX_BATCH_BYTES) {
  1099→  1099→    compiler.fatal(
  1100→  1100→      "batch_too_large",
  1101→  1101→      `the compiled batch exceeds the ${MAX_BATCH_BYTES}-byte limit (${serializedBatch.length} bytes)`,
  1102→  1102→    );
  1103→  1103→    return { ok: false, issues: compiler.fatals };
  1104→  1104→  }
  1105→  1105→
  1106→  1106→  // -- 6. risk + credits recomputed server-side (never the model's numbers).
  1107→  1107→  // G-4: the recomputed risk is a floor-safeguarded value -- the model's
  1108→  1108→  // declared risk is never LOWERED (the card must never understate risk),
  1109→  1109→  // but the command-based floors still raise it.
  1110→  1110→  const risk = riskMax(recomputePlanRisk(steps, compiler.toolRiskFloor()), editPlan.risk);
  1111→  1111→  const estimatedCredits = estimateCredits({ risk, steps: editPlan.steps });
  1112→  1112→
  1113→  1113→  // -- 7. dedupe advisory issues (unknown/param warns can double-report).
  1114→  1114→  const seen = new Set<string>();
  1115→  1115→  const warnings = compiler.warnings.filter((issue) => {
  1116→  1116→    const key = `${issue.code}|${issue.step ?? ""}|${issue.batchStep ?? ""}|${issue.command ?? ""}|${issue.param ?? ""}`;
  1117→  1117→    if (seen.has(key)) return false;
  1118→  1118→    seen.add(key);
  1119→  1119→    return true;
  1120→  1120→  });
  1121→  1121→
  1122→  1122→  const base: CompilePlanResult = {
  1123→  1123→    ok: true,
  1124→  1124→    batch,
  1125→  1125→    risk,
  1126→  1126→    estimatedCredits,
  1127→  1127→    resolutionRequests: requests,
  1128→  1128→    placeholders,
  1129→  1129→    warnings,
  1130→  1130→  };
  1131→  1131→  const annotation = compiler.annotation();
  1132→  1132→  return annotation !== undefined ? { ...base, saveVersion: annotation } : base;
  1133→  1133→}
  1134→  1134→
  1135→  1135→/**
  1136→  1136→ * Compile an EditPlan into ONE atomic AgentBatch (§8.5 bounds enforced), with
  1137→  1137→ * layer-name resolution requests, recomputed risk + credits, and typed issues.
  1138→  1138→ * Accepts a bare EditPlan or a SEAMS CompilePlanInput. NEVER throws (law #1).
  1139→  1139→ */
  1140→  1140→export function compilePlan(
  1141→  1141→  planOrInput: EditPlan | CompilePlanInput,
  1142→  1142→  deps: CompileDeps,
  1143→  1143→): CompilePlanResult {
  1144→  1144→  try {
  1145→  1145→    const plan = isCompileInput(planOrInput) ? planOrInput.plan : planOrInput;
  1146→  1146→    return compilePlanInner(plan, deps);
  1147→  1147→  } catch {
  1148→  1148→    // Absolute backstop: no stack traces, no details (law #1).
  1149→  1149→    return {
  1150→  1150→      ok: false,
  1151→  1151→      issues: [
  1152→  1152→        { severity: "fatal", code: "internal", message: "plan compilation failed unexpectedly" },
  1153→  1153→      ],
  1154→  1154→    };
  1155→  1155→  }
  1156→  1156→}
  1157→  1157→
  1158→  1158→/**
  1159→  1159→ * Live deps from the M5-A/M5-J contract surfaces, read at CALL time (ESM live
  1160→  1160→ * bindings) so peer stubs landing mid-wave are picked up. With today's stubs
  1161→  1161→ * the registry is empty — every compiled command fails the allowlist — wire
  1162→  1162→ * the real flow only after M5-J lands, or inject a registry fixture.
  1163→  1163→ */
  1164→  1164→export function defaultCompileDeps(): CompileDeps {
  1165→  1165→  return { registry: AGENT_COMMAND_REGISTRY, writeTools: WRITE_TOOL_ARGS_SCHEMAS };
  1166→  1166→}
