   151→/**
   152→ * One §7.4 event → one SSE frame, byte-shaped like the gateway's `sseFrame`
   153→ * (services/gateway/src/routes/chat.ts): `id: <seq>\ndata: <json>\n\n`.
   154→ * The client parses WHATWG frames and validates each payload against
   155→ * STUDIO_STREAM_EVENT_SCHEMA (packages/api-client/src/sse.ts) — so every
   156→ * fixture payload below must be schema-exact.
   157→ */
   158→function sseFrames(events: ReadonlyArray<Record<string, unknown>>): string {
   159→  return events
   160→    .map((event) => `id: ${String(event.seq)}\ndata: ${JSON.stringify(event)}\n\n`)
   161→    .join("");
   162→}
   163→
   164→/**
   165→ * The scripted-gateway leg: intercept the chat run POST
   166→ * (`POST /v1/threads/:id/chat`, same-origin through the dev edge) and
   167→ * fulfill it with a complete, deterministic SSE body. The REAL client
   168→ * reader/reducer/thread fold the frames exactly like a server stream; the
   169→ * real gateway (auth, threads, models, ledger) serves everything else.
   170→ */
   171→async function installScriptedRun(
   172→  page: Page,
   173→  events: ReadonlyArray<Record<string, unknown>>,
   174→): Promise<void> {
   175→  await page.route("**/v1/threads/*/chat", async (route) => {
   176→    await route.fulfill({
   177→      status: 200,
   178→      headers: {
   179→        "content-type": "text/event-stream; charset=utf-8",
   180→        "cache-control": "no-store",
   181→      },
   182→      body: sseFrames(events),
   183→    });
   184→  });
   185→}
