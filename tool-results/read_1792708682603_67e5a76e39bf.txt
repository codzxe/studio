   262→/**
   263→ * Runtime-narrowed subscription view of an event object. StripeEventLike is
   264→ * deliberately minimal (the seam's structural contract), so the period/price
   265→ * fields are validated HERE before use (law #2: webhook input is external).

   267→interface SubscriptionEventView {
   268→  readonly customer: string | null;
   269→  readonly status: string;
   270→  readonly trialEnd: number | null;
   271→  readonly cancelAtPeriodEnd: boolean;
   272→  readonly currentPeriodEnd: number | null;
   273→  readonly priceId: string | null;

   276→function priceIdOf(value: unknown): string | null {
   277→  if (typeof value === "string") return value;
   278→  if (isRecord(value)) return nonEmptyString(value.id);
   279→  return null;
   280→}
   281→
   282→function subscriptionViewOf(
   283→  object: StripeEventLike["data"]["object"],
   284→): SubscriptionEventView | null {
   285→  const record = object as unknown as Record<string, unknown>;
   286→  if (!isRecord(record)) return null;
   287→  const status = nonEmptyString(record.status);
   288→  if (status === null) return null;
   289→  let currentPeriodEnd: number | null = null;
   290→  let priceId: string | null = null;
   291→  const items = record.items;
   292→  if (isRecord(items) && Array.isArray(items.data) && items.data.length > 0) {
   293→    const item = items.data[0];
   294→    if (isRecord(item)) {
   295→      const periodEnd = item.current_period_end;
   296→      if (typeof periodEnd === "number" && Number.isFinite(periodEnd)) {
   297→        currentPeriodEnd = periodEnd;
   298→      }
   299→      priceId = priceIdOf(item.price);
   300→    }
   301→  }
   302→  const trialEnd = record.trial_end;
   303→  return {
   304→    customer: nonEmptyString(record.customer),
   305→    status,
   306→    trialEnd: typeof trialEnd === "number" && Number.isFinite(trialEnd) ? trialEnd : null,
   307→    cancelAtPeriodEnd: record.cancel_at_period_end === true,
   308→    currentPeriodEnd,
   309→    priceId,
   310→  };
   311→}

   339→/** Maps a typed seam failure ({ok:false, code, reason}) to the nearest wire
   340→ * code (law #1: the seam's reason is human-safe; it stays server-side in cause). */
   341→function seamFailure(
   342→  c: AppContext,
   343→  failure: { readonly code: string; readonly reason: string },
   344→): GatewayError {
   345→  const requestId = c.get("requestId");
   346→  const options = { requestId, cause: failure };
   347→  switch (failure.code) {
   348→    case "not_configured":
   349→      return new GatewayError(
   350→        "BILLING_NOT_CONFIGURED",
   351→        "Billing is not configured in this environment.",
   352→        { status: 400, ...options },
   353→      );
   354→    case "not_found":
   355→      return new GatewayError("NOT_FOUND", "That billing object was not found.", {
   356→        status: 404,
   357→        ...options,
   358→      });
   359→    case "invalid_request":
   360→      return new GatewayError("VALIDATION_FAILED", "The billing provider rejected that request.", {
   361→        status: 400,
   362→        ...options,
   363→      });
   364→    default:
   365→      return new GatewayError(
   366→        "SERVICE_UNAVAILABLE",
   367→        "The billing provider is unavailable right now.",
   368→        { status: 503, retryable: true, ...options },
   369→      );
   370→  }
   371→}

   377→export function billingRoutes(deps: GatewayDeps): Hono<AppBindings> {
   378→  const app = new Hono<AppBindings>();
   379→  app.use("/v1/billing/*", requireAuth());
   380→

   386→    subscriptions === null

   391→  function requireStripe(c: AppContext): NonNullable<GatewayDeps["stripe"]> {
   392→    if (deps.stripe === null || deps.env.stripe.secretKey === undefined) {
   393→      throw new StudioError("VALIDATION_FAILED", {
   394→        hint: "billing not configured in this environment",
   395→        requestId: c.get("requestId"),
   396→      });
   397→    }
   398→    return deps.stripe;
   399→  }
   400→
   401→  function requireBillingRole(c: AppContext): WorkspaceContext {
   402→    const workspace = c.get("workspace");
   403→    if (!can(workspace.role, "manageBilling")) {
   404→      throw new StudioError("FORBIDDEN", { requestId: c.get("requestId") });
   405→    }
   406→    return workspace;

   408→
   409→  function appBase(): string {
   410→    return deps.env.app.url.replace(/\/+$/, "");

   412→
   413→  /**
   414→   * Resolves the workspace's Stripe customer id (the durable repo row first,
   415→   * legacy org-metadata adoption, else creates the customer). The repo row is
   416→   * the source of truth for customer → workspace billing state.
   417→   */
   418→  async function ensureWorkspaceCustomer(
   419→    c: AppContext,
   420→    workspace: WorkspaceContext,
   421→    stripe: StripeSeam,
   422→  ): Promise<string> {
   423→    if (subscriptions !== null) {
   424→      const existing = await subscriptions.findStripeCustomer(workspace.id);
   425→      if (existing !== null && existing.stripeCustomerId !== "") {
   426→        return existing.stripeCustomerId;
   427→      }
   428→    }
   429→    const legacy = parseOrgMetadata(workspace.organization.metadata).stripeCustomerId;
   430→    if (typeof legacy === "string" && legacy !== "") {
   431→      if (subscriptions !== null) {
   432→        await subscriptions.ensureStripeCustomer(workspace.id, { stripeCustomerId: legacy });
   433→      }
   434→      return legacy;
   435→    }
   436→    const customer = await stripe.createCustomer({
   437→      email: c.get("auth").user.email,
   438→      metadata: { organizationId: workspace.id },
   439→    });
   440→    if (subscriptions !== null) {
   441→      await subscriptions.ensureStripeCustomer(workspace.id, { stripeCustomerId: customer.id });
   442→    }
   443→    return customer.id;
   444→  }

   446→  /**
   447→   * Price-id lookup through the M6-A helper (typed problem vocabulary).
   448→   * StudioEnv parses STRIPE_PRICE_IDS into a record at boot; the helper
   449→   * consumes the raw JSON value, so the parsed map is re-serialized here —
   450→   * values are env-validated strings, the round-trip is exact.

   452→  function planPriceId(planId: string): StripePriceIdLookup {

   456→  function requirePlanPriceId(c: AppContext, planId: string): string {
   457→    const lookup = planPriceId(planId);
   458→    if (lookup.ok) return lookup.priceId;
   459→    switch (lookup.problem.kind) {
   460→      case "plan-without-price":
   461→        throw new GatewayError(
   462→          "VALIDATION_FAILED",
   463→          "That plan has no price configured in this environment.",
   464→          {
   465→            status: 400,
   466→            hint: "The STRIPE_PRICE_IDS map has no price for that plan.",
   467→            requestId: c.get("requestId"),
   468→          },
   469→        );
   470→      case "unknown-plan":
   471→      case "plan-not-purchasable":
   472→        // Unreachable behind CHECKOUT_SCHEMA (D-4: trial is never a checkout
   473→        // target) — kept fail-closed anyway.
   474→        throw new GatewayError("VALIDATION_FAILED", "That plan is not available for checkout.", {
   475→          status: 400,
   476→          requestId: c.get("requestId"),
   477→        });
   478→      default:
   479→        throw new GatewayError(
   480→          "BILLING_NOT_CONFIGURED",
   481→          "Billing is not configured in this environment.",
   482→          { status: 400, requestId: c.get("requestId"), cause: lookup.problem },
   483→        );
   484→    }
   485→  }
   486→
   487→  // ── POST /v1/billing/checkout — plan subscription / plan change ──────────
   488→
   489→  app.post(
   490→    "/v1/billing/checkout",
   491→    requireWorkspace("admin"),
   492→    async (c: AppContext): Promise<Response> => {
   493→      const workspace = requireBillingRole(c);
   494→      const stripe = requireStripe(c);
   495→      const parsed = CHECKOUT_SCHEMA.safeParse(await c.req.json().catch(() => null));
   496→      if (!parsed.success) {
   497→        throw new StudioError("VALIDATION_FAILED", { requestId: c.get("requestId") });
   498→      }
   499→      const planId = parsed.data.planId;
   500→
   501→      // Server-side entitlement pre-checks (§M6 task 1): the target plan must
   502→      // fit the CURRENT workspace shape — checked here, never trusted to the
   503→      // client (the seats gate on invites is the sibling enforcement).
   504→      const entitlements = entitlementsFor(planId);
   505→      if (workspace.members.length > entitlements.seats) {
   506→        throw new GatewayError(
   507→          "VALIDATION_FAILED",
   508→          "This workspace has more members than that plan allows.",
   509→          {
   510→            status: 400,
   511→            hint: "Remove members before switching to a smaller plan.",
   512→            requestId: c.get("requestId"),
   513→          },
   514→        );
   515→      }
   516→
   517→      const priceId = requirePlanPriceId(c, planId);
   518→      const row =
   519→        subscriptions !== null ? await subscriptions.getByOrganization(workspace.id) : null;
   520→
   521→      if (row !== null && row.status !== "canceled") {
   522→        // A live subscription: Checkout sessions always create NEW
   523→        // subscriptions, so a plan change goes through subscriptions.update.
   524→        // DOWNGRADES ARE NEVER INSTANT (§M6 task 2): a smaller plan is
   525→        // scheduled at period end from the customer portal (Stripe's native
   526→        // semantics) — current entitlements stay until the period ends.
   527→        if (planRank(planId) < planRank(row.planId)) {
   528→          throw new GatewayError(
   529→            "VALIDATION_FAILED",
   530→            "Downgrades take effect at the end of the current billing period.",
   531→            {
   532→              status: 400,
   533→              hint: "Use the billing portal to schedule a downgrade; your current features stay until the period ends.",
   534→              requestId: c.get("requestId"),
   535→            },
   536→          );
   537→        }
   538→        const customerId = await ensureWorkspaceCustomer(c, workspace, stripe);
   539→        const updated = await stripe.updateSubscription(customerId, { planPriceKey: priceId });

   541→          throw seamFailure(c, updated);

   839→  /**
   840→   * Guarded transition wrapper (law #4: never double-apply). The repo's
   841→   * UPDATE re-checks `status IN from` — the racing/stale delivery loses with
   842→   * a typed CONFLICT, reported here as "guarded", never an error: a stale
   843→   * webhook is still a DELIVERED webhook (Stripe must see 2xx, not a retry loop).
   844→   */
   845→  async function guardedTransition(
   846→    organizationId: string,
   847→    input: SubscriptionTransitionInput,
   848→  ): Promise<boolean> {
   849→    if (subscriptions === null) return false;
   850→    try {
   851→      await subscriptions.transitionStatus(organizationId, input);
   852→      return true;
   853→    } catch (error) {
   854→      if (isDbError(error) && (error.code === "CONFLICT" || error.code === "NOT_FOUND")) {
   855→        return false;
   856→      }
   857→      throw error;
   858→    }
   859→  }

   861→  /** Ledger grant for a top-up from external µc metadata (D-12 + int32 ceiling). */
   862→  async function grantTopUp(
   863→    organizationId: string,
   864→    event: StripeEventLike,
   865→    metadata: Record<string, string> | null | undefined,
   866→    idempotencyKey: string,
   867→  ): Promise<boolean> {
   868→    const raw = metadata?.microcredits;
   869→    if (typeof raw !== "string" || raw === "") return false;
   870→    const microcredits = Number.parseInt(raw, 10);
   871→    if (!Number.isInteger(microcredits) || microcredits <= 0 || microcredits > MAX_LEDGER_AMOUNT) {
   872→      // Hostile/overflowing amounts are audited-ignored — never a poison retry.
   873→      return false;
   874→    }
   875→    await deps.repos.ledger.grant(organizationId, {
   876→      amount: microcredits,
   877→      idempotencyKey,
   878→      metadata: {
   879→        stripeEventId: event.id,
   880→        kind: "topup",
   881→        packId: metadata?.packId ?? null,
   882→      },
   883→    });
   884→    return true;
   885→  }

   918→  async function handleCheckoutCompleted(
   919→    stripe: StripeSeam,
   920→    event: StripeEventLike,
   921→  ): Promise<WebhookOutcome> {
   922→    const object = event.data.object;
   923→    const organizationId = organizationIdOf(object);
   924→    if (organizationId === null) return { ignored: "unattributed" };
   925→
   926→    // Customer + payment-method bookkeeping (§M6 task 5): the repo is the
   927→    // durable customer mapping; cardFingerprint metadata is the fake's card
   928→    // model (real Stripe attaches the fingerprint with the payment method).
   929→    await recordCustomer(organizationId, object);
   930→
   931→    const kind = object.metadata?.kind ?? null;
   932→    if (kind === "topup") {
   933→      const granted = await grantTopUp(
   934→        organizationId,
   935→        event,
   936→        object.metadata,
   937→        `stripe:${event.id}`,
   938→      );
   939→      return granted ? { handled: "topup.granted" } : { ignored: "topup.amount-invalid" };
   940→    }
   941→    if (kind !== "subscription") {
   942→      return { ignored: "unknown-kind" };
   943→    }
   944→
   945→    // The session object carries no period data — resync the canonical
   946→    // snapshot from the provider (D-4: the trial status rides on the sub).
   947→    const customer = nonEmptyString(object.customer);
   948→    if (customer === null) return { ignored: "no-customer" };
   949→    const snapshot = await stripe.subscription(customer);
   950→    if (!snapshot.ok) return { ignored: "subscription-unavailable" };
   951→    const planId = planIdFromMetadata(object) ?? planIdFromPrice(deps.env, snapshot.data.priceId);
   952→    if (planId === null) return { ignored: "unknown-plan" };
   953→    if (subscriptions !== null) {
   954→      // First row (or resubscription): upsert is race-proof on the unique
   955→      // organization_id — concurrent completions converge on one row.
   956→      await subscriptions.upsertState(organizationId, {
   957→        planId,
   958→        status: mapStripeStatus(snapshot.data.status),
   959→        stripeSubscriptionId: snapshot.data.subscriptionId,
   960→        currentPeriodEnd: toDate(snapshot.data.currentPeriodEnd),
   961→        trialEnd: toDate(snapshot.data.trialEnd ?? null),
   962→        cancelAtPeriodEnd: snapshot.data.cancelAtPeriodEnd,
   963→      });
   964→    }
   965→    return { handled: "subscription.synced" };
   966→  }

   968→  /**
   969→   * Shared guarded-transition driver for subscription lifecycle events
   970→   * (customer.subscription.updated/deleted). The event's object is the
   971→   * HISTORICAL truth (webhooks replay history); minimal objects resync from
   972→   * the provider.
   973→   */
   974→  async function applySubscriptionTransition(
   975→    stripe: StripeSeam,
   976→    event: StripeEventLike,
   977→    forcedTarget?: SubscriptionStatus,
   978→  ): Promise<WebhookOutcome> {
   979→    const object = event.data.object;
   980→    const organizationId = organizationIdOf(object);
   981→    if (organizationId === null) return { ignored: "unattributed" };
   982→    if (subscriptions === null) return { ignored: "no-persistence" };
   983→
   984→    let view = subscriptionViewOf(object);
   985→    if (view === null || view.currentPeriodEnd === null) {
   986→      const customer = view?.customer ?? nonEmptyString(object.customer);
   987→      if (customer !== null) {
   988→        view = await resyncView(stripe, customer);
   989→      }
   990→    }
   991→    if (view === null) return { ignored: "unparsable" };
   992→    const target = forcedTarget ?? mapStripeStatus(view.status);
   993→
   994→    const row = await subscriptions.getByOrganization(organizationId);
   995→    if (row === null) {
   996→      // A subscription event for a workspace we never tracked: attribution
   997→      // exists (metadata) but there is no state to transition — audited.
   998→      return { ignored: "no-subscription" };
   999→    }
  1000→
  1001→    // From-lists (the guarded state machine; heals land via invoice.paid):
  1002→    //  active   ← trialing | active | incomplete   (trial conversion, renewal)
  1003→    //  grace    ← active | grace | past_due | incomplete (past-due dunning)
  1004→    //  canceled ← any live state (BP §M6.2: status only — NO work deletion)
  1005→    //  trialing ← trialing | incomplete (idempotent re-pin)
  1006→    //  incomplete ← trialing | active | incomplete
  1007→    //  past_due ← grace (legacy rows only; Stripe past_due maps to grace)
  1008→    const FROM_LISTS: Record<SubscriptionStatus, readonly SubscriptionStatus[]> = {
  1009→      active: ["trialing", "active", "incomplete"],
  1010→      grace: ["active", "grace", "past_due", "incomplete"],
  1011→      canceled: ["trialing", "active", "grace", "past_due", "incomplete"],
  1012→      trialing: ["trialing", "incomplete"],
  1013→      incomplete: ["trialing", "active", "incomplete"],
  1014→      past_due: ["grace"],
  1015→    };
  1016→    const from = FROM_LISTS[target];
  1017→    if (!from.includes(row.status)) {
  1018→      return { ignored: "transition-guarded" };
  1019→    }
  1020→
  1021→    const periodEndMs = view.currentPeriodEnd !== null ? view.currentPeriodEnd * 1000 : null;
  1022→    // Never move the period BACKWARDS (an out-of-order renewal must not
  1023→    // regress the row), and never write a null over a known period.
  1024→    const periodAdvances =
  1025→      periodEndMs !== null &&
  1026→      (row.currentPeriodEnd === null || periodEndMs >= row.currentPeriodEnd.getTime());
  1027→
  1028→    // Plan pinning (§M6 task 2 "never instant feature loss"): the event's new
  1029→    // price only becomes the row's plan when it is an UPGRADE (features grow)
  1030→    // or the period actually advanced (the change took effect). A downgrade
  1031→    // scheduled at period end keeps the row's plan until then.
  1032→    const targetPlan = planIdFromPrice(deps.env, view.priceId) ?? planIdFromMetadata(object);
  1033→    const planId =
  1034→      targetPlan !== null &&
  1035→      target !== "canceled" &&
  1036→      (planRank(targetPlan) > planRank(row.planId) || periodAdvances)
  1037→        ? targetPlan
  1038→        : null;
  1039→
  1040→    const graceEnd =
  1041→      target === "grace"
  1042→        ? periodEndMs !== null
  1043→          ? new Date(periodEndMs + GRACE_PERIOD_DAYS * DAY_MS)
  1044→          : undefined
  1045→        : target === "active" || target === "canceled"
  1046→          ? null
  1047→          : undefined;
  1048→    const applied = await guardedTransition(organizationId, {
  1049→      from,
  1050→      to: target,
  1051→      ...(planId !== null ? { planId } : {}),
  1052→      ...(periodAdvances ? { currentPeriodEnd: new Date(periodEndMs) } : {}),
  1053→      ...(view.trialEnd !== null ? { trialEnd: new Date(view.trialEnd * 1000) } : {}),
  1054→      ...(graceEnd !== undefined ? { graceEnd } : {}),
  1055→      cancelAtPeriodEnd: view.cancelAtPeriodEnd,
  1056→    });
  1057→    return applied ? { handled: `subscription.${target}` } : { ignored: "transition-guarded" };
  1058→  }

  1067→  async function handleSubscriptionDeleted(
  1068→    stripe: StripeSeam,

  1076→  async function handleInvoice(event: StripeEventLike): Promise<WebhookOutcome> {
  1077→    const object = event.data.object;
  1078→    const organizationId = organizationIdOf(object);
  1079→    if (organizationId === null) return { ignored: "unattributed" };
  1080→    const invoices = invoicesRepo(deps.repos);
  1081→    if (invoices === null) return { ignored: "no-persistence" };
  1082→
  1083→    const record = object as unknown as Record<string, unknown>;
  1084→    const invoiceId = nonEmptyString(record.id);
  1085→    if (invoiceId === null) return { ignored: "invoice-unparsable" };
  1086→    const amountRaw = record.amount_paid ?? record.amount_due;
  1087→    const amountCents =
  1088→      typeof amountRaw === "number" && Number.isFinite(amountRaw) && amountRaw >= 0
  1089→        ? Math.round(amountRaw)
  1090→        : 0;
  1091→    const currencyRaw = nonEmptyString(record.currency);
  1092→    const periodStart = toDate(
  1093→      typeof record.period_start === "number" ? record.period_start : null,
  1094→    );
  1095→    const periodEnd = toDate(typeof record.period_end === "number" ? record.period_end : null);
  1096→
  1097→    const status = event.type === "invoice.paid" ? ("paid" as const) : ("void" as const);
  1098→    try {
  1099→      await invoices.record(organizationId, {
  1100→        stripeInvoiceId: invoiceId,
  1101→        amountCents,
  1102→        currency: currencyRaw ?? "usd",
  1103→        status,
  1104→        periodStart,
  1105→        periodEnd,
  1106→        hostedUrl: nonEmptyString(record.hosted_invoice_url),
  1107→        pdfUrl: nonEmptyString(record.invoice_pdf),
  1108→      });
  1109→    } catch (error) {
  1110→      if (isDbError(error) && (error.code === "VALIDATION_FAILED" || error.code === "CONFLICT")) {
  1111→        // A foreign/malformed invoice is audited-ignored — never a retry loop.
  1112→        return { ignored: "invoice-invalid" };
  1113→      }
  1114→      throw error;
  1115→    }
  1116→
  1117→    if (status !== "paid") {
  1118→      return { handled: "invoice.voided" };
  1119→    }
  1120→
  1121→    // Top-up grants invoiced outside Checkout (§M6 task 2): keyed on the
  1122→    // INVOICE id, so re-deliveries under fresh event ids cannot double-grant.
  1123→    let handled = "invoice.paid";
  1124→    if (object.metadata?.kind === "topup") {
  1125→      const granted = await grantTopUp(
  1126→        organizationId,
  1127→        event,
  1128→        object.metadata,
  1129→        `stripe:invoice:${invoiceId}:topup`,
  1130→      );
  1131→      if (granted) handled = "invoice.topup-granted";
  1132→    }
  1133→
  1134→    // Dunning heal (§M6 task 2 trial/renewal lifecycle): a PAID invoice while
  1135→    // the row sits in the grace window returns the subscription to active.
  1136→    if (subscriptions !== null) {
  1137→      const row = await subscriptions.getByOrganization(organizationId);
  1138→      if (row !== null && (row.status === "grace" || row.status === "past_due")) {
  1139→        const healed = await guardedTransition(organizationId, {
  1140→          from: ["grace", "past_due"],
  1141→          to: "active",
  1142→          ...(periodEnd !== null ? { currentPeriodEnd: periodEnd } : {}),
  1143→          graceEnd: null,
  1144→        });
  1145→        if (healed) handled = "invoice.paid-healed";
  1146→      }
  1147→    }
  1148→    return { handled };
  1149→  }
  1150→
  1151→  async function handleStripeEvent(
  1152→    stripe: StripeSeam,
  1153→    event: StripeEventLike,
  1154→  ): Promise<WebhookOutcome> {
  1155→    switch (event.type) {
  1156→      case "checkout.session.completed":
  1157→        return handleCheckoutCompleted(stripe, event);
  1158→      case "customer.subscription.updated":
  1159→        return handleSubscriptionUpdated(stripe, event);
  1160→      case "customer.subscription.deleted":
  1161→        return handleSubscriptionDeleted(stripe, event);
  1162→      case "invoice.paid":
  1163→      case "invoice.voided":
  1164→        return handleInvoice(event);
  1165→      default:
  1166→        // Unknown event types are 200 + audited-ignored: Stripe retries on
  1167→        // non-2xx, so an unhandled type must NEVER be a 500.
  1168→        return { ignored: "unhandled-type" };
  1169→    }
  1170→  }
  1171→
  1172→  return app;
  1173→}
