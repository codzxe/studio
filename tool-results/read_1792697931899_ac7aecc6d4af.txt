   104→function sseResponse(frames: readonly string[]): Response {
   105→  const body = new ReadableStream<Uint8Array>({
   106→    start(controller) {
   107→      for (const chunk of frames) controller.enqueue(encoder.encode(chunk));
   108→      controller.close();
   109→    },
   110→  });
   111→  return new Response(body, { status: 200, headers: { "content-type": "text/event-stream" } });
   112→}

   124→function scriptedFetch(script: ReadonlyArray<Response>): typeof fetch & { calls: RecordedCall[] } {
   125→  const calls: RecordedCall[] = [];
   126→  let index = 0;
   127→  const stub = (async (url: unknown, init?: RequestInit): Promise<Response> => {
   128→    calls.push({
   129→      url: String(url),
   130→      body: JSON.parse(String(init?.body)) as RecordedCall["body"],
   131→    });
   132→    const next = script[index];
   133→    index += 1;
   134→    if (next === undefined) throw new Error("script exhausted");
   135→    return next;
   136→  }) as typeof fetch & { calls: RecordedCall[] };
   137→  stub.calls = calls;
   138→  return stub;
   139→}

   145→class FakeStream implements AgentRunStream {
   146→  readonly events: RunStreamPayload[] = [];
   147→  readonly expected: Array<{ toolCallId: string; tool: string }> = [];
   148→  readonly controller = new AbortController();
   149→  waitMode: "pending" | "auto" = "pending";
   150→  private readonly waits = new Map<string, (result: ToolWaitResult) => void>();
   151→  private readonly early = new Map<string, unknown>();
   152→
   153→  get signal(): AbortSignal {
   154→    return this.controller.signal;
   155→  }
   156→
   157→  emit(payload: RunStreamPayload): StudioStreamEvent {
   158→    this.events.push(payload);
   159→    return { ...payload, runId: "run_inj", seq: this.events.length - 1 };
   160→  }
   161→
   162→  async yieldToStream(): Promise<void> {
   163→    await new Promise((resolve) => setImmediate(resolve));
   164→  }
   165→
   166→  expectToolCall(toolCallId: string, tool: string): void {
   167→    this.expected.push({ toolCallId, tool });
   168→  }
   169→
   170→  waitForToolResult(toolCallId: string): Promise<ToolWaitResult> {
   171→    if (this.waitMode === "auto") return Promise.resolve({ ok: true, result: { stub: true } });
   172→    if (this.early.has(toolCallId)) {
   173→      return Promise.resolve({ ok: true, result: this.early.get(toolCallId) });
   174→    }
   175→    return new Promise((resolve) => {
   176→      this.waits.set(toolCallId, resolve);
   177→    });
   178→  }
   179→
   180→  resolveToolResult(toolCallId: string, result: unknown): void {
   181→    const resolve = this.waits.get(toolCallId);
   182→    if (resolve === undefined) {
   183→      this.early.set(toolCallId, result);
   184→      return;
   185→    }
   186→    this.waits.delete(toolCallId);
   187→    resolve({ ok: true, result });
   188→  }
   189→}

   191→async function runToOutcome(
   192→  runner: ReturnType<typeof createLiteLLMRunner>,
   193→  run: AgentRun,
   194→  stream: FakeStream,
   195→): Promise<AgentRunOutcome> {
   196→  const outcome = await runner.start(run, stream);
   197→  if (outcome.kind === "failed") {
   198→    stream.emit({
   199→      type: "run.failed",
   200→      code: outcome.code,
   201→      message: outcome.message,
   202→      retryable: outcome.retryable,
   203→    });
   204→  } else if (outcome.kind === "cancelled") {
   205→    stream.emit({ type: "run.cancelled" });
   206→  } else {
   207→    stream.emit({ type: "run.completed" });
   208→  }
   209→  return outcome;
   210→}
