   115→class FakeOpenAIServer {
   116→  private readonly queue: ScriptEntry[] = [];
   117→  private server: Server | null = null;
   118→  url = "";
   119→  /** Every /v1/chat/completions request the runner made, in order. */
   120→  readonly calls: Array<{ body: ChatCompletionBody; authorization: string }> = [];
   121→
   122→  enqueue(...entries: ScriptEntry[]): void {
   123→    this.queue.push(...entries);
   124→  }
   125→
   126→  reset(): void {
   127→    this.queue.length = 0;
   128→    this.calls.length = 0;
   129→  }
   130→
   131→  async start(): Promise<void> {
   132→    const server = createServer((request, response) => {
   133→      void this.handle(request, response);
   134→    });
   135→    this.server = server;
   136→    await new Promise<void>((resolve) => {
   137→      server.listen(0, "127.0.0.1", resolve);
   138→    });
   139→    const address = server.address() as AddressInfo;
   140→    this.url = `http://127.0.0.1:${address.port}`;
   141→  }
   142→
   143→  async stop(): Promise<void> {
   144→    await new Promise<void>((resolve, reject) => {
   145→      const server = this.server;
   146→      if (server === null) {
   147→        resolve();
   148→        return;
   149→      }
   150→      server.close((error) => {
   151→        if (error !== undefined) reject(error);
   152→        else resolve();
   153→      });
   154→    });
   155→  }
   156→
   157→  private async handle(request: IncomingMessage, response: ServerResponse): Promise<void> {
   158→    // Swallow socket errors from aborted runs — the runner's fetch is torn
   159→    // down whenever a run cancels mid-round; the fake server must not crash.
   160→    response.on("error", () => undefined);
   161→    request.on("error", () => undefined);
   162→    const chunks: Buffer[] = [];
   163→    for await (const chunk of request) {
   164→      chunks.push(chunk as Buffer);
   165→      if (response.destroyed) return;
   166→    }
   167→    let body: ChatCompletionBody = { model: "", messages: [] };
   168→    try {
   169→      body = JSON.parse(Buffer.concat(chunks).toString("utf8")) as ChatCompletionBody;
   170→    } catch {
   171→      body = { model: "", messages: [] };
   172→    }
   173→    const authorization = String(request.headers.authorization ?? "");
   174→    this.calls.push({ body, authorization });
   175→
   176→    const next = this.queue.shift();
   177→    if (next === undefined) {
   178→      response.writeHead(500, { "content-type": "application/json" });
   179→      response.end(JSON.stringify({ error: { message: "fake script exhausted" } }));
   180→      return;
   181→    }
   182→    const frames = next(body);
   183→    response.writeHead(200, {
   184→      "content-type": "text/event-stream; charset=utf-8",
   185→      "cache-control": "no-store",
   186→    });
   187→    for (const wireFrame of frames) {
   188→      if (response.destroyed) return;
   189→      response.write(encoder.encode(wireFrame));
   190→      await new Promise<void>((resolveTick) => {
   191→        setImmediate(resolveTick);
   192→      });
   193→    }
   194→    response.end();
   195→  }
   196→}

   255→interface WireFrame {
   256→  /** The exact bytes between \n\n frame boundaries (no trailing \n\n). */
   257→  readonly raw: string;
   258→  /** Parsed `id:` line — the server seq. */
   259→  readonly id: number;
   260→  /** Parsed `data:` JSON. */
   261→  readonly event: StudioStreamEvent;
   262→}
   263→
   264→function parseFrame(raw: string): WireFrame {
   265→  const lines = raw.split("\n");
   266→  const idLine = lines[0];
   267→  const dataLine = lines[1];
   268→  if (
   269→    lines.length !== 2 ||
   270→    idLine === undefined ||
   271→    dataLine === undefined ||
   272→    !idLine.startsWith("id: ") ||
   273→    !dataLine.startsWith("data: ")
   274→  ) {
   275→    throw new Error(`malformed SSE frame on the wire: ${JSON.stringify(raw)}`);
   276→  }
   277→  const id = Number.parseInt(idLine.slice(4), 10);
   278→  const event = JSON.parse(dataLine.slice("data: ".length)) as StudioStreamEvent;
   279→  return { raw, id, event };
   280→}

   282→class SseStream {
   283→  private readonly reader: ReadableStreamDefaultReader<Uint8Array>;
   284→  private readonly decoder = new TextDecoder();
   285→  private buffer = "";
   286→  private eof = false;
   287→
   288→  constructor(response: Response) {
   289→    const body = response.body;
   290→    if (body === null) throw new Error("chat response had no SSE body");
   291→    this.reader = body.getReader();
   292→  }
   293→
   294→  /** Next complete frame; null at clean EOF. Throws on stall (test bug guard). */
   295→  async next(timeoutMs = 5000): Promise<WireFrame | null> {
   296→    if (this.eof) return null;
   297→    const deadline = Date.now() + timeoutMs;
   298→    for (;;) {
   299→      const boundary = this.buffer.indexOf("\n\n");
   300→      if (boundary !== -1) {
   301→        const raw = this.buffer.slice(0, boundary);
   302→        this.buffer = this.buffer.slice(boundary + 2);
   303→        if (raw.trim() === "") continue;
   304→        return parseFrame(raw);
   305→      }
   306→      const remaining = deadline - Date.now();
   307→      if (remaining <= 0) {
   308→        await this.cancel();
   309→        throw new Error("timed out waiting for the next SSE frame");
   310→      }
   311→      const read = await Promise.race([
   312→        this.reader.read(),
   313→        new Promise<never>((_resolve, reject) => {
   314→          const timer = setTimeout(() => {
   315→            reject(new Error("timed out waiting for the next SSE frame"));
   316→          }, remaining);
   317→          timer.unref?.();
   318→        }),
   319→      ]);
   320→      if (read.done) {
   321→        this.eof = true;
   322→        const tail = this.buffer;
   323→        this.buffer = "";
   324→        if (tail.trim() === "") return null;
   325→        return parseFrame(tail);
   326→      }
   327→      this.buffer += this.decoder.decode(read.value, { stream: true });
   328→    }
   329→  }
   330→
   331→  async cancel(): Promise<void> {
   332→    this.eof = true;
   333→    await this.reader.cancel().catch(() => undefined);
   334→  }
   335→
   336→  /** Reads frames until predicate(event) is true (inclusive) and returns them. */
   337→  async until(
   338→    predicate: (event: StudioStreamEvent) => boolean,
   339→    description: string,
   340→    timeoutMs = 5000,
   341→  ): Promise<WireFrame[]> {
   342→    const frames: WireFrame[] = [];
   343→    for (;;) {
   344→      const wireFrame = await this.next(timeoutMs);
   345→      if (wireFrame === null) {
   346→        throw new Error(`SSE stream ended before ${description}`);
   347→      }
   348→      frames.push(wireFrame);
   349→      if (predicate(wireFrame.event)) return frames;
   350→    }
   351→  }
   352→
   353→  /** All remaining frames through clean EOF. */
   354→  async rest(timeoutMs = 5000): Promise<WireFrame[]> {
   355→    const frames: WireFrame[] = [];
   356→    for (;;) {
   357→      const wireFrame = await this.next(timeoutMs);
   358→      if (wireFrame === null) return frames;
   359→      frames.push(wireFrame);
   360→    }
   361→  }
   362→}

   386→/**
   387→ * §7.4 wire contract over a collected frame list: exact `id:`/`data:` frame
   388→ * format, contiguous monotonic seq from `firstSeq`, schema-valid events, one
   389→ * runId, and (optionally) exactly one terminal event as the last frame.
   390→ */
   391→function expectValidStream(
   392→  frames: readonly WireFrame[],
   393→  options: { readonly requireTerminal?: boolean; readonly firstSeq?: number } = {},
   394→): StudioStreamEvent[] {
   395→  const requireTerminal = options.requireTerminal ?? true;
   396→  const firstSeq = options.firstSeq ?? 0;
   397→  expect(frames.length).toBeGreaterThan(0);
   398→  let previousSeq: number | null = null;
   399→  const events: StudioStreamEvent[] = [];
   400→  let terminals = 0;
   401→  for (const wireFrame of frames) {
   402→    // Exact wire format: `id: <seq>\ndata: <json>` — byte-identical modulo
   403→    // JSON key order, which JSON.parse/stringify round-trips losslessly here.
   404→    expect(wireFrame.raw).toBe(
   405→      `id: ${wireFrame.event.seq}\ndata: ${JSON.stringify(wireFrame.event)}`,
   406→    );
   407→    expect(wireFrame.id).toBe(wireFrame.event.seq);
   408→    if (previousSeq === null) expect(wireFrame.event.seq).toBe(firstSeq);
   409→    else expect(wireFrame.event.seq).toBe(previousSeq + 1);
   410→    previousSeq = wireFrame.event.seq;
   411→    const checked = STUDIO_STREAM_EVENT_SCHEMA.safeParse(wireFrame.event);
   412→    if (!checked.success) {
   413→      throw new Error(
   414→        `event seq ${wireFrame.event.seq} failed the §7.4 schema: ${checked.error.message}`,
   415→      );
   416→    }
   417→    events.push(checked.data);
   418→    if (TERMINAL_TYPES.has(checked.data.type)) terminals += 1;
   419→  }
   420→  if (firstSeq === 0) expect(events[0]?.type).toBe("run.started");
   421→  if (requireTerminal) {
   422→    expect(terminals).toBe(1);
   423→    const last = frames.at(-1)?.event;
   424→    if (last === undefined || !TERMINAL_TYPES.has(last.type)) {
   425→      throw new Error("the last frame on the wire must be the terminal event");
   426→    }
   427→  } else {
   428→    expect(terminals).toBe(0);
   429→  }
   430→  // One run per stream: every event carries the same runId.
   431→  const runIds = new Set(events.map((event) => event.runId));
   432→  expect(runIds.size).toBe(1);
   433→  return events;
   434→}

   572→async function chatStream(
   573→  fx: Fixture,
   574→  body: ChatBody,
   575→  options: ChatCallOptions = {},
   576→): Promise<Response> {
   577→  const headers: Record<string, string> = {
   578→    origin: TEST_APP_ORIGIN,
   579→    "content-type": "application/json",
   580→    accept: "text/event-stream",
   581→    cookie: fx.jar.header,
   582→    "x-workspace-id": fx.workspaceId,
   583→  };
   584→  if (options.lastEventId !== undefined) headers["last-event-id"] = options.lastEventId;
   585→  return fetch(`${server.base}/v1/threads/${fx.threadId}/chat`, {
   586→    method: "POST",
   587→    headers,
   588→    body: JSON.stringify(body),
   589→    signal: options.signal,
   590→    redirect: "manual",
   591→  });
   592→}

   646→/** Poll until probe() is true (the route emits run.completed BEFORE its DB writes). */
   647→async function waitFor(
   648→  description: string,
   649→  probe: () => Promise<boolean>,
   650→  timeoutMs = 4000,
   651→): Promise<void> {
   652→  const deadline = Date.now() + timeoutMs;
   653→  for (;;) {
   654→    if (await probe()) return;
   655→    if (Date.now() > deadline) throw new Error(`timed out waiting for ${description}`);
   656→    await new Promise((resolve) => {
   657→      setTimeout(resolve, 25);
   658→    });
   659→  }
   660→}
