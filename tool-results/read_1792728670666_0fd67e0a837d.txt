    96→    96→/** The pieces of a composition the estimator needs (spike §4/§7 shapes). */
    97→    97→interface CompMeta {
    98→    98→  readonly durationSeconds: number;
    99→    99→  readonly width: number;
   100→   100→  readonly height: number;
   101→   101→  readonly frameRate: number;
   102→   102→}
   103→   103→
   104→   104→function asRecord(value: unknown): Record<string, unknown> | null {
   105→   105→  return typeof value === "object" && value !== null && !Array.isArray(value)
   106→   106→    ? (value as Record<string, unknown>)
   107→   107→    : null;
   108→   108→}
   109→   109→
   110→   110→function positiveInt(value: unknown): number | null {
   111→   111→  return typeof value === "number" && Number.isInteger(value) && value > 0 ? value : null;
   112→   112→}
   113→   113→
   114→   114→function positiveNumber(value: unknown): number | null {
   115→   115→  return typeof value === "number" && Number.isFinite(value) && value > 0 ? value : null;
   116→   116→}
   117→   117→
   118→   118→/** The real editor wire shape: items{"<id>": {id, name, kind:{type:"Comp",…}}}. */
   119→   119→function compFromItems(items: unknown, compId: string): CompMeta | null {
   120→   120→  const map = asRecord(items);
   121→   121→  if (map === null) return null;
   122→   122→  for (const raw of Object.values(map)) {
   123→   123→    const item = asRecord(raw);
   124→   124→    if (item === null) continue;
   125→   125→    // The contract compId is the editor comp id OR its display name — match
   126→   126→    // either (tolerant by design; ids serialize as numbers in .ecproj).
   127→   127→    const idMatches = item.id !== undefined && String(item.id) === compId;
   128→   128→    const nameMatches = typeof item.name === "string" && item.name === compId;
   129→   129→    if (!idMatches && !nameMatches) continue;
   130→   130→    const kind = asRecord(item.kind);
   131→   131→    if (kind === null) continue;
   132→   132→    // serde tag is the VARIANT name ("Comp"); "Composition" is accepted for
   133→   133→    // tolerance (it is the editor's DISPLAY name for the same kind).
   134→   134→    if (kind.type !== "Comp" && kind.type !== "Composition") continue;
   135→   135→    const width = positiveInt(kind.width);
   136→   136→    const height = positiveInt(kind.height);
   137→   137→    const duration = positiveNumber(kind.duration);
   138→   138→    const rate = asRecord(kind.frame_rate) ?? asRecord(kind.frameRate);
   139→   139→    const fps =
   140→   140→      rate !== null && positiveNumber(rate.num) !== null
   141→   141→        ? Number(rate.num) / (positiveNumber(rate.den) ?? 1)
   142→   142→        : positiveNumber(kind.frameRate);
   143→   143→    if (width === null || height === null || duration === null || fps === null) continue;
   144→   144→    return { durationSeconds: duration / EC_TICKS_PER_SECOND, width, height, frameRate: fps };
   145→   145→  }
   146→   146→  return null;
   147→   147→}
   148→   148→
   149→   149→/**
   150→   150→ * Tolerance fallback for the synthetic/legacy `compositions: []` shape used by
   151→   151→ * older fixtures: durations/fps in SECONDS/Hz, ids/names as strings.
   152→   152→ */
   153→   153→function compFromCompositions(root: Record<string, unknown>, compId: string): CompMeta | null {
   154→   154→  if (!Array.isArray(root.compositions)) return null;
   155→   155→  for (const raw of root.compositions) {
   156→   156→    const comp = asRecord(raw);
   157→   157→    if (comp === null) continue;
   158→   158→    const idMatches = comp.id !== undefined && String(comp.id) === compId;
   159→   159→    const nameMatches = typeof comp.name === "string" && comp.name === compId;
   160→   160→    if (!idMatches && !nameMatches) continue;
   161→   161→    const width = positiveInt(comp.width);
   162→   162→    const height = positiveInt(comp.height);
   163→   163→    const duration = positiveNumber(comp.durationSeconds) ?? positiveNumber(comp.duration);
   164→   164→    const fps = positiveNumber(comp.frameRate);
   165→   165→    if (width === null || height === null || duration === null || fps === null) continue;
   166→   166→    return { durationSeconds: duration, width, height, frameRate: fps };
   167→   167→  }
   168→   168→  return null;
   169→   169→}
   170→   170→
   171→   171→/** Locate the requested comp inside a parsed .ecproj, or null. */
   172→   172→function findComp(ecproj: unknown, compId: string): CompMeta | null {
   173→   173→  const root = asRecord(ecproj);
   174→   174→  if (root === null) return null;
   175→   175→  return compFromItems(root.items, compId) ?? compFromCompositions(root, compId);
   176→   176→}

   182→   182→/** Normalized view of a render request the queue and the estimate share. */
   183→   183→interface RenderQuote {
   184→   184→  readonly project: Project;
   185→   185→  /** The pinned version — ALWAYS the project head, never client-supplied. */
   186→   186→  readonly versionId: string;
   187→   187→  readonly outputSeconds: number;
   188→   188→  readonly longEdgePx: number;
   189→   189→  readonly quality: RenderQualityId;
   190→   190→  /**
   191→   191→   * The settings stored on the row: the client's settings plus the server's
   192→   192→   * materialized effective frameRate (settings.frameRate ?? the comp's —
   193→   193→   * renderSecondsUsed reads it for the monthly-minutes accounting) and the
   194→   194→   * range clamped to the comp timeline (price == stored == rendered).
   195→   195→   */
   196→   196→  readonly storedSettings: Record<string, unknown>;
   197→   197→}
   198→   198→
   199→   199→/**
   200→   200→ * Resolve a queue/estimate request against the project's PINNED version: load
   201→   201→ * the project (tenant-scoped — cross-tenant ids are 404s), read the head
   202→   202→ * version's .ecproj bytes from storage, and derive duration/size/fps/quality
   203→   203→ * from the actual comp metadata. Shared by BOTH routes so the estimate and
   204→   204→ * the reserve can never disagree.
   205→   205→ */
   206→   206→async function resolveRenderQuote(
   207→   207→  deps: GatewayDeps,
   208→   208→  workspaceId: string,
   209→   209→  requestId: string,
   210→   210→  input: { readonly projectId: string; readonly compId: string },
   211→   211→  settings: {
   212→   212→    readonly timeFrom?: number;
   213→   213→    readonly timeTo?: number;
   214→   214→    readonly width?: number;
   215→   215→    readonly height?: number;
   216→   216→    readonly frameRate?: number;
   217→   217→    readonly quality?: RenderQualityId;
   218→   218→  },
   219→   219→): Promise<RenderQuote> {
   220→   220→  const project = await deps.repos.projects.get(workspaceId, input.projectId);
   221→   221→  if (project.currentVersionId === null) {
   222→   222→    // A project with no versions has nothing renderable (§9.3).
   223→   223→    throw new StudioError("RENDER_PROJECT_INVALID", {
   224→   224→      message: "This project can't be rendered as-is.",
   225→   225→      hint: "Save a version of the project first, then render.",
   226→   226→      requestId,
   227→   227→    });
   228→   228→  }
   229→   229→  const versionId = project.currentVersionId; // SERVER-pinned, §9.3
   230→   230→  const version = await deps.repos.versions.get(workspaceId, project.id, versionId);
   231→   231→
   232→   232→  const bytes = await deps.storage.read(version.objectKey);
   233→   233→  if (bytes === null) {
   234→   234→    // The version bytes are missing from storage — an infrastructure fault,
   235→   235→    // never the client's project. (LocalFS serves reads; the S3 driver's
   236→   236→    // read() is a documented null — S3 estimate/queue pricing needs a
   237→   237→    // signed-URL fetch seam, flagged to the worker wave.)
   238→   238→    throw new StudioError("RENDER_INTERNAL", { requestId });
   239→   239→  }
   240→   240→  let ecproj: unknown;
   241→   241→  try {
   242→   242→    ecproj = JSON.parse(bytes.toString("utf8")) as unknown;
   243→   243→  } catch {
   244→   244→    // The stored snapshot is not valid JSON — the project cannot be rendered
   245→   245→    // as-is (the upload only ever verified its sha256, not its schema).
   246→   246→    throw new StudioError("RENDER_PROJECT_INVALID", {
   247→   247→      message: "This project can't be rendered as-is.",
   248→   248→      hint: "Re-save the project from the editor and try again.",
   249→   249→      requestId,
   250→   250→    });
   251→   251→  }
   252→   252→  const comp = findComp(ecproj, input.compId);
   253→   253→  if (comp === null) {
   254→   254→    throw new StudioError("RENDER_PROJECT_INVALID", {
   255→   255→      message: "This project can't be rendered as-is.",
   256→   256→      hint: "The requested composition is not in this version of the project.",
   257→   257→      requestId,
   258→   258→    });
   259→   259→  }
   260→   260→
   261→   261→  // Effective output size: settings override the comp's native size one axis
   262→   262→  // at a time; the long edge is what the pricing tiers (and the plan cap) use.
   263→   263→  const width = settings.width ?? comp.width;
   264→   264→  const height = settings.height ?? comp.height;
   265→   265→  const longEdgePx = Math.max(width, height);
   266→   266→
   267→   267→  // Effective frame rate for the ROW (the monthly-minutes accounting reads
   268→   268→  // settingsJson.frameRate — materializing the comp's native fps keeps the
   269→   269→  // usage sum honest when the client omits one).
   270→   270→  const frameRate = settings.frameRate ?? comp.frameRate;
   271→   271→
   272→   272→  // Range: the comp timeline is the bound — an out-point beyond the duration
   273→   273→  // is clamped so the priced range, the STORED range and the worker's range
   274→   274→  // are the same seconds. An in-point at/past the duration is refused by
   275→   275→  // renderOutputSeconds (empty range) as VALIDATION_FAILED.
   276→   276→  const clampedTo =
   277→   277→    settings.timeTo !== undefined ? Math.min(settings.timeTo, comp.durationSeconds) : undefined;
   278→   278→  const outputSeconds = renderOutputSeconds({
   279→   279→    durationSeconds: comp.durationSeconds,
   280→   280→    timeFrom: settings.timeFrom,
   281→   281→    timeTo: clampedTo,
   282→   282→  });
   283→   283→
   284→   284→  return {
   285→   285→    project,
   286→   286→    versionId,
   287→   287→    outputSeconds,
   288→   288→    longEdgePx,
   289→   289→    quality: settings.quality ?? "good",
   290→   290→    storedSettings: {
   291→   291→      ...settings,
   292→   292→      frameRate,
   293→   293→      ...(clampedTo !== undefined ? { timeTo: clampedTo } : {}),
   294→   294→    },
   295→   295→  };
   296→   296→}

   298→   298→// ---------------------------------------------------------------------------
   299→   299→// The monthly render-minutes window (the billing convention, cited)
   300→   300→// ---------------------------------------------------------------------------
   301→   301→
   302→   302→/**
   303→   303→ * The render-minutes billing window starts at the FIRST INSTANT of the
   304→   304→ * current UTC month. CONVENTION CITATION (M8-G): no gateway billing surface
   305→   305→ * computes a period START — GET /v1/billing/state reports only
   306→   306→ * `currentPeriodEnd` (the Stripe anchor), the credits reconciliation window
   307→   307→ * takes EXPLICIT caller-supplied bounds, and the subscriptions row persists
   308→   308→ * `currentPeriodEnd` only — there is no stored period start, and trial
   309→   309→ * workspaces (no subscription row at all) gate on the trial catalog tier
   310→   310→ * through resolvePlan. The catalog says "render minutes per billing month",
   311→   311→ * so the derivable, workspace-uniform window is the UTC calendar month.
   312→   312→ * If a later billing wave anchors windows to the Stripe period, swap this
   313→   313→ * ONE helper (and the renderSecondsUsed call site) — nothing else changes.
   314→   314→ */
   315→   315→function utcMonthStart(now: Date): Date {
   316→   316→  return new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), 1));
   317→   317→}

   320→   320→// The routes
   321→   321→// ---------------------------------------------------------------------------
   322→   322→
   323→   323→export function rendersRoutes(deps: GatewayDeps): Hono<AppBindings> {
   324→   324→  const app = new Hono<AppBindings>();
   325→   325→  app.use("/v1/renders/*", requireAuth());
   326→   326→
   327→   327→  const toJob = async (render: Render): Promise<RenderJob> => {
   328→   328→    let resultUrl: string | null = null;
   329→   329→    // Signed download ONLY while the artifact exists: completed + objectKey
   330→   330→    // present. Retention-expired rows have objectKey cleared (M8-A
   331→   331→    // markOutputExpired) → resultUrl null, the row stays as history.
   332→   332→    if (render.status === "completed" && render.objectKey !== null) {
   333→   333→      const signed = await deps.storage.getSignedUrl(render.objectKey, SIGNED_URL_TTL_SECONDS, {
   334→   334→        method: "GET",
   335→   335→      });
   336→   336→      resultUrl = signed.url;
   337→   337→    }
   338→   338→    // D-11: the DB status vocabulary IS the wire vocabulary — completed, not
   339→   339→    // "succeeded"; the v1 completed→succeeded mapping is gone.
   340→   340→    const framesDone = render.framesDone;
   341→   341→    const framesTotal = render.framesTotal;
   342→   342→    const percent =
   343→   343→      framesTotal > 0 ? Math.min(100, Math.floor((framesDone * 100) / framesTotal)) : 0;
   344→   344→    return {
   345→   345→      id: render.id,
   346→   346→      projectId: render.projectId,
   347→   347→      compId: render.compId,
   348→   348→      versionId: render.versionId,
   349→   349→      format: (RENDER_FORMATS as readonly string[]).includes(render.format)
   350→   350→        ? (render.format as RenderJob["format"])
   351→   351→        : "mp4", // unreachable: the queue only admits declared formats
   352→   352→      settings: (render.settingsJson as Record<string, unknown> | null) ?? {},
   353→   353→      status: render.status as RenderJob["status"],
   354→   354→      cancelRequested: render.cancelRequested,
   355→   355→      progress: { framesDone, framesTotal, percent },
   356→   356→      // renders.credits is INTEGER µc (see the module header) → the D-12
   357→   357→      // 2-decimal display number at this boundary; null until priced.
   358→   358→      credits: render.credits > 0 ? microcreditsToDisplayCredits(render.credits) : null,
   359→   359→      resultUrl,
   360→   360→      errorCode: render.errorCode,
   361→   361→      error: render.error,
   362→   362→      sizeBytes: render.sizeBytes,
   363→   363→      createdBy: render.createdBy,
   364→   364→      createdAt: render.createdAt.toISOString(),
   365→   365→      startedAt: render.startedAt !== null ? render.startedAt.toISOString() : null,
   366→   366→      completedAt: render.completedAt !== null ? render.completedAt.toISOString() : null,
   367→   367→      expiresAt: render.expiresAt !== null ? render.expiresAt.toISOString() : null,
   368→   368→    };
   369→   369→  };
   370→   370→
   371→   371→  // ── POST /v1/renders — queue (editor) ─────────────────────────────────────
   372→   372→  app.post("/v1/renders", requireWorkspace("editor"), async (c: AppContext): Promise<Response> => {
   373→   373→    const requestId = c.get("requestId");
   374→   374→    const workspace = c.get("workspace");
   375→   375→
   376→   376→    const parsed = QUEUE_RENDER_SCHEMA.safeParse(await c.req.json().catch(() => null));
   377→   377→    if (!parsed.success) {
   378→   378→      throw new StudioError("VALIDATION_FAILED", { requestId });
   379→   379→    }
   380→   380→    const request = parsed.data;
   381→   381→
   382→   382→    // The comp metadata comes from the PINNED version — priced BEFORE any
   383→   383→    // gate so every entitlement check sees the real output (and so the
   384→   384→    // estimate route, which reuses this derivation, agrees exactly).
   385→   385→    const quote = await resolveRenderQuote(
   386→   386→      deps,
   387→   387→      workspace.id,
   388→   388→      requestId,
   389→   389→      request,
   390→   390→      request.settings,
   391→   391→    );
   392→   392→
   393→   393→    // Entitlement gates, in order — each a typed StudioError (plan comes from
   394→   394→    // the workspace's subscription row via the resolvePlan seam; entitlements
   395→   395→    // from the catalog, never inlined).
   396→   396→    const plan = await resolvePlan(workspace.id);
   397→   397→    const entitlements = entitlementsFor(plan);
   398→   398→
   399→   399→    // (a) D-2: only MP4 + PNG sequence are enabled in v1; the other declared
   400→   400→    // formats stay priced but refused at queue time.
   401→   401→    if (!isRenderFormatEnabled(request.format)) {
   402→   402→      throw new StudioError("RENDER_UNSUPPORTED_FORMAT", { requestId });
   403→   403→    }
   404→   404→
   405→   405→    // (b) The resolution cap. PINNED MAPPING (M8-G): the config catalog
   406→   406→    // documents RENDER_QUOTA_EXCEEDED (429) as the MONTHLY-MINUTES gate only
   407→   407→    // ("workspace out of render minutes this cycle", M8-B) — it says nothing
   408→   408→    // about a resolution cap, and the cap is validation-shaped (fix the
   409→   409→    // request, not the plan), so it maps to VALIDATION_FAILED. Status pinned
   410→   410→    // at 422 (Unprocessable Content): the payload is schema-valid but
   411→   411→    // semantically over the plan's maxRenderLongEdgePx, and the house's 400
   412→   412→    // stays reserved for boundary/schema failures. The row below overrides
   413→   413→    // the STATUS_BY_CODE 400 for this one gate only.
   414→   414→    if (quote.longEdgePx > entitlements.maxRenderLongEdgePx) {
   415→   415→      throw new GatewayError(
   416→   416→        "VALIDATION_FAILED",
   417→   417→        "That render size is larger than your plan allows.",
   418→   418→        {
   419→   419→          status: 422,
   420→   420→          hint: `Renders are capped at a ${entitlements.maxRenderLongEdgePx}px long edge on this plan. Lower the resolution or upgrade.`,
   421→   421→          requestId,
   422→   422→        },
   423→   423→      );
   424→   424→    }
   425→   425→
   426→   426→    // (c) Concurrency: queued + running rows against maxConcurrentRenders.
   427→   427→    const active = await deps.repos.renders.countActive(workspace.id);
   428→   428→    if (active >= entitlements.maxConcurrentRenders) {
   429→   429→      throw new StudioError("RENDER_CONCURRENCY_LIMIT", { requestId });
   430→   430→    }
   431→   431→
   432→   432→    // (d) Monthly render minutes: Σ completed output seconds this billing
   433→   433→    // month (renderSecondsUsed — the honest fallback is 0 for rows without a
   434→   434→    // usable frameRate) against renderMinutesPerMonth, via the catalog helper
   435→   435→    // so the threshold can never drift. The requested render must FIT the
   436→   436→    // remaining minutes: renders are charged per output second and no partial
   437→   437→    // policy is documented, so a job whose full output exceeds the remainder
   438→   438→    // is refused up-front (a fully-used quota is the remaining === 0 case).
   439→   439→    const windowStart = utcMonthStart(new Date());
   440→   440→    const secondsUsed = await deps.repos.renders.renderSecondsUsed(workspace.id, windowStart);
   441→   441→    const remainingMinutes = renderMinutesRemaining(plan, secondsUsed / 60);
   442→   442→    if (quote.outputSeconds > remainingMinutes * 60) {
   443→   443→      throw new StudioError("RENDER_QUOTA_EXCEEDED", { requestId });
   444→   444→    }
   445→   445→
   446→   446→    // (e) Credits: the ONE formula's µc — the exact hold this route takes and
   447→   447→    // the worker later settles. §8.8: refuse on a balance that cannot cover
   448→   448→    // the hold BEFORE any row exists (INSUFFICIENT_CREDITS 402, no ledger row).
   449→   449→    const detail = estimateRenderCostDetailUc({
   450→   450→      format: request.format,
   451→   451→      outputSeconds: quote.outputSeconds,
   452→   452→      longEdgePx: quote.longEdgePx,
   453→   453→      quality: quote.quality,
   454→   454→    });
   455→   455→    const balance = await deps.repos.ledger.balance(workspace.id);
   456→   456→    if (balance < detail.totalUc) {
   457→   457→      throw new StudioError("INSUFFICIENT_CREDITS", { requestId });
   458→   458→    }
   459→   459→
   460→   460→    // (f)+(g) Create the row FIRST — it is the idempotency anchor and the
   461→   461→    // ledger scope needs its id. versionId is the SERVER-pinned project head
   462→   462→    // (never client-supplied — the body schema rejects a versionId outright);
   463→   463→    // objectKey is pre-created at queue time with the `{renderId}` placeholder
   464→   464→    // so the worker never guesses §9.2 layout (repos/renders.ts resolves it
   465→   465→    // against the generated id); credits stores the integer µc cost (the
   466→   466→    // column dialect — module header), so the wire display and the ledger
   467→   467→    // hold below are the same number.
   468→   468→    const objectKey = renderKey(
   469→   469→      workspace.id,
   470→   470→      quote.project.id,
   471→   471→      "{renderId}",
   472→   472→      request.format === "png-seq" ? "zip" : "mp4",
   473→   473→    );
   474→   474→    const render = await deps.repos.renders.create(workspace.id, {
   475→   475→      projectId: quote.project.id,
   476→   476→      versionId: quote.versionId,
   477→   477→      compId: request.compId,
   478→   478→      format: request.format,
   479→   479→      settingsJson: quote.storedSettings,
   480→   480→      credits: detail.totalUc,
   481→   481→      createdBy: c.get("auth").user.id,
   482→   482→      objectKey,
   483→   483→    });
   484→   484→
   485→   485→    // (h) Hold the credits, scoped to the render. A refusal (the balance can
   486→   486→    // dip between the precheck and the hold — a rare race) must never orphan
   487→   487→    // the queued row for a worker to claim an UNPAID job: terminal-fail it
   488→   488→    // with error_code RENDER_INTERNAL and surface the original typed error
   489→   489→    // (402 INSUFFICIENT_CREDITS / 500) to the caller.
   490→   490→    try {
   491→   491→      await deps.repos.ledger.reserve(workspace.id, {
   492→   492→        renderId: render.id,
   493→   493→        amount: detail.totalUc,
   494→   494→        idempotencyKey: `render:reserve:${render.id}`,
   495→   495→        metadata: {
   496→   496→          versionId: quote.versionId,
   497→   497→          format: request.format,
   498→   498→          outputSeconds: quote.outputSeconds,
   499→   499→          longEdgePx: quote.longEdgePx,
   500→   500→          quality: quote.quality,
   501→   501→        },
   502→   502→      });
   503→   503→    } catch (error) {
   504→   504→      // NOTE (M8-G finding): the RENDER status machine (packages/db
   505→   505→      // statuses.ts, M8-A) has no queued→failed carve-out — unlike the RUN
   506→   506→      // machine's F-J5 row — so markFailed on the fresh queued row would
   507→   507→      // CONFLICT. The legal path to the same terminal state is the two-step
   508→   508→      // queued→running→failed below (the intermediate running is unobservable
   509→   509→      // — this single request terminal-fails before any claim or stale sweep
   510→   510→      // can see it; claimNext only takes queued rows). Follow-up for the db
   511→   511→      // owner: add "failed" to RENDER_TRANSITIONS.queued and simplify this to
   512→   512→      // one markFailed call.
   513→   513→      try {
   514→   514→        await deps.repos.renders.markRunning(workspace.id, render.id);
   515→   515→        await deps.repos.renders.markFailed(workspace.id, render.id, {
   516→   516→          error: "Could not hold the credits for this render.",
   517→   517→          errorCode: "RENDER_INTERNAL",
   518→   518→        });
   519→   519→      } catch (cleanupError) {
   520→   520→        deps.logger.error("render_reserve_cleanup_failed", {
   521→   521→          requestId,
   522→   522→          renderId: render.id,
   523→   523→          detail: cleanupError instanceof Error ? cleanupError.message : "unknown",
   524→   524→        });
   525→   525→      }
   526→   526→      throw error;
   527→   527→    }
   528→   528→
   529→   529→    await audit(deps, c, workspace.id, "render.queued", quote.project.id, {
   530→   530→      // The audit TARGET is the project (the activity feed resolves
   531→   531→      // project-targeted rows by targetId — a render id would never match);
   532→   532→      // the render rides metadata.
   533→   533→      renderId: render.id,
   534→   534→      versionId: quote.versionId,
   535→   535→      format: request.format,
   536→   536→      objectKey: render.objectKey,
   537→   537→      outputSeconds: quote.outputSeconds,
   538→   538→      longEdgePx: quote.longEdgePx,
   539→   539→    });
   540→   540→
   541→   541→    return ok(c, await toJob(render), 201);
   542→   542→  });
