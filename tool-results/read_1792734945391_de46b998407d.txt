     1→/**
     2→ * `attachEditorBridge` — the typed wrapper around the frozen editor iframe's
     3→ * `window.effectcraft` global (PRODUCT_ARCHITECTURE §6.2).
     4→ *
     5→ * Verified against the shipped dist (READ-ONLY sources, v0.13.0):
     6→ * - `window.effectcraft` + `window.effectcraftLoad = {}` are defined synchronously
     7→ *   by the inline module script in `apps/web/public/editor/index.html` (from
     8→ *   `editor/apps/effectcraft-web/web/index.html`) before the wasm runs.
     9→ * - `effectcraftLoad.wasmMs` is set once the module compiles, `readyMs` once
    10→ *   `ec.start()` returns; on failure the page sets `effectcraftLoad.error =

    12→ *   (`editor/apps/effectcraft-web/tests/smoke.mjs`) waits for
    13→ *   `effectcraftLoad.readyMs || effectcraftLoad.error`.

    15→ *   (`editor/apps/effectcraft-web/src/api.rs`): "Studio is not running yet"
    16→ *   (de-branded dist string) when the control channel is not up (transient,
    17→ *   retry), "request dropped" when it disconnects. Errors from the control
    18→ *   channel itself reject with that channel's `error` message.
    19→ * - Upstream documents a 60 s control-request timeout
    20→ *   (`editor/docs/control-protocol.md`); the web dist keeps requests pending
    21→ *   until they are answered or dropped, so the bridge enforces its own

    63→/** Transient startup rejections from api.rs ("Studio is not running yet" in the shipped dist) — retried once. */
    64→const NOT_READY_RE = /not running yet|not ready|still (loading|starting)|starting up/i;

    67→export interface AttachOptions {
    68→  /** Per-call timeout in ms (default 15 s). */
    69→  readonly callTimeoutMs?: number;
    70→  /** Handshake timeout in ms (default 60 s). */
    71→  readonly readyTimeoutMs?: number;
    72→  /** Expected editor version prefix; a mismatch emits an "error" event (attach still resolves). */
    73→  readonly version?: string;
    74→  /**
    75→   * Pre-attach event handler: sees the early "ready" emission, the
    76→   * version-mismatch "error" and the handshake "error"s (`EDITOR_LOAD_FAILED`
    77→   * when the dist's wasm failed to boot, `EDITOR_NOT_READY` on timeout) that
    78→   * fire before `attachEditorBridge` resolves. Routed into the same bus as
    79→   * `on(…)`.
    80→   */
    81→  readonly onEvent?: (event: EditorEventName, payload: unknown) => void;
    82→}

    84→class EventBus {
    85→  private readonly listeners = new Map<EditorEventName, Set<EditorEventHandler>>();
    86→  /** "ready" is sticky: late subscribers get the (single) emission replayed. */
    87→  private stickyReady: { payload: unknown } | null = null;
    88→
    89→  on(event: EditorEventName, cb: EditorEventHandler): () => void {
    90→    let set = this.listeners.get(event);
    91→    if (!set) {
    92→      set = new Set();
    93→      this.listeners.set(event, set);
    94→    }
    95→    set.add(cb);
    96→    if (event === "ready" && this.stickyReady) cb(this.stickyReady.payload);
    97→    return () => {
    98→      set?.delete(cb);
    99→    };
   100→  }
   101→
   102→  emit(event: EditorEventName, payload: unknown): void {
   103→    if (event === "ready") this.stickyReady = { payload };
   104→    for (const cb of [...(this.listeners.get(event) ?? [])]) cb(payload);
   105→  }
   106→
   107→  clear(): void {
   108→    this.listeners.clear();
   109→    this.stickyReady = null;
   110→  }
   111→}

   124→function zodMessage(err: unknown): string {
   125→  if (err instanceof z.ZodError) {
   126→    return err.issues
   127→      .map((issue) => `${issue.path.join(".") || "(root)"}: ${issue.message}`)

   134→  try {

   146→function isNotReadyMessage(message: string): boolean {
   147→  return NOT_READY_RE.test(message);
   148→}
   149→
   150→function classify(label: string, err: unknown): BridgeError {
   151→  if (err instanceof BridgeError) return err;
   152→  const message = messageOf(err);
   153→  if (DISCONNECTED_RE.test(message)) {
   154→    return new BridgeError("EDITOR_DISCONNECTED", `${label}: ${message}`, true, err);
   155→  }
   156→  if (isNotReadyMessage(message)) {
   157→    return new BridgeError("EDITOR_NOT_READY", `${label}: ${message}`, true, err);
   158→  }
   159→  return new BridgeError("EDITOR_CALL_FAILED", `${label}: ${message}`, false, err);
   160→}
   161→
   162→function withTimeout<T>(label: string, promise: Promise<T>, timeoutMs: number): Promise<T> {
   163→  return new Promise<T>((resolve, reject) => {
   164→    const timer = setTimeout(() => {
   165→      reject(new BridgeError("EDITOR_TIMEOUT", `${label}: timed out after ${timeoutMs} ms`, true));
   166→    }, timeoutMs);
   167→    const settle = (fn: (v: T | PromiseLike<T>) => void, value: T) => {
   168→      clearTimeout(timer);
   169→      fn(value);
   170→    };
   171→    promise.then(
   172→      (value) => settle(resolve, value),
   173→      (err) => {
   174→        clearTimeout(timer);
   175→        reject(err);
   176→      },
   177→    );
   178→  });
   179→}

   181→/**
   182→ * Poll for `effectcraftLoad.readyMs` (the dist sets it once `ec.start()`
   183→ * returns) or fail fast on `effectcraftLoad.error` — the exact two signals
   184→ * the shipped host page writes (`apps/web/public/editor/index.html`), and the
   185→ * same pair the upstream smoke test waits for.
   186→ *
   187→ * The handshake is the one place the bridge polls: the dist exposes no ready
   188→ * event, and `window.effectcraft` exists before the app runs. The `on(…)`
   189→ * event bus below never polls.
   190→ */
   191→async function waitForReady(iframe: HTMLIFrameElement, readyTimeoutMs: number): Promise<void> {
   192→  const deadline = Date.now() + readyTimeoutMs;
   193→  for (;;) {
   194→    const w = contentWindowOf(iframe);
   195→    const ec = w?.effectcraft;
   196→    const load = w?.effectcraftLoad;
   197→    if (ec && load) {
   198→      // Order matters: the dist sets `error` on any boot failure and never
   199→      // `readyMs`; if both are somehow present, the failure wins.
   200→      if (load.error !== undefined) {
   201→        throw new BridgeError("EDITOR_LOAD_FAILED", `editor failed to start: ${load.error}`, false);
   202→      }
   203→      if (load.readyMs !== undefined) return;
   204→      // A ready() hook (not in upstream v0.13): awaited once, bounded by the deadline.
   205→      const ready = (ec as { ready?: unknown }).ready;
   206→      if (typeof ready === "function") {
   207→        const remaining = Math.max(deadline - Date.now(), 0);
   208→        try {
   209→          await withTimeout("ready()", Promise.resolve(ready.call(ec)), remaining);
   210→          return;
   211→        } catch (err) {
   212→          throw classify("ready()", err);
   213→        }
   214→      }
   215→    }
   216→    if (Date.now() >= deadline) {
   217→      throw new BridgeError(
   218→        "EDITOR_NOT_READY",
   219→        `editor did not become ready within ${readyTimeoutMs} ms`,
   220→        true,
   221→      );
   222→    }
   223→    await delay(READY_POLL_MS);
   224→  }
   225→}

   231→/** The editor wasm's boot phase, read from the dist's `effectcraftLoad`. */
   232→export type EditorLoadStatus = "loading" | "ready" | "failed";
   233→
   234→/**
   235→ * A snapshot of the editor's boot state, taken from the shipped dist's
   236→ * `window.effectcraftLoad` (`apps/web/public/editor/index.html`):
   237→ * `{}` while loading, `wasmMs` once the module compiled, `readyMs` once
   238→ * `ec.start()` returned, `error` when the wasm failed to boot.
   239→ */
   240→export interface EditorLoadState {
   241→  /** `loading` until `readyMs`/`error` appears (or the globals are not yet visible). */
   242→  readonly status: EditorLoadStatus;
   243→  /** Present when `status === "failed"` — the dist's `String(e)`. */
   244→  readonly error?: string;
   245→  /** Set once the wasm module compiled (present on both ready and failed boots). */
   246→  readonly wasmMs?: number;
   247→  /** Set once `ec.start()` returned (present only on a ready boot). */
   248→  readonly readyMs?: number;
   249→}
   250→
   251→/**
   252→ * Read the editor's boot state synchronously — the public seam for "the wasm
   253→ * failed to boot" (M3.7). Works before (and without) `attachEditorBridge`:
   254→ * the host page can poll this while it shows its loading screen, then render
   255→ * the failure + retry UI the moment `status === "failed"`, and call
   256→ * `attachEditorBridge` once `status === "ready"`.
   257→ *
   258→ * - Returns `{ status: "loading" }` while the iframe has not exposed the
   259→ *   dist globals yet (including cross-origin reads, which throw and are
   260→ *   swallowed — the state is genuinely unknowable from this frame).
   261→ * - Same-origin only, like the rest of the bridge (§6.1).
   262→ *
   263→ * The `attachEditorBridge` rejection is the async counterpart: it fails with
   264→ * `BridgeError` code `EDITOR_LOAD_FAILED` carrying the same `error` string.
   265→ */
   266→export function editorLoadState(iframe: HTMLIFrameElement): EditorLoadState {
   267→  let load: EffectcraftLoad | undefined;
   268→  try {
   269→    load = (iframe.contentWindow as EffectcraftWindow | null)?.effectcraftLoad;
   270→  } catch {
   271→    return { status: "loading" }; // cross-origin: the dist's globals are not readable
   272→  }
   273→  if (!load) return { status: "loading" };
   274→  if (load.error !== undefined) {
   275→    return { status: "failed", error: load.error, wasmMs: load.wasmMs, readyMs: load.readyMs };
   276→  }
   277→  if (load.readyMs !== undefined) {
   278→    return { status: "ready", wasmMs: load.wasmMs, readyMs: load.readyMs };
   279→  }
   280→  return { status: "loading", wasmMs: load.wasmMs };
   281→}

   284→export function sameOrigin(iframe: HTMLIFrameElement): boolean {
   285→  const src = iframe.getAttribute("src") ?? "";
   286→  try {
   287→    const url = new URL(src, globalThis.location.href);
   288→    return url.origin === globalThis.location.origin;
   289→  } catch {
   290→    return true; // no/relative src → same origin
   291→  }

   294→/**

   296→ * assigns is what callers must keep — upstream `addFile` basename-es the
   297→ * request (`editor/apps/effectcraft-web/src/files.rs` `path_for`).
   298→ *
   299→ * Exported (additive, M8-F) so the render worker writes hydrated asset files

   304→export function sanitizeAssetName(name: string): string {
   305→  const cleaned = name
   306→    .replace(/[/\\]/g, "-")
   307→    .replace(/[^A-Za-z0-9._-]+/g, "-")
   308→    .replace(/-{2,}/g, "-")
   309→    .replace(/^[.-]+/, "")
   310→    .slice(0, 120);
   311→  return cleaned.length > 0 ? cleaned : "asset";
   312→}

   314→/**
   315→ * Attach to the editor iframe. Resolves once `window.effectcraft` exists and
   316→ * reports ready (`effectcraftLoad.readyMs`, or an optional `ready()` hook).
   317→ *
   318→ * Rejects with a `BridgeError` when the editor cannot boot:
   319→ * - `EDITOR_LOAD_FAILED` — the dist's `effectcraftLoad.error` is set (the wasm
   320→ *   failed to boot; the `error` string is in the message). Use
   321→ *   {@link editorLoadState} to observe this without awaiting the handshake.
   322→ * - `EDITOR_NOT_READY` — the handshake timed out (default 60 s).
   323→ *
   324→ * A load failure or handshake timeout is also emitted on the event bus as an
   325→ * `"error"` event (visible pre-attach via `AttachOptions.onEvent`).
   326→ *
   327→ * Every resolved method call is wrapped with a timeout (default 15 s), a
   328→ * single retry on transient "not ready" rejections, and `BridgeError`
   329→ * normalization (`EDITOR_TIMEOUT` | `EDITOR_CALL_FAILED` | `EDITOR_NOT_READY`
   330→ * | `EDITOR_LOAD_FAILED` | `EDITOR_DISCONNECTED`).
   331→ */
   332→export function attachEditorBridge(
   333→  iframe: HTMLIFrameElement,
   334→  opts: AttachOptions = {},
   335→): Promise<EditorBridge> {
   336→  const callTimeoutMs = opts.callTimeoutMs ?? DEFAULT_CALL_TIMEOUT_MS;
   337→  const readyTimeoutMs = opts.readyTimeoutMs ?? DEFAULT_READY_TIMEOUT_MS;
   338→  const bus = new EventBus();
   339→  if (opts.onEvent) {
   340→    for (const event of ["ready", "focus", "resize", "error"] as const) {
   341→      bus.on(event, (payload) => opts.onEvent?.(event, payload));
   342→    }
   343→  }
   344→  let disposed = false;
   345→
   346→  /** The handshake, shared by attach and `bridge.ready()`: emits bus "error"s. */
   347→  async function awaitReady(): Promise<void> {
   348→    try {
   349→      await waitForReady(iframe, readyTimeoutMs);
   350→    } catch (err) {
   351→      const bridgeErr = err instanceof BridgeError ? err : classify("ready", err);
   352→      bus.emit("error", bridgeErr);
   353→      throw bridgeErr;
   354→    }
   355→  }
   356→
   357→  async function call<T>(
   358→    label: string,
   359→    fn: (ec: EffectcraftGlobal, w: EffectcraftWindow) => T | Promise<T>,
   360→    timeoutMs: number = callTimeoutMs,
   361→  ): Promise<T> {
   362→    try {
   363→      if (disposed) {
   364→        throw new BridgeError("EDITOR_DISCONNECTED", `${label}: bridge is disposed`, false);
   365→      }
   366→      const w = contentWindowOf(iframe);
   367→      if (!w) {
   368→        throw new BridgeError(
   369→          "EDITOR_DISCONNECTED",
   370→          `${label}: iframe contentWindow is gone`,
   371→          false,
   372→        );
   373→      }
   374→      const ec = w.effectcraft;
   375→      if (!ec) {
   376→        throw new BridgeError(
   377→          "EDITOR_NOT_READY",
   378→          `${label}: window.effectcraft is missing (editor reloading?)`,
   379→          true,
   380→        );
   381→      }
   382→
   383→      let lastErr: unknown = undefined;
   384→      for (let attempt = 0; ; attempt++) {
   385→        if (attempt > 0) await delay(RETRY_DELAY_MS);
   386→        try {
   387→          // fn may throw synchronously (readFile/info are sync upstream) — Promise.resolve captures it.
   388→          return await withTimeout(label, Promise.resolve(fn(ec, w)), timeoutMs);
   389→        } catch (err) {
   390→          lastErr = err;
   391→          const retryable = !(err instanceof BridgeError) && isNotReadyMessage(messageOf(err));
   392→          if (attempt >= 1 || !retryable) break;
   393→        }
   394→      }
   395→      throw lastErr;
   396→    } catch (err) {
   397→      const bridgeErr = classify(label, err);
   398→      bus.emit("error", bridgeErr);
   399→      throw bridgeErr;
   400→    }
   401→  }
   402→
   403→  function rawInfo(): EditorInfo {
   404→    if (disposed) {
   405→      throw new BridgeError("EDITOR_DISCONNECTED", "info(): bridge is disposed", false);
   406→    }
   407→    const w = contentWindowOf(iframe);
   408→    const ec = w?.effectcraft;
   409→    if (!w || !ec) {
   410→      throw new BridgeError("EDITOR_NOT_READY", "info(): window.effectcraft is missing", true);
   411→    }
   412→    // info() is synchronous upstream; validate the bits the bridge relies on.
   413→    const parsed = parseOr(EditorInfoSchema, "info", ec.info());
   414→    const load = parseOr(EffectcraftLoadInfoSchema, "info.load", w.effectcraftLoad ?? {});
   415→    return { ...parsed, load };
   416→  }

   420→  function attachWindowEvents(w: EffectcraftWindow): void {
   421→    const onFocus = () => bus.emit("focus", { focused: true, window: w });
   422→    const onBlur = () => bus.emit("focus", { focused: false, window: w });
   423→    const onResize = () =>
   424→      bus.emit("resize", { width: w.innerWidth, height: w.innerHeight, window: w });
   425→    w.addEventListener("focus", onFocus);
   426→    w.addEventListener("blur", onBlur);
   427→    w.addEventListener("resize", onResize);
   428→    detachFns.push(() => {
   429→      w.removeEventListener("focus", onFocus);
   430→      w.removeEventListener("blur", onBlur);
   431→      w.removeEventListener("resize", onResize);
   432→    });
   433→  }
   434→
   435→  const detachFns: Array<() => void> = [];
   436→
   437→  return (async (): Promise<EditorBridge> => {
   438→    await awaitReady();
   439→
   440→    const w = contentWindowOf(iframe);
   441→    if (!w) {
   442→      throw new BridgeError("EDITOR_DISCONNECTED", "iframe contentWindow is gone", false);
   443→    }
   444→    attachWindowEvents(w);
   445→
   446→    const info = rawInfo();
   447→    bus.emit("ready", {
   448→      readyMs: w.effectcraftLoad?.readyMs,
   449→      wasmMs: w.effectcraftLoad?.wasmMs,
   450→      version: info.version,
   451→    });
   452→    if (opts.version !== undefined && !info.version.startsWith(opts.version)) {
   453→      bus.emit(
   454→        "error",
   455→        new Error(
   456→          `editor version ${JSON.stringify(info.version)} does not match expected ${JSON.stringify(opts.version)}`,
   457→        ),
   458→      );
   459→    }
   460→
   461→    const bridge: EditorBridge = {
   462→      async ready(): Promise<void> {
   463→        if (disposed) {
   464→          throw new BridgeError("EDITOR_DISCONNECTED", "bridge is disposed", false);
   465→        }
   466→        await awaitReady();
   467→      },
   468→
   469→      inspect(): Promise<UiInspect> {
   470→        return call("inspect", (ec) => ec.inspect()).then((v) =>
   471→          parseOr(UiInspectSchema, "inspect", v),
   472→        );
   473→      },
   474→
   475→      commands(filter?: string): Promise<CommandInfo[]> {
   476→        const params = filter === undefined ? {} : { filter };
   477→        return call("commands", (ec) => ec.request("engine.commands", params)).then((v) =>
   478→          parseOr(z.array(CommandInfoSchema), "commands", v),
   479→        );
   480→      },
   481→
   482→      describe(command: string): Promise<CommandSchema> {
   483→        return call("describe", (ec) => ec.execute("command.describe", { command })).then((v) =>
   484→          parseOr(CommandSchemaSchema, "describe", v),
   485→        );
   486→      },
   487→
   488→      query<T = unknown>(command: string, params?: unknown): Promise<T> {
   489→        return call(
   490→          `query ${command}`,
   491→          async (ec) => (await ec.execute(command, params ?? {})) as T,
   492→        );
   493→      },
   494→
   495→      execute<T = unknown>(command: string, params?: unknown): Promise<T> {
   496→        return call(
   497→          `execute ${command}`,
   498→          async (ec) => (await ec.execute(command, params ?? {})) as T,

   500→      },
   501→
   502→      request<T = unknown>(method: string, params?: unknown): Promise<T> {

   509→      async batch(input: Parameters<EditorBridge["batch"]>[0]): Promise<BatchResult> {
   510→        let parsed: AgentBatch;
   511→        try {
   512→          // §8.5 bounds, `$N.key` refs and the 2 MB cap — via @studio/contracts
   513→          parsed = normalizeAgentBatch(input);
   514→        } catch (err) {
   515→          const bridgeErr = new BridgeError(
   516→            "EDITOR_CALL_FAILED",
   517→            `invalid agent batch: ${zodMessage(err)}`,
   518→            false,
   519→            err,
   520→          );
   521→          bus.emit("error", bridgeErr);
   522→          throw bridgeErr;
   523→        }
   524→        const payload = { label: parsed.label, atomic: true, steps: parsed.steps };
   525→        const timeoutMs = Math.min(parsed.timeoutMs ?? callTimeoutMs, MAX_BATCH_TIMEOUT_MS);
   526→        return call(
   527→          `batch ${JSON.stringify(parsed.label)}`,
   528→          (ec) => ec.execute("engine.batch", payload),
   529→          timeoutMs,
   530→        ).then((v) => parseOr(BatchResultSchema, "batch", v));
   531→      },
   532→
   533→      renderFrame(params: RenderFrameParams): Promise<RenderFrameResult> {
   534→        return call("renderFrame", (ec) => ec.renderFrame(params)).then((v) =>
   535→          parseOr(RenderFrameResultSchema, "renderFrame", v),
   536→        );
   537→      },

   545→      async addAsset(asset: CloudAsset): Promise<string> {
   546→        const virtualPath = `/assets/${encodeURIComponent(asset.assetId)}/${sanitizeAssetName(asset.name)}`;
   547→        let blob: Blob;
   548→        if (asset.bytes !== undefined) {
   549→          blob =
   550→            asset.bytes instanceof Blob
   551→              ? asset.bytes
   552→              : new Blob([new Uint8Array(asset.bytes)], { type: "application/octet-stream" });
   553→        } else if (asset.url !== undefined) {
   554→          try {
   555→            const res = await fetch(asset.url);
   556→            if (!res.ok) {
   557→              throw new BridgeError(
   558→                "EDITOR_CALL_FAILED",
   559→                `addAsset: fetching ${asset.url} failed with HTTP ${res.status}`,
   560→                false,
   561→              );
   562→            }
   563→            blob = await res.blob();
   564→          } catch (err) {
   565→            if (err instanceof BridgeError) throw err;
   566→            throw new BridgeError(
   567→              "EDITOR_CALL_FAILED",
   568→              `addAsset: fetching ${asset.url} failed (${messageOf(err)})`,
   569→              false,
   570→              err,
   571→            );
   572→          }
   573→        } else {
   574→          const bridgeErr = new BridgeError(
   575→            "EDITOR_CALL_FAILED",
   576→            "addAsset: CloudAsset needs `url` or `bytes`",
   577→            false,
   578→          );
   579→          bus.emit("error", bridgeErr);
   580→          throw bridgeErr;
   581→        }
   582→        return call("addAsset", (ec) => ec.addFile(blob, virtualPath)).then((v) =>
   583→          parseOr(z.string(), "addAsset", v),
   584→        );
   585→      },
   586→
   587→      saveToBrowser(path: string): Promise<{ path: string; bytes: number }> {
   588→        return call("saveToBrowser", (ec) => ec.saveToBrowser(path)).then((v) =>
   589→          parseOr(SaveResultSchema, "saveToBrowser", v),
   590→        );
   591→      },
   592→
   593→      readFile(path: string): Promise<Uint8Array> {
   594→        return call("readFile", (ec) => ec.readFile(path)).then((v) => {
   595→          if (!(v instanceof Uint8Array)) {
   596→            const bridgeErr = new BridgeError(
   597→              "EDITOR_CALL_FAILED",
   598→              `readFile: no file at ${JSON.stringify(path)}`,
   599→              false,
   600→            );
   601→            bus.emit("error", bridgeErr);
   602→            throw bridgeErr;
   603→          }
   604→          return v;
   605→        });
   606→      },
   607→
   608→      screenshot(params?: ScreenshotParams): Promise<ScreenshotResult> {
   609→        return call("screenshot", (ec) => ec.screenshot(params ?? {})).then((v) =>
   610→          parseOr(ScreenshotResultSchema, "screenshot", v),
   611→        );
   612→      },
   613→
   614→      on(event: EditorEventName, cb: EditorEventHandler): () => void {
   615→        return bus.on(event, cb);
   616→      },
   617→
   618→      version(): string {
   619→        return rawInfo().version;
   620→      },
   621→
   622→      info(): EditorInfo {
   623→        return rawInfo();
   624→      },
   625→
   626→      capabilities(): Promise<AppCapabilities> {
   627→        return call("capabilities", (ec) => ec.execute("app.capabilities")).then((v) =>
   628→          parseOr(AppCapabilitiesSchema, "capabilities", v),
   629→        );
   630→      },
   631→
   632→      dispose(): void {
   633→        if (disposed) return;
   634→        disposed = true;
   635→        for (const detach of detachFns) detach();
   636→        detachFns.length = 0;
   637→        bus.clear();
   638→      },
   639→    };
   640→
   641→    return bridge;
   642→  })();

   648→
   649→export type EditorShortcutName =
   650→  | "toggleChat" // Cmd/Ctrl+L
   651→  | "inlineEdit" // Cmd/Ctrl+K
   652→  | "agent" // Cmd/Ctrl+I
   653→  | "stop" // Cmd/Ctrl+.
   654→  | "newChat" // Cmd/Ctrl+Shift+L
   655→  | "undoAgentRun" // Cmd/Ctrl+Shift+U
   656→  | "escape" // Esc
   657→  | "send"; // Cmd/Ctrl+Enter
   658→
   659→export interface EditorShortcutEvent {
   660→  readonly shortcut: EditorShortcutName;
   661→  readonly key: string;
   662→  readonly window: Window;
   663→  readonly native: KeyboardEvent;
   664→  preventDefault(): void;
   665→  stopPropagation(): void;
   666→}
   667→
   668→// biome-ignore lint/suspicious/noConfusingVoidType: handlers may return nothing (handled) or `false` (declined)
   669→export type EditorShortcutHandler = (event: EditorShortcutEvent) => boolean | void;
   670→
   671→export interface ShortcutHandlers {
   672→  readonly toggleChat?: EditorShortcutHandler;
   673→  readonly inlineEdit?: EditorShortcutHandler;
   674→  readonly agent?: EditorShortcutHandler;
   675→  readonly stop?: EditorShortcutHandler;
   676→  readonly newChat?: EditorShortcutHandler;
   677→  readonly undoAgentRun?: EditorShortcutHandler;
   678→  readonly escape?: EditorShortcutHandler;
   679→  readonly send?: EditorShortcutHandler;
   680→}
   681→
   682→function matchShortcut(e: KeyboardEvent): EditorShortcutName | undefined {
   683→  const mod = e.metaKey || e.ctrlKey; // `command` is ⌘ on macOS, Ctrl elsewhere (control-protocol.md)
   684→  const key = e.key.toLowerCase();
   685→  if (mod && !e.shiftKey && !e.altKey) {
   686→    if (key === "l") return "toggleChat";
   687→    if (key === "k") return "inlineEdit";
   688→    if (key === "i") return "agent";
   689→    if (key === ".") return "stop";
   690→    if (key === "enter") return "send";
   691→  }
   692→  if (mod && e.shiftKey && !e.altKey) {
   693→    if (key === "l") return "newChat";
   694→    if (key === "u") return "undoAgentRun";
   695→  }
   696→  if (!mod && !e.shiftKey && !e.altKey && !e.ctrlKey && !e.metaKey && key === "escape") {
   697→    return "escape";
   698→  }
   699→  return undefined;
   700→}
   701→
   702→/**
   703→ * Attach a same-origin key listener on the iframe's `contentWindow` and
   704→ * forward the chat shortcuts (EDITOR_CHAT_UX §9.2): the parent handles
   705→ * L / K / I / `.` and their Shift combos while the editor keeps every other
   706→ * shortcut. `Cmd/Ctrl+Enter` (send/apply) and `Esc` (close popover) forward
   707→ * too — the editor is not asked to know about the chat.
   708→ *
   709→ * A handler that returns `false` explicitly declines the event (nothing is
   710→ * prevented, the editor sees it as usual); anything else — `undefined` or
   711→ * `true` — marks it handled for the parent and the native event gets
   712→ * `preventDefault()` + `stopPropagation()`.
   713→ *
   714→ * Returns a detach function.
   715→ */
   716→export function forwardShortcut(iframe: HTMLIFrameElement, handlers: ShortcutHandlers): () => void {
   717→  const w = contentWindowOf(iframe);
   718→  if (!w) return () => {};
   719→  const onKeyDown = (e: KeyboardEvent) => {
   720→    if (e.repeat) return;
   721→    const shortcut = matchShortcut(e);
   722→    const handler = shortcut !== undefined ? handlers[shortcut] : undefined;
   723→    if (shortcut === undefined || handler === undefined) return;
   724→    let used = false;
   725→    const event: EditorShortcutEvent = {
   726→      shortcut,
   727→      key: e.key,
   728→      window: w,
   729→      native: e,
   730→      preventDefault: () => {
   731→        used = true;
   732→        e.preventDefault();
   733→      },
   734→      stopPropagation: () => {
   735→        used = true;
   736→        e.stopPropagation();
   737→      },
   738→    };
   739→    const handled = handler(event);
   740→    if (handled !== false) {
   741→      used = true;
   742→      e.preventDefault();
   743→      e.stopPropagation();
   744→    } else if (used) {
   745→      e.preventDefault();
   746→      e.stopPropagation();
   747→    }
   748→  };
   749→  w.addEventListener("keydown", onKeyDown);
   750→  return () => {
   751→    w.removeEventListener("keydown", onKeyDown);
   752→  };
   753→}
