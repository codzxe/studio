    45→function frame(data: unknown): string {
    46→  return `data: ${JSON.stringify(data)}\n\n`;
    47→}
    48→function textDelta(content: string): string {
    49→  return frame({ choices: [{ delta: { content } }] });
    50→}
    51→function toolCallStart(index: number, id: string, name: string): string {
    52→  return frame({
    53→    choices: [
    54→      {
    55→        delta: { tool_calls: [{ index, id, type: "function", function: { name, arguments: "" } }] },
    56→      },
    57→    ],
    58→  });
    59→}
    60→function toolCallArgs(index: number, args: string): string {
    61→  return frame({
    62→    choices: [{ delta: { tool_calls: [{ index, function: { arguments: args } }] } }],
    63→  });
    64→}
    65→function finish(reason: string): string {
    66→  return frame({ choices: [{ delta: {}, finish_reason: reason }] });
    67→}
    68→function usageFrame(promptTokens: number, completionTokens: number): string {
    69→  return frame({
    70→    choices: [],
    71→    usage: { prompt_tokens: promptTokens, completion_tokens: completionTokens },
    72→  });
    73→}
    74→const DONE = "data: [DONE]\n\n";

    76→function sseResponse(frames: readonly string[]): Response {
    77→  const body = new ReadableStream<Uint8Array>({
    78→    start(controller) {
    79→      for (const chunk of frames) controller.enqueue(encoder.encode(chunk));
    80→      controller.close();
    81→    },
    82→  });
    83→  return new Response(body, { status: 200, headers: { "content-type": "text/event-stream" } });
    84→}

    86→interface RecordedCall {
    87→  readonly url: string;
    88→  readonly headers: Record<string, string>;
    89→  readonly body: {
    90→    model: string;
    91→    messages: Array<Record<string, unknown>>;
    92→    tools?: Array<{ function: { name: string } }>;
    93→    tool_choice?: string;

    97→function scriptedFetch(script: ReadonlyArray<Response>): typeof fetch & { calls: RecordedCall[] } {
    98→  const calls: RecordedCall[] = [];
    99→  let index = 0;
   100→  const stub = (async (url: unknown, init?: RequestInit): Promise<Response> => {
   101→    calls.push({
   102→      url: String(url),
   103→      headers: (init?.headers ?? {}) as Record<string, string>,
   104→      body: JSON.parse(String(init?.body)) as RecordedCall["body"],
   105→    });
   106→    const next = script[index];
   107→    index += 1;
   108→    if (next === undefined) throw new Error("script exhausted");
   109→    return next;
   110→  }) as typeof fetch & { calls: RecordedCall[] };
   111→  stub.calls = calls;
   112→  return stub;
   113→}

   119→type WaitMode = "pending" | "timeout" | "aborted" | "auto";
   120→
   121→class FakeStream implements AgentRunStream {
   122→  readonly events: RunStreamPayload[] = [];
   123→  readonly expected: Array<{ toolCallId: string; tool: string }> = [];
   124→  readonly controller = new AbortController();
   125→  /** "pending" → the TEST resolves each wait (realistic client POST). */
   126→  waitMode: WaitMode = "pending";
   127→  private readonly waits = new Map<string, (result: ToolWaitResult) => void>();
   128→  /** Results that arrive before the runner starts waiting (client POST race). */
   129→  private readonly early = new Map<string, unknown>();
   130→
   131→  get signal(): AbortSignal {
   132→    return this.controller.signal;
   133→  }
   134→
   135→  emit(payload: RunStreamPayload): StudioStreamEvent {
   136→    this.events.push(payload);
   137→    return { ...payload, runId: "run_test", seq: this.events.length - 1 };
   138→  }
   139→
   140→  async yieldToStream(): Promise<void> {
   141→    await new Promise((resolve) => setImmediate(resolve));
   142→  }
   143→
   144→  expectToolCall(toolCallId: string, tool: string): void {
   145→    this.expected.push({ toolCallId, tool });
   146→  }
   147→
   148→  waitForToolResult(toolCallId: string): Promise<ToolWaitResult> {
   149→    if (this.waitMode === "timeout") return Promise.resolve({ ok: false, reason: "timeout" });
   150→    if (this.waitMode === "aborted") return Promise.resolve({ ok: false, reason: "aborted" });
   151→    if (this.waitMode === "auto") return Promise.resolve({ ok: true, result: { stub: true } });
   152→    if (this.early.has(toolCallId)) {
   153→      return Promise.resolve({ ok: true, result: this.early.get(toolCallId) });
   154→    }
   155→    return new Promise((resolve) => {
   156→      this.waits.set(toolCallId, resolve);
   157→    });
   158→  }
   159→
   160→  resolveToolResult(toolCallId: string, result: unknown): void {
   161→    const resolve = this.waits.get(toolCallId);
   162→    if (resolve === undefined) {
   163→      this.early.set(toolCallId, result);
   164→      return;
   165→    }
   166→    this.waits.delete(toolCallId);
   167→    resolve({ ok: true, result });
   168→  }
   169→}

   171→/** Mirrors the chat route's finalize emission order (§7.4) for assertions. */
   172→async function runToOutcome(
   173→  runner: ReturnType<typeof createLiteLLMRunner>,
   174→  run: AgentRun,
   175→  stream: FakeStream,
   176→  reserve = 10,
   177→): Promise<AgentRunOutcome> {
   178→  const outcome = await runner.start(run, stream);
   179→  if (outcome.kind === "completed") {
   180→    stream.emit({
   181→      type: "credits",
   182→      reserved: reserve,
   183→      committed: outcome.committedCredits,
   184→      balance: Math.max(0, reserve - outcome.committedCredits),
   185→    });
   186→    stream.emit({ type: "run.completed" });
   187→  } else if (outcome.kind === "failed") {
   188→    stream.emit({
   189→      type: "run.failed",
   190→      code: outcome.code,
   191→      message: outcome.message,
   192→      retryable: outcome.retryable,
   193→    });
   194→  } else {
   195→    stream.emit({ type: "run.cancelled" });
   196→  }
   197→  return outcome;
   198→}

   242→async function waitUntil(predicate: () => boolean, timeoutMs = 2_000): Promise<void> {
   243→  const deadline = Date.now() + timeoutMs;
   244→  while (!predicate()) {
   245→    if (Date.now() > deadline) throw new Error("waitUntil timed out");
   246→    await new Promise((resolve) => setImmediate(resolve));
   247→  }
   248→}
