     1→/**
     2→ * Client-side apply/undo engine — BUILD_PLAN §M5.4/§M5.5, DECISIONS.md D-6,
     3→ * docs/phases/M5-SEAMS.md (`ApplyController`, pinned).
     4→ *
     5→ * ONE user gesture = ONE atomic `engine.batch` = ONE undo step = ONE version
     6→ * (AGENTS.md hard rule 6). The whole flow NEVER throws — every outcome is a
     7→ * typed union the plan card (M5-G) renders:
     8→ *
     9→ *  `apply(plan)`:
    10→ *    1. resolve each `resolutionRequests` entry against the LIVE layer list
    11→ *       (the executor's list_layers path: `comp.info {}` → `layers`) — a name
    12→ *       matching exactly one layer binds it; ambiguous or unknown names fail
    13→ *       BEFORE any engine mutation;
    14→ *    2. patch the `compiledBatch` placeholders with the resolved layer ids
    15→ *       (deterministic deep clone; null-prototype request map — the M4-N F-1
    16→ *       lesson: model-adjacent keys never touch `Object.prototype`);
    17→ *    3. `bridge.execute("engine.batch", batch)` — ONE call. Verified wire:
    18→ *       the bridge passes the params object through `effectcraft.execute` →
    19→ *       `engine.execute` → `batch.rs`, which reads `{steps, label?, atomic?}`
    20→ *       (extra keys such as `timeoutMs` ride inert upstream — `timeoutMs` is
    21→ *       §8.5 bound metadata the compiled batch carries, not an engine param);
    22→ *    4. on success `await autosave.saveNow("agent-apply")` (D-6) — this
    23→ *       uploads a NEW immutable version; the outcome's `version.id` is the
    24→ *       applied versionId;
    25→ *    5. `api.runs.applyRun(runId, {batchSha256, versionId, idempotencyKey})` —
    26→ *       sha256 over the exact patched batch payload (WebCrypto via the
    27→ *       editor-session sha256Hex, with its vector-tested sync fallback for
    28→ *       hosts without `crypto.subtle` — never a new dependency), idempotency
    29→ *       key generated ONCE per user gesture. Gateway errors map: 409 →
    30→ *       conflict (explains, never re-applies), 402 → insufficient credits,
    31→ *       else typed failure.
    32→ *
    33→ *  `undo(runId)`: `bridge.execute("edit.undo")` (the whole run folded into ONE
    34→ *   undo step by `batch.rs`) THEN `api.runs.undoRun(runId, {idempotencyKey})`.
    35→ *   If `edit.undo` fails ("nothing to undo" — the upstream `can_undo` guard,
    36→ *   edit.rs:12) the gateway is NOT told — a typed `{ok:false}` comes back and
    37→ *   the version rows are never touched (§9.3). The post-undo state is left to
    38→ *   the D-6 cadence + the before-close save (undo is not a D-6 moment).
    39→ *
    40→ *  `reapply(plan)`: only when the client's CURRENT version id equals the plan's
    41→ *   `baseVersionId` (the session port reads it); anything else is
    42→ *   `{ok:false, reason:"base-version-changed"}` — BP §M5.5 "otherwise explain".
    43→ *   NOTE for the integrator: after undo the session's current version is the
    44→ *   APPLIED version's successor chain, so re-apply-after-undo needs the host
    45→ *   page to restore the session's version pointer to the plan's base (a
    46→ *   session-level concern — see the worklog M5-F record).
    47→ *
    48→ * Cross-agent pins (wave 1, reconciled by the integrator):
    49→ *  - placeholder token: a whole-string `"$request:<requestId>"` in the compiled
    50→ *    batch params, replaced with the resolved engine layer id (number);
    51→ *  - `api.runs.applyRun/undoRun` — @studio/api-client names (M5-E adds them);
    52→ *  - the plan event carries `compiledBatch` + `resolutionRequests` +
    53→ *    `baseVersionId` (M5-C adds them to the wire; the SSE reader strips
    54→ *    unknown keys until that lands, so the client treats them as optional).
    55→ */

   326→/** Bridge/editor failure → typed outcome; disconnect codes are named as such. */
   327→function bridgeOutcome(error: unknown): {
   328→  reason: "engine-failed" | "editor-disconnected";
   329→  message: string;
   330→} {
   331→  const message = messageOf(error);
   332→  if (isBridgeError(error) && DISCONNECT_CODES.has(error.code)) {
   333→    return { reason: "editor-disconnected", message };
   334→  }
   335→  return { reason: "engine-failed", message };
   336→}

   338→/** The typed fields of an api-client throw (ApiError-shaped), without zod. */
   339→function apiFailure(error: unknown): {
   340→  status: number | null;
   341→  code: string | undefined;
   342→  message: string;
   343→} {
   344→  if (typeof error === "object" && error !== null && "status" in error) {
   345→    const record = error as { status?: unknown; code?: unknown; message?: unknown };
   346→    const status = typeof record.status === "number" ? record.status : null;
   347→    const code = typeof record.code === "string" && record.code !== "" ? record.code : undefined;
   348→    const message =
   349→      typeof record.message === "string" && record.message !== "" ? record.message : "";
   350→    return { status, code, message };
   351→  }
   352→  return { status: null, code: undefined, message: "" };
   353→}

   359→interface CompLayerEntry {
   360→  readonly id: number;
   361→  readonly name: string;
   362→}
   363→
   364→/** `comp.info {}` → the active composition's layer list (query.rs; read-only). */
   365→async function listLayers(bridge: EditorBridge): Promise<readonly CompLayerEntry[]> {
   366→  const info = await bridge.query<{ layers?: unknown }>("comp.info", {});
   367→  const raw = info?.layers;
   368→  if (!Array.isArray(raw)) return [];
   369→  const out: CompLayerEntry[] = [];
   370→  for (const entry of raw) {
   371→    if (typeof entry !== "object" || entry === null) continue;
   372→    const record = entry as Record<string, unknown>;
   373→    if (typeof record.id !== "number" || typeof record.name !== "string") continue;
   374→    out.push({ id: record.id, name: record.name });
   375→  }
   376→  return out;
   377→}
   378→
   379→/**
   380→ * Resolves every request against the live layer list. Read-only: the only
   381→ * bridge traffic is ONE `comp.info` query however many requests there are.
   382→ * Ambiguous/unknown names fail BEFORE anything executes.
   383→ */
   384→async function resolveRequests(
   385→  requests: readonly LayerResolutionRequest[],
   386→  bridge: EditorBridge,
   387→): Promise<
   388→  { ok: true; resolved: readonly ResolvedLayer[] } | { ok: false; outcome: ApplyOutcome }
   389→> {
   390→  let layers: readonly CompLayerEntry[];
   391→  try {
   392→    layers = await listLayers(bridge);
   393→  } catch (error) {
   394→    const failure = bridgeOutcome(error);
   395→    const outcome: ApplyOutcome =
   396→      failure.reason === "editor-disconnected"
   397→        ? { ok: false, reason: "editor-disconnected", message: failure.message }
   398→        : {
   399→            ok: false,
   400→            reason: "resolution-failed",
   401→            message: "The layers could not be listed to resolve this plan. Try again in a moment.",
   402→          };
   403→    return { ok: false, outcome };
   404→  }
   405→
   406→  const resolved: ResolvedLayer[] = [];
   407→  for (const request of requests) {
   408→    // Exact-name matches only — a fuzzy match could bind the WRONG layer.
   409→    const matches = layers.filter((layer) => layer.name === request.layerName);
   410→    const first = matches[0];
   411→    if (matches.length === 1 && first !== undefined) {
   412→      resolved.push({ requestId: request.requestId, layerId: first.id, layerName: first.name });
   413→      continue;
   414→    }
   415→    if (matches.length > 1) {
   416→      return {
   417→        ok: false,
   418→        outcome: {
   419→          ok: false,
   420→          reason: "ambiguous",
   421→          candidates: matches.map((layer) => `${layer.name} (layer ${layer.id})`),
   422→          message: `Layer "${request.layerName}" matches ${matches.length} layers. Say which one you mean.`,
   423→        },
   424→      };
   425→    }
   426→    return {
   427→      ok: false,
   428→      outcome: {
   429→        ok: false,
   430→        reason: "not-found",
   431→        layerName: request.layerName,
   432→        message: `No layer named "${request.layerName}" is in the composition.`,
   433→      },
   434→    };
   435→  }
   436→  return { ok: true, resolved };
   437→}

   439→// ---------------------------------------------------------------------------
   440→// Placeholder patch (deterministic deep clone; F-1-safe accumulators)
   441→// ---------------------------------------------------------------------------
   442→
   443→/**
   444→ * Replaces every whole-string `"$request:<requestId>"` placeholder in the batch
   445→ * params with the resolved layer id. Deterministic: only exact matches swap
   446→ * and the clone preserves object key order, so the sha256 is stable.
   447→ *
   448→ * The cloned objects are NULL-PROTOTYPE (M4-N F-1): plan params are
   449→ * model-adjacent — a parsed `{"__proto__": …}` key must ride as OWN data, not
   450→ * rebind the clone's prototype.
   451→ */
   452→function patchPlaceholders(
   453→  value: unknown,
   454→  resolvedById: Readonly<Record<string, number>>,
   455→): { patched: unknown; unresolved: string[] } {
   456→  const unresolved: string[] = [];
   457→  const walk = (node: unknown): unknown => {
   458→    if (typeof node === "string") {
   459→      if (!node.startsWith(RESOLVE_TOKEN_PREFIX)) return node;
   460→      const requestId = node.slice(RESOLVE_TOKEN_PREFIX.length);
   461→      const layerId = resolvedById[requestId];
   462→      if (layerId === undefined) {
   463→        unresolved.push(node);
   464→        return node;
   465→      }
   466→      return layerId;
   467→    }
   468→    if (Array.isArray(node)) return node.map(walk);
   469→    if (typeof node === "object" && node !== null) {
   470→      const out: Record<string, unknown> = Object.create(null);
   471→      for (const [key, child] of Object.entries(node)) out[key] = walk(child);
   472→      return out;
   473→    }
   474→    return node;
   475→  };
   476→  return { patched: walk(value), unresolved };
   477→}

   479→/** Narrow hand-rolled validation of the compiled batch (no zod in apps/web). */
   480→function validCompiledBatch(batch: unknown): AgentBatch | null {
   481→  if (typeof batch !== "object" || batch === null) return null;
   482→  const record = batch as Record<string, unknown>;
   483→  if (typeof record.label !== "string" || record.label.length === 0) return null;
   484→  if (!Array.isArray(record.steps) || record.steps.length === 0) return null;
   485→  const steps: BatchStep[] = [];
   486→  for (const step of record.steps) {
   487→    if (typeof step !== "object" || step === null) return null;
   488→    const stepRecord = step as Record<string, unknown>;
   489→    if (typeof stepRecord.command !== "string" || stepRecord.command.length === 0) return null;
   490→    if (typeof stepRecord.params !== "object" || stepRecord.params === null) return null;
   491→    steps.push({
   492→      command: stepRecord.command,
   493→      params: stepRecord.params as Record<string, unknown>,
   494→    });
   495→  }
   496→  const timeoutMs =
   497→    typeof record.timeoutMs === "number" &&
   498→    Number.isInteger(record.timeoutMs) &&
   499→    record.timeoutMs > 0
   500→      ? record.timeoutMs
   501→      : undefined;
   502→  // Agent batches are ALWAYS atomic (§8.5) — forced regardless of the wire.
   503→  return timeoutMs === undefined
   504→    ? { label: record.label, atomic: true, steps }
   505→    : { label: record.label, atomic: true, steps, timeoutMs };
   506→}

   512→/** The exact params object handed to `bridge.execute("engine.batch", …)`. */
   513→export function engineBatchPayload(batch: AgentBatch): {
   514→  label: string;
   515→  atomic: true;
   516→  steps: readonly BatchStep[];
   517→  timeoutMs?: number;
   518→} {
   519→  return batch.timeoutMs === undefined
   520→    ? { label: batch.label, atomic: true, steps: batch.steps }
   521→    : { label: batch.label, atomic: true, steps: batch.steps, timeoutMs: batch.timeoutMs };
   522→}

   524→/**
   525→ * sha256 over the serialized engine payload — the exact shape the editor
   526→ * receives (JSON key order is deterministic: the payload object is built with
   527→ * a fixed key order and the patched params preserve theirs).
   528→ */
   529→export async function batchSha256Hex(payload: unknown): Promise<string> {
   530→  const serialized = JSON.stringify(payload) ?? "null";
   531→  return sha256Hex(new TextEncoder().encode(serialized));
   532→}

   540→
   541→  const notify = (event: ApplyEngineEvent): void => {
   542→    const callback = options.onEvent;
   543→    if (callback === undefined) return;
   544→    try {
   545→      callback(event);
   546→    } catch {
   547→      /* a host callback must never break the apply engine */
   548→    }
   549→  };
   550→
   551→  const failApply = (runId: string, outcome: ApplyOutcome): ApplyOutcome => {
   552→    notify({ type: "apply.failed", runId, outcome });
   553→    return outcome;
   554→  };
   555→
   556→  /** The shared execute-then-save-then-report path (apply + reapply). */
   557→  const runApply = async (plan: ApplicablePlan): Promise<ApplyOutcome> => {
   558→    // One gesture = one idempotency key (law #4), whatever happens below.
   559→    const idempotencyKey = crypto.randomUUID();
   560→    notify({ type: "apply.started", runId: plan.runId });
   561→
   562→    const compiled = validCompiledBatch(plan.compiledBatch);
   563→    if (compiled === null) {
   564→      return failApply(plan.runId, {
   565→        ok: false,
   566→        reason: "invalid-batch",
   567→        message: "The plan's compiled batch is malformed. Ask the agent to plan again.",
   568→      });
   569→    }
   570→
   571→    // 1. Resolve layer names against the live list — read-only; on any
   572→    //    ambiguity NOTHING executes (no engine.batch, no save, no applyRun).
   573→    let resolvedById: Readonly<Record<string, number>> = {};
   574→    let resolvedForWire: readonly PostedResolvedLayer[] = [];
   575→    const requests = plan.resolutionRequests ?? [];
   576→    if (requests.length > 0) {
   577→      const resolution = await resolveRequests(requests, bridge);
   578→      if (!resolution.ok) return failApply(plan.runId, resolution.outcome);
   579→      notify({ type: "apply.resolved", runId: plan.runId, resolved: resolution.resolved });
   580→      // Null-prototype accumulator (M4-N F-1): request ids are model-adjacent
   581→      // strings — on a plain `{}` a `__proto__` id would hit the accessor.
   582→      const map = Object.create(null) as Record<string, number>;
   583→      for (const entry of resolution.resolved) map[entry.requestId] = entry.layerId;
   584→      resolvedById = map;
   585→      // §8.7/M5-E: the name→id bindings ride the applyRun body (auditable
   586→      // server-side even though the batch already ran client-side).
   587→      resolvedForWire = resolution.resolved.map((entry) => ({
   588→        requestId: entry.requestId,
   589→        layerId: entry.layerId,
   590→      }));
   591→    }
   592→
   593→    // 2. Patch the placeholders with the resolved ids (deterministic clone).
   594→    //    An unbound `$request:` token must never reach the engine as a string.
   595→    const patchedSteps: BatchStep[] = [];
   596→    const unresolved: string[] = [];
   597→    for (const step of compiled.steps) {
   598→      const patch = patchPlaceholders(step.params, resolvedById);
   599→      unresolved.push(...patch.unresolved);
   600→      patchedSteps.push({
   601→        command: step.command,
   602→        params: patch.patched as Record<string, unknown>,
   603→      });
   604→    }
   605→    if (unresolved.length > 0) {
   606→      return failApply(plan.runId, {
   607→        ok: false,
   608→        reason: "resolution-failed",
   609→        message: `The plan still references unresolved layers (${unresolved.join(", ")}). Ask the agent to name the layer explicitly.`,
   610→      });
   611→    }
   612→
   613→    const payload = engineBatchPayload({ ...compiled, steps: patchedSteps });
   614→    const batchSha256 = await batchSha256Hex(payload);
   615→
   616→    // 3. ONE engine.batch — law #6 (ONE call, atomic, one undo step).
   617→    try {
   618→      await bridge.execute("engine.batch", payload);
   619→    } catch (error) {
   620→      const failure = bridgeOutcome(error);
   621→      return failApply(plan.runId, {
   622→        ok: false,
   623→        reason: failure.reason,
   624→        message: failure.message,
   625→      });
   626→    }
   627→
   628→    // 4. D-6: upload the applied state as a NEW version (awaited BEFORE the
   629→    //    applyRun report so the gateway always sees a version that exists).
   630→    autosave.markDirty?.(AGENT_APPLY_SAVE_REASON);
   631→    const saved: SaveOutcome = await autosave.saveNow(AGENT_APPLY_SAVE_REASON);
   632→    if (saved.saved !== true || saved.version === undefined) {
   633→      return failApply(
   634→        plan.runId,
   635→        saved.conflict === true
   636→          ? {
   637→              ok: false,
   638→              reason: "save-conflict",
   639→              message:
   640→                "The project changed elsewhere while the agent was working. Reload the latest version to continue — the edit is in the editor only.",
   641→            }
   642→          : {
   643→              ok: false,
   644→              reason: "save-failed",
   645→              message:
   646→                "The edit was applied but could not be saved as a version. It is in the editor only.",
   647→            },
   648→      );
   649→    }
   650→    const versionId = saved.version.id;
   651→
   652→    // 5. Report to the gateway (ledger commit + run.applied, idempotent).
   653→    try {
   654→      const response = await api.runs.applyRun(plan.runId, {
   655→        batchSha256,
   656→        versionId,
   657→        idempotencyKey,
   658→        ...(resolvedForWire.length > 0 ? { resolvedLayers: resolvedForWire } : {}),
   659→      });
   660→      const outcome: ApplyOutcome = {
   661→        ok: true,
   662→        versionId: typeof response.versionId === "string" ? response.versionId : versionId,
   663→        creditsCommitted: response.creditsCommitted,
   664→      };
   665→      notify({
   666→        type: "apply.completed",
   667→        runId: plan.runId,
   668→        versionId: outcome.versionId,
   669→        creditsCommitted: outcome.creditsCommitted,
   670→      });
   671→      return outcome;
   672→    } catch (error) {
   673→      const failure = apiFailure(error);
   674→      if (failure.status === 409) {
   675→        return failApply(plan.runId, {
   676→          ok: false,
   677→          reason: "conflict",
   678→          message:
   679→            "This plan was already applied or the run changed on the server. Nothing was re-executed.",
   680→        });
   681→      }
   682→      if (failure.status === 402) {
   683→        return failApply(plan.runId, {
   684→          ok: false,
   685→          reason: "insufficient-credits",
   686→          message:
   687→            "The edit and its version are saved, but there were not enough credits to commit this run.",
   688→        });
   689→      }
   690→      return failApply(plan.runId, {
   691→        ok: false,
   692→        reason: "failed",
   693→        message:
   694→          failure.message !== ""
   695→            ? failure.message
   696→            : "The edit and its version are saved, but reporting it to the server failed.",
   697→        code: failure.code,
   698→      });
   699→    }
   700→  };
   701→
   702→  return {
   703→    apply: (plan: ApplicablePlan): Promise<ApplyOutcome> => runApply(plan),
   704→
   705→    async undo(runId: string): Promise<UndoOutcome> {
   706→      notify({ type: "undo.started", runId });
   707→      // ONE edit.undo reverts the WHOLE run (batch.rs folds the steps).
   708→      try {
   709→        await bridge.execute("edit.undo");
   710→      } catch (error) {
   711→        const failure = bridgeOutcome(error);
   712→        const outcome: UndoOutcome = NOTHING_TO_UNDO_RE.test(failure.message)
   713→          ? { ok: false, reason: "nothing-to-undo", message: "There is nothing to undo." }
   714→          : { ok: false, reason: failure.reason, message: failure.message };
   715→        // The gateway is NOT told — undoRun would 409 on an unapplied run and
   716→        // the version rows must stay untouched (§9.3).
   717→        notify({ type: "undo.failed", runId, outcome });
   718→        return outcome;
   719→      }
   720→      try {
   721→        await api.runs.undoRun(runId, { idempotencyKey: crypto.randomUUID() });
   722→        notify({ type: "undo.completed", runId });
   723→        return { ok: true, undone: true };
   724→      } catch (error) {
   725→        const failure = apiFailure(error);
   726→        const outcome: UndoOutcome =
   727→          failure.status === 409
   728→            ? {
   729→                ok: false,
   730→                reason: "conflict",
   731→                message:
   732→                  "The server says this run was not applied, so there is nothing to undo there.",
   733→              }
   734→            : {
   735→                ok: false,
   736→                reason: "failed",
   737→                message:
   738→                  failure.message !== ""
   739→                    ? failure.message
   740→                    : "The undo happened in the editor but could not be reported.",
   741→                code: failure.code,
   742→              };
   743→        notify({ type: "undo.failed", runId, outcome });
   744→        return outcome;
   745→      }
   746→    },
   747→
   748→    async reapply(plan: ApplicablePlan): Promise<ApplyOutcome> {
   749→      // §M5.5: only while the session still sits on the plan's base version.
   750→      const current = options.session?.currentVersionId() ?? null;
   751→      if (plan.baseVersionId === undefined || current === null || current !== plan.baseVersionId) {
   752→        return failApply(plan.runId, {
   753→          ok: false,
   754→          reason: "base-version-changed",
   755→          message:
   756→            "The project changed since this plan was made, so re-applying it is not safe. Ask the agent to plan again.",
   757→        });
   758→      }
   759→      return await runApply(plan);
   760→    },
   761→  };
   762→}
