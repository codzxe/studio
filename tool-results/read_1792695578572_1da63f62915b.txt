    64→const encoder = new TextEncoder();

    66→function frame(data: unknown): string {
    67→  return `data: ${JSON.stringify(data)}\n\n`;
    68→}
    69→function textDelta(content: string): string {
    70→  return frame({ choices: [{ delta: { content } }] });
    71→}
    72→function toolCallStart(index: number, id: string, name: string): string {
    73→  return frame({
    74→    choices: [
    75→      {
    76→        delta: {
    77→          tool_calls: [{ index, id, type: "function", function: { name, arguments: "" } }],
    78→        },
    79→      },
    80→    ],
    81→  });
    82→}
    83→function toolCallArgs(index: number, args: string): string {
    84→  return frame({
    85→    choices: [{ delta: { tool_calls: [{ index, function: { arguments: args } }] } }],
    86→  });
    87→}
    88→function finish(reason: string): string {
    89→  return frame({ choices: [{ delta: {}, finish_reason: reason }] });
    90→}
    91→const DONE = "data: [DONE]\n\n";

    93→function sseResponse(frames: readonly string[]): Response {
    94→  const body = new ReadableStream<Uint8Array>({
    95→    start(controller) {
    96→      for (const chunk of frames) controller.enqueue(encoder.encode(chunk));
    97→      controller.close();
    98→    },
    99→  });
   100→  return new Response(body, { status: 200, headers: { "content-type": "text/event-stream" } });
   101→}

   104→  readonly url: string;
   105→  readonly body: {
   106→    model: string;

   113→function scriptedFetch(script: ReadonlyArray<Response>): typeof fetch & { calls: RecordedCall[] } {
   114→  const calls: RecordedCall[] = [];
   115→  let index = 0;
   116→  const stub = (async (url: unknown, init?: RequestInit): Promise<Response> => {
   117→    calls.push({
   118→      url: String(url),
   119→      body: JSON.parse(String(init?.body)) as RecordedCall["body"],
   120→    });
   121→    const next = script[index];
   122→    index += 1;
   123→    if (next === undefined) throw new Error("script exhausted");
   124→    return next;
   125→  }) as typeof fetch & { calls: RecordedCall[] };
   126→  stub.calls = calls;
   127→  return stub;
   128→}

   134→class FakeStream implements AgentRunStream {
   135→  readonly events: RunStreamPayload[] = [];
   136→  readonly expected: Array<{ toolCallId: string; tool: string }> = [];
   137→  readonly controller = new AbortController();
   138→  waitMode: "pending" | "auto" = "pending";
   139→  private readonly waits = new Map<string, (result: ToolWaitResult) => void>();
   140→  private readonly early = new Map<string, unknown>();
   141→
   142→  get signal(): AbortSignal {
   143→    return this.controller.signal;
   144→  }
   145→
   146→  emit(payload: RunStreamPayload): StudioStreamEvent {
   147→    this.events.push(payload);
   148→    return { ...payload, runId: "run_inj", seq: this.events.length - 1 };
   149→  }
   150→
   151→  async yieldToStream(): Promise<void> {
   152→    await new Promise((resolve) => setImmediate(resolve));
   153→  }
   154→
   155→  expectToolCall(toolCallId: string, tool: string): void {
   156→    this.expected.push({ toolCallId, tool });
   157→  }
   158→
   159→  waitForToolResult(toolCallId: string): Promise<ToolWaitResult> {
   160→    if (this.waitMode === "auto") return Promise.resolve({ ok: true, result: { stub: true } });
   161→    if (this.early.has(toolCallId)) {
   162→      return Promise.resolve({ ok: true, result: this.early.get(toolCallId) });
   163→    }
   164→    return new Promise((resolve) => {
   165→      this.waits.set(toolCallId, resolve);
   166→    });
   167→  }
   168→
   169→  resolveToolResult(toolCallId: string, result: unknown): void {
   170→    const resolve = this.waits.get(toolCallId);
   171→    if (resolve === undefined) {
   172→      this.early.set(toolCallId, result);
   173→      return;
   174→    }
   175→    this.waits.delete(toolCallId);
   176→    resolve({ ok: true, result });
   177→  }
   178→}

   180→async function runToOutcome(
   181→  runner: ReturnType<typeof createLiteLLMRunner>,
   182→  run: AgentRun,
   183→  stream: FakeStream,
   184→): Promise<AgentRunOutcome> {
   185→  const outcome = await runner.start(run, stream);
   186→  if (outcome.kind === "failed") {
   187→    stream.emit({
   188→      type: "run.failed",
   189→      code: outcome.code,
   190→      message: outcome.message,
   191→      retryable: outcome.retryable,
   192→    });
   193→  } else if (outcome.kind === "cancelled") {
   194→    stream.emit({ type: "run.cancelled" });
   195→  } else {
   196→    stream.emit({ type: "run.completed" });
   197→  }
   198→  return outcome;
   199→}

   243→const READ_TOOL_IDS = TOOL_LIST.filter((tool) => tool.kind === "read").map((tool) => tool.id);
   244→const WRITE_TOOL_IDS = TOOL_LIST.filter((tool) => tool.kind === "write").map((tool) => tool.id);

   246→/** §12.2 law: the provider is ONLY ever offered the 5 curated read tools. */
   247→function expectOnlyReadToolsEverSent(calls: readonly RecordedCall[]): void {
   248→  expect(calls.length).toBeGreaterThan(0);
   249→  for (const call of calls) {
   250→    const tools = call.body.tools;
   251→    if (tools === undefined) continue; // structured-output calls carry no tools
   252→    expect(tools.map((tool) => tool.function.name)).toEqual(READ_TOOL_IDS);
   253→  }
   254→  // Belt and braces: no write tool id ever appears in any tools array.
   255→  for (const call of calls) {
   256→    const serialized = JSON.stringify(call.body.tools ?? []);
   257→    for (const writeId of WRITE_TOOL_IDS) {
   258→      expect(serialized.includes(writeId)).toBe(false);
   259→    }
   260→  }
   261→}

   263→/** §12.2 law: no raw engine command string is ever put on the wire. */
   264→function expectNoEngineCommandStrings(calls: readonly RecordedCall[]): void {
   265→  for (const call of calls) {
   266→    const serialized = JSON.stringify(call.body);
   267→    expect(serialized.includes("engine.execute")).toBe(false);
   268→    expect(serialized.includes("engine.batch")).toBe(false);
   269→  }

   276→  expect(({} as Record<string, unknown>).polluted).toBeUndefined();
   277→  expect(Object.getPrototypeOf(probe).polluted).toBeUndefined();
