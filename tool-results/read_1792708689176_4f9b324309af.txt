   282→   282→function subscriptionViewOf(
   283→   283→  object: StripeEventLike["data"]["object"],
   284→   284→): SubscriptionEventView | null {
   285→   285→  const record = object as unknown as Record<string, unknown>;
   286→   286→  if (!isRecord(record)) return null;
   287→   287→  const status = nonEmptyString(record.status);
   288→   288→  if (status === null) return null;
   289→   289→  let currentPeriodEnd: number | null = null;
   290→   290→  let priceId: string | null = null;
   291→   291→  const items = record.items;
   292→   292→  if (isRecord(items) && Array.isArray(items.data) && items.data.length > 0) {
   293→   293→    const item = items.data[0];
   294→   294→    if (isRecord(item)) {
   295→   295→      const periodEnd = item.current_period_end;
   296→   296→      if (typeof periodEnd === "number" && Number.isFinite(periodEnd)) {
   297→   297→        currentPeriodEnd = periodEnd;
   298→   298→      }
   299→   299→      priceId = priceIdOf(item.price);
   300→   300→    }
   301→   301→  }
   302→   302→  const trialEnd = record.trial_end;
   303→   303→  return {
   304→   304→    customer: nonEmptyString(record.customer),
   305→   305→    status,
   306→   306→    trialEnd: typeof trialEnd === "number" && Number.isFinite(trialEnd) ? trialEnd : null,
   307→   307→    cancelAtPeriodEnd: record.cancel_at_period_end === true,
   308→   308→    currentPeriodEnd,
   309→   309→    priceId,
   310→   310→  };
   311→   311→}

   418→   418→  async function ensureWorkspaceCustomer(
   419→   419→    c: AppContext,
   420→   420→    workspace: WorkspaceContext,
   421→   421→    stripe: StripeSeam,
   422→   422→  ): Promise<string> {
   423→   423→    if (subscriptions !== null) {

   426→   426→        return existing.stripeCustomerId;

   428→   428→    }
   429→   429→    const legacy = parseOrgMetadata(workspace.organization.metadata).stripeCustomerId;
   430→   430→    if (typeof legacy === "string" && legacy !== "") {
   431→   431→      if (subscriptions !== null) {

   439→   439→    });
   440→   440→    if (subscriptions !== null) {

   845→   845→  async function guardedTransition(
   846→   846→    organizationId: string,
   847→   847→    input: SubscriptionTransitionInput,
   848→   848→  ): Promise<boolean> {
   849→   849→    if (subscriptions === null) return false;
   850→   850→    try {
   851→   851→      await subscriptions.transitionStatus(organizationId, input);
   852→   852→      return true;
   853→   853→    } catch (error) {
   854→   854→      if (isDbError(error) && (error.code === "CONFLICT" || error.code === "NOT_FOUND")) {
   855→   855→        return false;
   856→   856→      }
   857→   857→      throw error;
   858→   858→    }

   861→   861→  /** Ledger grant for a top-up from external µc metadata (D-12 + int32 ceiling). */
   862→   862→  async function grantTopUp(
   863→   863→    organizationId: string,
   864→   864→    event: StripeEventLike,
   865→   865→    metadata: Record<string, string> | null | undefined,
   866→   866→    idempotencyKey: string,
   867→   867→  ): Promise<boolean> {
   868→   868→    const raw = metadata?.microcredits;
   869→   869→    if (typeof raw !== "string" || raw === "") return false;
   870→   870→    const microcredits = Number.parseInt(raw, 10);
   871→   871→    if (!Number.isInteger(microcredits) || microcredits <= 0 || microcredits > MAX_LEDGER_AMOUNT) {
   872→   872→      // Hostile/overflowing amounts are audited-ignored — never a poison retry.
   873→   873→      return false;
   874→   874→    }
   875→   875→    await deps.repos.ledger.grant(organizationId, {
   876→   876→      amount: microcredits,
   877→   877→      idempotencyKey,
   878→   878→      metadata: {
   879→   879→        stripeEventId: event.id,
   880→   880→        kind: "topup",
   881→   881→        packId: metadata?.packId ?? null,
   882→   882→      },
   883→   883→    });
   884→   884→    return true;
   885→   885→  }

   890→   890→    object: StripeEventLike["data"]["object"],
   891→   891→  ): Promise<void> {
   892→   892→    const customerId = nonEmptyString(object.customer);
   893→   893→    if (subscriptions === null || customerId === null) return;
   894→   894→    const fingerprint = nonEmptyString(object.metadata?.cardFingerprint);
   895→   895→    await subscriptions.ensureStripeCustomer(organizationId, {
   896→   896→      stripeCustomerId: customerId,
   897→   897→      ...(fingerprint !== null ? { cardFingerprint: fingerprint } : {}),
   898→   898→    });
   899→   899→  }

   903→   903→    stripe: StripeSeam,

   905→   905→  ): Promise<SubscriptionEventView | null> {
   906→   906→    const snapshot = await stripe.subscription(customer);
   907→   907→    if (!snapshot.ok) return null;
   908→   908→    return {
   909→   909→      customer,
   910→   910→      status: snapshot.data.status,
   911→   911→      trialEnd: snapshot.data.trialEnd ?? null,
   912→   912→      cancelAtPeriodEnd: snapshot.data.cancelAtPeriodEnd,
   913→   913→      currentPeriodEnd: snapshot.data.currentPeriodEnd,
   914→   914→      priceId: snapshot.data.priceId,
   915→   915→    };
   916→   916→  }

   918→   918→  async function handleCheckoutCompleted(
   919→   919→    stripe: StripeSeam,
   920→   920→    event: StripeEventLike,
   921→   921→  ): Promise<WebhookOutcome> {
   922→   922→    const object = event.data.object;
   923→   923→    const organizationId = organizationIdOf(object);
   924→   924→    if (organizationId === null) return { ignored: "unattributed" };
   925→   925→
   926→   926→    // Customer + payment-method bookkeeping (§M6 task 5): the repo is the
   927→   927→    // durable customer mapping; cardFingerprint metadata is the fake's card
   928→   928→    // model (real Stripe attaches the fingerprint with the payment method).
   929→   929→    await recordCustomer(organizationId, object);
   930→   930→
   931→   931→    const kind = object.metadata?.kind ?? null;
   932→   932→    if (kind === "topup") {
   933→   933→      const granted = await grantTopUp(
   934→   934→        organizationId,
   935→   935→        event,
   936→   936→        object.metadata,
   937→   937→        `stripe:${event.id}`,
   938→   938→      );
   939→   939→      return granted ? { handled: "topup.granted" } : { ignored: "topup.amount-invalid" };
   940→   940→    }
   941→   941→    if (kind !== "subscription") {
   942→   942→      return { ignored: "unknown-kind" };
   943→   943→    }
   944→   944→
   945→   945→    // The session object carries no period data — resync the canonical
   946→   946→    // snapshot from the provider (D-4: the trial status rides on the sub).
   947→   947→    const customer = nonEmptyString(object.customer);
   948→   948→    if (customer === null) return { ignored: "no-customer" };
   949→   949→    const snapshot = await stripe.subscription(customer);
   950→   950→    if (!snapshot.ok) return { ignored: "subscription-unavailable" };
   951→   951→    const planId = planIdFromMetadata(object) ?? planIdFromPrice(deps.env, snapshot.data.priceId);
   952→   952→    if (planId === null) return { ignored: "unknown-plan" };
   953→   953→    if (subscriptions !== null) {
   954→   954→      // First row (or resubscription): upsert is race-proof on the unique
   955→   955→      // organization_id — concurrent completions converge on one row.
   956→   956→      await subscriptions.upsertState(organizationId, {
   957→   957→        planId,
   958→   958→        status: mapStripeStatus(snapshot.data.status),
   959→   959→        stripeSubscriptionId: snapshot.data.subscriptionId,
   960→   960→        currentPeriodEnd: toDate(snapshot.data.currentPeriodEnd),
   961→   961→        trialEnd: toDate(snapshot.data.trialEnd ?? null),
   962→   962→        cancelAtPeriodEnd: snapshot.data.cancelAtPeriodEnd,
   963→   963→      });
   964→   964→    }
   965→   965→    return { handled: "subscription.synced" };
   966→   966→  }
   967→   967→
   968→   968→  /**
   969→   969→   * Shared guarded-transition driver for subscription lifecycle events
   970→   970→   * (customer.subscription.updated/deleted). The event's object is the
   971→   971→   * HISTORICAL truth (webhooks replay history); minimal objects resync from
   972→   972→   * the provider.
   973→   973→   */
   974→   974→  async function applySubscriptionTransition(
   975→   975→    stripe: StripeSeam,
   976→   976→    event: StripeEventLike,
   977→   977→    forcedTarget?: SubscriptionStatus,
   978→   978→  ): Promise<WebhookOutcome> {
   979→   979→    const object = event.data.object;
   980→   980→    const organizationId = organizationIdOf(object);
   981→   981→    if (organizationId === null) return { ignored: "unattributed" };
   982→   982→    if (subscriptions === null) return { ignored: "no-persistence" };
   983→   983→
   984→   984→    let view = subscriptionViewOf(object);
   985→   985→    if (view === null || view.currentPeriodEnd === null) {
   986→   986→      const customer = view?.customer ?? nonEmptyString(object.customer);
   987→   987→      if (customer !== null) {
   988→   988→        view = await resyncView(stripe, customer);
   989→   989→      }
   990→   990→    }
   991→   991→    if (view === null) return { ignored: "unparsable" };
   992→   992→    const target = forcedTarget ?? mapStripeStatus(view.status);
   993→   993→
   994→   994→    const row = await subscriptions.getByOrganization(organizationId);
   995→   995→    if (row === null) {
   996→   996→      // A subscription event for a workspace we never tracked: attribution
   997→   997→      // exists (metadata) but there is no state to transition — audited.
   998→   998→      return { ignored: "no-subscription" };
   999→   999→    }
  1000→  1000→
  1001→  1001→    // From-lists (the guarded state machine; heals land via invoice.paid):
  1002→  1002→    //  active   ← trialing | active | incomplete   (trial conversion, renewal)
  1003→  1003→    //  grace    ← active | grace | past_due | incomplete (past-due dunning)
  1004→  1004→    //  canceled ← any live state (BP §M6.2: status only — NO work deletion)
  1005→  1005→    //  trialing ← trialing | incomplete (idempotent re-pin)
  1006→  1006→    //  incomplete ← trialing | active | incomplete
  1007→  1007→    //  past_due ← grace (legacy rows only; Stripe past_due maps to grace)
  1008→  1008→    const FROM_LISTS: Record<SubscriptionStatus, readonly SubscriptionStatus[]> = {
  1009→  1009→      active: ["trialing", "active", "incomplete"],
  1010→  1010→      grace: ["active", "grace", "past_due", "incomplete"],
  1011→  1011→      canceled: ["trialing", "active", "grace", "past_due", "incomplete"],
  1012→  1012→      trialing: ["trialing", "incomplete"],
  1013→  1013→      incomplete: ["trialing", "active", "incomplete"],
  1014→  1014→      past_due: ["grace"],
  1015→  1015→    };
  1016→  1016→    const from = FROM_LISTS[target];
  1017→  1017→    if (!from.includes(row.status)) {
  1018→  1018→      return { ignored: "transition-guarded" };
  1019→  1019→    }
  1020→  1020→
  1021→  1021→    const periodEndMs = view.currentPeriodEnd !== null ? view.currentPeriodEnd * 1000 : null;
  1022→  1022→    // Never move the period BACKWARDS (an out-of-order renewal must not
  1023→  1023→    // regress the row), and never write a null over a known period.
  1024→  1024→    const periodAdvances =
  1025→  1025→      periodEndMs !== null &&
  1026→  1026→      (row.currentPeriodEnd === null || periodEndMs >= row.currentPeriodEnd.getTime());
  1027→  1027→
  1028→  1028→    // Plan pinning (§M6 task 2 "never instant feature loss"): the event's new
  1029→  1029→    // price only becomes the row's plan when it is an UPGRADE (features grow)
  1030→  1030→    // or the period actually advanced (the change took effect). A downgrade
  1031→  1031→    // scheduled at period end keeps the row's plan until then.
  1032→  1032→    const targetPlan = planIdFromPrice(deps.env, view.priceId) ?? planIdFromMetadata(object);
  1033→  1033→    const planId =
  1034→  1034→      targetPlan !== null &&
  1035→  1035→      target !== "canceled" &&
  1036→  1036→      (planRank(targetPlan) > planRank(row.planId) || periodAdvances)
  1037→  1037→        ? targetPlan
  1038→  1038→        : null;
  1039→  1039→
  1040→  1040→    const graceEnd =
  1041→  1041→      target === "grace"
  1042→  1042→        ? periodEndMs !== null
  1043→  1043→          ? new Date(periodEndMs + GRACE_PERIOD_DAYS * DAY_MS)
  1044→  1044→          : undefined
  1045→  1045→        : target === "active" || target === "canceled"
  1046→  1046→          ? null
  1047→  1047→          : undefined;
  1048→  1048→    const applied = await guardedTransition(organizationId, {
  1049→  1049→      from,
  1050→  1050→      to: target,
  1051→  1051→      ...(planId !== null ? { planId } : {}),
  1052→  1052→      ...(periodAdvances ? { currentPeriodEnd: new Date(periodEndMs) } : {}),
  1053→  1053→      ...(view.trialEnd !== null ? { trialEnd: new Date(view.trialEnd * 1000) } : {}),
  1054→  1054→      ...(graceEnd !== undefined ? { graceEnd } : {}),
  1055→  1055→      cancelAtPeriodEnd: view.cancelAtPeriodEnd,
  1056→  1056→    });
  1057→  1057→    return applied ? { handled: `subscription.${target}` } : { ignored: "transition-guarded" };
  1058→  1058→  }

  1060→  1060→  async function handleSubscriptionUpdated(
  1061→  1061→    stripe: StripeSeam,
  1062→  1062→    event: StripeEventLike,
  1063→  1063→  ): Promise<WebhookOutcome> {
  1064→  1064→    return applySubscriptionTransition(stripe, event);
  1065→  1065→  }

  1068→  1068→    stripe: StripeSeam,
  1069→  1069→    event: StripeEventLike,
  1070→  1070→  ): Promise<WebhookOutcome> {
  1071→  1071→    // BP §M6.2: cancellation writes STATUS ONLY. Projects, versions and
  1072→  1072→    // ledger rows are never touched by the webhook — no work deletion, ever.

  1076→  1076→  async function handleInvoice(event: StripeEventLike): Promise<WebhookOutcome> {
  1077→  1077→    const object = event.data.object;
  1078→  1078→    const organizationId = organizationIdOf(object);
  1079→  1079→    if (organizationId === null) return { ignored: "unattributed" };
  1080→  1080→    const invoices = invoicesRepo(deps.repos);
  1081→  1081→    if (invoices === null) return { ignored: "no-persistence" };
  1082→  1082→
  1083→  1083→    const record = object as unknown as Record<string, unknown>;
  1084→  1084→    const invoiceId = nonEmptyString(record.id);
  1085→  1085→    if (invoiceId === null) return { ignored: "invoice-unparsable" };
  1086→  1086→    const amountRaw = record.amount_paid ?? record.amount_due;
  1087→  1087→    const amountCents =
  1088→  1088→      typeof amountRaw === "number" && Number.isFinite(amountRaw) && amountRaw >= 0
  1089→  1089→        ? Math.round(amountRaw)
  1090→  1090→        : 0;
  1091→  1091→    const currencyRaw = nonEmptyString(record.currency);
  1092→  1092→    const periodStart = toDate(
  1093→  1093→      typeof record.period_start === "number" ? record.period_start : null,
  1094→  1094→    );
  1095→  1095→    const periodEnd = toDate(typeof record.period_end === "number" ? record.period_end : null);
  1096→  1096→
  1097→  1097→    const status = event.type === "invoice.paid" ? ("paid" as const) : ("void" as const);
  1098→  1098→    try {
  1099→  1099→      await invoices.record(organizationId, {
  1100→  1100→        stripeInvoiceId: invoiceId,
  1101→  1101→        amountCents,
  1102→  1102→        currency: currencyRaw ?? "usd",
  1103→  1103→        status,
  1104→  1104→        periodStart,
  1105→  1105→        periodEnd,
  1106→  1106→        hostedUrl: nonEmptyString(record.hosted_invoice_url),
  1107→  1107→        pdfUrl: nonEmptyString(record.invoice_pdf),
  1108→  1108→      });
  1109→  1109→    } catch (error) {
  1110→  1110→      if (isDbError(error) && (error.code === "VALIDATION_FAILED" || error.code === "CONFLICT")) {
  1111→  1111→        // A foreign/malformed invoice is audited-ignored — never a retry loop.
  1112→  1112→        return { ignored: "invoice-invalid" };
  1113→  1113→      }
  1114→  1114→      throw error;
  1115→  1115→    }
  1116→  1116→
  1117→  1117→    if (status !== "paid") {
  1118→  1118→      return { handled: "invoice.voided" };
  1119→  1119→    }
  1120→  1120→
  1121→  1121→    // Top-up grants invoiced outside Checkout (§M6 task 2): keyed on the
  1122→  1122→    // INVOICE id, so re-deliveries under fresh event ids cannot double-grant.
  1123→  1123→    let handled = "invoice.paid";
  1124→  1124→    if (object.metadata?.kind === "topup") {
  1125→  1125→      const granted = await grantTopUp(
  1126→  1126→        organizationId,
  1127→  1127→        event,
  1128→  1128→        object.metadata,
  1129→  1129→        `stripe:invoice:${invoiceId}:topup`,
  1130→  1130→      );
  1131→  1131→      if (granted) handled = "invoice.topup-granted";
  1132→  1132→    }
  1133→  1133→
  1134→  1134→    // Dunning heal (§M6 task 2 trial/renewal lifecycle): a PAID invoice while
  1135→  1135→    // the row sits in the grace window returns the subscription to active.
  1136→  1136→    if (subscriptions !== null) {
  1137→  1137→      const row = await subscriptions.getByOrganization(organizationId);
  1138→  1138→      if (row !== null && (row.status === "grace" || row.status === "past_due")) {
  1139→  1139→        const healed = await guardedTransition(organizationId, {
  1140→  1140→          from: ["grace", "past_due"],
  1141→  1141→          to: "active",
  1142→  1142→          ...(periodEnd !== null ? { currentPeriodEnd: periodEnd } : {}),
  1143→  1143→          graceEnd: null,
  1144→  1144→        });
  1145→  1145→        if (healed) handled = "invoice.paid-healed";
  1146→  1146→      }
  1147→  1147→    }
  1148→  1148→    return { handled };
  1149→  1149→  }
  1150→  1150→
  1151→  1151→  async function handleStripeEvent(
  1152→  1152→    stripe: StripeSeam,
  1153→  1153→    event: StripeEventLike,
  1154→  1154→  ): Promise<WebhookOutcome> {
  1155→  1155→    switch (event.type) {
  1156→  1156→      case "checkout.session.completed":
  1157→  1157→        return handleCheckoutCompleted(stripe, event);
  1158→  1158→      case "customer.subscription.updated":
  1159→  1159→        return handleSubscriptionUpdated(stripe, event);
  1160→  1160→      case "customer.subscription.deleted":
  1161→  1161→        return handleSubscriptionDeleted(stripe, event);
  1162→  1162→      case "invoice.paid":
  1163→  1163→      case "invoice.voided":
  1164→  1164→        return handleInvoice(event);
  1165→  1165→      default:
  1166→  1166→        // Unknown event types are 200 + audited-ignored: Stripe retries on
  1167→  1167→        // non-2xx, so an unhandled type must NEVER be a 500.
  1168→  1168→        return { ignored: "unhandled-type" };
  1169→  1169→    }
  1170→  1170→  }
  1171→  1171→
  1172→  1172→  return app;
  1173→  1173→}
