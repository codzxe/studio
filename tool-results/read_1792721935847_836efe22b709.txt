     1→     1→//! EffectCraft headless CLI: one-shot agent commands, frame rendering and the MCP server.
     2→     2→//!
     3→     3→//! ```text
     4→     4→//! effectcraft-cli info                                        project + engine summary

    52→    52→/// `println!` that never panics: see [`write_line`].
    53→    53→macro_rules! say {
    54→    54→    ($($t:tt)*) => {
    55→    55→        write_line(format_args!($($t)*))
    56→    56→    };
    57→    57→}

    59→    59→/// `eprintln!` that never panics: a diagnostic that can't be written is dropped.
    60→    60→macro_rules! note {
    61→    61→    ($($t:tt)*) => {{
    62→    62→        let _ = writeln!(std::io::stderr(), $($t)*);
    63→    63→    }};
    64→    64→}

    66→    66→/// Why stdout stopped taking output, once it has.
    67→    67→static STDOUT_FAILED: std::sync::OnceLock<std::io::ErrorKind> = std::sync::OnceLock::new();
    68→    68→
    69→    69→/// Write a line to stdout. Once that fails, later output is dropped but the command still
    70→    70→/// finishes its work (a `run … --save` sequence isn't cut short). A reader that closes stdout
    71→    71→/// early (`info --json | head`) has read what it wanted; any other failure is reported and makes
    72→    72→/// the exit status 1 ([`stdout_broken`]).
    73→    73→fn write_line(text: std::fmt::Arguments) {
    74→    74→    if STDOUT_FAILED.get().is_some() {
    75→    75→        return;
    76→    76→    }
    77→    77→    let r = {
    78→    78→        let mut out = std::io::stdout().lock();
    79→    79→        out.write_fmt(text).and_then(|()| out.write_all(b"\n"))
    80→    80→    };
    81→    81→    if let Err(e) = r {
    82→    82→        let _ = STDOUT_FAILED.set(e.kind());
    83→    83→        if e.kind() != std::io::ErrorKind::BrokenPipe {
    84→    84→            note!("effectcraft-cli: cannot write to stdout: {e}");
    85→    85→        }
    86→    86→    }
    87→    87→}
    88→    88→
    89→    89→/// Whether stdout failed for a reason other than its reader closing it early.
    90→    90→fn stdout_broken() -> bool {
    91→    91→    STDOUT_FAILED.get().is_some_and(|k| *k != std::io::ErrorKind::BrokenPipe)
    92→    92→}

   199→   199→struct Args {
   200→   200→    pos: Vec<String>,
   201→   201→    opts: Vec<(String, Option<String>)>,
   202→   202→    /// `--project F` or a positional `*.ecproj`.
   203→   203→    project: Option<String>,
   204→   204→}
   205→   205→
   206→   206→impl Args {
   207→   207→    fn parse(raw: Vec<String>) -> Result<Args, String> {
   208→   208→        let (mut pos, mut opts, mut unknown) = (vec![], vec![], vec![]);
   209→   209→        let mut it = raw.into_iter();
   210→   210→        while let Some(a) = it.next() {
   211→   211→            if a.starts_with("--") && a.len() > 2 {
   212→   212→                let (k, inline) = match a.split_once('=') {
   213→   213→                    Some((k, v)) => (k.to_string(), Some(v.to_string())),
   214→   214→                    None => (a, None),
   215→   215→                };
   216→   216→                if VALUED.contains(&k.as_str()) {
   217→   217→                    let v = inline.or_else(|| it.next()).ok_or_else(|| format!("{k} needs a value"))?;
   218→   218→                    opts.push((k, Some(v)));
   219→   219→                } else if !FLAGS.contains(&k.as_str()) {
   220→   220→                    unknown.push(format!("`{k}`"));
   221→   221→                } else if inline.is_some() {
   222→   222→                    return Err(format!("{k} takes no value"));
   223→   223→                } else {
   224→   224→                    opts.push((k, None));
   225→   225→                }
   226→   226→            } else {
   227→   227→                pos.push(a);
   228→   228→            }
   229→   229→        }
   230→   230→        // Before anything runs: a misspelt option must not run the command with defaults, or
   231→   231→        // leave its value behind as a positional (`--saveas x.ecproj` opening x.ecproj).
   232→   232→        if !unknown.is_empty() {
   233→   233→            let s = if unknown.len() == 1 { "" } else { "s" };
   234→   234→            return Err(format!("unknown option{s} {}", unknown.join(", ")));
   235→   235→        }
   236→   236→        let mut a = Args { pos, opts, project: None };
   237→   237→        a.project = match a.opt("--project") {
   238→   238→            Some(p) => Some(p.to_string()),
   239→   239→            None => a.pos.iter().position(|x| x.ends_with(".ecproj")).map(|i| a.pos.remove(i)),
   240→   240→        };
   241→   241→        Ok(a)
   242→   242→    }
   243→   243→    fn flag(&self, k: &str) -> bool {
   244→   244→        self.opts.iter().any(|(o, _)| o == k)
   245→   245→    }
   246→   246→    /// Every value of a repeatable option, in order.
   247→   247→    fn opts(&self, k: &str) -> Vec<&str> {
   248→   248→        self.opts.iter().filter(|(o, _)| o == k).filter_map(|(_, v)| v.as_deref()).collect()
   249→   249→    }
   250→   250→    fn opt(&self, k: &str) -> Option<&str> {
   251→   251→        self.opts.iter().rev().find(|(o, _)| o == k).and_then(|(_, v)| v.as_deref())
   252→   252→    }
   253→   253→    fn num(&self, k: &str) -> Result<Option<f64>, String> {
   254→   254→        self.opt(k).map(|v| v.parse::<f64>().map_err(|_| format!("{k}: not a number: {v}"))).transpose()
   255→   255→    }
   256→   256→}

   258→   258→/// A comp/layer reference: number → id, `-` → none (active), else a name / `#n`.
   259→   259→fn reference(s: &str) -> Option<Value> {
   260→   260→    match s {
   261→   261→        "-" | "" => None,
   262→   262→        _ => Some(s.parse::<u64>().map(Value::from).unwrap_or_else(|_| json!(s))),
   263→   263→    }
   264→   264→}
   265→   265→
   266→   266→/// JSON if it parses, else the bare string.
   267→   267→fn value_arg(s: &str) -> Value {
   268→   268→    serde_json::from_str(s).unwrap_or_else(|_| json!(s))
   269→   269→}

   271→   271→/// The command line as text. `std::env::args` panics on an argument that is not Unicode (a
   272→   272→/// Latin-1 file name on Linux). Every argument is used as a string, and a lossy copy would read or
   273→   273→/// write another file than the one named, so such an argument is a usage error naming it.
   274→   274→fn utf8_args(raw: impl IntoIterator<Item = std::ffi::OsString>) -> Result<Vec<String>, String> {
   275→   275→    raw.into_iter().map(|a| a.into_string().map_err(|a| format!("argument `{}` is not valid Unicode", a.to_string_lossy()))).collect()
   276→   276→}

   278→   278→fn main() {
   279→   279→    let raw = match utf8_args(std::env::args_os().skip(1)) {
   280→   280→        Ok(raw) => raw,
   281→   281→        Err(e) => fail_usage(&e),
   282→   282→    };
   283→   283→    if matches!(raw.first().map(String::as_str), Some("--version" | "-V" | "version")) {
   284→   284→        say!("effectcraft-cli {}", env!("CARGO_PKG_VERSION"));
   285→   285→        return;
   286→   286→    }
   287→   287→    if raw.is_empty() || raw.iter().any(|a| a == "-h" || a == "--help") {
   288→   288→        say!("{USAGE}");
   289→   289→        std::process::exit(if raw.is_empty() { 2 } else { 0 });
   290→   290→    }
   291→   291→    let mut args = match Args::parse(raw) {
   292→   292→        Ok(a) => a,
   293→   293→        Err(e) => fail_usage(&e),
   294→   294→    };
   295→   295→    let json_out = args.flag("--json");
   296→   296→    if args.pos.is_empty() {
   297→   297→        fail_usage("missing subcommand");
   298→   298→    }
   299→   299→    let cmd = args.pos.remove(0);
   300→   300→    match run(&cmd, &args, json_out) {
   301→   301→        Ok(()) if stdout_broken() => std::process::exit(1),
   302→   302→        Ok(()) => {}
   303→   303→        Err(Failure::Usage(e)) => fail_usage(&e),
   304→   304→        Err(Failure::Error(e)) => {
   305→   305→            if json_out {
   306→   306→                say!("{}", json!({"error": e}));
   307→   307→            }
   308→   308→            note!("effectcraft-cli {cmd}: {e}");
   309→   309→            std::process::exit(1);
   310→   310→        }
   311→   311→    }
   312→   312→}
   313→   313→
   314→   314→fn fail_usage(e: &str) -> ! {
   315→   315→    note!("effectcraft-cli: {e}\n{USAGE}");
   316→   316→    std::process::exit(2)
   317→   317→}

   338→   338→/// Build the backend: a bridge to the app, or a headless session with the requested project.
   339→   339→fn backend(args: &Args, default_demo: bool) -> Result<Backend, Failure> {
   340→   340→    if let Some(addr) = args.opt("--bridge") {
   341→   341→        if args.project.is_some() || args.flag("--demo") {
   342→   342→            return usage_err("--bridge drives the running app's project; use `exec file.open` to open another");
   343→   343→        }
   344→   344→        return Ok(Backend::bridge(addr)?);
   345→   345→    }
   346→   346→    let mut b = Backend::headless(session(args)?);
   347→   347→    if let Some(p) = &args.project {
   348→   348→        b.exec("file.open", json!({"path": p}))?;
   349→   349→    } else if args.flag("--demo") || (default_demo && !args.flag("--empty")) {
   350→   350→        b.exec("file.openDemoProject", json!({}))?;
   351→   351→    }
   352→   352→    Ok(b)
   353→   353→}

   355→   355→/// A wired session; `--gpu` attaches the GPU compositor (renders then follow the project's
   356→   356→/// renderer setting, Mercury GPU Acceleration by default).
   357→   357→fn session(args: &Args) -> Result<effectcraft_engine::Session, Failure> {
   358→   358→    let mut s = effectcraft_host::session();
   359→   359→    if args.flag("--autosave") {
   360→   360→        let dir = effectcraft_host::config_dir().ok_or_else(|| Failure::Error("--autosave: no config directory; set EFFECTCRAFT_CONFIG_DIR".into()))?;
   361→   361→        // Read only auto-save settings. Opting in must not change headless rendering or load
   362→   362→        // the desktop's models, shortcuts, disk cache or recovery sentinel.
   363→   363→        use effectcraft_engine::config::ConfigStore;
   364→   364→        let config = effectcraft_engine::config::DirConfig::new(dir);
   365→   365→        if let Some(text) = config.read(effectcraft_engine::prefs::PREFS_FILE) {
   366→   366→            s.prefs.auto_save = effectcraft_engine::prefs::Prefs::from_json(&text).auto_save;
   367→   367→        }
   368→   368→        s.prefs.normalize();
   369→   369→    }
   370→   370→    if args.flag("--gpu") {
   371→   371→        let g = effectcraft_gpu::Gpu::try_headless().map_err(|e| Failure::Error(format!("--gpu: no usable GPU adapter ({e})")))?;
   372→   372→        s.accel = Some(std::sync::Arc::new(g));
   373→   373→    } else {
   374→   374→        s.accel_note = Some("headless: renders on the CPU unless started with --gpu".into());
   375→   375→    }
   376→   376→    Ok(s)
   377→   377→}

   379→   379→/// `--save` / `--save-as` after a mutating command.
   380→   380→fn maybe_save(b: &mut Backend, args: &Args) -> Result<Option<String>, Failure> {
   381→   381→    let r = if let Some(p) = args.opt("--save-as") {
   382→   382→        b.exec("file.saveAs", json!({"path": p}))?
   383→   383→    } else if args.flag("--save") {
   384→   384→        b.exec("file.save", json!({})).map_err(|e| Failure::Error(format!("{e} (use --project F to save in place, or --save-as F)")))?
   385→   385→    } else {
   386→   386→        return Ok(None);
   387→   387→    };
   388→   388→    Ok(r.get("path").and_then(Value::as_str).map(str::to_string).or_else(|| args.opt("--save-as").map(str::to_string)))
   389→   389→}

   391→   391→fn tool(b: &mut Backend, name: &str, a: Value) -> Result<Value, Failure> {
   392→   392→    match tools::run(b, name, &a)? {
   393→   393→        Reply::Json(v) => Ok(v),
   394→   394→        Reply::Image { info, .. } => Ok(info),
   395→   395→    }

   398→   398→/// `script FILE.jsx [project]` / `script --eval CODE`: run JavaScript against a headless session
   399→   399→/// (an empty project unless one is given, or `--demo`) or the running app (`--bridge`).
   400→   400→fn script_cmd(args: &Args, json_out: bool) -> Result<(), Failure> {
   401→   401→    let (code, name) = match (args.opt("--eval"), args.pos.first()) {
   402→   402→        (Some(c), _) => (c.to_string(), "eval".to_string()),
   403→   403→        (None, Some(f)) => (std::fs::read_to_string(f).map_err(|e| Failure::Error(format!("cannot read {f}: {e}")))?, f.clone()),
   404→   404→        (None, None) => return usage_err("script needs a .jsx/.js file or --eval CODE"),
   405→   405→    };
   406→   406→    let mut b = backend(args, false)?;
   407→   407→    let r = b.exec("script.run", json!({"code": code, "name": name}))?;
   408→   408→    let saved = maybe_save(&mut b, args)?;
   409→   409→    if json_out {
   410→   410→        let mut o = r.clone();
   411→   411→        if let Some(p) = &saved {
   412→   412→            o["saved"] = json!(p);
   413→   413→        }
   414→   414→        emit(&o, true);
   415→   415→    } else {
   416→   416→        if let Some(out) = r["output"].as_str().filter(|o| !o.is_empty()) {
   417→   417→            say!("{out}");
   418→   418→        }
   419→   419→        if !r["result"].is_null() && r["error"].is_null() {
   420→   420→            say!("{}", r["result"]);
   421→   421→        }
   422→   422→        if let Some(p) = saved {
   423→   423→            note!("saved {p}");
   424→   424→        }
   425→   425→    }
   426→   426→    if let Some(e) = r.get("error").filter(|e| !e.is_null()) {
   427→   427→        let pos = match (e["line"].as_u64(), e["column"].as_u64()) {
   428→   428→            (Some(l), Some(c)) => format!(":{l}:{c}"),
   429→   429→            (Some(l), None) => format!(":{l}"),
   430→   430→            _ => String::new(),
   431→   431→        };
   432→   432→        let msg = format!("{}{pos}: {}", e["file"].as_str().unwrap_or(&name), e["message"].as_str().unwrap_or("script error"));
   433→   433→        if json_out {
   434→   434→            std::process::exit(1);
   435→   435→        }
   436→   436→        return Err(Failure::Error(msg));
   437→   437→    }
   438→   438→    Ok(())
   439→   439→}

   445→   445→    } else {
   446→   446→        say!("{}", serde_json::to_string_pretty(v).unwrap_or_default());
   447→   447→    }
   448→   448→}

   450→   450→fn run(cmd: &str, args: &Args, json_out: bool) -> Result<(), Failure> {
   451→   451→    if args.flag("--autosave") && cmd != "mcp" {
   452→   452→        return usage_err("--autosave is only supported by mcp");
   453→   453→    }
   454→   454→    template::validate(cmd, args)?;
   455→   455→    match cmd {
   456→   456→        "render" => render(args, json_out)?,
   457→   457→        "bench" => bench_cmd(args)?,
   458→   458→        "info" => {
   459→   459→            let mut b = backend(args, true)?;
   460→   460→            let p = tool(&mut b, "get_project", json!({}))?;
   461→   461→            let comp = tool(&mut b, "get_comp", json!({})).ok();
   462→   462→            let info = json!({
   463→   463→                "version": env!("CARGO_PKG_VERSION"),
   464→   464→                "mode": if b.is_bridge() { "bridge" } else { "headless" },
   465→   465→                "commands": effectcraft_engine::command_specs().len(),
   466→   466→                "effects": effectcraft_engine::effects::registry().len(),
   467→   467→                "project": p,
   468→   468→                "activeComp": comp,
   469→   469→            });
   470→   470→            if json_out {
   471→   471→                emit(&info, true);
   472→   472→            } else {
   473→   473→                say!(
   474→   474→                    "EffectCraft {} ({}), {} commands, {} effects",
   475→   475→                    info["version"].as_str().unwrap_or(""),
   476→   476→                    info["mode"].as_str().unwrap_or(""),
   477→   477→                    info["commands"],
   478→   478→                    info["effects"]
   479→   479→                );
   480→   480→                say!("project: {}", p["path"].as_str().unwrap_or("(unsaved)"));
   481→   481→                for i in p["items"].as_array().into_iter().flatten() {
   482→   482→                    say!("  item {:>3}  {:<12} {}", i["id"], i["type"].as_str().unwrap_or(""), i["name"].as_str().unwrap_or(""));
   483→   483→                }
   484→   484→                if let Some(c) = comp {
   485→   485→                    say!(
   486→   486→                        "active comp {} \"{}\" {}x{} @ {} fps, {} s",
   487→   487→                        c["id"],
   488→   488→                        c["name"].as_str().unwrap_or(""),
   489→   489→                        c["width"],
   490→   490→                        c["height"],
   491→   491→                        c["frameRate"],
   492→   492→                        c["duration"]
   493→   493→                    );
   494→   494→                    for l in c["layers"].as_array().into_iter().flatten() {
   495→   495→                        say!("  #{:<3} id {:<4} {:<10} {}", l["index"], l["id"], l["type"].as_str().unwrap_or(""), l["name"].as_str().unwrap_or(""));
   496→   496→                    }
   497→   497→                }
   498→   498→            }
   499→   499→        }
   500→   500→        "commands" => list_commands_cmd(args, json_out)?,
   501→   501→        "exec" if args.flag("--list") => list_commands_cmd(args, json_out)?,
   502→   502→        "script" => script_cmd(args, json_out)?,
   503→   503→        "exec" => {
   504→   504→            let Some(id) = args.pos.first().cloned() else { return usage_err("exec needs a command id") };
   505→   505→            let params = match (args.opt("--params"), args.pos.get(1).map(String::as_str)) {
   506→   506→                (Some(p), _) | (None, Some(p)) => serde_json::from_str(p).map_err(|e| Failure::Usage(format!("params are not valid JSON: {e}")))?,
   507→   507→                (None, None) => json!({}),
   508→   508→            };
   509→   509→            let mut b = backend(args, true)?;
   510→   510→            let r = b.exec(&id, params)?;
   511→   511→            if let Some(e) = failed_render(&id, &r) {
   512→   512→                return Err(Failure::Error(e));
   513→   513→            }
   514→   514→            let saved = maybe_save(&mut b, args)?;
   515→   515→            emit(&with_saved(r, saved), json_out);
   516→   516→        }
   517→   517→        "run" => {
   518→   518→            if args.pos.is_empty() {
   519→   519→                return usage_err("run needs <command-id> [<json>] pairs");
   520→   520→            }
   521→   521→            let pairs = args.pos.clone();
   522→   522→            let mut b = backend(args, true)?;
   523→   523→            let mut results = vec![];
   524→   524→            let mut i = 0;
   525→   525→            while i < pairs.len() {
   526→   526→                let id = &pairs[i];
   527→   527→                let params = match pairs.get(i + 1).filter(|p| p.trim_start().starts_with('{')) {
   528→   528→                    Some(p) => {
   529→   529→                        i += 1;
   530→   530→                        serde_json::from_str(p).map_err(|e| Failure::Usage(format!("{id}: params are not valid JSON: {e}")))?
   531→   531→                    }
   532→   532→                    None => json!({}),
   533→   533→                };
   534→   534→                i += 1;
   535→   535→                let r = b.exec(id, params).map_err(|e| Failure::Error(format!("{id}: {e}")))?;
   536→   536→                if let Some(e) = failed_render(id, &r) {
   537→   537→                    return Err(Failure::Error(format!("{id}: {e}")));
   538→   538→                }
   539→   539→                if !json_out {
   540→   540→                    emit(&r, false);
   541→   541→                }
   542→   542→                results.push(json!({"command": id, "result": r}));
   543→   543→            }
   544→   544→            let saved = maybe_save(&mut b, args)?;
   545→   545→            if json_out {
   546→   546→                emit(&with_saved(json!(results), saved), true);
   547→   547→            }
   548→   548→        }
   549→   549→        "props" => {
   550→   550→            let [comp, layer] = positional::<2>(args, "props <comp> <layer>")?;
   551→   551→            let mut b = backend(args, true)?;
   552→   552→            let a = json!({"comp": reference(&comp), "layer": reference(&layer), "flat": args.flag("--flat") || !json_out, "time": args.num("--time")?, "depth": args.num("--depth")?});
   553→   553→            let v = tool(&mut b, "get_layer", a)?;
   554→   554→            if json_out {
   555→   555→                emit(&v, true);
   556→   556→            } else {
   557→   557→                say!("layer {} \"{}\" ({}) at {} s", v["id"], v["name"].as_str().unwrap_or(""), v["type"].as_str().unwrap_or(""), v["time"]);
   558→   558→                for p in v["properties"].as_array().into_iter().flatten() {
   559→   559→                    let mut extra = String::new();
   560→   560→                    if let Some(k) = p.get("keys") {
   561→   561→                        extra += &format!("  [{k} keys]");
   562→   562→                    }
   563→   563→                    if let Some(e) = p.get("expression").and_then(Value::as_str) {
   564→   564→                        extra += &format!("  expr: {e}");
   565→   565→                    }
   566→   566→                    say!("  {:<44} {:<8} {}{extra}", p["path"].as_str().unwrap_or(""), p["type"].as_str().unwrap_or(""), p["value"]);
   567→   567→                }
   568→   568→            }
   569→   569→        }
   570→   570→        "get" => {
   571→   571→            let [comp, layer, path] = positional::<3>(args, "get <comp> <layer> <path>")?;
   572→   572→            let mut b = backend(args, true)?;
   573→   573→            let v = tool(&mut b, "get_property", json!({"comp": reference(&comp), "layer": reference(&layer), "path": path, "time": args.num("--time")?}))?;
   574→   574→            emit(&v, json_out);
   575→   575→        }
   576→   576→        "set" => {
   577→   577→            let expr = args.opt("--expression").map(str::to_string);
   578→   578→            let need = if expr.is_some() { 3 } else { 4 };
   579→   579→            if args.pos.len() < need {
   580→   580→                return usage_err("set <comp> <layer> <path> <value> [--time S] [--expression E]");
   581→   581→            }
   582→   582→            let (comp, layer, path) = (args.pos[0].clone(), args.pos[1].clone(), args.pos[2].clone());
   583→   583→            let value = args.pos.get(3).map(|v| value_arg(v));
   584→   584→            let mut b = backend(args, true)?;
   585→   585→            let a =
   586→   586→                json!({"comp": reference(&comp), "layer": reference(&layer), "path": path, "value": value, "time": args.num("--time")?, "expression": expr});
   587→   587→            let v = tool(&mut b, "set_property", a)?;
   588→   588→            let saved = maybe_save(&mut b, args)?;
   589→   589→            emit(&with_saved(v, saved), json_out);
   590→   590→        }
   591→   591→        "render-frame" | "frame" => {
   592→   592→            let mut b = if args.opt("--template").is_some() {
   593→   593→                let mut s = session(args)?;
   594→   594→                template::prepare(&mut s, args)?;
   595→   595→                Backend::headless(s)
   596→   596→            } else {
   597→   597→                backend(args, true)?
   598→   598→            };
   599→   599→            let comp = args.opt("--comp").and_then(reference);
   600→   600→            let mut time = args.num("--time")?;
   601→   601→            let mut max_side = args.num("--max-side")?.map(|m| m as u32).unwrap_or(0);
   602→   602→            let needs_info = args.opt("--frame").is_some() || args.opt("--scale").is_some();
   603→   603→            if needs_info {
   604→   604→                let c = tool(&mut b, "get_comp", json!({"comp": comp}))?;
   605→   605→                if let Some(f) = args.num("--frame")? {
   606→   606→                    time = Some(f / c["frameRate"].as_f64().unwrap_or(30.0));
   607→   607→                }
   608→   608→                if let Some(k) = args.num("--scale")? {
   609→   609→                    let long = c["width"].as_f64().unwrap_or(0.0).max(c["height"].as_f64().unwrap_or(0.0));
   610→   610→                    max_side = (long * k).round().max(1.0) as u32;
   611→   611→                }
   612→   612→            }
   613→   613→            let out = args.opt("--out").unwrap_or("frame.png").to_string();
   614→   614→            let t0 = std::time::Instant::now();
   615→   615→            let f = b.render_with(comp.as_ref(), time, max_side, args.flag("--transparent"))?;
   616→   616→            let ms = t0.elapsed().as_secs_f64() * 1000.0;
   617→   617→            std::fs::write(&out, &f.png).map_err(|e| format!("cannot write {out}: {e}"))?;
   618→   618→            let info = json!({"path": out, "comp": f.comp, "time": f.time, "width": f.width, "height": f.height, "ms": (ms * 10.0).round() / 10.0});
   619→   619→            if json_out {
   620→   620→                emit(&info, true);
   621→   621→            } else {
   622→   622→                note!("rendered {}x{} at {:.3}s in {ms:.1} ms -> {out}", f.width, f.height, f.time);
   623→   623→            }
   624→   624→        }
   625→   625→        "mcp" => {
   626→   626→            if args.flag("--autosave") && args.opt("--bridge").is_some() {
   627→   627→                return usage_err("--autosave is for headless MCP; the bridged app owns its auto-saves");
   628→   628→            }
   629→   629→            let mut b = backend(args, false)?;
   630→   630→            let (read, write) = (args.opts("--automation-read-root"), args.opts("--automation-write-root"));
   631→   631→            if !read.is_empty() || !write.is_empty() {
   632→   632→                let roots = |r: Vec<&str>| r.into_iter().map(std::path::PathBuf::from).collect::<Vec<_>>();
   633→   633→                b.confine(effectcraft_automation::confine::Confinement::new(&roots(read), &roots(write))?);
   634→   634→            }
   635→   635→            let server = McpServer::new(b);
   636→   636→            let mut server = if args.flag("--autosave") {
   637→   637→                let root =
   638→   638→                    effectcraft_host::config_dir().ok_or_else(|| Failure::Error("--autosave: no config directory; set EFFECTCRAFT_CONFIG_DIR".into()))?;
   639→   639→                server.with_autosave(&root)?
   640→   640→            } else {
   641→   641→                server
   642→   642→            };
   643→   643→            // A client that closes its end of stdout has ended the session.
   644→   644→            match server.serve_stdio() {
   645→   645→                Err(e) if e.kind() != std::io::ErrorKind::BrokenPipe => return Err(e.to_string().into()),
   646→   646→                _ => {}
   647→   647→            }
   648→   648→        }
   649→   649→        other => return usage_err(&format!("unknown subcommand `{other}`")),
   650→   650→    }
   651→   651→    Ok(())
   652→   652→}

   655→   655→    if args.pos.len() < N {

   661→   661→/// `renderQueue.render` reports a failed item in its result, not as an error. Under `run`/`exec`
   662→   662→/// that item fails the command (#1426), as it fails `render --queue`.
   663→   663→fn failed_render(id: &str, r: &Value) -> Option<String> {
   664→   664→    if id != "renderQueue.render" {
   665→   665→        return None;
   666→   666→    }
   667→   667→    let failed: Vec<String> = r["items"]
   668→   668→        .as_array()
   669→   669→        .into_iter()
   670→   670→        .flatten()
   671→   671→        .filter_map(|i| Some(format!("{}: {}", i["compName"].as_str().unwrap_or("?"), i["status"].get("Failed")?.as_str().unwrap_or("failed"))))
   672→   672→        .collect();
   673→   673→    (!failed.is_empty()).then(|| failed.join("; "))
   674→   674→}

   683→   683→/// A relative `--out` is relative to the working directory, like `--project` and every other CLI
   684→   684→/// path. The Render Queue alone would resolve it against the project's folder.
   685→   685→fn from_cwd(path: &str) -> Result<String, Failure> {
   686→   686→    std::path::absolute(path).map(|p| p.to_string_lossy().into_owned()).map_err(|e| Failure::Error(format!("render: cannot resolve --out {path}: {e}")))
   687→   687→}

   689→   689→/// `render`: queue `--comp` (or the active comp) with the given settings unless `--queue`, then
   690→   690→/// render the queue with a progress line on stderr. Fails if any item fails.
   691→   691→fn render(args: &Args, json_out: bool) -> Result<(), Failure> {
   692→   692→    use effectcraft_engine::project::render_queue::{OutputFormat, RenderStatus};
   693→   693→    if args.opt("--bridge").is_some() {
   694→   694→        return usage_err("render runs headless; use `exec renderQueue.add` / `renderQueue.render` with --bridge");
   695→   695→    }
   696→   696→    let mut s = session(args)?;
   697→   697→    if args.opt("--template").is_some() {
   698→   698→        template::prepare(&mut s, args)?;
   699→   699→    } else if let Some(p) = &args.project {
   700→   700→        s.execute("file.open", json!({"path": p})).map_err(|e| Failure::Error(e.to_string()))?;
   701→   701→    } else if args.flag("--demo") || !args.flag("--empty") {
   702→   702→        s.execute("file.openDemoProject", json!({})).map_err(|e| Failure::Error(e.to_string()))?;
   703→   703→    }
   704→   704→    let err = |e: effectcraft_engine::EngineError| Failure::Error(e.to_string());
   705→   705→    if !args.flag("--queue") {
   706→   706→        let Some(out) = args.opt("--out") else { return usage_err("render: --out FILE is required (or --queue)") };
   707→   707→        let out = &from_cwd(out)?;
   708→   708→        let mut p = json!({"output": out});
   709→   709→        if let Some(c) = args.opt("--comp") {
   710→   710→            p["comp"] = json!(c);
   711→   711→        }
   712→   712→        let ext = std::path::Path::new(out).extension().map(|e| e.to_string_lossy().to_string());
   713→   713→        if let Some(f) = args.opt("--format").map(str::to_string).or(ext) {
   714→   714→            p["format"] = json!(f);
   715→   715→        }
   716→   716→        match (args.num("--start")?, args.num("--end")?) {
   717→   717→            (None, None) => p["timeSpan"] = json!(if args.flag("--work-area") { "workArea" } else { "comp" }),
   718→   718→            (a, b) => {
   719→   719→                p["start"] = json!(a.unwrap_or(0.0));
   720→   720→                if let Some(b) = b {
   721→   721→                    p["end"] = json!(b);
   722→   722→                }
   723→   723→            }
   724→   724→        }
   725→   725→        if let Some(v) = args.num("--fps")? {
   726→   726→            p["frameRate"] = json!(v);
   727→   727→        }
   728→   728→        if let Some(r) = args.opt("--resolution").or(args.opt("--scale")) {
   729→   729→            p["resolution"] = r.parse::<f64>().map(|v| json!(v)).unwrap_or(json!(r));
   730→   730→        }
   731→   731→        for (flag, key) in [("--quality", "quality"), ("--channels", "channels"), ("--prores", "proresProfile"), ("--audio", "audio")] {
   732→   732→            if let Some(v) = args.opt(flag) {
   733→   733→                p[key] = json!(v);
   734→   734→            }
   735→   735→        }
   736→   736→        if let Some(v) = args.num("--jpeg-quality")? {
   737→   737→            p["quality"] = json!(v);
   738→   738→        }
   739→   739→        if let Some(v) = args.num("--bitrate")? {
   740→   740→            p["bitrate"] = json!(v);
   741→   741→            // VP9 WebM encodes by quality unless told to target the bitrate (#440).
   742→   742→            // The format parser decides what is WebM, so its aliases (`vp9`, `webmvp9opus`) count.
   743→   743→            if p["format"].as_str().and_then(OutputFormat::from_name) == Some(OutputFormat::WebM) {
   744→   744→                p["webmBitrate"] = json!(true);
   745→   745→            }
   746→   746→        }
   747→   747→        for (flag, key) in [
   748→   748→            ("--profile", "profile"),
   749→   749→            ("--level", "level"),
   750→   750→            ("--rate-control", "rateControl"),
   751→   751→            ("--webm-codec", "webmCodec"),
   752→   752→            ("--opus-app", "opusApplication"),
   753→   753→        ] {
   754→   754→            if let Some(v) = args.opt(flag) {
   755→   755→                p[key] = json!(v);
   756→   756→            }
   757→   757→        }

   937→   937→    v.sort_by(f64::total_cmp);

   945→   945→/// CPU time of the whole process (all threads) in ms: a load-independent measure of work.
   946→   946→fn cpu_ms() -> f64 {

   948→   948→}

   950→   950→/// One frame: (wall ms, CPU ms, per-layer timings).
   951→   951→fn render_timed(s: &Session, cid: ItemId, t: Tick, opts: RenderOpts, cache: Option<&LayerCache>) -> (f64, f64, Vec<LayerTiming>) {
   952→   952→    let prof = std::sync::Mutex::new(Vec::<LayerTiming>::new());
   953→   953→    let mut r = Renderer::new(&s.project, s.footage.as_ref(), opts);
   954→   954→    r.expr = s.expr.as_deref();
   955→   955→    r.cache = cache;
   956→   956→    r.profile = Some(&prof);
   957→   957→    let c0 = cpu_ms();
   958→   958→    let t0 = std::time::Instant::now();
   959→   959→    let img = r.comp_frame(cid, t);
   960→   960→    let ms = t0.elapsed().as_secs_f64() * 1e3;
   961→   961→    let cpu = cpu_ms() - c0;
   962→   962→    std::hint::black_box(&img);
   963→   963→    (ms, cpu, prof.into_inner().unwrap_or_default())
   964→   964→}

   966→   966→type LayerRow = (usize, String, Vec<f64>, Vec<f64>, Vec<(String, Vec<f64>)>);

   968→   968→/// `frame --bench N`: render the same frame N times without the layer cache (the full cost of a
   969→   969→/// frame) and print min/median frame time plus min per-layer and per-effect times.
   970→   970→fn bench(s: &Session, cid: ItemId, t: Tick, opts: RenderOpts, n: usize) {
   971→   971→    let mut totals = Vec::with_capacity(n);
   972→   972→    let mut cpus = Vec::with_capacity(n);
   973→   973→    let mut per_layer: Vec<LayerRow> = Vec::new();
   974→   974→    for _ in 0..n {
   975→   975→        let (ms, cpu, prof) = render_timed(s, cid, t, opts, None);
   976→   976→        totals.push(ms);
   977→   977→        cpus.push(cpu);
   978→   978→        for (i, lt) in prof.into_iter().enumerate() {
   979→   979→            if per_layer.len() <= i {
   980→   980→                per_layer.push((lt.depth, lt.layer.clone(), vec![], vec![], vec![]));
   981→   981→            }
   982→   982→            let e = &mut per_layer[i];
   983→   983→            e.2.push(lt.process_ms);
   984→   984→            e.3.push(lt.composite_ms);
   985→   985→            for (j, (id, ms)) in lt.effects.into_iter().enumerate() {
   986→   986→                if e.4.len() <= j {
   987→   987→                    e.4.push((id, vec![]));
   988→   988→                }
   989→   989→                e.4[j].1.push(ms);
   990→   990→            }
   991→   991→        }
   992→   992→    }
   993→   993→    note!(
   994→   994→        "bench (no cache): {n} runs at scale {}: wall min {:.2} ms, median {:.2} ms; CPU (all threads) min {:.2} ms",
   995→   995→        opts.scale,
   996→   996→        min(&totals),
   997→   997→        median(&mut totals),
   998→   998→        min(&cpus)
   999→   999→    );
  1000→  1000→    for (d, name, p, c, fx) in per_layer {
  1001→  1001→        note!("  {:indent$}{name:<28} process {:8.2} ms  composite {:8.2} ms  (min)", "", min(&p), min(&c), indent = d * 2);
  1002→  1002→        for (id, v) in fx {
  1003→  1003→            note!("  {:indent$}  fx {id:<30} {:8.2} ms", "", min(&v), indent = d * 2);
  1004→  1004→        }
  1005→  1005→    }
  1006→  1006→}

  1008→  1008→/// `frame --bench-play N`: render N consecutive frames from `t` as playback does, without and
  1009→  1009→/// with the layer cache, and print per-frame times.
  1010→  1010→fn bench_play(s: &Session, cid: ItemId, t: Tick, opts: RenderOpts, n: usize) {
  1011→  1011→    let Some(comp) = s.project.comp(cid) else { return };
  1012→  1012→    let f0 = comp.frame_rate.frame_at(t);
  1013→  1013→    let cache = LayerCache::default();
  1014→  1014→    for (label, c) in [("no cache", None), ("layer cache", Some(&cache))] {
  1015→  1015→        let runs: Vec<(f64, f64)> = (0..n as i64)
  1016→  1016→            .map(|i| {
  1017→  1017→                let (ms, cpu, _) = render_timed(s, cid, comp.frame_rate.tick_of(f0 + i), opts, c);
  1018→  1018→                (ms, cpu)
  1019→  1019→            })
  1020→  1020→            .collect();
  1021→  1021→        let mut v: Vec<f64> = runs.iter().map(|r| r.0).collect();
  1022→  1022→        let mean = v.iter().sum::<f64>() / n as f64;
  1023→  1023→        let cpu = runs.iter().map(|r| r.1).sum::<f64>() / n as f64;
  1024→  1024→        note!(
  1025→  1025→            "bench-play ({label}): {n} frames at scale {}: wall mean {mean:.2} ms, median {:.2} ms, min {:.2} ms; CPU mean {cpu:.2} ms",
  1026→  1026→            opts.scale,
  1027→  1027→            median(&mut v),
  1028→  1028→            min(&v)
  1029→  1029→        );
  1030→  1030→    }
  1031→  1031→    let st = cache.stats();
  1032→  1032→    note!("  layer cache: {} hits, {} misses, {} entries, {:.1} MB", st.hits, st.misses, st.entries, st.bytes as f64 / 1e6);
  1033→  1033→}

  1188→  1188→/// `bench --gpu`: CPU vs GPU ms/frame for every comp (or `--comp`) at Full and Half. "cold"
  1189→  1189→/// renders everything (no layer cache); "warm" reuses the layer cache (playback / scrubbing of
  1190→  1190→/// unchanged layers: compositing cost); "viewer" is the GPU display path without readback;
  1191→  1191→/// "auto" is Backend::Auto (Mercury GPU Acceleration) choosing per comp from measured times,
  1192→  1192→/// warm; "up/dn MB" are the GPU's uploads / readbacks per warm GPU frame.
  1193→  1193→fn bench_gpu(s: &Session, args: &Args) -> Result<(), Failure> {
  1194→  1194→    use effectcraft_render::Backend;
  1195→  1195→    let gpu = effectcraft_gpu::Gpu::try_headless().map_err(|e| Failure::Error(format!("--gpu: no usable GPU adapter ({e})")))?;
  1196→  1196→    let n = args.num("--n")?.unwrap_or(10.0).max(1.0) as usize;
  1197→  1197→    // Every comp, plus an adjustment-layer comp built for the benchmark (the main comp under a
  1198→  1198→    // full-frame adjustment layer with a GPU effect stack) and an Advanced 3D comp (`--adv3d`:
  1199→  1199→    // that one only).
  1200→  1200→    let mut project = (*s.project).clone();
  1201→  1201→    project.settings.gpu_acceleration = true;
  1202→  1202→    let comps: Vec<ItemId> = match args.opt("--comp") {
  1203→  1203→        _ if args.flag("--adv3d") => adv3d_bench_comp(&mut project).into_iter().collect(),
  1204→  1204→        Some(_) => s.active_comp_id().into_iter().collect(),
  1205→  1205→        None => {
  1206→  1206→            let mut v: Vec<ItemId> = s.project.comps().map(|(id, _)| *id).collect();
  1207→  1207→            v.extend(adjustment_bench_comp(&mut project));
  1208→  1208→            v.extend(adv3d_bench_comp(&mut project));
  1209→  1209→            v
  1210→  1210→        }
  1211→  1211→    };
  1212→  1212→    let project = &project;
  1213→  1213→    note!("GPU: {} — median of {n} runs (ms/frame)", effectcraft_render::Accelerator::name(&gpu));
  1214→  1214→    note!(
  1215→  1215→        "{:<28} {:>5} {:>10} {:>9} {:>9} {:>9} {:>9} {:>9} {:>8} {:>9} {:>9} {:>11}",
  1216→  1216→        "comp",
  1217→  1217→        "res",
  1218→  1218→        "size",
  1219→  1219→        "cpu cold",
  1220→  1220→        "gpu cold",
  1221→  1221→        "cpu warm",
  1222→  1222→        "gpu warm",
  1223→  1223→        "gpu view",
  1224→  1224→        "speedup",
  1225→  1225→        "auto",
  1226→  1226→        "gpu≠cpu",
  1227→  1227→        "up/dn MB"
  1228→  1228→    );
  1229→  1229→    for cid in comps {
  1230→  1230→        let Some(comp) = project.comp(cid) else { continue };
  1231→  1231→        let name = project.item(cid).map(|i| i.name.clone()).unwrap_or_default();
  1232→  1232→        let t = Tick::from_seconds_f64(args.num("--time")?.unwrap_or(3.0).min(comp.duration.seconds() * 0.5));
  1233→  1233→        for (label, scale) in [("Full", 1.0), ("Half", 0.5)] {
  1234→  1234→            let mk = |backend: Backend, cache: Option<&LayerCache>| {
  1235→  1235→                let mut r = Renderer::new(project, s.footage.as_ref(), RenderOpts { scale, backend, ..Default::default() });
  1236→  1236→                r.expr = s.expr.as_deref();
  1237→  1237→                r.cache = cache;
  1238→  1238→                r.accel = Some(&gpu);
  1239→  1239→                r.comp_frame(cid, t)
  1240→  1240→            };
  1241→  1241→            let cpu_cold = time_ms(n, || {
  1242→  1242→                std::hint::black_box(mk(Backend::Cpu, None));
  1243→  1243→            });
  1244→  1244→            let gpu_cold = time_ms(n, || {
  1245→  1245→                std::hint::black_box(mk(Backend::Gpu, None));
  1246→  1246→            });
  1247→  1247→            let cache = LayerCache::default();
  1248→  1248→            let cpu_warm = time_ms(n, || {
  1249→  1249→                std::hint::black_box(mk(Backend::Cpu, Some(&cache)));
  1250→  1250→            });
  1251→  1251→            let gcache = LayerCache::default();
  1252→  1252→            // Traffic in steady state: after the frame that fills the layer cache.
  1253→  1253→            std::hint::black_box(mk(Backend::Gpu, Some(&gcache)));
  1254→  1254→            let t0 = gpu.context().transfer_stats();
  1255→  1255→            let gpu_warm = time_ms(n, || {
  1256→  1256→                std::hint::black_box(mk(Backend::Gpu, Some(&gcache)));
  1257→  1257→            });
  1258→  1258→            let t1 = gpu.context().transfer_stats();
  1259→  1259→            let per = |b: u64| b as f64 / (n + 1) as f64 / (1u64 << 20) as f64;
  1260→  1260→            let traffic = format!("{:.1}/{:.1}", per(t1.upload_bytes - t0.upload_bytes), per(t1.readback_bytes - t0.readback_bytes));
  1261→  1261→            // Auto: warm-up frames on both sides, then the measured choice.
  1262→  1262→            for _ in 0..2 * effectcraft_render::AutoPick::WARMUP {
  1263→  1263→                std::hint::black_box(mk(Backend::Auto, Some(&gcache)));
  1264→  1264→            }
  1265→  1265→            let auto = time_ms(n, || {
  1266→  1266→                std::hint::black_box(mk(Backend::Auto, Some(&gcache)));
  1267→  1267→            });
  1268→  1268→            let view = time_ms(n, || {
  1269→  1269→                let mut r = Renderer::new(project, s.footage.as_ref(), RenderOpts { scale, backend: Backend::Gpu, ..Default::default() });
  1270→  1270→                r.expr = s.expr.as_deref();
  1271→  1271→                r.cache = Some(&gcache);
  1272→  1272→                r.accel = Some(&gpu);
  1273→  1273→                std::hint::black_box(gpu.render_display(&r, cid, t));
  1274→  1274→                gpu.wait();
  1275→  1275→            });
  1276→  1276→            let size = format!("{}x{}", (comp.width as f64 * scale).round(), (comp.height as f64 * scale).round());
  1277→  1277→            // Agreement: share of pixels where the GPU frame differs from the CPU reference by
  1278→  1278→            // more than 1/255 on any channel.
  1279→  1279→            let (a, b) = (mk(Backend::Cpu, Some(&cache)), mk(Backend::Gpu, Some(&gcache)));
  1280→  1280→            let off = a
  1281→  1281→                .data
  1282→  1282→                .iter()
  1283→  1283→                .zip(&b.data)
  1284→  1284→                .filter(|(p, q)| {
  1285→  1285→                    (0..4).any(|c| {
  1286→  1286→                        let d = (p[c] - q[c]).abs();
  1287→  1287→                        d.is_nan() || d > 1.0 / 255.0 + 1e-6
  1288→  1288→                    })
  1289→  1289→                })
  1290→  1290→                .count();
  1291→  1291→            let pct = 100.0 * off as f64 / a.data.len().max(1) as f64;
  1292→  1292→            let speedup = cpu_warm / gpu_warm.max(1e-9);
  1293→  1293→            note!(
  1294→  1294→                "{name:<28} {label:>5} {size:>10} {cpu_cold:>9.2} {gpu_cold:>9.2} {cpu_warm:>9.2} {gpu_warm:>9.2} {view:>9.2} {speedup:>7.2}x {auto:>9.2} {pct:>8.3}% {traffic:>11}"
  1295→  1295→            );
  1296→  1296→        }
  1297→  1297→    }
  1298→  1298→    Ok(())
  1299→  1299→}
