   105→function makeWire(options: { readonly models?: "ok" | "fail" } = {}) {
   106→  const segments: WireSegment[] = [];
   107→  const postEvents: PostedEvent[] = [];
   108→
   109→  const active = (): WireSegment => {
   110→    const segment = segments[segments.length - 1];
   111→    if (segment === undefined) throw new Error("no open stream segment");
   112→    return segment;
   113→  };
   114→
   115→  const openSegment = (
   116→    threadId: string,
   117→    body: ChatRequest,
   118→    handlers: WireHandlers,
   119→  ): WireSegment => {
   120→    const segment: WireSegment = {
   121→      threadId,
   122→      body,
   123→      handlers,
   124→      paused: false,
   125→      cancel: vi.fn(async () => undefined),
   126→      abort: vi.fn(),
   127→      reconnect: vi.fn(async () => {
   128→        openSegment(threadId, body, handlers); // sse.ts resume: same body re-POST
   129→      }),
   130→    };
   131→    segments.push(segment);
   132→    return segment;
   133→  };
   134→
   135→  const client = {
   136→    threads: {
   137→      create: async (projectId: string): Promise<Thread> => ({
   138→        id: THREAD_ID,
   139→        projectId,
   140→        title: null,
   141→        createdBy: "usr_test",
   142→        createdAt: "2026-10-24T00:00:00.000Z",
   143→        updatedAt: "2026-10-24T00:00:00.000Z",
   144→      }),
   145→    },
   146→    models: {
   147→      list: async () => {
   148→        if (options.models === "fail") throw new Error("models endpoint down");
   149→        return MODELS_OK;
   150→      },
   151→    },
   152→    chat: {
   153→      stream: (threadId: string, body: ChatRequest, handlers?: WireHandlers) =>
   154→        openSegment(threadId, body, handlers ?? {}),
   155→    },
   156→    runs: {
   157→      postEvent: async (
   158→        runId: string,
   159→        body: Record<string, unknown>,
   160→        options?: { idempotencyKey?: string },
   161→      ) => {
   162→        postEvents.push({ runId, body, idempotencyKey: options?.idempotencyKey });
   163→        return null;
   164→      },
   165→    },
   166→  } as unknown as ApiClient;
   167→
   168→  /** Feed normalized §7.4 events through the CURRENT segment (foldEvent runs for real). */
   169→  const feed = (...events: StudioStreamEvent[]): void => {
   170→    act(() => {
   171→      for (const event of events) active().handlers.onEvent?.(event);
   172→    });
   173→  };
   174→
   175→  /** The sse.ts terminal summary (fired AFTER the terminal event was folded). */
   176→  const finishRun = (runId: string): void => {
   177→    act(() => {
   178→      active().handlers.onDone?.({ reason: "terminal", runId, status: "applied" });
   179→    });
   180→  };
   181→
   182→  /** One transport drop: onReconnect → sse.ts re-POSTs the SAME body (a new segment). */
   183→  const dropOnce = (): void => {
   184→    const segment = active();
   185→    act(() => {
   186→      segment.handlers.onReconnect?.({ attempt: 1, delayMs: 1000, lastEventId: null });
   187→      openSegment(segment.threadId, segment.body, segment.handlers);
   188→    });
   189→  };
   190→
   191→  /** §8.4 exhaustion: 3 attempts (each onReconnect + same-key re-POST), then pause. */
   192→  const dropToPause = (): void => {
   193→    const segment = active();
   194→    act(() => {
   195→      for (let attempt = 1; attempt <= 3; attempt++) {
   196→        segment.handlers.onReconnect?.({ attempt, delayMs: 1000 * attempt, lastEventId: null });
   197→        openSegment(segment.threadId, segment.body, segment.handlers);
   198→      }
   199→      segment.handlers.onPause?.({ attempts: 3 });
   200→      segment.handlers.onDone?.({ reason: "paused", runId: null, status: "streaming" });
   201→    });
   202→  };
   203→
   204→  return { client, segments, postEvents, active, feed, finishRun, dropOnce, dropToPause };
   205→}

   226→function makeRecordedBridge(fake: FakeEditor): { bridge: EditorBridge; calls: BridgeCalls } {
   227→  const inner = attachFakeBridge(fake);
   228→  const calls: BridgeCalls = { inspect: 0, queries: [], executes: [], batches: 0, renderFrames: 0 };
   229→  const bridge: EditorBridge = {
   230→    ...inner,
   231→    inspect: async () => {
   232→      calls.inspect += 1;
   233→      return inner.inspect();
   234→    },
   235→    query: ((command: string, params?: unknown) => {
   236→      calls.queries.push(command);
   237→      return inner.query(command, params);
   238→    }) as EditorBridge["query"],
   239→    execute: ((command: string, params?: unknown) => {
   240→      calls.executes.push(command);
   241→      return inner.execute(command, params);
   242→    }) as EditorBridge["execute"],
   243→    batch: (input) => {
   244→      calls.batches += 1;
   245→      return inner.batch(input);
   246→    },
   247→    renderFrame: (params) => {
   248→      calls.renderFrames += 1;
   249→      return inner.renderFrame(params);
   250→    },
   251→  };
   252→  return { bridge, calls };
   253→}
