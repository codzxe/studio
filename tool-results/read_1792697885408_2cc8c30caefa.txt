     1→     1→/**
     2→     2→ * Client-side apply/undo engine — BUILD_PLAN §M5.4/§M5.5, DECISIONS.md D-6,
     3→     3→ * docs/phases/M5-SEAMS.md (`ApplyController`, pinned).
     4→     4→ *
     5→     5→ * ONE user gesture = ONE atomic `engine.batch` = ONE undo step = ONE version
     6→     6→ * (AGENTS.md hard rule 6). The whole flow NEVER throws — every outcome is a
     7→     7→ * typed union the plan card (M5-G) renders:
     8→     8→ *
     9→     9→ *  `apply(plan)`:
    10→    10→ *    1. resolve each `resolutionRequests` entry against the LIVE layer list
    11→    11→ *       (the executor's list_layers path: `comp.info {}` → `layers`) — a name
    12→    12→ *       matching exactly one layer binds it; ambiguous or unknown names fail
    13→    13→ *       BEFORE any engine mutation;
    14→    14→ *    2. patch the `compiledBatch` placeholders with the resolved layer ids
    15→    15→ *       (deterministic deep clone; null-prototype request map — the M4-N F-1
    16→    16→ *       lesson: model-adjacent keys never touch `Object.prototype`);
    17→    17→ *    3. `bridge.execute("engine.batch", batch)` — ONE call. Verified wire:
    18→    18→ *       the bridge passes the params object through `effectcraft.execute` →
    19→    19→ *       `engine.execute` → `batch.rs`, which reads `{steps, label?, atomic?}`
    20→    20→ *       (extra keys such as `timeoutMs` ride inert upstream — `timeoutMs` is
    21→    21→ *       §8.5 bound metadata the compiled batch carries, not an engine param);
    22→    22→ *    4. on success `await autosave.saveNow("agent-apply")` (D-6) — this
    23→    23→ *       uploads a NEW immutable version; the outcome's `version.id` is the
    24→    24→ *       applied versionId;
    25→    25→ *    5. `api.runs.applyRun(runId, {batchSha256, versionId, idempotencyKey})` —
    26→    26→ *       sha256 over the exact patched batch payload (WebCrypto via the
    27→    27→ *       editor-session sha256Hex, with its vector-tested sync fallback for
    28→    28→ *       hosts without `crypto.subtle` — never a new dependency), idempotency
    29→    29→ *       key generated ONCE per user gesture. Gateway errors map: 409 →
    30→    30→ *       conflict (explains, never re-applies), 402 → insufficient credits,
    31→    31→ *       else typed failure.
    32→    32→ *
    33→    33→ *  `undo(runId)`: `bridge.execute("edit.undo")` (the whole run folded into ONE
    34→    34→ *   undo step by `batch.rs`) THEN `api.runs.undoRun(runId, {idempotencyKey})`.
    35→    35→ *   If `edit.undo` fails ("nothing to undo" — the upstream `can_undo` guard,
    36→    36→ *   edit.rs:12) the gateway is NOT told — a typed `{ok:false}` comes back and
    37→    37→ *   the version rows are never touched (§9.3). The post-undo state is left to
    38→    38→ *   the D-6 cadence + the before-close save (undo is not a D-6 moment).
    39→    39→ *
    40→    40→ *  `reapply(plan)`: only when the client's CURRENT version id equals the plan's
    41→    41→ *   `baseVersionId` (the session port reads it); anything else is
    42→    42→ *   `{ok:false, reason:"base-version-changed"}` — BP §M5.5 "otherwise explain".
    43→    43→ *   NOTE for the integrator: after undo the session's current version is the
    44→    44→ *   APPLIED version's successor chain, so re-apply-after-undo needs the host
    45→    45→ *   page to restore the session's version pointer to the plan's base (a
    46→    46→ *   session-level concern — see the worklog M5-F record).
    47→    47→ *
    48→    48→ * Cross-agent pins (wave 1, reconciled by the integrator):
    49→    49→ *  - placeholder token: a whole-string `"$request:<requestId>"` in the compiled
    50→    50→ *    batch params, replaced with the resolved engine layer id (number);
    51→    51→ *  - `api.runs.applyRun/undoRun` — @studio/api-client names (M5-E adds them);
    52→    52→ *  - the plan event carries `compiledBatch` + `resolutionRequests` +
    53→    53→ *    `baseVersionId` (M5-C adds them to the wire; the SSE reader strips

   341→   341→  code: string | undefined;
   342→   342→  message: string;
   343→   343→} {
   344→   344→  if (typeof error === "object" && error !== null && "status" in error) {

   348→   348→    const message =
   349→   349→      typeof record.message === "string" && record.message !== "" ? record.message : "";

   359→   359→interface CompLayerEntry {
   360→   360→  readonly id: number;
   361→   361→  readonly name: string;

   363→   363→
   364→   364→/** `comp.info {}` → the active composition's layer list (query.rs; read-only). */
   365→   365→async function listLayers(bridge: EditorBridge): Promise<readonly CompLayerEntry[]> {
   366→   366→  const info = await bridge.query<{ layers?: unknown }>("comp.info", {});
   367→   367→  const raw = info?.layers;
   368→   368→  if (!Array.isArray(raw)) return [];
   369→   369→  const out: CompLayerEntry[] = [];
   370→   370→  for (const entry of raw) {
   371→   371→    if (typeof entry !== "object" || entry === null) continue;
   372→   372→    const record = entry as Record<string, unknown>;
   373→   373→    if (typeof record.id !== "number" || typeof record.name !== "string") continue;
   374→   374→    out.push({ id: record.id, name: record.name });
   375→   375→  }
   376→   376→  return out;
   377→   377→}
   378→   378→
   379→   379→/**
   380→   380→ * Resolves every request against the live layer list. Read-only: the only
   381→   381→ * bridge traffic is ONE `comp.info` query however many requests there are.
   382→   382→ * Ambiguous/unknown names fail BEFORE anything executes.
   383→   383→ */
   384→   384→async function resolveRequests(
   385→   385→  requests: readonly LayerResolutionRequest[],
   386→   386→  bridge: EditorBridge,
   387→   387→): Promise<
   388→   388→  { ok: true; resolved: readonly ResolvedLayer[] } | { ok: false; outcome: ApplyOutcome }
   389→   389→> {
   390→   390→  let layers: readonly CompLayerEntry[];
   391→   391→  try {
   392→   392→    layers = await listLayers(bridge);
   393→   393→  } catch (error) {
   394→   394→    const failure = bridgeOutcome(error);
   395→   395→    const outcome: ApplyOutcome =
   396→   396→      failure.reason === "editor-disconnected"
   397→   397→        ? { ok: false, reason: "editor-disconnected", message: failure.message }
   398→   398→        : {
   399→   399→            ok: false,
   400→   400→            reason: "resolution-failed",
   401→   401→            message: "The layers could not be listed to resolve this plan. Try again in a moment.",
   402→   402→          };
   403→   403→    return { ok: false, outcome };
   404→   404→  }
   405→   405→
   406→   406→  const resolved: ResolvedLayer[] = [];
   407→   407→  for (const request of requests) {
   408→   408→    // Exact-name matches only — a fuzzy match could bind the WRONG layer.
   409→   409→    const matches = layers.filter((layer) => layer.name === request.layerName);
   410→   410→    const first = matches[0];
   411→   411→    if (matches.length === 1 && first !== undefined) {
   412→   412→      resolved.push({ requestId: request.requestId, layerId: first.id, layerName: first.name });
   413→   413→      continue;
   414→   414→    }
   415→   415→    if (matches.length > 1) {
   416→   416→      return {
   417→   417→        ok: false,
   418→   418→        outcome: {
   419→   419→          ok: false,
   420→   420→          reason: "ambiguous",
   421→   421→          candidates: matches.map((layer) => `${layer.name} (layer ${layer.id})`),
   422→   422→          message: `Layer "${request.layerName}" matches ${matches.length} layers. Say which one you mean.`,
   423→   423→        },
   424→   424→      };
   425→   425→    }
   426→   426→    return {
   427→   427→      ok: false,
   428→   428→      outcome: {
   429→   429→        ok: false,
   430→   430→        reason: "not-found",
   431→   431→        layerName: request.layerName,
   432→   432→        message: `No layer named "${request.layerName}" is in the composition.`,
   433→   433→      },
   434→   434→    };
   435→   435→  }
   436→   436→  return { ok: true, resolved };
   437→   437→}

   447→   447→ *
   448→   448→ * The cloned objects are NULL-PROTOTYPE (M4-N F-1): plan params are
   449→   449→ * model-adjacent — a parsed `{"__proto__": …}` key must ride as OWN data, not
   450→   450→ * rebind the clone's prototype.
   451→   451→ */
   452→   452→function patchPlaceholders(
   453→   453→  value: unknown,
   454→   454→  resolvedById: Readonly<Record<string, number>>,
   455→   455→): { patched: unknown; unresolved: string[] } {
   456→   456→  const unresolved: string[] = [];
   457→   457→  const walk = (node: unknown): unknown => {
   458→   458→    if (typeof node === "string") {
   459→   459→      if (!node.startsWith(RESOLVE_TOKEN_PREFIX)) return node;
   460→   460→      const requestId = node.slice(RESOLVE_TOKEN_PREFIX.length);
   461→   461→      const layerId = resolvedById[requestId];
   462→   462→      if (layerId === undefined) {
   463→   463→        unresolved.push(node);
   464→   464→        return node;
   465→   465→      }
   466→   466→      return layerId;
   467→   467→    }
   468→   468→    if (Array.isArray(node)) return node.map(walk);
   469→   469→    if (typeof node === "object" && node !== null) {
   470→   470→      const out: Record<string, unknown> = Object.create(null);
   471→   471→      for (const [key, child] of Object.entries(node)) out[key] = walk(child);
   472→   472→      return out;
   473→   473→    }
   474→   474→    return node;
   475→   475→  };
   476→   476→  return { patched: walk(value), unresolved };
   477→   477→}
   478→   478→
   479→   479→/** Narrow hand-rolled validation of the compiled batch (no zod in apps/web). */
   480→   480→function validCompiledBatch(batch: unknown): AgentBatch | null {
   481→   481→  if (typeof batch !== "object" || batch === null) return null;
   482→   482→  const record = batch as Record<string, unknown>;
   483→   483→  if (typeof record.label !== "string" || record.label.length === 0) return null;
   484→   484→  if (!Array.isArray(record.steps) || record.steps.length === 0) return null;
   485→   485→  const steps: BatchStep[] = [];
   486→   486→  for (const step of record.steps) {
   487→   487→    if (typeof step !== "object" || step === null) return null;
   488→   488→    const stepRecord = step as Record<string, unknown>;
   489→   489→    if (typeof stepRecord.command !== "string" || stepRecord.command.length === 0) return null;
   490→   490→    if (typeof stepRecord.params !== "object" || stepRecord.params === null) return null;
   491→   491→    steps.push({
   492→   492→      command: stepRecord.command,
   493→   493→      params: stepRecord.params as Record<string, unknown>,
   494→   494→    });
   495→   495→  }
   496→   496→  const timeoutMs =
   497→   497→    typeof record.timeoutMs === "number" &&
   498→   498→    Number.isInteger(record.timeoutMs) &&
   499→   499→    record.timeoutMs > 0
   500→   500→      ? record.timeoutMs
   501→   501→      : undefined;
   502→   502→  // Agent batches are ALWAYS atomic (§8.5) — forced regardless of the wire.
   503→   503→  return timeoutMs === undefined
   504→   504→    ? { label: record.label, atomic: true, steps }
   505→   505→    : { label: record.label, atomic: true, steps, timeoutMs };
   506→   506→}

   511→   511→
   512→   512→/** The exact params object handed to `bridge.execute("engine.batch", …)`. */
   513→   513→export function engineBatchPayload(batch: AgentBatch): {
   514→   514→  label: string;
   515→   515→  atomic: true;
   516→   516→  steps: readonly BatchStep[];
   517→   517→  timeoutMs?: number;
   518→   518→} {
   519→   519→  return batch.timeoutMs === undefined
   520→   520→    ? { label: batch.label, atomic: true, steps: batch.steps }
   521→   521→    : { label: batch.label, atomic: true, steps: batch.steps, timeoutMs: batch.timeoutMs };
   522→   522→}
   523→   523→
   524→   524→/**
   525→   525→ * sha256 over the serialized engine payload — the exact shape the editor
   526→   526→ * receives (JSON key order is deterministic: the payload object is built with
   527→   527→ * a fixed key order and the patched params preserve theirs).
   528→   528→ */
   529→   529→export async function batchSha256Hex(payload: unknown): Promise<string> {
   530→   530→  const serialized = JSON.stringify(payload) ?? "null";
   531→   531→  return sha256Hex(new TextEncoder().encode(serialized));
   532→   532→}

   538→   538→export function createApplyController(options: CreateApplyControllerOptions): ApplyController {
   539→   539→  const { bridge, autosave, api } = options;
   540→   540→
   541→   541→  const notify = (event: ApplyEngineEvent): void => {
   542→   542→    const callback = options.onEvent;
   543→   543→    if (callback === undefined) return;
   544→   544→    try {
   545→   545→      callback(event);
   546→   546→    } catch {
   547→   547→      /* a host callback must never break the apply engine */
   548→   548→    }
   549→   549→  };
   550→   550→
   551→   551→  const failApply = (runId: string, outcome: ApplyOutcome): ApplyOutcome => {
   552→   552→    notify({ type: "apply.failed", runId, outcome });
   553→   553→    return outcome;
   554→   554→  };
   555→   555→
   556→   556→  /** The shared execute-then-save-then-report path (apply + reapply). */
   557→   557→  const runApply = async (plan: ApplicablePlan): Promise<ApplyOutcome> => {
   558→   558→    // One gesture = one idempotency key (law #4), whatever happens below.
   559→   559→    const idempotencyKey = crypto.randomUUID();
   560→   560→    notify({ type: "apply.started", runId: plan.runId });
   561→   561→
   562→   562→    const compiled = validCompiledBatch(plan.compiledBatch);
   563→   563→    if (compiled === null) {
   564→   564→      return failApply(plan.runId, {
   565→   565→        ok: false,
   566→   566→        reason: "invalid-batch",
   567→   567→        message: "The plan's compiled batch is malformed. Ask the agent to plan again.",
   568→   568→      });
   569→   569→    }
   570→   570→
   571→   571→    // 1. Resolve layer names against the live list — read-only; on any
   572→   572→    //    ambiguity NOTHING executes (no engine.batch, no save, no applyRun).
   573→   573→    let resolvedById: Readonly<Record<string, number>> = {};
   574→   574→    let resolvedForWire: readonly PostedResolvedLayer[] = [];
   575→   575→    const requests = plan.resolutionRequests ?? [];
   576→   576→    if (requests.length > 0) {
   577→   577→      const resolution = await resolveRequests(requests, bridge);
   578→   578→      if (!resolution.ok) return failApply(plan.runId, resolution.outcome);
   579→   579→      notify({ type: "apply.resolved", runId: plan.runId, resolved: resolution.resolved });
   580→   580→      // Null-prototype accumulator (M4-N F-1): request ids are model-adjacent
   581→   581→      // strings — on a plain `{}` a `__proto__` id would hit the accessor.
   582→   582→      const map = Object.create(null) as Record<string, number>;
   583→   583→      for (const entry of resolution.resolved) map[entry.requestId] = entry.layerId;
   584→   584→      resolvedById = map;
   585→   585→      // §8.7/M5-E: the name→id bindings ride the applyRun body (auditable
   586→   586→      // server-side even though the batch already ran client-side).
   587→   587→      resolvedForWire = resolution.resolved.map((entry) => ({
   588→   588→        requestId: entry.requestId,
   589→   589→        layerId: entry.layerId,
   590→   590→      }));
   591→   591→    }
   592→   592→
   593→   593→    // 2. Patch the placeholders with the resolved ids (deterministic clone).
   594→   594→    //    An unbound `$request:` token must never reach the engine as a string.
   595→   595→    const patchedSteps: BatchStep[] = [];
   596→   596→    const unresolved: string[] = [];
   597→   597→    for (const step of compiled.steps) {
   598→   598→      const patch = patchPlaceholders(step.params, resolvedById);
   599→   599→      unresolved.push(...patch.unresolved);
   600→   600→      patchedSteps.push({
   601→   601→        command: step.command,
   602→   602→        params: patch.patched as Record<string, unknown>,
   603→   603→      });
   604→   604→    }
   605→   605→    if (unresolved.length > 0) {
   606→   606→      return failApply(plan.runId, {
   607→   607→        ok: false,
   608→   608→        reason: "resolution-failed",
   609→   609→        message: `The plan still references unresolved layers (${unresolved.join(", ")}). Ask the agent to name the layer explicitly.`,
   610→   610→      });
   611→   611→    }
   612→   612→
   613→   613→    const payload = engineBatchPayload({ ...compiled, steps: patchedSteps });
   614→   614→    const batchSha256 = await batchSha256Hex(payload);
   615→   615→
   616→   616→    // 3. ONE engine.batch — law #6 (ONE call, atomic, one undo step).
   617→   617→    try {
   618→   618→      await bridge.execute("engine.batch", payload);
   619→   619→    } catch (error) {
   620→   620→      const failure = bridgeOutcome(error);
   621→   621→      return failApply(plan.runId, {
   622→   622→        ok: false,
   623→   623→        reason: failure.reason,
   624→   624→        message: failure.message,
   625→   625→      });
   626→   626→    }
   627→   627→
   628→   628→    // 4. D-6: upload the applied state as a NEW version (awaited BEFORE the
   629→   629→    //    applyRun report so the gateway always sees a version that exists).
   630→   630→    autosave.markDirty?.(AGENT_APPLY_SAVE_REASON);
   631→   631→    const saved: SaveOutcome = await autosave.saveNow(AGENT_APPLY_SAVE_REASON);
   632→   632→    if (saved.saved !== true || saved.version === undefined) {
   633→   633→      return failApply(
   634→   634→        plan.runId,
   635→   635→        saved.conflict === true
   636→   636→          ? {
   637→   637→              ok: false,
   638→   638→              reason: "save-conflict",
   639→   639→              message:
   640→   640→                "The project changed elsewhere while the agent was working. Reload the latest version to continue — the edit is in the editor only.",
   641→   641→            }
   642→   642→          : {
   643→   643→              ok: false,
   644→   644→              reason: "save-failed",
   645→   645→              message:
   646→   646→                "The edit was applied but could not be saved as a version. It is in the editor only.",
   647→   647→            },
   648→   648→      );
   649→   649→    }
   650→   650→    const versionId = saved.version.id;
   651→   651→
   652→   652→    // 5. Report to the gateway (ledger commit + run.applied, idempotent).
   653→   653→    try {
   654→   654→      const response = await api.runs.applyRun(plan.runId, {
   655→   655→        batchSha256,
   656→   656→        versionId,
   657→   657→        idempotencyKey,
   658→   658→        ...(resolvedForWire.length > 0 ? { resolvedLayers: resolvedForWire } : {}),
   659→   659→      });
   660→   660→      const outcome: ApplyOutcome = {
   661→   661→        ok: true,
   662→   662→        versionId: typeof response.versionId === "string" ? response.versionId : versionId,
   663→   663→        creditsCommitted: response.creditsCommitted,
   664→   664→      };
   665→   665→      notify({
   666→   666→        type: "apply.completed",
   667→   667→        runId: plan.runId,
   668→   668→        versionId: outcome.versionId,
   669→   669→        creditsCommitted: outcome.creditsCommitted,
   670→   670→      });
   671→   671→      return outcome;
   672→   672→    } catch (error) {
   673→   673→      const failure = apiFailure(error);
   674→   674→      if (failure.status === 409) {
   675→   675→        return failApply(plan.runId, {
   676→   676→          ok: false,
   677→   677→          reason: "conflict",
   678→   678→          message:
   679→   679→            "This plan was already applied or the run changed on the server. Nothing was re-executed.",
   680→   680→        });
   681→   681→      }
   682→   682→      if (failure.status === 402) {
   683→   683→        return failApply(plan.runId, {
   684→   684→          ok: false,
   685→   685→          reason: "insufficient-credits",
   686→   686→          message:
   687→   687→            "The edit and its version are saved, but there were not enough credits to commit this run.",
   688→   688→        });
   689→   689→      }
   690→   690→      return failApply(plan.runId, {
   691→   691→        ok: false,
   692→   692→        reason: "failed",
   693→   693→        message:
   694→   694→          failure.message !== ""
   695→   695→            ? failure.message
   696→   696→            : "The edit and its version are saved, but reporting it to the server failed.",
   697→   697→        code: failure.code,
   698→   698→      });
   699→   699→    }
   700→   700→  };
   701→   701→
   702→   702→  return {
   703→   703→    apply: (plan: ApplicablePlan): Promise<ApplyOutcome> => runApply(plan),
   704→   704→
   705→   705→    async undo(runId: string): Promise<UndoOutcome> {
   706→   706→      notify({ type: "undo.started", runId });
   707→   707→      // ONE edit.undo reverts the WHOLE run (batch.rs folds the steps).
   708→   708→      try {
   709→   709→        await bridge.execute("edit.undo");
   710→   710→      } catch (error) {
   711→   711→        const failure = bridgeOutcome(error);
   712→   712→        const outcome: UndoOutcome = NOTHING_TO_UNDO_RE.test(failure.message)
   713→   713→          ? { ok: false, reason: "nothing-to-undo", message: "There is nothing to undo." }
   714→   714→          : { ok: false, reason: failure.reason, message: failure.message };
   715→   715→        // The gateway is NOT told — undoRun would 409 on an unapplied run and
   716→   716→        // the version rows must stay untouched (§9.3).
   717→   717→        notify({ type: "undo.failed", runId, outcome });
   718→   718→        return outcome;
   719→   719→      }
   720→   720→      try {
   721→   721→        await api.runs.undoRun(runId, { idempotencyKey: crypto.randomUUID() });
   722→   722→        notify({ type: "undo.completed", runId });
   723→   723→        return { ok: true, undone: true };
   724→   724→      } catch (error) {
   725→   725→        const failure = apiFailure(error);
   726→   726→        const outcome: UndoOutcome =
   727→   727→          failure.status === 409
   728→   728→            ? {
   729→   729→                ok: false,
   730→   730→                reason: "conflict",
   731→   731→                message:
   732→   732→                  "The server says this run was not applied, so there is nothing to undo there.",
   733→   733→              }
   734→   734→            : {
   735→   735→                ok: false,
   736→   736→                reason: "failed",
   737→   737→                message:
   738→   738→                  failure.message !== ""
   739→   739→                    ? failure.message
   740→   740→                    : "The undo happened in the editor but could not be reported.",
   741→   741→                code: failure.code,
   742→   742→              };
   743→   743→        notify({ type: "undo.failed", runId, outcome });
   744→   744→        return outcome;
   745→   745→      }
   746→   746→    },
   747→   747→
   748→   748→    async reapply(plan: ApplicablePlan): Promise<ApplyOutcome> {
   749→   749→      // §M5.5: only while the session still sits on the plan's base version.
   750→   750→      const current = options.session?.currentVersionId() ?? null;
   751→   751→      if (plan.baseVersionId === undefined || current === null || current !== plan.baseVersionId) {
   752→   752→        return failApply(plan.runId, {
   753→   753→          ok: false,
   754→   754→          reason: "base-version-changed",
   755→   755→          message:
   756→   756→            "The project changed since this plan was made, so re-applying it is not safe. Ask the agent to plan again.",
   757→   757→        });
   758→   758→      }
   759→   759→      return await runApply(plan);
   760→   760→    },
   761→   761→  };
   762→   762→}
