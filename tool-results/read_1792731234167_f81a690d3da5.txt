   292→ * The stateful scripted lease UX (the M5 spec's scenario-B pattern, mounted
   293→ * BEFORE the goto so the banner's mount poll reads it):
   294→ *  - `GET /v1/projects/:id/lease-state` answers "viewer" until the request
   295→ *    lands, then "queued" (my own queue entry rides the queuedAt the scripted
   296→ *    requestLease response pinned — the banner's position fallback), and back

   298→ *  - `POST /v1/projects/:id/lease-request` answers the join outcome;
   299→ *  - `DELETE /v1/projects/:id/lease-request` answers the cancel outcome.
   300→ */
   301→async function installScriptedLeaseFlow(page: Page): Promise<void> {
   302→  const myQueuedAt = new Date(Date.now() - 30_000).toISOString();
   303→  let phase: "viewer" | "queued" = "viewer";
   304→
   305→  await page.route("**/v1/projects/*/lease-state", async (route) => {
   306→    const state =
   307→      phase === "queued"
   308→        ? {
   309→            holder: scriptedHolder(),
   310→            queue: [
   311→              {
   312→                userId: "usr_sam",
   313→                displayName: "Sam Vega",
   314→                queuedAt: new Date(Date.now() - 90_000).toISOString(),
   315→              },
   316→              { userId: "usr_demo_queued", displayName: "Demo User", queuedAt: myQueuedAt },
   317→            ],
   318→            youAre: "queued",
   319→            canAcquire: false,
   320→          }
   321→        : { holder: scriptedHolder(), queue: [], youAre: "viewer", canAcquire: false };
   322→    await route.fulfill(okEnvelope(state));
   323→  });
   324→
   325→  await page.route("**/v1/projects/*/lease-request", async (route) => {
   326→    const method = route.request().method();
   327→    if (method === "DELETE") {
   328→      phase = "viewer";
   329→      await route.fulfill(okEnvelope({ cancelled: true }));
   330→      return;
   331→    }
   332→    phase = "queued";
   333→    await route.fulfill(
   334→      okEnvelope({
   335→        queued: true,
   336→        position: 2,
   337→        queueLength: 2,
   338→        queuedAt: myQueuedAt,
   339→        duplicate: false,
   340→      }),
   341→    );
   342→  });
   343→}

   380→    createdAt: created,
   381→  };
