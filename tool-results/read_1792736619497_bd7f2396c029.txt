     1→/**
     2→ * Client-side apply/undo engine — BUILD_PLAN §M5.4/§M5.5, DECISIONS.md D-6,
     3→ * docs/phases/M5-SEAMS.md (`ApplyController`, pinned).
     4→ *
     5→ * ONE user gesture = ONE atomic `engine.batch` = ONE undo step = ONE version
     6→ * (AGENTS.md hard rule 6). The whole flow NEVER throws — every outcome is a
     7→ * typed union the plan card (M5-G) renders:
     8→ *
     9→ *  `apply(plan)`:
    10→ *    1. validate the `compiledBatch` shape AND allowlist every step id
    11→ *       against the curated agent registry (F-6: `isAgentCommand` via the
    12→ *       api-client runtime-contracts seam) — a non-member id is a typed
    13→ *       `invalid-batch` BEFORE any engine traffic (the engine's registry +
    14→ *       NOT_IN_BATCH rejections stay the defense-in-depth backstop);
    15→ *    2. resolve each `resolutionRequests` entry against the LIVE layer list
    16→ *       (the executor's list_layers path: `comp.info {}` → `layers`) — a name
    17→ *       matching exactly one layer binds it; ambiguous or unknown names fail
    18→ *       BEFORE any engine mutation;
    19→ *    3. patch the `compiledBatch` placeholders with the resolved layer ids
    20→ *       (deterministic deep clone; null-prototype request map — the M4-N F-1
    21→ *       lesson: model-adjacent keys never touch `Object.prototype`);
    22→ *    4. `bridge.execute("engine.batch", batch)` — ONE call. Verified wire:
    23→ *       the bridge passes the params object through `effectcraft.execute` →
    24→ *       `engine.execute` → `batch.rs`, which reads `{steps, label?, atomic?}`
    25→ *       (extra keys such as `timeoutMs` ride inert upstream — `timeoutMs` is
    26→ *       §8.5 bound metadata the compiled batch carries, not an engine param);
    27→ *    5. on success `await autosave.saveNow("agent-apply")` (D-6) — this
    28→ *       uploads a NEW immutable version; the outcome's `version.id` is the
    29→ *       applied versionId;
    30→ *    6. `api.runs.applyRun(runId, {batchSha256, versionId, idempotencyKey})` —
    31→ *       sha256 over the exact patched batch payload (WebCrypto via the
    32→ *       editor-session sha256Hex, with its vector-tested sync fallback for
    33→ *       hosts without `crypto.subtle` — never a new dependency). Gateway
    34→ *       errors map: 409 → conflict (explains, never re-applies) UNLESS the
    35→ *       code is LEASE_HELD (M7/D-23 → the typed `lease-held` outcome, the
    36→ *       server-side read-only guard is authoritative), 402 → insufficient
    37→ *       credits, else typed failure. A LOCAL LEASE_HELD rejection from the
    38→ *       M7 read-only gate (lib/editor-session read-only.ts) maps to the same
    39→ *       `lease-held` outcome before anything reaches the engine.
    40→ *
    41→ *  Retention (law #4; F-L1/F-L2/F-L4 — the `toolresult:<runId>:<callId>`
    42→ *  retention in tool-executor.ts is the precedent): once a run's batch has
    43→ *  EXECUTED, the controller retains per run —
    44→ *   * the gesture's idempotency key + the report pieces (batchSha256,
    45→ *     resolvedLayers), so a RETRY of the same gesture reuses the key
    46→ *     (a fresh gesture after a failure that never executed — or after a
    47→ *     terminal outcome — mints a fresh key), and
    48→ *   * the applied batch's label ("Agent: <summary>", from the compiledBatch),
    49→ *     which `undo` compares against the engine's `{undone: label}` answer.
    50→ *
    51→ *  `retrySave(runId)` (F-L1): the re-save-only recovery for a run whose batch
    52→ *   executed but whose version upload / gateway report did not land — steps
    53→ *   5+6 ONLY; `engine.batch` is NEVER re-executed. The plan card's save-failed
    54→ *   state wires "Retry save" to this; a typed `failed` answers when no
    55→ *   applied-batch context is retained.
    56→ *
    57→ *  `undo(runId, hint?)`: `bridge.execute("edit.undo")` (the whole run folded
    58→ *   into ONE undo step by `batch.rs`). The engine answers `{undone: label}`
    59→ *   (edit.rs undo), which is compared against the run's batch label — the
    60→ *   retained label, or `hint.batchLabel` when the thread layer passes the
    61→ *   planRuntime's compiledBatch label for a controller that never saw the
    62→ *   apply. An UNKNOWN or MISMATCHED pop is a typed
    63→ *   `{ok:false, reason:"undo-stack-mismatch"}` and the gateway is NOT told
    64→ *   (F-L4: the row never claims an undo that didn't happen; the popped
    65→ *   batch sits on the redo stack and the outcome says so). Only a verified
    66→ *   pop reaches `api.runs.undoRun(runId, {idempotencyKey})`. If `edit.undo`
    67→ *   fails ("nothing to undo" — the upstream `can_undo` guard, edit.rs:12)
    68→ *   the gateway is NOT told either (§9.3). The post-undo state is left to
    69→ *   the D-6 cadence + the before-close save (undo is not a D-6 moment).
    70→ *
    71→ *  `reapply(plan)`: only when the client's CURRENT version id equals the plan's
    72→ *   `baseVersionId` (the session port reads it); anything else is
    73→ *   `{ok:false, reason:"base-version-changed"}` — BP §M5.5 "otherwise explain".
    74→ *   NOTE for the integrator: after undo the session's current version is the
    75→ *   APPLIED version's successor chain, so re-apply-after-undo needs the host
    76→ *   page to restore the session's version pointer to the plan's base (a
    77→ *   session-level concern — see the worklog M5-F record).
    78→ *
    79→ * Cross-agent pins (wave 1, reconciled by the integrator):
    80→ *  - placeholder token: a whole-string `"$request:<requestId>"` in the compiled
    81→ *    batch params, replaced with the resolved engine layer id (number);
    82→ *  - `api.runs.applyRun/undoRun` — @studio/api-client names (M5-E adds them);
    83→ *  - the plan event carries `compiledBatch` + `resolutionRequests` +
    84→ *    `baseVersionId` (M5-C adds them to the wire; the SSE reader strips
    85→ *    unknown keys until that lands, so the client treats them as optional).
    86→ */

   415→/** Bridge/editor failure → typed outcome; disconnect codes are named as such. */
   416→function bridgeOutcome(error: unknown): {
   417→  reason: "engine-failed" | "editor-disconnected";
   418→  message: string;
   419→} {
   420→  const message = messageOf(error);
   421→  if (isBridgeError(error) && DISCONNECT_CODES.has(error.code)) {
   422→    return { reason: "editor-disconnected", message };
   423→  }
   424→  return { reason: "engine-failed", message };
   425→}
   426→
   427→/** The typed fields of an api-client throw (ApiError-shaped), without zod. */
   428→function apiFailure(error: unknown): {
   429→  status: number | null;
   430→  code: string | undefined;
   431→  message: string;
   432→} {
   433→  if (typeof error === "object" && error !== null && "status" in error) {
   434→    const record = error as { status?: unknown; code?: unknown; message?: unknown };
   435→    const status = typeof record.status === "number" ? record.status : null;
   436→    const code = typeof record.code === "string" && record.code !== "" ? record.code : undefined;
   437→    const message =
   438→      typeof record.message === "string" && record.message !== "" ? record.message : "";
   439→    return { status, code, message };
   440→  }
   441→  return { status: null, code: undefined, message: "" };
   442→}

   448→interface CompLayerEntry {
   449→  readonly id: number;
   450→  readonly name: string;
   451→}
   452→
   453→/** `comp.info {}` → the active composition's layer list (query.rs; read-only). */
   454→async function listLayers(bridge: EditorBridge): Promise<readonly CompLayerEntry[]> {
   455→  const info = await bridge.query<{ layers?: unknown }>("comp.info", {});
   456→  const raw = info?.layers;
   457→  if (!Array.isArray(raw)) return [];
   458→  const out: CompLayerEntry[] = [];
   459→  for (const entry of raw) {
   460→    if (typeof entry !== "object" || entry === null) continue;
   461→    const record = entry as Record<string, unknown>;
   462→    if (typeof record.id !== "number" || typeof record.name !== "string") continue;
   463→    out.push({ id: record.id, name: record.name });
   464→  }
   465→  return out;
   466→}
   467→
   468→/**
   469→ * Resolves every request against the live layer list. Read-only: the only
   470→ * bridge traffic is ONE `comp.info` query however many requests there are.
   471→ * Ambiguous/unknown names fail BEFORE anything executes.
   472→ */
   473→async function resolveRequests(
   474→  requests: readonly LayerResolutionRequest[],
   475→  bridge: EditorBridge,
   476→): Promise<
   477→  { ok: true; resolved: readonly ResolvedLayer[] } | { ok: false; outcome: ApplyOutcome }
   478→> {
   479→  let layers: readonly CompLayerEntry[];
   480→  try {
   481→    layers = await listLayers(bridge);
   482→  } catch (error) {
   483→    const failure = bridgeOutcome(error);
   484→    const outcome: ApplyOutcome =
   485→      failure.reason === "editor-disconnected"
   486→        ? { ok: false, reason: "editor-disconnected", message: failure.message }
   487→        : {
   488→            ok: false,
   489→            reason: "resolution-failed",
   490→            message: "The layers could not be listed to resolve this plan. Try again in a moment.",
   491→          };
   492→    return { ok: false, outcome };
   493→  }
   494→
   495→  const resolved: ResolvedLayer[] = [];
   496→  for (const request of requests) {
   497→    // Exact-name matches only — a fuzzy match could bind the WRONG layer.
   498→    const matches = layers.filter((layer) => layer.name === request.layerName);
   499→    const first = matches[0];
   500→    if (matches.length === 1 && first !== undefined) {
   501→      resolved.push({ requestId: request.requestId, layerId: first.id, layerName: first.name });
   502→      continue;
   503→    }
   504→    if (matches.length > 1) {
   505→      return {
   506→        ok: false,
   507→        outcome: {
   508→          ok: false,
   509→          reason: "ambiguous",
   510→          candidates: matches.map((layer) => `${layer.name} (layer ${layer.id})`),
   511→          message: `Layer "${request.layerName}" matches ${matches.length} layers. Say which one you mean.`,
   512→        },
   513→      };
   514→    }
   515→    return {
   516→      ok: false,
   517→      outcome: {
   518→        ok: false,
   519→        reason: "not-found",
   520→        layerName: request.layerName,
   521→        message: `No layer named "${request.layerName}" is in the composition.`,
   522→      },
   523→    };
   524→  }
   525→  return { ok: true, resolved };
   526→}
   527→
   528→// ---------------------------------------------------------------------------
   529→// Placeholder patch (deterministic deep clone; F-1-safe accumulators)
   530→// ---------------------------------------------------------------------------
   531→
   532→/**
   533→ * Replaces every whole-string `"$request:<requestId>"` placeholder in the batch
   534→ * params with the resolved layer id. Deterministic: only exact matches swap
   535→ * and the clone preserves object key order, so the sha256 is stable.
   536→ *
   537→ * The cloned objects are NULL-PROTOTYPE (M4-N F-1): plan params are
   538→ * model-adjacent — a parsed `{"__proto__": …}` key must ride as OWN data, not
   539→ * rebind the clone's prototype.
   540→ */
   541→function patchPlaceholders(
   542→  value: unknown,
   543→  resolvedById: Readonly<Record<string, number>>,
   544→): { patched: unknown; unresolved: string[] } {
   545→  const unresolved: string[] = [];
   546→  const walk = (node: unknown): unknown => {
   547→    if (typeof node === "string") {
   548→      if (!node.startsWith(RESOLVE_TOKEN_PREFIX)) return node;
   549→      const requestId = node.slice(RESOLVE_TOKEN_PREFIX.length);
   550→      const layerId = resolvedById[requestId];
   551→      if (layerId === undefined) {
   552→        unresolved.push(node);
   553→        return node;
   554→      }
   555→      return layerId;
   556→    }
   557→    if (Array.isArray(node)) return node.map(walk);
   558→    if (typeof node === "object" && node !== null) {
   559→      const out: Record<string, unknown> = Object.create(null);
   560→      for (const [key, child] of Object.entries(node)) out[key] = walk(child);
   561→      return out;
   562→    }
   563→    return node;
   564→  };
   565→  return { patched: walk(value), unresolved };
   566→}
   567→
   568→/** Narrow hand-rolled validation of the compiled batch (no zod in apps/web). */
   569→function validCompiledBatch(batch: unknown): AgentBatch | null {
   570→  if (typeof batch !== "object" || batch === null) return null;
   571→  const record = batch as Record<string, unknown>;
   572→  if (typeof record.label !== "string" || record.label.length === 0) return null;
   573→  if (!Array.isArray(record.steps) || record.steps.length === 0) return null;
   574→  const steps: BatchStep[] = [];
   575→  for (const step of record.steps) {
   576→    if (typeof step !== "object" || step === null) return null;
   577→    const stepRecord = step as Record<string, unknown>;
   578→    if (typeof stepRecord.command !== "string" || stepRecord.command.length === 0) return null;
   579→    if (typeof stepRecord.params !== "object" || stepRecord.params === null) return null;
   580→    steps.push({
   581→      command: stepRecord.command,
   582→      params: stepRecord.params as Record<string, unknown>,
   583→    });
   584→  }
   585→  const timeoutMs =
   586→    typeof record.timeoutMs === "number" &&
   587→    Number.isInteger(record.timeoutMs) &&
   588→    record.timeoutMs > 0
   589→      ? record.timeoutMs
   590→      : undefined;
   591→  // Agent batches are ALWAYS atomic (§8.5) — forced regardless of the wire.
   592→  return timeoutMs === undefined
   593→    ? { label: record.label, atomic: true, steps }
   594→    : { label: record.label, atomic: true, steps, timeoutMs };
   595→}

   598→// sha256 of the exact engine payload

   600→
   601→/** The exact params object handed to `bridge.execute("engine.batch", …)`. */
   602→export function engineBatchPayload(batch: AgentBatch): {
   603→  label: string;
   604→  atomic: true;
   605→  steps: readonly BatchStep[];
   606→  timeoutMs?: number;
   607→} {
   608→  return batch.timeoutMs === undefined
   609→    ? { label: batch.label, atomic: true, steps: batch.steps }
   610→    : { label: batch.label, atomic: true, steps: batch.steps, timeoutMs: batch.timeoutMs };
   611→}
   612→
   613→/**
   614→ * sha256 over the serialized engine payload — the exact shape the editor
   615→ * receives (JSON key order is deterministic: the payload object is built with
   616→ * a fixed key order and the patched params preserve theirs).
   617→ */
   618→export async function batchSha256Hex(payload: unknown): Promise<string> {
   619→  const serialized = JSON.stringify(payload) ?? "null";
   620→  return sha256Hex(new TextEncoder().encode(serialized));
   621→}
   622→
   623→// ---------------------------------------------------------------------------
   624→// The controller
   625→// ---------------------------------------------------------------------------
   626→
   627→/**
   628→ * What the controller retains per run ONCE ITS BATCH EXECUTED (F-L1/F-L2;
   629→ * the `toolresult:<runId>:<callId>` retention in tool-executor.ts is the
   630→ * precedent): the apply gesture's idempotency key — a RETRY of the same
   631→ * gesture reuses it; a fresh gesture after a failure that never executed, or
   632→ * after a terminal outcome (applied / 409 / 402), mints a fresh key — plus
   633→ * the pieces the save + report steps need. Cleared on every terminal outcome;
   634→ * retained while a save or report is still owed.
   635→ */
   636→interface RunApplyRecord {
   637→  readonly idempotencyKey: string;
   638→  readonly batchSha256: string;
   639→  readonly resolvedForWire: readonly PostedResolvedLayer[];
   640→}
   641→
   642→export function createApplyController(options: CreateApplyControllerOptions): ApplyController {
   643→  const { bridge, autosave, api } = options;
   644→
   645→  /** Retained per-run apply-gesture records (F-L1/F-L2). */
   646→  const runRecords = new Map<string, RunApplyRecord>();
   647→  /** Retained per-run applied-batch labels (F-L4) — they OUTLIVE a successful
   648→   * apply (undo needs them) and are spent by a VERIFIED pop. */
   649→  const appliedLabels = new Map<string, string>();
   650→
   651→  const notify = (event: ApplyEngineEvent): void => {
   652→    const callback = options.onEvent;
   653→    if (callback === undefined) return;
   654→    try {
   655→      callback(event);
   656→    } catch {
   657→      /* a host callback must never break the apply engine */
   658→    }
   659→  };
   660→
   661→  const failApply = (runId: string, outcome: ApplyOutcome): ApplyOutcome => {
   662→    notify({ type: "apply.failed", runId, outcome });
   663→    return outcome;
   664→  };
   665→
   666→  /**
   667→   * Steps 5+6 of the apply flow — shared by apply/reapply AND the F-L1
   668→   * re-save-only retry (which starts here, never touching the engine):
   669→   * upload the applied state as a NEW version (awaited BEFORE the applyRun
   670→   * report so the gateway always sees a version that exists), then report.
   671→   */
   672→  const saveAndReport = async (runId: string, record: RunApplyRecord): Promise<ApplyOutcome> => {
   673→    const saved: SaveOutcome = await autosave.saveNow(AGENT_APPLY_SAVE_REASON);
   674→    if (saved.saved !== true || saved.version === undefined) {
   675→      // No version ⇒ the gateway is never told "applied". The divergence
   676→      // (engine applied, version not) is surfaced honestly via
   677→      // `batchApplied` + the copy, and the record STAYS so the card's
   678→      // Retry save (retrySave) re-runs ONLY these steps (F-L1).
   679→      return failApply(
   680→        runId,
   681→        saved.conflict === true
   682→          ? {
   683→              ok: false,
   684→              reason: "save-conflict",
   685→              batchApplied: true,
   686→              message:
   687→                "The project changed elsewhere while the agent was working. Reload the latest version to continue — the edit is in the editor only.",
   688→            }
   689→          : {
   690→              ok: false,
   691→              reason: "save-failed",
   692→              batchApplied: true,
   693→              message:
   694→                "The edit was applied but could not be saved as a version. It is in the editor only.",
   695→            },
   696→      );
   697→    }
   698→    const versionId = saved.version.id;
   699→
   700→    try {
   701→      const response = await api.runs.applyRun(runId, {
   702→        batchSha256: record.batchSha256,
   703→        versionId,
   704→        idempotencyKey: record.idempotencyKey,
   705→        ...(record.resolvedForWire.length > 0 ? { resolvedLayers: record.resolvedForWire } : {}),
   706→      });
   707→      // Terminal success: the gesture reported — release the retained key.
   708→      runRecords.delete(runId);
   709→      const outcome: ApplyOutcome = {
   710→        ok: true,
   711→        versionId: typeof response.versionId === "string" ? response.versionId : versionId,
   712→        creditsCommitted: response.creditsCommitted,
   713→      };
   714→      notify({
   715→        type: "apply.completed",
   716→        runId,
   717→        versionId: outcome.versionId,
   718→        creditsCommitted: outcome.creditsCommitted,
   719→      });
   720→      return outcome;
   721→    } catch (error) {
   722→      const failure = apiFailure(error);
   723→      if (failure.status === 409 || failure.status === 402) {
   724→        // Terminal server answers: release the retained key (a retry after
   725→        // these is a NEW gesture — a retained key would make the idempotency
   726→        // middleware replay the cached 409/402 answer forever).
   727→        runRecords.delete(runId);
   728→      }
   729→      if (failure.status === 409) {
   730→        // M7/D-23 (additive conflict arm): a 409 carrying the wire's
   731→        // LEASE_HELD code is the server-side read-only guard refusing the
   732→        // report — the server is AUTHORITATIVE, so surface it with its own
   733→        // typed outcome and the gateway's copy (which names the holder).
   734→        // The batch + version DID land; only the report is owed (the
   735→        // idempotency middleware replays it once the lease is ours again).
   736→        if (failure.code === "LEASE_HELD") {
   737→          return failApply(runId, {
   738→            ok: false,
   739→            reason: "lease-held",
   740→            batchApplied: true,
   741→            message:
   742→              failure.message !== ""
   743→                ? `${failure.message} The edit and its version are saved, but the run could not be recorded — take the edit lease to finish reporting it.`
   744→                : "Another member is editing this project right now. The edit and its version are saved, but the run could not be recorded — take the edit lease to finish reporting it.",
   745→          });
   746→        }
   747→        return failApply(runId, {
   748→          ok: false,
   749→          reason: "conflict",
   750→          message:
   751→            "This plan was already applied or the run changed on the server. Nothing was re-executed.",
   752→        });
   753→      }
   754→      if (failure.status === 402) {
   755→        return failApply(runId, {
   756→          ok: false,
   757→          reason: "insufficient-credits",
   758→          message:
   759→            "The edit and its version are saved, but there were not enough credits to commit this run.",
   760→        });
   761→      }
   762→      // Network/other failures keep the record: a retry of THIS gesture
   763→      // reuses the key (F-L2) and the gateway's {batchSha256, versionId}
   764→      // replay branch dedupes server-side.
   765→      return failApply(runId, {
   766→        ok: false,
   767→        reason: "failed",
   768→        message:
   769→          failure.message !== ""
   770→            ? failure.message
   771→            : "The edit and its version are saved, but reporting it to the server failed.",
   772→        code: failure.code,
   773→      });
   774→    }
   775→  };
   776→
   777→  /** The shared execute-then-save-then-report path (apply + reapply). */
   778→  const runApply = async (plan: ApplicablePlan): Promise<ApplyOutcome> => {
   779→    notify({ type: "apply.started", runId: plan.runId });
   780→
   781→    const compiled = validCompiledBatch(plan.compiledBatch);
   782→    if (compiled === null) {
   783→      return failApply(plan.runId, {
   784→        ok: false,
   785→        reason: "invalid-batch",
   786→        message: "The plan's compiled batch is malformed. Ask the agent to plan again.",
   787→      });
   788→    }
   789→    // F-6 — the client allowlist gate: every step id must be a curated agent
   790→    // command BEFORE anything reaches the engine (bounds/shape checks alone
   791→    // let hostile ids through to the engine's own registry, the finding).
   792→    if (compiled.steps.some((step) => !isAgentCommand(step.command))) {
   793→      return failApply(plan.runId, {
   794→        ok: false,
   795→        reason: "invalid-batch",
   796→        message:
   797→          "The plan's compiled batch contains a command the agent may not run. Ask the agent to plan again.",
   798→      });
   799→    }
   800→
   801→    // One gesture = one idempotency key (law #4). F-L2: a retained record
   802→    // means this is a RETRY of a gesture whose batch already executed — reuse
   803→    // its key (the toolresult:<runId>:<callId> precedent). Fresh gestures
   804→    // after pre-execution failures / terminal outcomes mint fresh keys.
   805→    const retained = runRecords.get(plan.runId);
   806→    const idempotencyKey = retained?.idempotencyKey ?? crypto.randomUUID();
   807→
   808→    // 1. Resolve layer names against the live list — read-only; on any
   809→    //    ambiguity NOTHING executes (no engine.batch, no save, no applyRun).
   810→    let resolvedById: Readonly<Record<string, number>> = {};
   811→    let resolvedForWire: readonly PostedResolvedLayer[] = [];
   812→    const requests = plan.resolutionRequests ?? [];
   813→    if (requests.length > 0) {
   814→      const resolution = await resolveRequests(requests, bridge);
   815→      if (!resolution.ok) return failApply(plan.runId, resolution.outcome);
   816→      notify({ type: "apply.resolved", runId: plan.runId, resolved: resolution.resolved });
   817→      // Null-prototype accumulator (M4-N F-1): request ids are model-adjacent
   818→      // strings — on a plain `{}` a `__proto__` id would hit the accessor.
   819→      const map = Object.create(null) as Record<string, number>;
   820→      for (const entry of resolution.resolved) map[entry.requestId] = entry.layerId;
   821→      resolvedById = map;
   822→      // §8.7/M5-E: the name→id bindings ride the applyRun body (auditable
   823→      // server-side even though the batch already ran client-side).
   824→      resolvedForWire = resolution.resolved.map((entry) => ({
   825→        requestId: entry.requestId,
   826→        layerId: entry.layerId,
   827→      }));
   828→    }
   829→
   830→    // 2. Patch the placeholders with the resolved ids (deterministic clone).
   831→    //    An unbound `$request:` token must never reach the engine as a string.
   832→    const patchedSteps: BatchStep[] = [];
   833→    const unresolved: string[] = [];
   834→    for (const step of compiled.steps) {
   835→      const patch = patchPlaceholders(step.params, resolvedById);
   836→      unresolved.push(...patch.unresolved);
   837→      patchedSteps.push({
   838→        command: step.command,
   839→        params: patch.patched as Record<string, unknown>,
   840→      });
   841→    }
   842→    if (unresolved.length > 0) {
   843→      return failApply(plan.runId, {
   844→        ok: false,
   845→        reason: "resolution-failed",
   846→        message: `The plan still references unresolved layers (${unresolved.join(", ")}). Ask the agent to name the layer explicitly.`,
   847→      });
   848→    }
   849→
   850→    const payload = engineBatchPayload({ ...compiled, steps: patchedSteps });
   851→    const batchSha256 = await batchSha256Hex(payload);
   852→
   853→    // 3. ONE engine.batch — law #6 (ONE call, atomic, one undo step).
   854→    try {
   855→      await bridge.execute("engine.batch", payload);
   856→    } catch (error) {
   857→      // M7/D-23 (additive): the LOCAL read-only gate (lib/editor-session
   858→      // read-only.ts) rejects the batch with the typed LEASE_HELD outcome
   859→      // BEFORE it reaches the engine — nothing executed, nothing saved, the
   860→      // gateway was never told. Honest copy names the holder (never an id).
   861→      if (isReadOnlyLeaseError(error)) {
   862→        return failApply(plan.runId, {
   863→          ok: false,
   864→          reason: "lease-held",
   865→          message: `${error.message} The plan was not applied — request edit access, or take the lease when it is free, then apply again.`,
   866→        });
   867→      }
   868→      const failure = bridgeOutcome(error);
   869→      return failApply(plan.runId, {
   870→        ok: false,
   871→        reason: failure.reason,
   872→        message: failure.message,
   873→      });
   874→    }
   875→
   876→    // The batch executed: retain the gesture key + report pieces for retries
   877→    // (F-L1/F-L2) and the batch label for the undo verification (F-L4).
   878→    const record: RunApplyRecord = { idempotencyKey, batchSha256, resolvedForWire };
   879→    runRecords.set(plan.runId, record);
   880→    appliedLabels.set(plan.runId, compiled.label);
   881→
   882→    // 4. D-6: upload the applied state as a NEW version, then report (the
   883→    // shared steps 5+6 — also the whole of the F-L1 re-save-only retry).
   884→    autosave.markDirty?.(AGENT_APPLY_SAVE_REASON);
   885→    return await saveAndReport(plan.runId, record);
   886→  };
   887→
   888→  return {
   889→    apply: (plan: ApplicablePlan): Promise<ApplyOutcome> => runApply(plan),
   890→
   891→    /**
   892→     * F-L1 — the RE-SAVE-ONLY recovery: the engine step is not in question
   893→     * (the batch already executed), so `engine.batch` is NEVER re-run; only
   894→     * the D-6 version upload + the gateway report retry, with the retained
   895→     * gesture key and the original batchSha256/resolvedLayers.
   896→     */
   897→    retrySave: async (runId: string): Promise<ApplyOutcome> => {
   898→      const record = runRecords.get(runId);
   899→      if (record === undefined) {
   900→        // Nothing executed-and-unsaved through THIS controller: an honest
   901→        // typed refusal — never a silent re-execute of anything.
   902→        return failApply(runId, {
   903→          ok: false,
   904→          reason: "failed",
   905→          message:
   906→            "This plan has no applied edit to re-save in this session. If the edit is still in the editor, save a version from the editor menu.",
   907→        });
   908→      }
   909→      return await saveAndReport(runId, record);
   910→    },
   911→
   912→    async undo(runId: string, hint?: UndoHint): Promise<UndoOutcome> {
   913→      notify({ type: "undo.started", runId });
   914→      // The run's applied-batch label (F-L4): the retained label wins; the
   915→      // optional hint (the thread layer passes the planRuntime's
   916→      // compiledBatch label) covers a controller that never saw the apply.
   917→      const expectedLabel = appliedLabels.get(runId) ?? hint?.batchLabel;
   918→      // ONE edit.undo reverts the WHOLE run (batch.rs folds the steps); the
   919→      // engine answers {undone: label} (edit.rs undo).
   920→      let popped: { undone?: unknown } | undefined;
   921→      try {
   922→        popped = await bridge.execute<{ undone?: unknown }>("edit.undo");
   923→      } catch (error) {
   924→        // M7/D-23 (additive): the LOCAL read-only gate refused the undo —
   925→        // the engine was never touched and the gateway is not told.
   926→        if (isReadOnlyLeaseError(error)) {
   927→          const outcome: UndoOutcome = {
   928→            ok: false,
   929→            reason: "lease-held",
   930→            message: `${error.message} The undo was not performed — take the edit lease first.`,
   931→          };
   932→          notify({ type: "undo.failed", runId, outcome });
   933→          return outcome;
   934→        }
   935→        const failure = bridgeOutcome(error);
   936→        const outcome: UndoOutcome = NOTHING_TO_UNDO_RE.test(failure.message)
   937→          ? { ok: false, reason: "nothing-to-undo", message: "There is nothing to undo." }
   938→          : { ok: false, reason: failure.reason, message: failure.message };
   939→        // The gateway is NOT told — undoRun would 409 on an unapplied run and
   940→        // the version rows must stay untouched (§9.3).
   941→        notify({ type: "undo.failed", runId, outcome });
   942→        return outcome;
   943→      }
   944→      const poppedLabel =
   945→        popped !== undefined && typeof popped.undone === "string" ? popped.undone : undefined;
   946→      // F-L4 — the stack-blind undo guard: verify the engine popped THE
   947→      // RUN'S batch before telling the gateway. An unknown or mismatched pop
   948→      // is surfaced honestly (the popped edit sits on the redo stack) and
   949→      // the gateway row is never told an undo that didn't happen.
   950→      if (poppedLabel === undefined || poppedLabel !== expectedLabel) {
   951→        // The popped batch belongs to a different gesture: its run's retained
   952→        // label + record (if any) are now stale — the applied state left the
   953→        // stack through this run's undo.
   954→        if (poppedLabel !== undefined) {
   955→          for (const [id, label] of appliedLabels) {
   956→            if (label === poppedLabel) {
   957→              appliedLabels.delete(id);
   958→              runRecords.delete(id);
   959→            }
   960→          }
   961→        }
   962→        const outcome: UndoOutcome = {
   963→          ok: false,
   964→          reason: "undo-stack-mismatch",
   965→          message:
   966→            poppedLabel === undefined
   967→              ? "The editor undid an unnamed edit, so this run's undo could not be verified. The server was not told — redo restores it."
   968→              : expectedLabel === undefined
   969→                ? "This plan's edit could not be matched in this session's undo stack, so its undo was not reported. The server was not told — redo restores the edit the editor undid."
   970→                : "A different edit was on top of the undo stack, so this run was not marked undone. Undo that edit first, or redo to restore it.",
   971→        };
   972→        notify({ type: "undo.failed", runId, outcome });
   973→        return outcome;
   974→      }
   975→      // Verified: THE RUN'S batch popped — the retained label is spent.
   976→      appliedLabels.delete(runId);
   977→      try {
   978→        await api.runs.undoRun(runId, { idempotencyKey: crypto.randomUUID() });
   979→        notify({ type: "undo.completed", runId });
   980→        return { ok: true, undone: true };
   981→      } catch (error) {
   982→        const failure = apiFailure(error);
   983→        let outcome: UndoOutcome;
   984→        if (failure.status === 409 && failure.code === "LEASE_HELD") {
   985→          // M7/D-23 (additive conflict arm): the server-side lease guard
   986→          // refused the undo report (someone else holds the lease). The undo
   987→          // DID happen in the engine; the gateway row was not touched —
   988→          // honest copy, no re-attempt while blocked.
   989→          outcome = {
   990→            ok: false,
   991→            reason: "lease-held",
   992→            message:
   993→              failure.message !== ""
   994→                ? `${failure.message} The undo happened in the editor, but the server could not record it — take the edit lease to finish reporting it.`
   995→                : "Another member is editing this project right now. The undo happened in the editor, but the server could not record it — take the edit lease to finish reporting it.",
   996→          };
   997→        } else if (failure.status === 409) {
   998→          outcome = {
   999→            ok: false,
  1000→            reason: "conflict",
  1001→            message: "The server says this run was not applied, so there is nothing to undo there.",
  1002→          };
  1003→        } else {
  1004→          outcome = {
  1005→            ok: false,
  1006→            reason: "failed",
  1007→            message:
  1008→              failure.message !== ""
  1009→                ? failure.message
  1010→                : "The undo happened in the editor but could not be reported.",
  1011→            code: failure.code,
  1012→          };
  1013→        }
  1014→        notify({ type: "undo.failed", runId, outcome });
  1015→        return outcome;
  1016→      }
  1017→    },

  1019→    async reapply(plan: ApplicablePlan): Promise<ApplyOutcome> {
