   194→function defaultSleep(ms: number): Promise<void> {
   195→  if (ms <= 0) return Promise.resolve();
   196→  return new Promise((resolve) => {
   197→    setTimeout(resolve, ms);
   198→  });
   199→}

   201→/** Serialized query string; `undefined`/`null` values are skipped. */
   202→export function buildQuery(
   203→  params: Record<string, string | number | boolean | null | undefined>,
   204→): string {
   205→  const search = new URLSearchParams();
   206→  for (const [key, value] of Object.entries(params)) {
   207→    if (value === undefined || value === null) continue;
   208→    search.set(key, String(value));
   209→  }
   210→  const text = search.toString();
   211→  return text === "" ? "" : `?${text}`;
   212→}

   229→interface ClientInternals {
   230→  readonly baseUrl: string;
   231→  readonly fetchImpl: FetchLike;
   232→  readonly extraHeaders: Record<string, string>;

   236→}

   241→
   242→class ApiClientImpl {
   243→  constructor(readonly core: ClientInternals) {}
   244→
   245→  /**
   246→   * One HTTP call: envelope unwrap, response validation, GET retry with
   247→   * backoff. Never surfaces raw response text — errors are typed ApiErrors.
   248→   */
   249→  async request<T>(plan: RequestPlan<T>): Promise<T> {
   250→    const url = `${this.core.baseUrl}${plan.path}${plan.query ? buildQuery(plan.query) : ""}`;
   251→
   252→    let attempt = 0;
   253→    for (;;) {
   254→      let status = 0;
   255→      let statusText = "";
   256→      let text = "";
   257→      try {
   258→        const init = this.buildInit(plan);
   259→        const response = await this.core.fetchImpl(url, init);
   260→        status = response.status;
   261→        statusText = response.statusText;
   262→        text = await response.text();
   263→      } catch (error) {
   264→        if (isAbortError(error) || plan.signal?.aborted) throw error;
   265→        const netError = networkError(error);
   266→        if (!this.shouldRetry(plan, netError, attempt)) throw netError;
   267→        await this.backoff(plan, attempt);
   268→        attempt += 1;
   269→        continue;
   270→      }
   271→
   272→      if (status < 200 || status > 299) {
   273→        const apiError = await fromResponse({
   274→          status,
   275→          statusText,
   276→          text: async () => text,
   277→        });
   278→        if (!this.shouldRetry(plan, apiError, attempt)) throw apiError;
   279→        await this.backoff(plan, attempt);
   280→        attempt += 1;
   281→        continue;
   282→      }
   283→
   284→      return this.unwrapOk<T>(status, text, plan);
   285→    }
   286→  }
   287→
   288→  private buildInit(plan: RequestPlan<unknown>): RequestInit {
   289→    const headers = new Headers(this.core.extraHeaders);
   290→    headers.set("Accept", "application/json");
   291→    if (plan.body !== undefined) headers.set("Content-Type", "application/json");
   292→    if (plan.idempotencyKey !== undefined) headers.set("Idempotency-Key", plan.idempotencyKey);
   293→    const init: RequestInit = {
   294→      method: plan.method,
   295→      headers,
   296→      signal: plan.signal,
   297→    };
   298→    if (plan.body !== undefined) init.body = JSON.stringify(plan.body);
   299→    return init;
   300→  }
   301→
   302→  /** Only idempotent-safe GETs retry, and only when the failure is retryable. */
   303→  private shouldRetry(plan: RequestPlan<unknown>, error: ApiError, attempt: number): boolean {
   304→    if (plan.method !== "GET") return false;
   305→    if (attempt >= this.core.retryDelays.length) return false;
   306→    if (error.status === 0) return true; // network error before any response
   307→    if (error.status === 408 || error.status === 429 || error.status >= 500) {
   308→      return error.retryable;
   309→    }
   310→    return false;
   311→  }
   312→
   313→  private backoff(plan: RequestPlan<unknown>, attempt: number): Promise<void> {
   314→    const delay = this.core.retryDelays[attempt] ?? 0;
   315→    return sleepWithAbort(this.core.sleep, delay, plan.signal);
   316→  }
   317→
   318→  /** 2xx handling: empty body → `null`; otherwise a §7.3 Ok envelope is required. */
   319→  private unwrapOk<T>(status: number, text: string, plan: RequestPlan<T>): T {
   320→    if (text === "") {
   321→      if (plan.responseSchema === undefined) return null as T;
   322→      throw malformedResponseError(status);
   323→    }
   324→    let json: unknown;
   325→    try {
   326→      json = JSON.parse(text);
   327→    } catch {
   328→      throw malformedResponseError(status);
   329→    }
   330→    if (isErr(json)) {
   331→      const error: Err["error"] = json.error;
   332→      // A 2xx status with an Err body is still the §7.3 failure arm.
   333→      throw new ApiError({
   334→        code: error.code,
   335→        message: error.message,
   336→        hint: error.hint,
   337→        retryable: error.retryable,
   338→        requestId: error.requestId,
   339→        status,
   340→        cause: json,
   341→      });
   342→    }
   343→    if (!isOk(json)) throw malformedResponseError(status);
   344→    const data = json.data;
   345→    if (plan.responseSchema === undefined) return data as T;
   346→    try {
   347→      return plan.responseSchema.parse(data);
   348→    } catch (error) {
   349→      throw malformedResponseError(status, error);
   350→    }
   351→  }
   352→
   353→  /** Client-side §7.2 request validation — fails fast, never sends. */
   354→  validateBody<T>(schema: z.ZodType<T>, body: unknown, path: string): T {
   355→    try {
   356→      return schema.parse(body);
   357→    } catch (error) {
   358→      if (isApiError(error)) throw error;
   359→      throw invalidRequestError(
   360→        Object.assign(new Error(`invalid body for ${path}`), { cause: error }),
   361→      );
   362→    }
   363→  }
   364→
   365→  autoKey(): string {
   366→    return this.core.newKey();
   367→  }
   368→
   369→  // -- GETs don't send Idempotency-Key; mutations always do (auto when absent).
   370→  async get<T>(
   371→    path: string,
   372→    query: Record<string, string | number | boolean | undefined> | undefined,
   373→    schema: z.ZodType<T>,
   374→    signal?: AbortSignal,
   375→  ): Promise<T> {
   376→    return this.request<T>({ method: "GET", path, query, responseSchema: schema, signal });
   377→  }
   378→
   379→  async mutate<T>(
   380→    method: "POST" | "PATCH" | "DELETE",
   381→    path: string,
   382→    plan: {
   383→      body?: unknown;
   384→      bodySchema?: z.ZodType<unknown>;
   385→      responseSchema?: z.ZodType<T>;
   386→      idempotencyKey?: string;
   387→      signal?: AbortSignal;
   388→    },
   389→  ): Promise<T> {
   390→    const body =
   391→      plan.bodySchema !== undefined
   392→        ? this.validateBody(plan.bodySchema, plan.body, path)
   393→        : plan.body;
   394→    return this.request<T>({
   395→      method,
   396→      path,
   397→      body,
   398→      responseSchema: plan.responseSchema,
   399→      idempotencyKey: plan.idempotencyKey ?? this.autoKey(),
   400→      signal: plan.signal,
   401→    });
   402→  }
   403→}

   551→/**
   552→ * Builds the typed API client. `baseUrl` includes the version segment
   553→ * (default `"/v1"` for same-origin browser use). All mutation methods send
   554→ * `Idempotency-Key` (§7.2) — pass your own per logical operation, or let the
   555→ * client derive one.
   556→ */
   557→export function createApiClient(options: ApiClientOptions = {}): ApiClient {
   558→  const core: ClientInternals = {
   559→    baseUrl: (options.baseUrl ?? DEFAULT_BASE_URL).replace(/\/+$/, ""),
   560→    fetchImpl: options.fetch ?? ((input, init) => fetch(input, init)),
   561→    extraHeaders: { ...(options.headers ?? {}) },
   562→    newKey: options.idempotencyKey ?? (() => crypto.randomUUID()),
   563→    sleep: options.sleep ?? defaultSleep,
   564→    retryDelays: options.retry?.delaysMs ?? RETRY_DELAYS_MS,
   565→  };
   566→  const impl = new ApiClientImpl(core);
   567→
   568→  const mutationKey = (options: CallOptions | undefined): string | undefined => {

   863→/** Two-phase version upload (§9.2/§9.3): descriptor POST + signed-URL byte PUT. */
   864→async function uploadVersion(
   865→  impl: ApiClientImpl,
   866→  projectId: string,
   867→  input: {
   868→    readonly baseVersionId: string;
   869→    readonly sha256: string;
   870→    readonly label?: string;
   871→    readonly agentRunId?: string;
   872→    readonly bytes: Blob | Uint8Array;
   873→  },
   874→  options: CallOptions | undefined,
   875→): Promise<ProjectVersion> {
   876→  const { bytes } = input;
   877→  if (!(bytes instanceof Blob) && !(bytes instanceof Uint8Array)) {
   878→    throw invalidRequestError(new Error("bytes must be a Blob or Uint8Array"));
   879→  }
   880→  const sizeBytes = bytes instanceof Blob ? bytes.size : bytes.byteLength;
   881→  const descriptor = impl.validateBody(
   882→    UPLOAD_VERSION_SCHEMA,
   883→    {
   884→      baseVersionId: input.baseVersionId,
   885→      sha256: input.sha256,
   886→      label: input.label,
   887→      agentRunId: input.agentRunId,
   888→      sizeBytes,
   889→    },
   890→    "upload version descriptor",
   891→  );
   892→
   893→  const response = await impl.mutate<UploadVersionResponse>(
   894→    "POST",
   895→    `/projects/${encodeId(projectId)}/versions`,
   896→    {
   897→      body: descriptor,
   898→      responseSchema: UPLOAD_VERSION_RESPONSE_SCHEMA,
   899→      idempotencyKey: options?.idempotencyKey ?? impl.autoKey(),
   900→      signal: options?.signal,
   901→    },
   902→  );
   903→
   904→  // Phase two: bytes go straight to object storage via the signed URL — the
   905→  // storage response is NOT a §7.3 envelope, so only the status is checked.
   906→  const { url, headers } = response.upload;
   907→  const put = await impl.core.fetchImpl(url, {
   908→    method: "PUT",
   909→    headers: { ...headers },
   910→    // TS 5.7's DOM BodyInit doesn't accept Uint8Array<ArrayBufferLike>
   911→    // (SharedArrayBuffer edge); runtime fetch (browser + undici) does.
   912→    body: bytes instanceof Blob ? bytes : (bytes as unknown as BodyInit),
   913→    signal: options?.signal,
   914→  });
   915→  if (!put.ok) {
   916→    throw uploadFailedError(put.status);
   917→  }
   918→  return response.version;
   919→}
