     1→/**
     2→ * The M4 adversarial injection corpus (Task M4-N) — exported, reusable
     3→ * fixtures proving that UNTRUSTED project text survives every boundary as
     4→ * DATA: gateway prompt assembly, the LiteLLM runner's tool allowlist, args
     5→ * Zod validation, EditPlan schema bounds, log redaction (M4.3), the web
     6→ * markdown renderer, tool cards, the client tool executor and the context
     7→ * strip (PRODUCT_ARCHITECTURE §8.3 "untrusted data" rule, §12.2 "Agent
     8→ * safety", BUILD_PLAN §M4 task 6 "Safety: no write tools registered;
     9→ * prompt-injection tests; tool allowlist; step/time limits").
    10→ *
    11→ * This module is PURE DATA + pure builders: zero imports so the same corpus
    12→ * is importable from BOTH `apps/web` tests and `services/gateway` tests
    13→ * (the gateway test reaches it via a relative path). Every fixture carries
    14→ * its threat model in `threat` and a distinctive `needle` used by sweep
    15→ * assertions — needles are always single-line and quote/backslash-free so
    16→ * they survive `JSON.stringify` contexts unchanged.
    17→ *
    18→ * Corpus-wide invariants (tests rely on them):
    19→ *   - every `needle` occurs inside `payload` (or the fixture's built value);
    20→ *   - no fixture may crash a correct implementation — hostile inputs must be
    21→ *     rejected or carried verbatim, never thrown on;
    22→ *   - the WRITE tool ids referenced in payloads exist in @studio/contracts
    23→ *     TOOL_LIST (kind "write") but are NOT registered in M4's runner
    24→ *     (M4.6: the hard backstop is that no write path exists at all).
    25→ */
    26→
    27→// ---------------------------------------------------------------------------
    28→// Fixture types
    29→// ---------------------------------------------------------------------------
    30→
    31→/** Where a payload can travel in the M4 pipeline. */
    32→export type InjectionSurface =
    33→  | "projectText" // layer names, text-layer content, comp names, comments, metadata
    34→  | "userMessage" // the chat message itself (the user may paste hostile text)
    35→  | "toolResult" // data fed back through the tool-role message / tool cards
    36→  | "args" // tool-call arguments the model emits
    37→  | "plan" // studio.plan_edit goals and structured-output plan payloads
    38→  | "markdown"; // anything rendered through the chat Markdown component
    39→
    40→export interface InjectionFixture {
    41→  /** Stable fixture id (referenced by name in test failures and M4-SAFETY.md). */
    42→  readonly id: string;
    43→  /** Threat model: what real-world attacker this simulates and why it is dangerous. */
    44→  readonly threat: string;
    45→  /** Pipeline surfaces this payload plausibly reaches. */
    46→  readonly surfaces: readonly InjectionSurface[];
    47→  /** The raw hostile string. */
    48→  readonly payload: string;
    49→  /** Distinctive single-line, quote-free marker embedded in the payload. */
    50→  readonly needle: string;
    51→}
    52→
    53→/** A hostile TOOL RESULT (data the model sees back — the §7.4 tool-role message). */
    54→export interface ToolResultFixture {
    55→  readonly id: string;
    56→  readonly threat: string;
    57→  /** The tool the result claims to answer. */
    58→  readonly tool: string;
    59→  /** Hostile result value (plain JSON-serializable data). */
    60→  readonly result: unknown;
    61→  /** Distinctive marker carried inside the result, used by sweep assertions. */
    62→  readonly needle: string;
    63→}
    64→
    65→/** Hostile tool-call ARGUMENTS as the raw JSON string a compromised model emits. */
    66→export interface HostileToolArgsFixture {
    67→  readonly id: string;
    68→  readonly threat: string;
    69→  readonly tool: string;
    70→  /** RAW JSON text exactly as it appears on the wire in `tool_calls[].function.arguments`. */
    71→  readonly argsJson: string;
    72→  /** Distinctive marker carried inside the args, used by sweep assertions. */
    73→  readonly needle: string;
    74→}
    75→
    76→/** A hostile structured-output EditPlan (what a compromised model streams back). */
    77→export interface HostilePlanFixture {
    78→  readonly id: string;
    79→  readonly threat: string;
    80→  /** The raw JSON string the model streams back for the structured plan call. */
    81→  readonly planJson: string;
    82→  /** Distinctive marker carried inside the plan JSON. */
    83→  readonly needle: string;
    84→  /**
    85→   * What a correct EDIT_PLAN_SCHEMA parse must do with this payload.
    86→   * "reject" — validation failure (bounds/enum/shape); "data" — the payload
    87→   * parses but every hostile fragment stays inert JSON data (M4 compiles
    88→   * nothing; the apply path does not exist yet).
    89→   */
    90→  readonly expectation: "reject" | "data";
    91→}
