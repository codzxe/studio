   125→class FakeOpenAIServer {
   126→  private readonly queue: ScriptEntry[] = [];
   127→  private server: Server | null = null;
   128→  url = "";
   129→
   130→  enqueue(...entries: ScriptEntry[]): void {
   131→    this.queue.push(...entries);
   132→  }
   133→
   134→  reset(): void {
   135→    this.queue.length = 0;
   136→  }

   138→  async start(): Promise<void> {
   139→    const server = createServer((request, response) => {
   140→      void this.handle(request, response);
   141→    });
   142→    this.server = server;
   143→    await new Promise<void>((resolve) => {
   144→      server.listen(0, "127.0.0.1", resolve);
   145→    });

   147→    this.url = `http://127.0.0.1:${address.port}`;
   148→  }

   150→  async stop(): Promise<void> {
   151→    const server = this.server;
   152→    if (server === null) return;
   153→    await new Promise<void>((resolve) => {
   154→      server.close(() => resolve());
   155→    });

   157→
   158→  private async handle(request: IncomingMessage, response: ServerResponse): Promise<void> {
   159→    // Swallow socket errors from aborted runs (the runner's fetch tears down).
   160→    response.on("error", () => undefined);
   161→    request.on("error", () => undefined);
   162→    const chunks: Buffer[] = [];
   163→    for await (const chunk of request) {
   164→      chunks.push(chunk as Buffer);
   165→      if (response.destroyed) return;
   166→    }
   167→    let body: ChatCompletionBody = { model: "", messages: [] };
   168→    try {
   169→      body = JSON.parse(Buffer.concat(chunks).toString("utf8")) as ChatCompletionBody;
   170→    } catch {
   171→      body = { model: "", messages: [] };
   172→    }
   173→    const next = this.queue.shift();
   174→    if (next === undefined) {
   175→      response.writeHead(500, { "content-type": "application/json" });
   176→      response.end(JSON.stringify({ error: { message: "fake script exhausted" } }));
   177→      return;
   178→    }
   179→    response.writeHead(200, {
   180→      "content-type": "text/event-stream; charset=utf-8",
   181→      "cache-control": "no-store",
   182→    });
   183→    for (const wireFrame of next(body)) {
   184→      if (response.destroyed) return;
   185→      response.write(encoder.encode(wireFrame));
   186→      await new Promise<void>((resolveTick) => {
   187→        setImmediate(resolveTick);
   188→      });
   189→    }
   190→    response.end();
   191→  }
   192→}

   198→interface WireFrame {
   199→  readonly raw: string;
   200→  readonly id: number;

   204→function parseFrame(raw: string): WireFrame {
   205→  const lines = raw.split("\n");
   206→  const idLine = lines[0];
   207→  const dataLine = lines[1];
   208→  if (
   209→    lines.length !== 2 ||
   210→    idLine === undefined ||
   211→    dataLine === undefined ||
   212→    !idLine.startsWith("id: ") ||
   213→    !dataLine.startsWith("data: ")
   214→  ) {
   215→    throw new Error(`malformed SSE frame on the wire: ${JSON.stringify(raw)}`);
   216→  }
   217→  const id = Number.parseInt(idLine.slice(4), 10);
   218→  const event = JSON.parse(dataLine.slice(6)) as StudioStreamEvent;
   219→  return { raw, id, event };
   220→}

   222→class SseStream {
   223→  private readonly reader: ReadableStreamDefaultReader<Uint8Array>;
   224→  private readonly decoder = new TextDecoder();
   225→  private buffer = "";
   226→  private eof = false;
   227→
   228→  constructor(response: Response) {
   229→    const body = response.body;
   230→    if (body === null) throw new Error("chat response had no SSE body");
   231→    this.reader = body.getReader();
   232→  }
   233→
   234→  async next(timeoutMs = 5000): Promise<WireFrame | null> {
   235→    if (this.eof) return null;
   236→    const deadline = Date.now() + timeoutMs;
   237→    for (;;) {
   238→      const boundary = this.buffer.indexOf("\n\n");
   239→      if (boundary !== -1) {
   240→        const raw = this.buffer.slice(0, boundary);
   241→        this.buffer = this.buffer.slice(boundary + 2);
   242→        if (raw.trim() === "") continue;
   243→        return parseFrame(raw);
   244→      }
   245→      const remaining = deadline - Date.now();
   246→      if (remaining <= 0) {
   247→        await this.cancel();
   248→        throw new Error("timed out waiting for the next SSE frame");
   249→      }
   250→      const read = await Promise.race([
   251→        this.reader.read(),
   252→        new Promise<never>((_resolve, reject) => {
   253→          const timer = setTimeout(() => {
   254→            reject(new Error("timed out waiting for the next SSE frame"));
   255→          }, remaining);
   256→          timer.unref?.();
   257→        }),
   258→      ]);
   259→      if (read.done) {
   260→        this.eof = true;
   261→        const tail = this.buffer;
   262→        this.buffer = "";
   263→        if (tail.trim() === "") return null;
   264→        return parseFrame(tail);
   265→      }
   266→      this.buffer += this.decoder.decode(read.value, { stream: true });
   267→    }
   268→  }
   269→
   270→  async cancel(): Promise<void> {
   271→    this.eof = true;
   272→    await this.reader.cancel().catch(() => undefined);
   273→  }
   274→
   275→  async rest(timeoutMs = 5000): Promise<WireFrame[]> {
   276→    const frames: WireFrame[] = [];
   277→    for (;;) {
   278→      const wireFrame = await this.next(timeoutMs);
   279→      if (wireFrame === null) return frames;
   280→      frames.push(wireFrame);
   281→    }
   282→  }
   283→}

   293→function expectValidStream(frames: readonly WireFrame[]): StudioStreamEvent[] {
   294→  expect(frames.length).toBeGreaterThan(0);
   295→  let previousSeq: number | null = null;
   296→  const events: StudioStreamEvent[] = [];
   297→  let terminals = 0;
   298→  for (const wireFrame of frames) {
   299→    expect(wireFrame.raw).toBe(
   300→      `id: ${wireFrame.event.seq}\ndata: ${JSON.stringify(wireFrame.event)}`,
   301→    );
   302→    expect(wireFrame.id).toBe(wireFrame.event.seq);
   303→    if (previousSeq === null) expect(wireFrame.event.seq).toBe(0);
   304→    else expect(wireFrame.event.seq).toBe(previousSeq + 1);
   305→    previousSeq = wireFrame.event.seq;
   306→    const checked = STUDIO_STREAM_EVENT_SCHEMA.safeParse(wireFrame.event);
   307→    if (!checked.success) {
   308→      throw new Error(
   309→        `event seq ${wireFrame.event.seq} failed the §7.4 schema: ${checked.error.message}`,
   310→      );
   311→    }
   312→    events.push(checked.data);
   313→    if (TERMINAL_TYPES.has(checked.data.type)) terminals += 1;
   314→  }
   315→  expect(events[0]?.type).toBe("run.started");
   316→  expect(terminals).toBe(1);
   317→  const last = frames.at(-1)?.event;
   318→  if (last === undefined || !TERMINAL_TYPES.has(last.type)) {
   319→    throw new Error("the last frame on the wire must be the terminal event");
   320→  }
   321→  const runIds = new Set(events.map((event) => event.runId));
   322→  expect(runIds.size).toBe(1);
   323→  return events;
   324→}

   581→/** Bounded poll (25ms ticks) — the M5-K waitFor precedent. */
   582→async function waitFor(description: string, probe: () => Promise<boolean>, timeoutMs = 4000) {
   583→  const deadline = Date.now() + timeoutMs;
   584→  for (;;) {
   585→    if (await probe()) return;
   586→    if (Date.now() > deadline) throw new Error(`timed out waiting for ${description}`);
   587→    await new Promise((resolve) => {
   588→      setTimeout(resolve, 25);
   589→    });
   590→  }
   591→}

   649→/**
   650→ * Drives ONE full planning chat through the real route (M5-K planRun): the
   651→ * model calls plan_edit, the server-side structured round returns the plan,
   652→ * the model wraps up. Returns the full valid stream + the plan event.
   653→ */
   654→async function planRun(
   655→  jar: CookieJar,
   656→  workspaceId: string,
   657→  threadId: string,
   658→  message = PLAN_MESSAGE,
   659→) {
   660→  openai.enqueue(
   661→    planEditRound("call_collab_plan", message),
   662→    planJsonRound(PLAN),
   663→    finalAnswerRound("Planned — apply it from the card whenever you're ready."),
   664→  );
   665→  const response = await chatStream(jar, workspaceId, threadId, message);
   666→  expect(response.status).toBe(200);
   667→  expect(response.headers.get("content-type")).toContain("text/event-stream");
   668→  const stream = new SseStream(response);
   669→  const frames = await stream.rest();
   670→  const events = expectValidStream(frames);
   671→  const runId = events[0]?.runId;
   672→  if (runId === undefined) throw new Error("no run.started event");
   673→  await waitFor("the run row to complete", async () => {
   674→    const run = await getRun(jar, workspaceId, runId);
   675→    return run.status === "completed";
   676→  });
   677→  const plan = events.find((event): event is PlanEvent => event.type === "plan");
   678→  if (plan === undefined) throw new Error("no plan event on the wire");
   679→  return { runId, events, plan };
   680→}

   682→/**
   683→ * The caller's full apply gesture (the client's saveNow + apply): upload the
   684→ * post-batch snapshot chained on the plan's base, then POST the apply with
   685→ * the compiled batch's sha + the resolved layer ids.
   686→ */
   687→async function applyPlan(
   688→  jar: CookieJar,
   689→  workspaceId: string,
   690→  projectId: string,
   691→  runId: string,
   692→  plan: PlanEvent,
   693→  transportKey: string,
   694→): Promise<{ status: number; body: Record<string, unknown> }> {
   695→  const batch = plan.compiledBatch;
   696→  if (batch === undefined) throw new Error("the plan event carried no compiledBatch");
   697→  const baseVersionId = plan.baseVersionId;
   698→  if (typeof baseVersionId !== "string" || baseVersionId === null) {
   699→    throw new Error("the plan event carried no baseVersionId");
   700→  }
   701→  const upload = await uploadVersion(jar, workspaceId, projectId, {
   702→    baseVersionId,
   703→    agentRunId: runId,
   704→    label: "agent-apply",
   705→  });
   706→  if (upload.status !== 201) {
   707→    return { status: upload.status, body: { error: { code: upload.code ?? "UPLOAD_FAILED" } } };
   708→  }
   709→  const resolved =
   710→    plan.resolutionRequests?.map((request, index) => ({
   711→      requestId: request.requestId,
   712→      layerId: `layer-${index + 1}`,
   713→    })) ?? [];
   714→  const response = await api(server.base, `/v1/runs/${runId}/apply`, {
   715→    method: "POST",
   716→    jar,
   717→    headers: { "x-workspace-id": workspaceId, "idempotency-key": transportKey },
   718→    body: {
   719→      batchSha256: shaOfCompiledBatch(batch),
   720→      versionId: upload.versionId,
   721→      idempotencyKey: transportKey,
   722→      resolvedLayers: resolved,
   723→    },
   724→  });
   725→  return {
   726→    status: response.status,
   727→    body: (await response.json()) as Record<string, unknown>,
   728→  };
   729→}
