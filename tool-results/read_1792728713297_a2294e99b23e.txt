    13→ * Auto-run only when DATABASE_URL is a postgres URL (same gate as the other
    14→ * integration suites):
    15→ *   `DATABASE_URL=postgres://… pnpm --filter @studio/db test`

    18→ * listExpiredOutputs are trusted-internal GLOBAL operations, and aborted

    20→ * (1) fails-or-requeues stale running rows, (2) drains the queue, and

    38→  async function makePinnedProject(
    39→    workspaceId: string,
    40→    name: string,
    41→  ): Promise<{ projectId: string; versionId: string }> {
    42→    const project = await repos.projects.create(workspaceId, { name, createdBy: userId });
    43→    const { version } = await repos.versions.upload(workspaceId, {
    44→      projectId: project.id,
    45→      expectedVersionId: null,
    46→      objectKey: `workspaces/${workspaceId}/projects/${project.id}/versions/v1/project.ecproj`,
    47→      sha256: randomUUID().replaceAll("-", "").slice(0, 8),
    48→      sizeBytes: 8,
    49→      createdBy: userId,
    50→    });
    51→    return { projectId: project.id, versionId: version.id };

   564→    const wsLedger = `ws_led_${randomUUID().slice(0, 8)}`;
   565→    await db
   566→      .insert(schema.organization)
   567→      .values({ id: wsLedger, name: "Led", slug: wsLedger })

   570→    const render = await repos.renders.create(wsLedger, {
   571→      projectId: pinned.projectId,
   572→      versionId: pinned.versionId,
   573→      compId: "comp_main",
   574→      format: "mp4",
   575→      createdBy: userId,
   576→    });
   577→
   578→    const grant = await repos.ledger.grant(wsLedger, {
   579→      amount: 1_000_000, // 1 credit in µc (D-12: integer microcredits only)
   580→      idempotencyKey: `m8a:grant:${render.id}`,
   581→    });
   582→    expect(grant.entry.renderId).toBeNull(); // unscoped grants stay unscoped
   583→
   584→    const held = await repos.ledger.reserve(wsLedger, {
   585→      renderId: render.id, // render-scoped hold — no runId
   586→      amount: 500,
   587→      idempotencyKey: `m8a:reserve:${render.id}`,
   588→    });
   589→    expect(held.entry.renderId).toBe(render.id);
   590→    expect(held.entry.runId).toBeNull();
   591→    expect(held.replayed).toBe(false);
   592→
   593→    // a ledger row may be scoped to a run OR a render — never both
   594→    await expect(
   595→      repos.ledger.reserve(wsLedger, {
   596→        runId: "run_x",
   597→        renderId: render.id,
   598→        amount: 10,
   599→        idempotencyKey: `m8a:xor:${render.id}`,
   600→      }),
   601→    ).rejects.toMatchObject({ code: "VALIDATION_FAILED" });
   602→    // …and hold operations require exactly one producer
   603→    await expect(
   604→      repos.ledger.reserve(wsLedger, {
   605→        amount: 10,
   606→        idempotencyKey: `m8a:noscope:${render.id}`,
   607→      }),
   608→    ).rejects.toMatchObject({ code: "VALIDATION_FAILED" });
   609→
   610→    // settle commits the actual cost and releases the full hold (D-20)
   611→    const settled = await repos.ledger.settle(wsLedger, {
   612→      renderId: render.id,
   613→      actualCost: 300,
   614→      idempotencyKey: `m8a:settle:${render.id}`,
   615→    });
   616→    expect(settled.commit?.renderId).toBe(render.id);
   617→    expect(settled.commit?.amount).toBe(-300);
   618→    expect(settled.refund?.amount).toBe(500);
   619→    expect(settled.replayed).toBe(false);
   620→
   621→    // nothing left to refund for this render

   624→      idempotencyKey: `m8a:refund-all:${render.id}`,
