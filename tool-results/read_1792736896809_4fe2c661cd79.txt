     1→/**
     2→ * Pure chat run-state reducer — Seam 4 of docs/phases/M4-SEAMS.md
     3→ * (EDITOR_CHAT_UX §8.1/§8.2/§8.4; PRODUCT_ARCHITECTURE.md §7.4).
     4→ *
     5→ * `foldEvent` folds ONE normalized §7.4 event (as parsed + validated by
     6→ * `@studio/api-client`'s SSE reader — never re-validated here) into the
     7→ * whole-thread `ChatUiState`. Total, immutable and replay-idempotent:
     8→ *
     9→ *  * New-run takeover — the first `run.started` for a run the thread has no
    10→ *    assistant message for creates the placeholder bound to that `runId`,
    11→ *    flips the phase to `streaming`, clears any stale error and RESETS the
    12→ *    seq cursor to that run's seq space (§7.4 seqs are per-run).
    13→ *  * Older runs stay frozen — events for any run other than `activeRunId`
    14→ *    are dropped, so a replay of a finished run can never mutate it.
    15→ *  * Replay dedupe — events at/below `lastSeq` for the active run are
    16→ *    dropped, so a `Last-Event-ID` replay (or re-feeding the same list) is a
    17→ *    no-op and the returned reference is unchanged.
    18→ *  * Terminal freeze — once the active run is done/failed/cancelled its
    19→ *    message no longer accepts content events, but a later TERMINAL event
    20→ *    for the same run still reconciles the phase (server truth: a
    21→ *    `run.completed` that lands after a local stop flips `cancelled` →
    22→ *    `done`; §8.3 "cancellation never auto-undoes an applied batch").
    23→ *  * `preview`/`approval.required` advance the cursor but change nothing
    24→ *    else in M4 (previews ride tool cards).
    25→ *
    26→ * M5 (BUILD_PLAN §M5.3–§M5.5): a `plan` event may additionally carry the
    27→ * server-compiled batch + layer-resolution requests + the base version
    28→ * (M5-C adds them to the wire; the SSE reader strips unknown keys until
    29→ * then). When present they fold into the message's `planRuntime` for the
    30→ * ApplyController (M5-F); the thread phase is untouched, so M4 behavior is
    31→ * byte-identical without them.
    32→ *
    33→ * Client-driven transitions (§8.1 states the USER causes — submit, stop, the
    34→ * §8.4 reconnect banner, the pause, and the M5 apply/undo gestures) never
    35→ * consume a seq; they live in `applyClientAction`.
    36→ */

    93→/**
    94→ * M6-H: the NEWEST §7.4 `credits` snapshot on the thread — the terminal
    95→ * `credits` event the gateway emits when it settles the run (reserved →
    96→ * commit/refund), which carries the post-run `balance`. Values are integer
    97→ * MICROcredits (D-12; the same units `GET /v1/credits` reports). Pure; `null`
    98→ * when no run has reported credits yet.
    99→ */
   100→export function latestCreditsSnapshot(
   101→  state: ChatUiState,
   102→): NonNullable<ChatUiMessage["credits"]> | null {
   103→  for (let index = state.messages.length - 1; index >= 0; index -= 1) {
   104→    const message = state.messages[index];
   105→    if (message?.credits !== undefined) return message.credits;
   106→  }
   107→  return null;
   108→}

   114→ * `chat.reconnecting` the §8.4 banner (also dispatched when resuming a

   155→/**
   156→ * Appends the user's message to the thread. Pure: the caller owns the id
   157→ * (`crypto.randomUUID()` in the controller). Does NOT touch the phase — the
   158→ * controller decides between `chat.submit` (wire now) and queueing.
   159→ */
   160→export function applyUserMessage(
   161→  state: ChatUiState,
   162→  message: { id: string; text: string },
   163→): ChatUiState {
   164→  const userMessage: ChatUiMessage = {
   165→    id: message.id,
   166→    role: "user",
   167→    text: message.text,
   168→    toolCards: [],
   169→  };
   170→  return { ...state, messages: [...state.messages, userMessage] };
   171→}

   173→/**
   174→ * §8.1 client-driven transitions. Total and pure: invalid transitions return
   175→ * the SAME state reference (a no-op).
   176→ */
   177→export function applyClientAction(state: ChatUiState, action: ChatClientAction): ChatUiState {
   178→  // The M5 `chat.plan.*` gestures fold the plan's per-message runtime only
   179→  // (applyPlanAction) — the thread phase keeps its exact M4 semantics so
   180→  // sends/steer/reconnect behave byte-identically.
   181→  if (isPlanGesture(action)) {
   182→    return applyPlanAction(state, action);
   183→  }
   184→  switch (action.type) {
   185→    case "chat.submit": {
   186→      // A new send only starts from idle or a terminal phase; a live run
   187→      // takes the steer queue instead.
   188→      if (LIVE_RUN_PHASES.has(state.phase)) return state;
   189→      // M6-H: a fresh send re-arms the credits block — the server re-enforces
   190→      // the §8.8 gate on the wire (honest client-side UX only, never a
   191→      // permanent client lock).
   192→      return { ...state, phase: "sending", error: undefined, blocked: undefined };
   193→    }
   194→    case "chat.stop": {
   195→      if (!LIVE_RUN_PHASES.has(state.phase)) return state;
   196→      return { ...state, phase: "cancelled" };
   197→    }
   198→    case "chat.reconnecting": {
   199→      // Dispatched ONLY while a resumable stream exists; "failed" here means
   200→      // RUN_PAUSED (resume), not a terminal run failure.
   201→      if (!LIVE_RUN_PHASES.has(state.phase) && state.phase !== "failed") return state;
   202→      return { ...state, phase: "reconnecting" };
   203→    }
   204→    case "chat.paused": {
   205→      if (!LIVE_RUN_PHASES.has(state.phase)) return state;
   206→      return { ...state, phase: "failed", error: RUN_PAUSED_ERROR };
   207→    }
   208→    case "chat.error": {
   209→      if (!LIVE_RUN_PHASES.has(state.phase)) return state;
   210→      // M6-H: a 402 INSUFFICIENT_CREDITS (§8.8 — the POST never started a
   211→      // run) additionally records the TYPED blocked state. It is NOT
   212→      // retryable (the §7.3 envelope says so — no retry loop); the way out
   213→      // is the top-up CTA, and `chat.submit` clears the block.
   214→      return {
   215→        ...state,
   216→        phase: "failed",
   217→        error: action.error,
   218→        ...(action.error.code === INSUFFICIENT_CREDITS_CODE
   219→          ? { blocked: blockedCreditsFrom(action.error) }
   220→          : {}),
   221→      };
   222→    }
   223→  }
   224→}

   253→function upsertStartedCard(
   254→  cards: readonly ToolCardState[],
   255→  started: { toolCallId: string; name: string; args: unknown },
   256→): readonly ToolCardState[] {
   257→  if (cards.some((card) => card.toolCallId === started.toolCallId)) {
   258→    return cards.map((card) =>
   259→      card.toolCallId === started.toolCallId
   260→        ? { ...card, name: started.name, args: started.args, status: "running" as const }
   261→        : card,
   262→    );
   263→  }
   264→  return [
   265→    ...cards,
   266→    { toolCallId: started.toolCallId, name: started.name, args: started.args, status: "running" },
   267→  ];
   268→}

   270→function appendDeltaText(
   271→  cards: readonly ToolCardState[],
   272→  delta: { toolCallId: string; delta: string },
   273→): readonly ToolCardState[] {
   274→  if (cards.some((card) => card.toolCallId === delta.toolCallId)) {

   278→        : card,

   282→  // card with an unknown name keeps the UI honest without guessing.

   284→    ...cards,
   285→    { toolCallId: delta.toolCallId, name: "", status: "running", deltaText: delta.delta },
   286→  ];
   287→}

   289→function applyToolResult(
   290→  cards: readonly ToolCardState[],
   291→  result: { toolCallId: string; result: unknown },
   292→): readonly ToolCardState[] {
   293→  const status: "done" | "failed" = isFailedToolResult(result.result) ? "failed" : "done";
   294→  if (cards.some((card) => card.toolCallId === result.toolCallId)) {
   295→    return cards.map((card) =>
   296→      card.toolCallId === result.toolCallId ? { ...card, status, result: result.result } : card,
   297→    );
   298→  }
   299→  // Result without a started card (ordering anomaly): still show the card.
   300→  return [...cards, { toolCallId: result.toolCallId, name: "", status, result: result.result }];
   301→}

   313→/**
   314→ * The `plan` event, extended with the M5 runtime fields (M5-C adds them to
   315→ * the wire; absent — and stripped by the SSE reader — until that lands).
   316→ * Presence-checked + narrowly validated below: a malformed runtime folds as
   317→ * a plain M4 plan (card data only, no apply surface).
   318→ */
   319→type PlanEventWithRuntime = Extract<StudioStreamEvent, { type: "plan" }> & {
   320→  readonly compiledBatch?: unknown;
   321→  readonly resolutionRequests?: unknown;
   322→  readonly baseVersionId?: unknown;
   323→  readonly risk?: unknown;
   324→  readonly estimatedCredits?: unknown;
   325→  readonly autoApply?: unknown;
   326→};

   328→/** `steps: [{command, params}]` with a non-empty label — the §8.5 apply payload. */
   329→function parseCompiledBatch(value: unknown): PlanRuntime["compiledBatch"] | null {
   330→  if (typeof value !== "object" || value === null) return null;
   331→  const record = value as Record<string, unknown>;
   332→  if (typeof record.label !== "string" || record.label.length === 0) return null;
   333→  if (!Array.isArray(record.steps) || record.steps.length === 0) return null;
   334→  const steps: Array<{ command: string; params: Record<string, unknown> }> = [];
   335→  for (const step of record.steps) {
   336→    if (typeof step !== "object" || step === null) return null;
   337→    const stepRecord = step as Record<string, unknown>;
   338→    if (typeof stepRecord.command !== "string" || stepRecord.command.length === 0) return null;
   339→    if (typeof stepRecord.params !== "object" || stepRecord.params === null) return null;
   340→    steps.push({
   341→      command: stepRecord.command,
   342→      params: stepRecord.params as Record<string, unknown>,
   343→    });
   344→  }
   345→  const timeoutMs =
   346→    typeof record.timeoutMs === "number" &&
   347→    Number.isInteger(record.timeoutMs) &&
   348→    record.timeoutMs > 0
   349→      ? record.timeoutMs
   350→      : undefined;
   351→  return timeoutMs === undefined
   352→    ? { label: record.label, atomic: true, steps }
   353→    : { label: record.label, atomic: true, steps, timeoutMs };
   354→}
   355→
   356→/** `[{requestId, layerName, disambiguation?}]` — the §8.7 client-side bindings. */
   357→function parseResolutionRequests(value: unknown): readonly LayerResolutionRequest[] | undefined {
   358→  if (!Array.isArray(value) || value.length === 0) return undefined;
   359→  const out: LayerResolutionRequest[] = [];
   360→  for (const entry of value) {
   361→    if (typeof entry !== "object" || entry === null) return undefined;
   362→    const record = entry as Record<string, unknown>;
   363→    if (typeof record.requestId !== "string" || record.requestId.length === 0) return undefined;
   364→    if (typeof record.layerName !== "string") return undefined;
   365→    const hints = Array.isArray(record.disambiguation)
   366→      ? record.disambiguation.filter((hint): hint is string => typeof hint === "string")
   367→      : [];
   368→    out.push({ requestId: record.requestId, layerName: record.layerName, disambiguation: hints });
   369→  }
   370→  return out;
   371→}
   372→
   373→/** The §7.3/D-22 verdict — `{mode: "auto"|"propose", reason}`, else undefined. */
   374→function parseAutoApply(value: unknown): PlanRuntime["autoApply"] {
   375→  if (typeof value !== "object" || value === null) return undefined;
   376→  const record = value as Record<string, unknown>;
   377→  if (record.mode !== "auto" && record.mode !== "propose") return undefined;
   378→  if (typeof record.reason !== "string" || record.reason.length === 0) return undefined;
   379→  return { mode: record.mode, reason: record.reason };
   380→}

   382→/** The server-recomputed risk (D-21: never the model's self-declared value). */
   383→function parseServerRisk(value: unknown): PlanRuntime["risk"] {
   384→  return value === "low" || value === "medium" || value === "high" ? value : undefined;
   385→}
   386→
   387→/** The server-recomputed deterministic credit estimate (§8.8). */
   388→function parseServerCredits(value: unknown): PlanRuntime["estimatedCredits"] {
   389→  return typeof value === "number" && Number.isFinite(value) && value >= 0 ? value : undefined;
   390→}

   392→/**
   393→ * Builds the message's `planRuntime` from a plan event when it carries the M5
   394→ * payload; `null` when it does not (the M4 card shape stays as-is).
   395→ */
   396→export function planRuntimeFromEvent(event: StudioStreamEvent): PlanRuntime | null {
   397→  if (event.type !== "plan") return null;
   398→  const extended = event as PlanEventWithRuntime;
   399→  const compiledBatch = parseCompiledBatch(extended.compiledBatch);
   400→  if (compiledBatch === null) return null;
   401→  const resolutionRequests = parseResolutionRequests(extended.resolutionRequests);
   402→  const baseVersionId =
   403→    typeof extended.baseVersionId === "string" && extended.baseVersionId.length > 0
   404→      ? extended.baseVersionId
   405→      : undefined;
   406→  // M5-Q: the SERVER-recomputed card values (risk/credits, D-21) + the D-22
   407→  // auto-apply verdict fold with the runtime — presence-checked so a hostile
   408→  // stream can only omit them, never fabricate client credit math.
   409→  const risk = parseServerRisk(extended.risk);
   410→  const estimatedCredits = parseServerCredits(extended.estimatedCredits);
   411→  const autoApply = parseAutoApply(extended.autoApply);
   412→  return {
   413→    compiledBatch,
   414→    ...(resolutionRequests !== undefined ? { resolutionRequests } : {}),
   415→    ...(baseVersionId !== undefined ? { baseVersionId } : {}),
   416→    ...(risk !== undefined ? { risk } : {}),
   417→    ...(estimatedCredits !== undefined ? { estimatedCredits } : {}),
   418→    ...(autoApply !== undefined ? { autoApply } : {}),
   419→    outcome: "idle",
   420→  };
   421→}

   440→/**
   441→ * Merges one user pick into the runtime's pick map (M5-Q, §8.7). The map is
   442→ * rebuilt NULL-PROTOTYPE every time (M4-N F-1): request ids are model-adjacent
   443→ * strings — on a plain `{}` a `__proto__` id would rebind the accumulator's
   444→ * prototype (a spread of a null-proto object would LOSE the null proto, so the
   445→ * merge is an explicit copy loop).
   446→ */
   447→function mergePick(
   448→  existing: Readonly<Record<string, number>> | undefined,
   449→  requestId: string,
   450→  layerId: number,
   451→): Readonly<Record<string, number>> {
   452→  const next = Object.create(null) as Record<string, number>;
   453→  if (existing !== undefined) {
   454→    for (const [key, value] of Object.entries(existing)) next[key] = value;
   455→  }
   456→  next[requestId] = layerId;
   457→  return next;
   458→}
   459→
   460→/**
   461→ * Replaces every whole-string `$request:<requestId>` placeholder whose
   462→ * requestId carries a user pick with the chosen layer id (M5-Q). Deterministic
   463→ * deep clone with NULL-PROTOTYPE objects (the M4-N F-1 rule — mirrors
   464→ * apply.ts's `patchPlaceholders`; keep in sync with it).
   465→ */
   466→function patchUserPicks(value: unknown, picks: Readonly<Record<string, number>>): unknown {
   467→  if (typeof value === "string") {
   468→    if (!value.startsWith(RESOLVE_TOKEN_PREFIX)) return value;
   469→    const requestId = value.slice(RESOLVE_TOKEN_PREFIX.length);
   470→    const picked = picks[requestId];
   471→    return picked === undefined ? value : picked;
   472→  }
   473→  if (Array.isArray(value)) return value.map((item) => patchUserPicks(item, picks));
   474→  if (typeof value === "object" && value !== null) {
   475→    const out: Record<string, unknown> = Object.create(null);
   476→    for (const [key, child] of Object.entries(value)) out[key] = patchUserPicks(child, picks);
   477→    return out;
   478→  }
   479→  return value;
   480→}

   482→/** Folds one `chat.plan.*` gesture onto the message's planRuntime (pure). */
   483→function applyPlanAction(state: ChatUiState, action: PlanGestureAction): ChatUiState {
   484→  const runId = action.runId;
   485→  const foldRuntime = (update: (runtime: PlanRuntime) => PlanRuntime): ChatUiState => ({
   486→    ...state,
   487→    messages: mapMessageOfRun(state, runId, (message) => {
   488→      if (message.planRuntime === undefined) return message; // no M5 payload on this card
   489→      return { ...message, planRuntime: update(message.planRuntime) };
   490→    }),
   491→  });
   492→
   493→  switch (action.type) {
   494→    case "chat.plan.apply":
   495→      return foldRuntime((runtime) =>
   496→        runtime.outcome === "idle" ||
   497→        runtime.outcome === "apply-failed" ||
   498→        runtime.outcome === "undone"
   499→          ? { ...runtime, outcome: "applying", error: undefined, candidates: undefined }
   500→          : runtime,
   501→      );
   502→    case "chat.plan.applied":
   503→      return foldRuntime((runtime) =>
   504→        runtime.outcome === "applying"
   505→          ? {
   506→              ...runtime,
   507→              outcome: "applied",
   508→              versionId: action.versionId,
   509→              error: undefined,
   510→              candidates: undefined,
   511→            }
   512→          : runtime,
   513→      );
   514→    case "chat.plan.apply-failed":
   515→      return foldRuntime((runtime) =>
   516→        runtime.outcome === "applying"
   517→          ? {
   518→              ...runtime,
   519→              outcome: "apply-failed",
   520→              error: action.error,
   521→              ...(action.candidates !== undefined ? { candidates: action.candidates } : {}),
   522→            }
   523→          : runtime,
   524→      );
   525→    case "chat.plan.undo":
   526→      return foldRuntime((runtime) =>
   527→        runtime.outcome === "applied" ? { ...runtime, outcome: "undoing" } : runtime,
   528→      );
   529→    case "chat.plan.undone":
   530→      return foldRuntime((runtime) =>
   531→        runtime.outcome === "undoing"
   532→          ? { ...runtime, outcome: "undone", error: undefined }
   533→          : runtime,
   534→      );
   535→    case "chat.plan.undo-failed":
   536→      return foldRuntime((runtime) =>
   537→        runtime.outcome === "undoing"
   538→          ? { ...runtime, outcome: "undo-failed", error: action.error }
   539→          : runtime,
   540→      );
   541→    case "chat.plan.resolve":
   542→      // §8.7 (M5-Q): the user's candidate pick binds requestId → layer id; the
   543→      // pick is durable for this plan (kept across the re-driven attempt) and
   544→      // folds WITHOUT touching the outcome — the re-drive's own `chat.plan.apply`
   545→      // gesture follows immediately.
   546→      return foldRuntime((runtime) => ({
   547→        ...runtime,
   548→        resolutionPicks: mergePick(runtime.resolutionPicks, action.requestId, action.layerId),
   549→      }));
   550→    case "chat.plan.dismiss":
   551→      return foldRuntime((runtime) => ({ ...runtime, dismissed: true }));
   552→  }
   553→}
   554→
   555→/**
   556→ * The apply inputs for one plan, straight out of its runtime — the pure core of
   557→ * {@link applicablePlanOf} (the thread layer passes a MESSAGE's runtime in,
   558→ * the same shape the store folds). M5-Q: the user's ambiguous-layer picks are
   559→ * PRE-PATCHED into the batch (`$request:` tokens → layer ids) and their
   560→ * requests drop out of `resolutionRequests`, so the re-driven apply skips name
   561→ * resolution for them (§8.7 — the pick is user authority).
   562→ */
   563→export function applicablePlanFromRuntime(runId: string, runtime: PlanRuntime): ApplicablePlan {
   564→  const picks = runtime.resolutionPicks;
   565→  const hasPicks = picks !== undefined && Object.keys(picks).length > 0;
   566→  const requests =
   567→    runtime.resolutionRequests?.filter((request) => picks?.[request.requestId] === undefined) ?? [];
   568→  const batch = hasPicks
   569→    ? {
   570→        ...runtime.compiledBatch,
   571→        steps: runtime.compiledBatch.steps.map((step) => ({
   572→          command: step.command,
   573→          params: patchUserPicks(step.params, picks ?? {}) as Record<string, unknown>,
   574→        })),
   575→      }
   576→    : runtime.compiledBatch;
   577→  return {
   578→    runId,
   579→    ...(runtime.baseVersionId !== undefined ? { baseVersionId: runtime.baseVersionId } : {}),
   580→    compiledBatch: batch,
   581→    ...(requests.length > 0 ? { resolutionRequests: requests } : {}),
   582→  };
   583→}

   585→/** The message bound to `runId`'s M5 plan runtime, if any (M5-Q helper). */
   586→export function planRuntimeOf(state: ChatUiState, runId: string): PlanRuntime | undefined {
   587→  return state.messages.find((entry) => entry.runId === runId)?.planRuntime;
   588→}
   589→
   590→/**
   591→ * The apply inputs for the plan card bound to `runId`, straight out of the
   592→ * folded state (the provider/card passes this to `controller.apply`);
   593→ * `null` when the message has no M5 runtime (no compiled batch to execute).
   594→ */
   595→export function applicablePlanOf(state: ChatUiState, runId: string): ApplicablePlan | null {
   596→  const runtime = planRuntimeOf(state, runId);
   597→  if (runtime === undefined) return null;
   598→  return applicablePlanFromRuntime(runId, runtime);
   599→}

   605→/**
   606→ * Folds one §7.4 event into the whole-thread state. Pure, total and
   607→ * replay-idempotent (see the module header for the exact rule set); dropped
   608→ * events return the SAME state reference.
   609→ */
   610→export function foldEvent(state: ChatUiState, event: StudioStreamEvent): ChatUiState {
   611→  // New-run takeover: the first run.started for a run with no message yet.
   612→  if (event.type === "run.started" && !state.messages.some((m) => m.runId === event.runId)) {
   613→    const placeholder: ChatUiMessage = {
   614→      id: `run:${event.runId}`,
   615→      role: "assistant",
   616→      text: "",
   617→      runId: event.runId,
   618→      toolCards: [],
   619→    };
   620→    return {
   621→      ...state,
   622→      messages: [...state.messages, placeholder],
   623→      phase: "streaming",
   624→      activeRunId: event.runId,
   625→      error: undefined,
   626→      // M6-H: the server accepted the run, so the workspace could fund it —
   627→      // any stale typed block is over.
   628→      blocked: undefined,
   629→      lastSeq: event.seq,
   630→    };
   631→  }
   632→
   633→  // Older runs stay frozen; events for unknown runs are dropped.
   634→  if (event.runId !== state.activeRunId) return state;
   635→  // Replay dedupe (per active run): seqs at/below the cursor were already folded.
   636→  if (event.seq <= state.lastSeq) return state;
   637→
   638→  const frozen = TERMINAL_RUN_PHASES.has(state.phase);
   639→  // Every accepted event advances the cursor so the Last-Event-ID stays truthful.
   640→  let next: ChatUiState = { ...state, lastSeq: event.seq };
   641→
   642→  switch (event.type) {
   643→    case "run.started": {
   644→      // Replay of the active run's start with a later seq — nothing to create.
   645→      if (!frozen) next = { ...next, phase: resumeConfirm(next.phase), blocked: undefined };
   646→      break;
   647→    }
   648→    case "message.delta": {
   649→      if (frozen) break;
   650→      next = {
   651→        ...next,
   652→        phase: resumeConfirm(next.phase),
   653→        messages: mapActiveMessage(next, (message) => ({
   654→          ...message,
   655→          text: `${message.text}${event.delta}`,
   656→        })),
   657→      };
   658→      break;
   659→    }
   660→    case "tool.started": {
   661→      if (frozen) break;
   662→      next = {
   663→        ...next,
   664→        phase: resumeConfirm(next.phase),
   665→        messages: mapActiveMessage(next, (message) => ({
   666→          ...message,
   667→          toolCards: upsertStartedCard(message.toolCards, event),
   668→        })),
   669→      };
   670→      break;
   671→    }
   672→    case "tool.delta": {
   673→      if (frozen) break;
   674→      next = {
   675→        ...next,
   676→        phase: resumeConfirm(next.phase),
   677→        messages: mapActiveMessage(next, (message) => ({
   678→          ...message,
   679→          toolCards: appendDeltaText(message.toolCards, event),
   680→        })),
   681→      };
   682→      break;
   683→    }
   684→    case "tool.result": {
   685→      if (frozen) break;
   686→      next = {
   687→        ...next,
   688→        phase: resumeConfirm(next.phase),
   689→        messages: mapActiveMessage(next, (message) => ({
   690→          ...message,
   691→          toolCards: applyToolResult(message.toolCards, event),
   692→        })),
   693→      };
   694→      break;
   695→    }
   696→    case "plan": {
   697→      if (frozen) break;
   698→      const planRuntime = planRuntimeFromEvent(event);
   699→      next = {
   700→        ...next,
   701→        phase: resumeConfirm(next.phase),
   702→        messages: mapActiveMessage(next, (message) => ({
   703→          ...message,
   704→          plan: event.plan,
   705→          // A NEW plan event resets the card's runtime (fresh compiled batch,
   706→          // fresh lifecycle); without the M5 payload the runtime clears.
   707→          ...(planRuntime === null ? { planRuntime: undefined } : { planRuntime }),
   708→        })),
   709→      };
   710→      break;
   711→    }
   712→    case "credits": {
   713→      if (frozen) break;
   714→      next = {
   715→        ...next,
   716→        phase: resumeConfirm(next.phase),
   717→        messages: mapActiveMessage(next, (message) => ({
   718→          ...message,
   719→          credits: {
   720→            reserved: event.reserved,
   721→            committed: event.committed,
   722→            balance: event.balance,
   723→          },
   724→        })),
   725→      };
   726→      break;
   727→    }
   728→    case "preview":
   729→    case "approval.required":
   730→      // M4: cursor-only events — previews ride tool cards (render_preview is
   731→      // a client tool whose result lands on the card); approval UI is M5.
   732→      break;
   733→    case "run.completed": {
   734→      next = { ...next, phase: "done", error: undefined, blocked: undefined };
   735→      break;
   736→    }
   737→    case "run.failed": {
   738→      next = {
   739→        ...next,
   740→        phase: "failed",
   741→        error: { code: event.code, message: event.message, retryable: event.retryable },
   742→        // M6-H (defensive symmetry): a terminal failure carrying the §8.8
   743→        // code records the typed block too — the shipped gateway 402s BEFORE
   744→        // streaming, but the fold stays honest for any stream that says so.
   745→        ...(event.code === INSUFFICIENT_CREDITS_CODE ? { blocked: blockedCreditsFrom(event) } : {}),
   746→      };
   747→      break;
   748→    }
   749→    case "run.cancelled": {
   750→      next = { ...next, phase: "cancelled" };
   751→      break;
   752→    }
   753→  }
   754→  return next;
   755→}
