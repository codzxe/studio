    55→async function connect() {
    56→  for (let i = 0; i < 100; i++) {
    57→    try {
    58→      const list = await (await fetch(`http://127.0.0.1:${port}/json/list`)).json();
    59→      const page = list.find((t) => t.type === "page");
    60→      if (page) {
    61→        ws = new WebSocket(page.webSocketDebuggerUrl);
    62→        await new Promise((res, rej) => { ws.onopen = res; ws.onerror = rej; });
    63→        ws.onmessage = (m) => {
    64→          const msg = JSON.parse(m.data);
    65→          if (msg.id && waiting.has(msg.id)) {
    66→            waiting.get(msg.id)(msg);
    67→            waiting.delete(msg.id);
    68→          } else if (msg.method === "Runtime.consoleAPICalled") {
    69→            logs.push(msg.params.args.map((a) => a.value ?? a.description ?? "").join(" "));
    70→          } else if (msg.method === "Runtime.exceptionThrown") {
    71→            logs.push("EXCEPTION " + JSON.stringify(msg.params.exceptionDetails.exception?.description ?? msg.params.exceptionDetails.text));
    72→          }
    73→        };
    74→        return;
    75→      }
    76→    } catch {}
    77→    await sleep(100);
    78→  }
    79→  throw new Error("Chrome did not start");
    80→}

    82→function send(method, params = {}) {
    83→  const id = nextId++;
    84→  ws.send(JSON.stringify({ id, method, params }));
    85→  return new Promise((res, rej) => waiting.set(id, (m) => (m.error ? rej(new Error(`${method}: ${m.error.message}`)) : res(m.result))));
    86→}

    88→async function js(expr, timeout = 300000) {
    89→  // (promise rejections carry plain strings: wrap them so the message survives)
    90→  const wrapped = `(async () => { try { return await (${expr}); } catch (e) { throw new Error(String(e && e.message || e)); } })()`;
    91→  const r = await send("Runtime.evaluate", { expression: wrapped, awaitPromise: true, returnByValue: true, timeout });
    92→  if (r.exceptionDetails) throw new Error(`${expr}: ${r.exceptionDetails.exception?.description ?? r.exceptionDetails.text}`);
    93→  return r.result.value;
    94→}

   119→async function until(expr, ms = 60000, step = 200) {
   120→  const t0 = Date.now();
   121→  while (Date.now() - t0 < ms) {
   122→    if (await js(expr)) return Date.now() - t0;
   123→    await sleep(step);
   124→  }
   125→  throw new Error(`timeout waiting for ${expr}`);
   126→}

   133→    await sleep(200);
   134→  }
   135→  throw new Error("no download");
   136→}
