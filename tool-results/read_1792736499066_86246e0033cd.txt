     1→// @studio/storage — upload validation (W2-O). Zero-dep, pure byte readers.
     2→//
     3→// Spec anchors:
     4→// - §20 acceptance criterion 2: JPEG/PNG upload, 25 MB cap, EXIF stripped,
     5→//   60 MP decompression bomb rejected with a clear error.
     6→// - §15.2 uploads: MIME sniffing by magic bytes (never trust headers), size caps,
     7→//   image decode validation, EXIF stripping (colour profiles kept).
     8→// - §21 Security row: upload fuzzing — polyglot files, zip bombs, EXIF payloads.
     9→//
    10→// Security posture (deliberate, documented):
    11→// - Dimensions come from HEADER PARSING ONLY (PNG IHDR / JPEG SOF). Pixel data
    12→//   is never inflated or decoded here, so zip-bomb IDAT payloads cannot hurt
    13→//   this module; the megapixel ceiling is enforced on the claimed dimensions
    14→//   before any worker touches pixels (§20.2 "rejected with a clear error").
    15→// - Polyglot policy: a PNG must terminate at IEND with nothing after it. The
    16→//   whole-buffer scan only looks for signatures ≥ 8 bytes long (PNG signature,
    17→//   RIFF…WEBP) because a 3-byte JPEG SOI is expected ~once per 16.7 MB of
    18→//   compressed payload data. A JPEG SOI therefore only counts in a region that
    19→//   must be empty (after PNG IEND); a JPEG's own nested SOI is legal (EXIF
    20→//   thumbnails) and is removed by the strip plan instead.
    21→// - All failures are typed envelopes `{ ok:false, code, message, hint,
    22→//   retryable }` (AGENTS.md style). Nothing throws on malformed input; only a
    23→//   caller passing nonsensical limit numbers (a programmer error) throws.
    24→

   169→// ---------------------------------------------------------------------------
   170→// Internal byte reader (bounds-checked DataView; every read is pre-validated so
   171→// malformed input can never throw — it fails with a typed envelope instead).
   172→// ---------------------------------------------------------------------------
   173→
   174→class ByteReader {
   175→  private readonly view: DataView;
   176→
   177→  constructor(readonly raw: Uint8Array) {
   178→    this.view = new DataView(raw.buffer, raw.byteOffset, raw.byteLength);
   179→  }
   180→
   181→  get length(): number {
   182→    return this.raw.byteLength;
   183→  }
   184→
   185→  /** True when `n` bytes are readable at `at` (and `at` is not negative). */
   186→  has(at: number, n: number): boolean {
   187→    return at >= 0 && at + n <= this.length;
   188→  }
   189→
   190→  u8(at: number): number {
   191→    return this.view.getUint8(at);
   192→  }
   193→
   194→  u16be(at: number): number {
   195→    return this.view.getUint16(at, false);
   196→  }
   197→
   198→  u32be(at: number): number {
   199→    return this.view.getUint32(at, false);
   200→  }
   201→}

   203→// ---------------------------------------------------------------------------
   204→// Magic-byte helpers (hand-rolled; NO image libraries — AGENTS.md "no installs")

   206→
   207→/** PNG signature: 89 50 4E 47 0D 0A 1A 0A. */
   208→function isPngSignature(b: ByteReader, at: number): boolean {
   209→  return (
   210→    b.has(at, 8) &&
   211→    b.u8(at) === 0x89 &&
   212→    b.u8(at + 1) === 0x50 &&
   213→    b.u8(at + 2) === 0x4e &&
   214→    b.u8(at + 3) === 0x47 &&
   215→    b.u8(at + 4) === 0x0d &&
   216→    b.u8(at + 5) === 0x0a &&
   217→    b.u8(at + 6) === 0x1a &&
   218→    b.u8(at + 7) === 0x0a
   219→  );
   220→}
   221→
   222→/** JPEG SOI plus the following marker's 0xFF: FF D8 FF. */
   223→function isJpegSoi(b: ByteReader, at: number): boolean {
   224→  return b.has(at, 3) && b.u8(at) === 0xff && b.u8(at + 1) === 0xd8 && b.u8(at + 2) === 0xff;
   225→}
   226→
   227→/** WEBP container: "RIFF" …(4-byte size)… "WEBP". */
   228→function isWebpSignature(b: ByteReader, at: number): boolean {
   229→  return (
   230→    b.has(at, 12) &&
   231→    b.u8(at) === 0x52 &&
   232→    b.u8(at + 1) === 0x49 &&
   233→    b.u8(at + 2) === 0x46 &&
   234→    b.u8(at + 3) === 0x46 &&
   235→    b.u8(at + 8) === 0x57 &&
   236→    b.u8(at + 9) === 0x45 &&
   237→    b.u8(at + 10) === 0x42 &&
   238→    b.u8(at + 11) === 0x50
   239→  );
   240→}
   241→
   242→type MagicHit = "png" | "jpeg" | "webp";
   243→
   244→/**
   245→ * Strong signatures only (PNG sig, RIFF…WEBP — both ≥ 8 bytes, collision-safe
   246→ * across the whole 25 MB cap). Used on the entire buffer where compressed
   247→ * payload lives. Never returns early on a weak JPEG hit.
   248→ */
   249→function scanStrongMagic(b: ByteReader, from: number): "png" | "webp" | null {
   250→  for (let i = from; i + 8 <= b.length; i++) {
   251→    const v = b.u8(i);
   252→    if (v === 0x89 && isPngSignature(b, i)) {
   253→      return "png";
   254→    }
   255→    if (v === 0x52 && isWebpSignature(b, i)) {
   256→      return "webp";
   257→    }
   258→  }
   259→  return null;
   260→}
   261→
   262→/**
   263→ * Any image magic, including the weak 3-byte JPEG SOI — only meaningful in a
   264→ * region that must be empty (PNG post-IEND trailing bytes).
   265→ */
   266→function scanAnyMagic(b: ByteReader, from: number): MagicHit | null {
   267→  // Need i, i+1, i+2 readable → i + 3 <= length (a bare 3-byte SOI tail counts).
   268→  for (let i = from; i + 3 <= b.length; i++) {
   269→    const v = b.u8(i);
   270→    if (v === 0x89 && isPngSignature(b, i)) {
   271→      return "png";
   272→    }
   273→    if (v === 0xff && b.u8(i + 1) === 0xd8 && b.u8(i + 2) === 0xff) {
   274→      return "jpeg";
   275→    }
   276→    if (v === 0x52 && isWebpSignature(b, i)) {
   277→      return "webp";
   278→    }
   279→  }
   280→  return null;
   281→}

   283→// ---------------------------------------------------------------------------
   284→// CRC-32 (PNG chunks). Bitwise, table-free: IHDR is 17 bytes, this is cheap and
   285→// avoids a lookup table in a hot module. Same polynomial as zlib (0xEDB88320).
   286→// ---------------------------------------------------------------------------
   287→
   288→function crc32Range(bytes: Uint8Array, start: number, end: number): number {
   289→  const view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
   290→  let crc = 0xffff_ffff;
   291→  for (let i = start; i < end; i++) {
   292→    crc ^= view.getUint8(i);
   293→    for (let k = 0; k < 8; k++) {
   294→      crc = (crc >>> 1) ^ (0xedb8_8320 & -(crc & 1));
   295→    }
   296→  }
   297→  return (crc ^ 0xffff_ffff) >>> 0;
   298→}

   301→// sniffMime
   302→// ---------------------------------------------------------------------------
   303→
   304→/**
   305→ * Detect the media type from magic bytes (§15.2: sniff, never trust headers).
   306→ * PNG → `image/png`, JPEG (FF D8 FF) → `image/jpeg`, WEBP (RIFF…WEBP) →
   307→ * `image/webp`; anything else — including SVG, GIF and empty input — fails
   308→ * with `unsupported_media_type` (SVG is deliberately not sniffable: §15.2 XSS
   309→ * note — only sanitised SVG may ever be accepted, and never via this path).
   310→ */
   311→export function sniffMime(bytes: Uint8Array): SniffResult {
   312→  const b = new ByteReader(bytes);
   313→  if (isPngSignature(b, 0)) {
   314→    return { ok: true, mime: "image/png" };
   315→  }
   316→  if (isJpegSoi(b, 0)) {
   317→    return { ok: true, mime: "image/jpeg" };
   318→  }
   319→  if (isWebpSignature(b, 0)) {
   320→    return { ok: true, mime: "image/webp" };
   321→  }
   322→  return fail(
   323→    "unsupported_media_type",
   324→    "Unsupported file type.",
   325→    "Magic bytes must match PNG, JPEG or WebP.",
   326→  );
   327→}

   329→// ---------------------------------------------------------------------------
   330→// PNG dimensions — IHDR only, CRC verified (header parse, never pixels)
   331→// ---------------------------------------------------------------------------
   332→
   333→/**
   334→ * Parse PNG width/height from the IHDR chunk (big-endian at offsets 16/20 —
   335→ * first chunk, 13-byte data, CRC32 over `IHDR` + data verified). Zero
   336→ * dimensions, a foreign first chunk, or a CRC mismatch → `file_corrupt`.
   337→ * Structural tail checks (IEND/polyglot) live in `validateUpload`.
   338→ */
   339→export function pngDimensions(bytes: Uint8Array): DimensionResult {
   340→  const b = new ByteReader(bytes);
   341→  if (!isPngSignature(b, 0)) {
   342→    return corrupt("PNG signature missing.");
   343→  }
   344→  // Signature(8) + length(4) + "IHDR"(4) + data(13) + crc(4) = 33 bytes minimum.
   345→  if (!b.has(8, 25)) {
   346→    return corrupt("PNG is truncated (IHDR chunk incomplete).");
   347→  }
   348→  const ihdrLength = b.u32be(8);
   349→  if (ihdrLength !== 13) {
   350→    return corrupt(`PNG IHDR length is ${ihdrLength}; expected 13.`);
   351→  }
   352→  const isIhdr = b.u8(12) === 0x49 && b.u8(13) === 0x48 && b.u8(14) === 0x44 && b.u8(15) === 0x52;
   353→  if (!isIhdr) {
   354→    return corrupt("PNG first chunk must be IHDR.");
   355→  }
   356→  const width = b.u32be(16);
   357→  const height = b.u32be(20);
   358→  if (width === 0 || height === 0) {
   359→    return corrupt("PNG width or height is zero.");
   360→  }
   361→  const storedCrc = b.u32be(29);
   362→  const computedCrc = crc32Range(b.raw, 12, 29);
   363→  if (storedCrc !== computedCrc) {
   364→    return corrupt("PNG IHDR CRC does not match.");
   365→  }
   366→  return { ok: true, width, height, megapixels: megapixelsOf(width, height) };
   367→}

   369→// ---------------------------------------------------------------------------
   370→// JPEG segment walk (shared by dimensions / hasExif / stripExifPlan)
   371→// ---------------------------------------------------------------------------
   372→
   373→const MARKER_SOI = 0xd8;
   374→const MARKER_EOI = 0xd9;
   375→const MARKER_SOS = 0xda;
   376→const MARKER_COM = 0xfe;
   377→const MARKER_APP1 = 0xe1;
   378→const MARKER_APP2 = 0xe2;
   379→const MARKER_SOF0 = 0xc0;
   380→const MARKER_SOF2 = 0xc2;
   381→
   382→/** A length-bearing header segment (anything except SOI/EOI/SOS/RST/TEM). */
   383→interface JpegSegment {
   384→  readonly code: number;
   385→  /** Offset of the first 0xFF of the marker (FF fill bytes included). */
   386→  readonly start: number;
   387→  /** Offset of the marker code byte. */
   388→  readonly codeAt: number;
   389→  /** First payload byte (after the 2-byte length field). */
   390→  readonly payloadAt: number;
   391→  /** Exclusive end of the segment. */
   392→  readonly end: number;
   393→}
   394→
   395→type JpegProblem = "truncated" | "bad_marker" | "bad_length" | "second_soi" | "second_sof";
   396→
   397→interface JpegHeader {
   398→  readonly segments: readonly JpegSegment[];
   399→  readonly sof: { readonly height: number; readonly width: number } | null;
   400→  /** Where header parsing stopped (SOS/EOI = well-formed end of the header). */
   401→  readonly stop: { readonly kind: "sos" | "eoi" | "eof" | "desync"; readonly at: number };
   402→  /** First structural problem, if any (fatal for parsing). */
   403→  readonly problem: JpegProblem | null;
   404→}
   405→
   406→function problemMessage(problem: JpegProblem): string {
   407→  switch (problem) {
   408→    case "truncated":
   409→      return "JPEG is truncated.";
   410→    case "bad_marker":
   411→      return "JPEG marker structure is invalid.";
   412→    case "bad_length":
   413→      return "JPEG segment length is invalid.";
   414→    case "second_soi":
   415→      return "JPEG contains a second SOI marker (double header).";
   416→    case "second_sof":
   417→      return "JPEG contains multiple SOF markers (conflicting dimensions).";
   418→  }
   419→}
   420→
   421→/**
   422→ * Walk length-bearing header segments starting right after SOI, skipping the FF
   423→ * fill bytes that may precede a marker code. Stops at SOS (scan data), EOI,
   424→ * end-of-buffer, or the first structural problem. Never reads inside
   425→ * entropy-coded data. A second SOI/SOF is recorded as a problem: it is the
   426→ * classic double-header / dimension-mismatch polyglot trick (§21).
   427→ */
   428→function walkJpegHeader(b: ByteReader): JpegHeader {
   429→  const segments: JpegSegment[] = [];
   430→  let sof: { height: number; width: number } | null = null;
   431→  let pos = 2; // after FF D8
   432→  for (;;) {
   433→    if (!b.has(pos, 1) || b.u8(pos) !== 0xff) {
   434→      const outOfBytes = !b.has(pos, 1);
   435→      return {
   436→        segments,
   437→        sof,
   438→        stop: { kind: outOfBytes ? "eof" : "desync", at: pos },
   439→        problem: outOfBytes ? "truncated" : "bad_marker",
   440→      };
   441→    }
   442→    let codeAt = pos;
   443→    while (b.has(codeAt, 1) && b.u8(codeAt) === 0xff) {
   444→      codeAt++; // skip fill bytes
   445→    }
   446→    if (!b.has(codeAt, 1)) {
   447→      return { segments, sof, stop: { kind: "eof", at: pos }, problem: "truncated" };
   448→    }
   449→    const code = b.u8(codeAt);
   450→    if (code === MARKER_SOI) {
   451→      return { segments, sof, stop: { kind: "desync", at: pos }, problem: "second_soi" };
   452→    }
   453→    if (code === MARKER_EOI) {
   454→      return { segments, sof, stop: { kind: "eoi", at: pos }, problem: null };
   455→    }
   456→    if (code === MARKER_SOS) {
   457→      return { segments, sof, stop: { kind: "sos", at: pos }, problem: null };
   458→    }
   459→    if (code === 0x01 || (code >= 0xd0 && code <= 0xd7)) {
   460→      // TEM / RST markers have no length and are illegal before SOS.
   461→      return { segments, sof, stop: { kind: "desync", at: pos }, problem: "bad_marker" };
   462→    }
   463→    if (!b.has(codeAt + 1, 2)) {
   464→      return { segments, sof, stop: { kind: "eof", at: pos }, problem: "truncated" };
   465→    }
   466→    const length = b.u16be(codeAt + 1);
   467→    if (length < 2) {
   468→      return { segments, sof, stop: { kind: "desync", at: pos }, problem: "bad_length" };
   469→    }
   470→    const end = codeAt + 1 + length;
   471→    if (end > b.length) {
   472→      return { segments, sof, stop: { kind: "eof", at: pos }, problem: "truncated" };
   473→    }
   474→    const segment: JpegSegment = { code, start: pos, codeAt, payloadAt: codeAt + 3, end };
   475→    if (code === MARKER_SOF0 || code === MARKER_SOF2) {
   476→      if (sof !== null) {
   477→        return { segments, sof, stop: { kind: "desync", at: pos }, problem: "second_sof" };
   478→      }
   479→      // Lf(2) P(1) Y(2) X(2) — need precision + height + width readable.
   480→      if (length < 8 || !b.has(codeAt + 3, 5)) {
   481→        return { segments, sof, stop: { kind: "desync", at: pos }, problem: "bad_length" };
   482→      }
   483→      sof = { height: b.u16be(codeAt + 4), width: b.u16be(codeAt + 6) };
   484→    }
   485→    segments.push(segment);
   486→    pos = end;
   487→  }
   488→}

   490→// ---------------------------------------------------------------------------
   491→// JPEG dimensions — SOF0/SOF2 marker scan (handle FF padding), reject malformed
   492→// ---------------------------------------------------------------------------
   493→
   494→/**
   495→ * Scan for the SOF0/SOF2 frame marker and read width/height from it (baseline
   496→ * and progressive JPEG, §20.2's JPEG surface). Truncated files, files with no
   497→ * SOF before SOS/EOI, a second SOI/SOF, or zero dimensions → `file_corrupt`.
   498→ */
   499→export function jpegDimensions(bytes: Uint8Array): DimensionResult {
   500→  const b = new ByteReader(bytes);
   501→  if (!b.has(0, 2) || b.u8(0) !== 0xff || b.u8(1) !== 0xd8) {
   502→    return corrupt("JPEG SOI marker missing.");
   503→  }
   504→  const header = walkJpegHeader(b);
   505→  if (header.problem !== null) {
   506→    return corrupt(problemMessage(header.problem));
   507→  }
   508→  const frame = header.sof;
   509→  if (frame === null) {
   510→    return corrupt("JPEG has no SOF0/SOF2 frame marker.");
   511→  }
   512→  if (frame.height === 0 || frame.width === 0) {
   513→    return corrupt("JPEG width or height is zero.");
   514→  }
   515→  return {
   516→    ok: true,
   517→    width: frame.width,
   518→    height: frame.height,
   519→    megapixels: megapixelsOf(frame.width, frame.height),
   520→  };
   521→}

   523→// ---------------------------------------------------------------------------
   524→// dimensions — dispatch
   525→// ---------------------------------------------------------------------------
   526→
   527→/**
   528→ * Parse dimensions by sniffed type. PNG/JPEG are parsed header-only; WebP is
   529→ * recognised by `sniffMime` but not dimension-parsed here (§20.2's upload
   530→ * surface is JPEG/PNG), failing with `unsupported_media_type`.
   531→ */
   532→export function dimensions(bytes: Uint8Array): DimensionResult {
   533→  const sniffed = sniffMime(bytes);
   534→  if (!sniffed.ok) {
   535→    return sniffed;
   536→  }
   537→  if (sniffed.mime === "image/png") {
   538→    return pngDimensions(bytes);
   539→  }
   540→  if (sniffed.mime === "image/jpeg") {
   541→    return jpegDimensions(bytes);
   542→  }
   543→  return fail("unsupported_media_type", "WebP dimension parsing is not supported.", HINT_SUPPORTED);
   544→}
   545→
   546→// ---------------------------------------------------------------------------
   547→// EXIF detection (JPEG only — PNG has no EXIF standard, always false)
   548→// ---------------------------------------------------------------------------
   549→
   550→/**
   551→ * True when an APP1/APP2 segment carries an EXIF payload: the `Exif\0\0`
   552→ * identifier followed by a TIFF byte-order mark `II*\0` (little-endian) or
   553→ * `MM\0*` (big-endian). XMP APP1 and ICC APP2 do not match. Malformed input
   554→ * never throws — it simply reports no EXIF.
   555→ */
   556→function isExifSegment(b: ByteReader, segment: JpegSegment): boolean {
   557→  if (segment.code !== MARKER_APP1 && segment.code !== MARKER_APP2) {
   558→    return false;
   559→  }
   560→  const { payloadAt, end } = segment;
   561→  // "Exif\0\0" (6 bytes) + byte-order mark (4 bytes).
   562→  if (payloadAt + 10 > end) {
   563→    return false;
   564→  }
   565→  const exifId =
   566→    b.u8(payloadAt) === 0x45 &&
   567→    b.u8(payloadAt + 1) === 0x78 &&
   568→    b.u8(payloadAt + 2) === 0x69 &&
   569→    b.u8(payloadAt + 3) === 0x66 &&
   570→    b.u8(payloadAt + 4) === 0x00 &&
   571→    b.u8(payloadAt + 5) === 0x00;
   572→  if (!exifId) {
   573→    return false;
   574→  }
   575→  const littleEndian =
   576→    b.u8(payloadAt + 6) === 0x49 &&
   577→    b.u8(payloadAt + 7) === 0x49 &&
   578→    b.u8(payloadAt + 8) === 0x2a &&
   579→    b.u8(payloadAt + 9) === 0x00;
   580→  const bigEndian =
   581→    b.u8(payloadAt + 6) === 0x4d &&
   582→    b.u8(payloadAt + 7) === 0x4d &&
   583→    b.u8(payloadAt + 8) === 0x00 &&
   584→    b.u8(payloadAt + 9) === 0x2a;
   585→  return littleEndian || bigEndian;
   586→}
   587→
   588→/**
   589→ * Whether the bytes contain EXIF metadata. JPEG only: APP1/APP2 `Exif\0\0`
   590→ * with an `II*`/`MM*` TIFF byte order. PNG has no EXIF standard → false; WebP
   591→ * and anything unrecognised → false. Detector semantics — never throws.
   592→ */
   593→export function hasExif(bytes: Uint8Array): boolean {
   594→  const b = new ByteReader(bytes);
   595→  if (!b.has(0, 2) || b.u8(0) !== 0xff || b.u8(1) !== 0xd8) {
   596→    return false;
   597→  }
   598→  const header = walkJpegHeader(b);
   599→  for (const segment of header.segments) {
   600→    if (isExifSegment(b, segment)) {
   601→      return true;
   602→    }
   603→  }
   604→  return false;
   605→}
   606→
   607→// ---------------------------------------------------------------------------
   608→// stripExifPlan — pure description; the worker performs the re-slice
   609→// ---------------------------------------------------------------------------
   610→
   611→/** Marker label for a removed segment (only APP1/APP2/COM are ever dropped). */
   612→function removedMarker(code: number): RemovedSegment["marker"] {
   613→  if (code === MARKER_APP1) {
   614→    return "APP1";
   615→  }
   616→  if (code === MARKER_APP2) {
   617→    return "APP2";
   618→  }
   619→  return "COM";
   620→}
   621→
   622→/**
   623→ * Describe which byte ranges to KEEP so the worker can rebuild the file with
   624→ * EXIF removed (§15.2: strip EXIF, keep colour profiles). JPEG only: APP1/APP2
   625→ * segments carrying EXIF and every COM (comment) segment are marked removed;
   626→ * SOI, APP0 (JFIF), ICC APP2, DQT/DHT, SOF, SOS + entropy + EOI are kept, so
   627→ * re-slicing `bytes.subarray(r.start, r.end)` over `keepBytes` yields the same
   628→ * image with metadata gone. Non-JPEG input needs no strip and plans a whole-file
   629→ * keep. A JPEG whose header cannot be walked fails `file_corrupt` (the worker
   630→ * never strips a file `validateUpload` has not already accepted).
   631→ */
   632→export function stripExifPlan(bytes: Uint8Array): StripExifResult {
   633→  const b = new ByteReader(bytes);
   634→  const wholeFile: StripExifPlan = {
   635→    ok: true,
   636→    needsStrip: false,
   637→    keepBytes: [{ start: 0, end: b.length }],
   638→    removed: [],
   639→  };
   640→  if (!b.has(0, 2) || b.u8(0) !== 0xff || b.u8(1) !== 0xd8) {
   641→    return wholeFile; // PNG (no EXIF standard) / WebP / unknown: nothing to strip.
   642→  }
   643→  const header = walkJpegHeader(b);
   644→  if (header.problem !== null) {
   645→    return fail("file_corrupt", problemMessage(header.problem), HINT_REEXPORT);
   646→  }
   647→  const keep: ByteRange[] = [];
   648→  const removed: RemovedSegment[] = [];
   649→  let cursor = 0; // SOI [0,2) is always kept; ranges stay contiguous.
   650→  for (const segment of header.segments) {
   651→    const isComment = segment.code === MARKER_COM;
   652→    if (!isComment && !isExifSegment(b, segment)) {
   653→      continue;
   654→    }
   655→    if (segment.start > cursor) {
   656→      keep.push({ start: cursor, end: segment.start });
   657→    }
   658→    removed.push({ marker: removedMarker(segment.code), start: segment.start, end: segment.end });
   659→    cursor = segment.end;
   660→  }
   661→  if (removed.length === 0) {
   662→    return wholeFile;
   663→  }
   664→  if (cursor < b.length) {
   665→    keep.push({ start: cursor, end: b.length });
   666→  }
   667→  return { ok: true, needsStrip: true, keepBytes: keep, removed };
   668→}
   669→
   670→// ---------------------------------------------------------------------------
   671→// validateUpload — the §20.2 gate
   672→// ---------------------------------------------------------------------------
   673→
   674→/**
   675→ * Structural PNG tail check: walk the chunk chain to IEND and reject anything
   676→ * after it. A second IHDR anywhere is a dimension-mismatch trick → corrupt.
   677→ * Trailing bytes containing an image signature → polyglot; other trailing junk
   678→ * → corrupt. (Only reached for files that already parsed and sit under the
   679→ * megapixel ceiling, so the walk is bounded by the 25 MB size cap.)
   680→ */
   681→function pngTailCheck(b: ByteReader): UploadFailure | null {
   682→  let pos = 8;
   683→  for (;;) {
   684→    if (!b.has(pos, 8)) {
   685→      return corrupt("PNG is truncated before IEND.");
   686→    }
   687→    const length = b.u32be(pos);
   688→    const typeIs = (t0: number, t1: number, t2: number, t3: number): boolean =>
   689→      b.u8(pos + 4) === t0 && b.u8(pos + 5) === t1 && b.u8(pos + 6) === t2 && b.u8(pos + 7) === t3;
   690→    if (typeIs(0x49, 0x48, 0x44, 0x52) && pos !== 8) {
   691→      return corrupt("PNG contains a duplicate IHDR chunk.");
   692→    }
   693→    // 4 length + 4 type + data + 4 crc must fit inside the file.
   694→    const end = pos + 12 + length;
   695→    if (end > b.length) {
   696→      return corrupt("PNG chunk exceeds the file length.");
   697→    }
   698→    if (typeIs(0x49, 0x45, 0x4e, 0x44)) {
   699→      if (length !== 0) {
   700→        return corrupt("PNG IEND chunk is invalid.");
   701→      }
   702→      if (end < b.length) {
   703→        // The trailing region must be empty; any image magic there = polyglot.
   704→        if (scanAnyMagic(b, end) !== null) {
   705→          return polyglot();
   706→        }
   707→        return corrupt("PNG has trailing data after IEND.");
   708→      }
   709→      return null;
   710→    }
   711→    pos = end;
   712→  }
   713→}
   714→
   715→/**
   716→ * Validate an upload against §20.2: MIME sniff by magic bytes, 25 MB cap
   717→ * (default), EXIF handled downstream (never rejected — §20.2 says stripped),
   718→ * 60 MP bomb ceiling enforced from header dimensions alone, and §21 polyglot
   719→ * guards (PNG double headers, trailing second-format data, duplicate IHDR,
   720→ * JPEG second SOI/SOF).
   721→ *
   722→ * Check order (cheap → expensive, and deliberate): size cap → sniff →
   723→ * strong-signature polyglot scan → header dimensions → megapixel ceiling →
   724→ * PNG structural tail walk. A 61 MP "header-only" stub therefore reports
   725→ * `resolution_exceeds_ceiling` (the §20.2 clear error) instead of a truncation
   726→ * detail; structural strictness applies to files under the ceiling.
   727→ *
   728→ * `planCeiling` (§14.1 per-plan canvas cap) narrows the megapixel ceiling when
   729→ * present: the effective cap is `min(maxMegapixels, planCeiling)`.
   730→ */
   731→export function validateUpload(bytes: Uint8Array, limits: UploadLimits = {}): UploadValidation {
   732→  const maxBytes = limits.maxBytes ?? DEFAULT_MAX_UPLOAD_BYTES;
   733→  const maxMegapixels = limits.maxMegapixels ?? DEFAULT_MAX_UPLOAD_MEGAPIXELS;
   734→  const { planCeiling } = limits;
   735→  const positiveFinite = (n: number): boolean => Number.isFinite(n) && n > 0;
   736→  if (!positiveFinite(maxBytes) || !positiveFinite(maxMegapixels)) {
   737→    // Programmer error (limits come from config, not user input): fail loudly.
   738→    throw new Error("[storage/validate] limits must be positive finite numbers");
   739→  }
   740→  if (planCeiling !== undefined && !positiveFinite(planCeiling)) {
   741→    throw new Error("[storage/validate] planCeiling must be a positive finite number");
   742→  }
   743→
   744→  const b = new ByteReader(bytes);
   745→
   746→  // 1. Size cap first — cheapest, and it stops all scans before they start.
   747→  if (b.length > maxBytes) {
   748→    return fail(
   749→      "upload_too_large",
   750→      `File is ${b.length} bytes; the upload cap is ${maxBytes} bytes.`,
   751→      `max ${maxBytes} bytes`,
   752→    );
   753→  }
   754→
   755→  // 2. Magic-byte sniff (§15.2: headers lie).
   756→  const sniffed = sniffMime(bytes);
   757→  if (!sniffed.ok) {
   758→    return sniffed;
   759→  }
   760→  const { mime } = sniffed;
   761→  if (mime === "image/webp") {
   762→    // Recognised on the wire, but §20.2's upload surface is JPEG/PNG only.
   763→    return fail("unsupported_media_type", "Unsupported upload type: image/webp.", HINT_SUPPORTED);
   764→  }
   765→
   766→  // 3. Polyglot scan for strong signatures (PNG sig, RIFF…WEBP — collision-safe
   767→  //    inside compressed payload at the 25 MB cap). The weak JPEG SOI is
   768→  //    handled by the PNG tail walk instead (see module header).
   769→  const foreign = scanStrongMagic(b, 1);
   770→  if (foreign !== null) {
   771→    return polyglot();
   772→  }
   773→
   774→  // 4. Header dimensions (never pixels).
   775→  const parsed = mime === "image/png" ? pngDimensions(bytes) : jpegDimensions(bytes);
   776→  if (!parsed.ok) {
   777→    return parsed;
   778→  }
   779→
   780→  // 5. Megapixel ceiling from header dimensions alone — the 60 MP bomb gate.
   781→  const ceiling = planCeiling === undefined ? maxMegapixels : Math.min(maxMegapixels, planCeiling);
   782→  if (parsed.megapixels > ceiling) {
   783→    const detail =
   784→      `Image is ${parsed.width}×${parsed.height} ` +
   785→      `(${fmtMp(parsed.megapixels)} megapixels); the cap is ${ceiling} megapixels.`;
   786→    return fail("resolution_exceeds_ceiling", detail, `max ${ceiling} MP per image`);
   787→  }
   788→
   789→  // 6. PNG structural tail: IEND termination, no duplicate IHDR, no trailing file.
   790→  if (mime === "image/png") {
   791→    const tail = pngTailCheck(b);
   792→    if (tail !== null) {
   793→      return tail;
   794→    }
   795→  }
   796→
   797→  return {
   798→    ok: true,
   799→    mime,
   800→    width: parsed.width,
   801→    height: parsed.height,
   802→    megapixels: parsed.megapixels,
   803→  };
   804→}
   805→
   806→// ---------------------------------------------------------------------------
   807→// checksum — sha256 via node:crypto (asset integrity/dedupe)
   808→// ---------------------------------------------------------------------------
   809→
   810→/** SHA-256 of the bytes as lowercase hex (64 chars). Used for asset dedupe and
   811→ * integrity records; never a secret. */
   812→export function checksum(bytes: Uint8Array): string {
   813→  return createHash("sha256").update(bytes).digest("hex");
   814→}
