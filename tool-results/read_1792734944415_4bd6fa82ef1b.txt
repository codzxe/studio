     2→// PRODUCT_ARCHITECTURE.md §8.5–§8.7, docs/phases/M5-SEAMS.md pinned seam,

     4→// 100 commands / 2 MB / 60 s).
     5→//
     6→// Mission (laws #5 + #6): the model proposes an EditPlan built from CURATED
     7→// write-tool args; this module is the only thing that turns a plan into an
     8→// engine `AgentBatch`. The per-step `batch` fields the model proposes are
     9→// validated and ADVISORY ONLY — the compiled batch always comes from tool
    10→// args translated against the curated command registry (deps.registry). One
    11→// plan ⇒ ONE atomic batch ⇒ ONE undo step ⇒ ONE version (§8.4/§8.5).
    12→//
    13→// Tool → command translation table (every id verified against the real
    14→// EffectCraft source at editor/crates/engine/src/commands/*.rs):
    15→//
    16→//   create_composition → comp.new {name, width, height, frameRate,
    17→//                                  duration: args.durationSeconds}
    18→//   create_text        → layer.newText {text, name?, position?, box?, font?,
    19→//                                  style?, size?, fill?, stroke?, strokeWidth?,
    20→//                                  tracking?, leading?} — colors take BOTH
    21→//                                  COLOR_SCHEMA forms (F-S1: a [r,g,b(,a)]
    22→//                                  channel array compiles to the engine's
    23→//                                  canonical #hex form, see engineColor)
    24→//   set_properties     → prop.set × update {layer, path, value}
    25→//                                  (path "name" → layer.rename, G-7/D-21)
    26→//   set_keyframes      → prop.addKey × keyframe {layer, path: propertyPath,
    27→//                                  time, value}
    28→//   add_effect         → effect.apply {effect, layers: [layer]}
    29→//   apply_preset       → anim.applyBuiltinPreset {preset, layers: targets}
    30→//   undo_run           → edit.undo {}
    31→//   save_version       → NO engine command — a compile-time annotation
    32→//                        (result.saveVersion) consumed by the apply flow.
    33→//
    34→// Ambiguity rulings (documented for the integrator — see the M5-B worklog):
    35→//   * apply_preset NEVER compiles to `anim.applyPreset`: that command takes a
    36→//     filesystem `path` (or an inline preset OBJECT), and a model-supplied
    37→//     path breaks law #5 ("the model never sees filesystem paths"). Path-like
    38→//     preset strings emit a `warn` and compile as builtin preset names.
    39→//   * add_effect `params` and set_keyframes `easing` are CLIENT STATE (the
    40→//     effect instance index and keyframe interpolation live in the editor);
    41→//     they cannot be compiled server-side without guessing ids. They emit
    42→//     `warn` issues instead of being silently dropped or fabricated.
    43→//   * Layer references stay NAME strings in the compiled batch (the engine
    44→//     accepts id|name|#n); `placeholders` + `resolutionRequests` let the
    45→//     client (M5-F) substitute resolved ids before Apply. The compiler NEVER
    46→//     fabricates ids; an injected `layerResolver` may substitute real ones.
    47→//   * SEAMS names the request field `disambiguation` (the brief said
    48→//     `candidates` — SEAMS wins per its own precedence rule). It carries the

    50→//   * Risk and credits are RECOMPUTED from the compiled commands (never the
    51→//     model's numbers). "High" is currently unreachable via the curated
    52→//     tools (they compile to additive/property commands only) — it becomes
    53→//     reachable when the registry widens past the eight write tools.

   234→function clip(text: string, max = MAX_MESSAGE_LENGTH): string {
   235→  return text.length <= max ? text : `${text.slice(0, max - 1)}…`;
   236→}

   238→function isPlainObject(value: unknown): value is Record<string, unknown> {
   239→  return typeof value === "object" && value !== null && !Array.isArray(value);
   240→}
   241→
   242→function isFiniteNumber(value: unknown): value is number {
   243→  return typeof value === "number" && Number.isFinite(value);
   244→}
   245→
   246→function readString(source: Record<string, unknown>, key: string): string | undefined {
   247→  const value = source[key];
   248→  return typeof value === "string" && value.length > 0 ? value : undefined;
   249→}

   251→/** Own-property registry lookup — a "__proto__" command id must NOT resolve
   252→ * through the prototype chain (M4-N F-1 defense, applied to the registry). */
   253→function registryHas(registry: CommandSchema, id: string): boolean {
   254→  return Object.prototype.hasOwnProperty.call(registry, id);
   255→}

   257→/** Deterministic, depth-capped canonical JSON (sorted object keys) used to
   258→ * compare the model's proposed batch with the compiled translation. */
   259→function stableStringify(value: unknown, depth = 0): string {
   260→  if (depth > 64) return '"<deep>"';
   261→  if (value === null) return "null";
   262→  switch (typeof value) {
   263→    case "string":
   264→      return JSON.stringify(value);
   265→    case "number":
   266→      return Number.isFinite(value) ? String(value) : '"<nan>"';
   267→    case "boolean":
   268→    case "bigint":
   269→      return String(value);
   270→    default: {
   271→      if (Array.isArray(value)) {
   272→        return `[${value.map((item) => stableStringify(item, depth + 1)).join(",")}]`;
   273→      }
   274→      if (typeof value !== "object") return '"<unserializable>"';
   275→      const record = value as Record<string, unknown>;
   276→      const body = Object.keys(record)
   277→        .sort()
   278→        .map((key) => `${JSON.stringify(key)}:${stableStringify(record[key], depth + 1)}`)
   279→        .join(",");
   280→      return `{${body}}`;
   281→    }
   282→  }
   283→}

   285→function riskMax(a: PlanRisk, b: PlanRisk): PlanRisk {
   286→  return PLAN_RISK_ORDER[a] >= PLAN_RISK_ORDER[b] ? a : b;

   289→// ---------------------------------------------------------------------------
   290→// Color (F-S1) — the engine's text-color convention, mirrored EXACTLY
   291→// ---------------------------------------------------------------------------
   292→
   293→/**
   294→ * The engine's text-color form (F-S1). `layer.newText` / `layer.setText`
   295→ * fill+stroke parse through the engine's own `color_json`
   296→ * (keyframe/src/text_doc.rs:956): a `#rrggbb` / `#rrggbbaa` string normalizes
   297→ * each channel as `u8 / 255`, while an ARRAY is read RAW into the internal
   298→ * 0..1 color space (no scaling — that arm exists for the engine's own
   299→ * 0..1-internal callers, e.g. script prelude `fill: [1, 1, 1, 1]`). The
   300→ * contracts COLOR_SCHEMA speaks 0..255 channels, so a schema-valid
   301→ * `[r, g, b]` / `[r, g, b, a]` array compiles into the engine's canonical hex
   302→ * string — the exact `/255` quantization the engine's own hex arm applies —
   303→ * and is NEVER passed through raw (a raw 0..255 array would land in a 0..1
   304→ * space and render as a blown-out color). Strings ride as-is (the common
   305→ * case and every existing golden).
   306→ */
   307→function engineColor(value: unknown): string | undefined {
   308→  if (typeof value === "string") return value.length > 0 ? value : undefined;
   309→  if (!Array.isArray(value)) return undefined;
   310→  if (value.length < 3 || value.length > 4) return undefined;
   311→  const channels: number[] = [];
   312→  for (const channel of value) {
   313→    if (typeof channel !== "number" || !Number.isFinite(channel)) return undefined;
   314→    // Clamp to the schema's 0..255 band before quantizing (the engine's hex
   315→    // arm parses exactly two hex digits per channel; Math.round halves up).
   316→    channels.push(Math.min(255, Math.max(0, Math.round(channel))));
   317→  }
   318→  return `#${channels.map((channel) => channel.toString(16).padStart(2, "0")).join("")}`;
   319→}

   321→// ---------------------------------------------------------------------------
   322→// Risk recompute (brief: from compiled commands — never the model's number)
   323→// ---------------------------------------------------------------------------
   324→
   325→/**
   326→ * Recomputed plan risk from the COMPILED commands plus an optional tool-catalog
   327→ * floor. delete-ish / effect.remove / render/export / edit.undo+redo ⇒ at
   328→ * least medium; nothing of the sort ⇒ low. Exported for tests and future
   329→ * registry widening.
   330→ */
   331→export function recomputePlanRisk(
   332→  steps: ReadonlyArray<Pick<BatchStep, "command">>,
   333→  toolFloor: PlanRisk = "low",
   334→): PlanRisk {
   335→  let risk: PlanRisk = "low";
   336→  for (const step of steps) {
   337→    const id = step.command;
   338→    if (DELETE_ISH_COMMAND.test(id) || RENDER_ISH_COMMAND.test(id) || HISTORY_COMMANDS.has(id)) {
   339→      risk = riskMax(risk, "medium");
   340→    }
   341→  }
   342→  return riskMax(risk, toolFloor);
   343→}

   367→class Compiler {
   368→  readonly warnings: PlanIssue[] = [];
   369→  readonly fatals: PlanIssue[] = [];
   370→  private readonly compiledSteps: BatchStep[] = [];
   371→  private readonly sites: Array<LocalLayerSite & { planStep: number; globalStep: number }> = [];
   372→  private readonly toolRisks: PlanRisk[] = [];
   373→  private saveVersion: SaveVersionAnnotation | undefined;
   374→  private saveVersionStep: number | undefined;
   375→
   376→  /** Plan context (D-21): the plan's affectedLayers names — apply_preset's
   377→   * no-layers fallback. Set by compilePlanInner before the step loop. */
   378→  planLayerNames: readonly string[] = [];
   379→
   380→  constructor(private readonly depsRegistry: CommandSchema) {}
   381→
   382→  private pushIssue(issue: PlanIssue): void {
   383→    const target = issue.severity === "fatal" ? this.fatals : this.warnings;
   384→    if (target.length >= MAX_ISSUES) return; // bounded, deterministic
   385→    target.push(issue);
   386→  }
   387→
   388→  fatal(
   389→    code: PlanIssueCode,
   390→    message: string,
   391→    extra?: Omit<PlanIssue, "severity" | "code" | "message">,
   392→  ): void {
   393→    this.pushIssue({ severity: "fatal", code, message: clip(message), ...extra });
   394→  }
   395→
   396→  warn(
   397→    code: PlanIssueCode,
   398→    message: string,
   399→    extra?: Omit<PlanIssue, "severity" | "code" | "message">,
   400→  ): void {
   401→    this.pushIssue({ severity: "warn", code, message: clip(message), ...extra });
   402→  }
   403→
   404→  /** Translate one plan step's curated args into registry command steps. */
   405→  translate(planStep: number, tool: string, args: unknown): StepTranslation {
   406→    const steps: BatchStep[] = [];
   407→    const sites: LocalLayerSite[] = [];
   408→    const argsObj: Record<string, unknown> = isPlainObject(args) ? args : {};
   409→    const add = (command: string, params: Record<string, unknown>): void => {
   410→      steps.push({ command, params });
   411→    };
   412→    const addSite = (param: string, layerName: string): void => {
   413→      sites.push({ localStep: steps.length - 1, param, layerName });
   414→    };
   415→    const unusable = (field: string, why: string): void => {
   416→      this.warn("unusable_args", `field \`${field}\` ${why}`, { step: planStep, tool });
   417→    };
   418→    const compilesToNothing = (): void => {
   419→      // G-3 completeness ruling: a step that translates to ZERO commands
   420→      // makes the WHOLE plan fatally incomplete — the user must never approve
   421→      // N edits and silently get fewer (M5-M gp-11 finding).
   422→      this.fatal(
   423→        "empty_step",
   424→        "step compiled to no engine commands; the plan is rejected as incomplete",
   425→        { step: planStep, tool },
   426→      );
   427→    };
   428→    /** Read `[{name}]`-shaped layer refs (M5-A LAYER_REFS_SCHEMA) as names. */
   429→    const readLayerNames = (value: unknown): string[] => {
   430→      const names: string[] = [];
   431→      if (!Array.isArray(value)) return names;
   432→      for (const entry of value) {
   433→        const name = isPlainObject(entry) ? readString(entry, "name") : undefined;
   434→        if (name !== undefined) names.push(name);
   435→        else unusable("layers", "entry has no usable name (skipped)");
   436→      }
   437→      return names;
   438→    };
   439→
   440→    switch (tool) {
   441→      case "create_composition": {
   442→        const params: Record<string, unknown> = {};
   443→        const name = readString(argsObj, "name");
   444→        if (name !== undefined) params.name = name;
   445→        for (const key of ["width", "height", "frameRate"] as const) {
   446→          const value = argsObj[key];
   447→          if (isFiniteNumber(value)) params[key] = value;
   448→          else if (value !== undefined) unusable(key, "is not a finite number");
   449→        }
   450→        const duration = argsObj.durationSeconds;
   451→        if (isFiniteNumber(duration)) params.duration = duration;
   452→        else if (duration !== undefined) unusable("durationSeconds", "is not a finite number");
   453→        if (params.name === undefined) {
   454→          compilesToNothing();
   455→          break;
   456→        }
   457→        add("comp.new", params);
   458→        break;
   459→      }
   460→
   461→      case "create_text": {
   462→        const params: Record<string, unknown> = {};
   463→        const text = readString(argsObj, "text");
   464→        if (text !== undefined) params.text = text;
   465→        else if ("text" in argsObj) unusable("text", "is not a non-empty string");
   466→        const name = readString(argsObj, "name");
   467→        if (name !== undefined) params.name = name;
   468→        const font = readString(argsObj, "font");
   469→        if (font !== undefined) params.font = font;
   470→        const style = readString(argsObj, "style");
   471→        if (style !== undefined) params.style = style;
   472→        // F-S1: fill/stroke accept BOTH COLOR_SCHEMA forms. Strings ride 1:1
   473→        // (the common case and every pre-existing golden); schema-valid
   474→        // [r, g, b(, a)] channel arrays are CONVERTED into the engine's
   475→        // canonical hex form (see engineColor) instead of being silently
   476→        // dropped — and anything else warns as unusable, never rides.
   477→        for (const key of ["fill", "stroke"] as const) {
   478→          const value = argsObj[key];
   479→          if (value === undefined) continue;
   480→          const color = engineColor(value);
   481→          if (color !== undefined) {
   482→            params[key] = color;
   483→          } else {
   484→            this.warn(
   485→              "unusable_args",
   486→              `field \`${key}\` is not a #hex color string or a [r, g, b(, a)] channel array`,
   487→              { step: planStep, tool, param: key },
   488→            );
   489→          }
   490→        }
   491→        for (const key of ["size", "strokeWidth", "tracking", "leading"] as const) {
   492→          const value = argsObj[key];
   493→          if (isFiniteNumber(value)) params[key] = value;
   494→          else if (value !== undefined) unusable(key, "is not a finite number");
   495→        }
   496→        const box = argsObj.box;
   497→        if (box !== undefined) {
   498→          const [bx, by, w, h] = Array.isArray(box) ? box : [];
   499→          if (isFiniteNumber(bx) && isFiniteNumber(by) && isFiniteNumber(w) && isFiniteNumber(h)) {
   500→            params.box = [bx, by, w, h];
   501→          } else {
   502→            unusable("box", "is not an [x, y, w, h] number tuple");
   503→          }
   504→        }
   505→        if (params.text === undefined) {
   506→          compilesToNothing();
   507→          break;
   508→        }
   509→        const position = argsObj.position;
   510→        if (position !== undefined) {
   511→          const [x, y] = Array.isArray(position) ? position : [];
   512→          if (isFiniteNumber(x) && isFiniteNumber(y)) {
   513→            params.position = [x, y];
   514→          } else {
   515→            unusable("position", "is not a [x, y] number pair");
   516→          }
   517→        }
   518→        if (params.text === undefined) {
   519→          compilesToNothing();
   520→          break;
   521→        }
   522→        add("layer.newText", params);
   523→        break;
   524→      }
   525→
   526→      case "set_properties": {
   527→        // M5-A shape: {layers: [{name}], properties: [{path, value}]} ->
   528→        // cross-product prop.set per layer x property; a "name" path maps to
   529→        // layer.rename (G-7 -- rename was otherwise inexpressible).
   530→        const layers = readLayerNames(argsObj.layers);
   531→        const properties = argsObj.properties;
   532→        if (layers.length === 0) {
   533→          unusable("layers", "no usable named layers");
   534→          compilesToNothing();
   535→          break;
   536→        }
   537→        if (!Array.isArray(properties)) {
   538→          unusable("properties", "is not an array");
   539→          compilesToNothing();
   540→          break;
   541→        }
   542→        for (const property of properties) {
   543→          if (!isPlainObject(property)) {
   544→            unusable("properties", "contains a non-object entry (skipped)");
   545→            continue;
   546→          }
   547→          const path = readString(property, "path");
   548→          if (path === undefined) {
   549→            unusable("properties", "entry has no path (skipped)");
   550→            continue;
   551→          }
   552→          for (const layer of layers) {
   553→            if (path === "name") {
   554→              const value = readString(property, "value");
   555→              if (value === undefined) {
   556→                unusable("properties", "a rename needs a string value (skipped)");
   557→                continue;
   558→              }
   559→              add("layer.rename", { layer, name: value });
   560→            } else {
   561→              const params: Record<string, unknown> = { layer, path };
   562→              if ("value" in property) params.value = property.value;
   563→              add("prop.set", params);
   564→            }
   565→            addSite("layer", layer);
   566→          }
   567→        }
   568→        if (steps.length === 0) compilesToNothing();
   569→        break;
   570→      }
   571→
   572→      case "set_keyframes": {
   573→        // M5-A shape: {layers: [{name}], propertyPath, keys: [{timeSeconds,
   574→        // value, interpolation?}]} -> cross-product prop.addKey per layer x key.
   575→        const layers = readLayerNames(argsObj.layers);
   576→        const keys = argsObj.keys;
   577→        const propertyPath = readString(argsObj, "propertyPath");
   578→        if (layers.length === 0) {
   579→          unusable("layers", "no usable named layers");
   580→          compilesToNothing();
   581→          break;
   582→        }
   583→        if (!Array.isArray(keys)) {
   584→          unusable("keys", "is not an array");
   585→          compilesToNothing();
   586→          break;
   587→        }
   588→        for (const key of keys) {
   589→          if (!isPlainObject(key)) {
   590→            unusable("keys", "contains a non-object entry (skipped)");
   591→            continue;
   592→          }
   593→          const time = key.timeSeconds;
   594→          const interpolation = readString(key, "interpolation");
   595→          for (const layer of layers) {
   596→            const params: Record<string, unknown> = { layer };
   597→            if (propertyPath !== undefined) params.path = propertyPath;
   598→            if (isFiniteNumber(time)) params.time = time;
   599→            else if (time !== undefined) {
   600→              unusable("keys", "entry timeSeconds is not a finite number");
   601→            }
   602→            if ("value" in key) params.value = key.value;
   603→            if (interpolation !== undefined && interpolation !== "linear") {
   604→              this.warn(
   605→                "uncompilable_arg",
   606→                `interpolation \`${clip(interpolation, 24)}\` is editor-side keyframe state; the key compiles with the engine default (set it via the timeline after apply)`,
   607→                { step: planStep, tool, param: "interpolation" },
   608→              );
   609→            }
   610→            add("prop.addKey", params);
   611→            addSite("layer", layer);
   612→          }
   613→        }
   614→        if (steps.length === 0) compilesToNothing();
   615→        break;
   616→      }
   617→
   618→      case "add_effect": {
   619→        // M5-A shape: {layers: [{name}], effect: {name}, params?} ->
   620→        // effect.apply {layers: [names], effect: name}. Effect instance
   621→        // params are editor state (uncompilable server-side; warn + set via
   622→        // set_properties after apply).
   623→        const layers = readLayerNames(argsObj.layers);
   624→        if (layers.length === 0) {
   625→          unusable("layers", "no usable named layers");
   626→          compilesToNothing();
   627→          break;
   628→        }
   629→        const effectObj = argsObj.effect;
   630→        const effectName = isPlainObject(effectObj) ? readString(effectObj, "name") : undefined;
   631→        const params: Record<string, unknown> = { layers: [...layers] };
   632→        if (effectName !== undefined) params.effect = effectName;
   633→        else unusable("effect", "has no usable name");
   634→        add("effect.apply", params);
   635→        layers.forEach((layer, index) => addSite(`layers[${index}]`, layer));
   636→        const effectParams = argsObj.params;
   637→        if (effectParams !== undefined) {
   638→          if (isPlainObject(effectParams)) {
   639→            for (const key of Object.keys(effectParams).sort()) {
   640→              this.warn(
   641→                "uncompilable_arg",
   642→                `effect parameter \`${clip(key, 40)}\` cannot be compiled server-side (the effect instance index is editor state); set it via set_properties after apply`,
   643→                { step: planStep, tool, param: key },
   644→              );
   645→            }
   646→          } else {
   647→            unusable("params", "is not an object");
   648→          }
   649→        }
   650→        break;
   651→      }
   652→
   653→      case "apply_preset": {
   654→        // M5-A shape: {preset: {name}|{builtinId}, layers? [{name}], args?}.
   655→        // D-21 ruling: when `layers` is omitted the plan's affectedLayers
   656→        // names are the targets (never the live editor selection -- a
   657→        // selection-capture hazard the compiler refuses).
   658→        const explicitLayers = readLayerNames(argsObj.layers);
   659→        const layers = explicitLayers.length > 0 ? explicitLayers : [...this.planLayerNames];
   660→        if (layers.length === 0) {
   661→          unusable(
   662→            "layers",
   663→            "no target layers: neither step layers nor plan affectedLayers name any",
   664→          );
   665→          compilesToNothing();
   666→          break;
   667→        }
   668→        const presetObj = argsObj.preset;
   669→        let preset: string | undefined;
   670→        if (isPlainObject(presetObj)) {
   671→          preset = readString(presetObj, "name") ?? readString(presetObj, "builtinId");
   672→        }
   673→        if (preset !== undefined && PRESET_PATH_LIKE.test(preset)) {
   674→          this.warn(
   675→            "uncompilable_arg",
   676→            "preset looks like a filesystem path; paths are never agent-compilable " +
   677→              "(compiled as a builtin preset name)",
   678→            { step: planStep, tool, param: "preset" },
   679→          );
   680→        }
   681→        if (preset === undefined) {
   682→          unusable("preset", "needs a name or builtinId");
   683→          compilesToNothing();
   684→          break;
   685→        }
   686→        const params: Record<string, unknown> = { layers: [...layers], preset };
   687→        const presetArgs = argsObj.args;
   688→        if (isPlainObject(presetArgs)) {
   689→          for (const key of Object.keys(presetArgs).sort()) {
   690→            this.warn(
   691→              "uncompilable_arg",
   692→              `preset arg \`${clip(key, 40)}\` is editor-side state; set it via set_properties after apply`,
   693→              { step: planStep, tool, param: key },
   694→            );
   695→          }
   696→        }
   697→        add("anim.applyBuiltinPreset", params);
   698→        layers.forEach((layer, index) => addSite(`layers[${index}]`, layer));
   699→        break;
   700→      }
   701→
   702→      case "undo_run": {
   703→        add("edit.undo", {});
   704→        if ("runId" in argsObj) {
   705→          this.warn(
   706→            "undo_run_target",
   707→            "undo_run named a specific run; the engine command reverts the most recently " +
   708→              "applied batch — the gateway checks run ordering at undo time",
   709→            { step: planStep, tool },
   710→          );
   711→        }
   712→        break;
   713→      }
   714→
   715→      case "save_version": {
   716→        const rawLabel = argsObj.label;
   717→        const label = typeof rawLabel === "string" ? rawLabel : undefined;
   718→        if (this.saveVersion !== undefined && this.saveVersionStep !== undefined) {
   719→          this.warn(
   720→            "duplicate_save_version",
   721→            "an earlier save_version step was superseded (the last labels the version)",
   722→            {
   723→              step: this.saveVersionStep,
   724→              tool,
   725→            },
   726→          );
   727→        }
   728→        this.saveVersion = label !== undefined ? { label } : {};
   729→        this.saveVersionStep = planStep;
   730→        break;
   731→      }
   732→
   733→      default: {
   734→        // Exhaustiveness guard — lookupWriteTool already rejected unknown ids.
   735→        this.fatal("unknown_tool", `tool \`${clip(tool, 64)}\` is not a write tool`, {
   736→          step: planStep,
   737→          tool,
   738→        });
   739→        break;
   740→      }
   741→    }
   742→
   743→    return { steps, sites };
   744→  }
   745→
   746→  /** Validate the model's PROPOSED batch for one step (advisory — warn only:
   747→   * the compiled translation wins, law #5). */
   748→  validateProposed(
   749→    planStep: number,
   750→    tool: string,
   751→    proposed: readonly BatchStep[],
   752→    translated: readonly BatchStep[],
   753→  ): void {
   754→    // Own-property allowlist scan — a "__proto__" id must not resolve through
   755→    // the prototype chain (SEAMS: "never invent a command id").
   756→    for (const [index, step] of proposed.entries()) {
   757→      if (!registryHas(this.depsRegistry, step.command)) {
   758→        this.warn(
   759→          "unknown_command",
   760→          `proposed command \`${clip(step.command, 64)}\` is not in the agent registry`,
   761→          {
   762→            step: planStep,
   763→            tool,
   764→            command: step.command,
   765→            batchStep: index + 1,
   766→          },
   767→        );
   768→      }
   769→    }
   770→    // Required params for the KNOWN proposed commands (reuse the contracts
   771→    // helper; unknown-command duplicates were handled above).
   772→    for (const issue of checkBatchCommands({ steps: [...proposed] }, this.depsRegistry)) {
   773→      if (issue.reason !== "missing_required_param") continue;
   774→      this.warn(
   775→        "missing_required_param",
   776→        `proposed \`${issue.command}\` is missing required param \`${issue.param}\``,
   777→        {
   778→          step: planStep,
   779→          tool,
   780→          command: issue.command,
   781→          batchStep: issue.step,
   782→          param: issue.param,
   783→        },
   784→      );
   785→    }
   786→    // Backwards-only $N.key refs (§8.5) within the proposed batch's own
   787→    // 1-based numbering.
   788→    for (const issue of validateBatchRefs(proposed)) {
   789→      this.warn(
   790→        "unresolvable_ref",
   791→        `proposed batch has $-reference \`${issue.ref.raw}\` (${issue.reason})`,
   792→        {
   793→          step: planStep,
   794→          tool,
   795→          batchStep: issue.step,
   796→        },
   797→      );
   798→    }
   799→    // Disagreement: the compiled translation wins — warn, never merge.
   800→    const agrees =
   801→      proposed.length === translated.length &&
   802→      proposed.every(
   803→        (step, index) =>
   804→          step.command === translated[index]?.command &&
   805→          stableStringify(step.params) === stableStringify(translated[index]?.params),
   806→      );
   807→    if (!agrees) {
   808→      this.warn(
   809→        "batch_mismatch",
   810→        "the model's proposed batch disagrees with the compiled translation; the translation stands",
   811→        {
   812→          step: planStep,
   813→          tool,
   814→        },
   815→      );
   816→    }
   817→  }
   818→
   819→  /** Merge one plan step's translation into the single atomic batch. */
   820→  merge(planStep: number, translation: StepTranslation, toolRisk: PlanRisk | undefined): void {
   821→    const base = this.compiledSteps.length;
   822→    for (const step of translation.steps) this.compiledSteps.push(step);
   823→    for (const site of translation.sites) {
   824→      this.sites.push({ ...site, planStep, globalStep: base + site.localStep });
   825→    }
   826→    this.toolRisks.push(toolRisk ?? "low");
   827→  }
   828→
   829→  steps(): readonly BatchStep[] {
   830→    return this.compiledSteps;
   831→  }
   832→
   833→  layerSites(): ReadonlyArray<LocalLayerSite & { planStep: number; globalStep: number }> {
   834→    return this.sites;
   835→  }
   836→
   837→  toolRiskFloor(): PlanRisk {
   838→    return this.toolRisks.reduce<PlanRisk>((floor, risk) => riskMax(floor, risk), "low");
   839→  }
   840→
   841→  annotation(): SaveVersionAnnotation | undefined {
   842→    return this.saveVersion;
   843→  }

   850→/** The M5-A seam may key tools by bare id (`create_text`) or with the
   851→ * `studio.` prefix — and the plan may use either form too. Deterministic
   852→ * candidate order, own-property lookup only. */
   853→function lookupWriteTool(deps: CompileDeps, tool: string): NormalizedTool | undefined {
   854→  const bare = tool.startsWith(STUDIO_TOOL_PREFIX) ? tool.slice(STUDIO_TOOL_PREFIX.length) : tool;
   855→  for (const candidate of [bare, `${STUDIO_TOOL_PREFIX}${bare}`, tool]) {
   856→    if (!Object.prototype.hasOwnProperty.call(deps.writeTools, candidate)) continue;
   857→    const entry = deps.writeTools[candidate];
   858→    if (
   859→      entry !== null &&
   860→      typeof entry === "object" &&
   861→      "safeParse" in entry &&
   862→      typeof entry.safeParse === "function"
   863→    ) {
   864→      return { bareId: bare, argsSchema: entry as ZodTypeAny };
   865→    }
   866→    if (isPlainObject(entry)) {
   867→      const def = entry as Record<string, unknown>;
   868→      const args = def.args;
   869→      const risk = def.risk;
   870→      return {
   871→        bareId: bare,
   872→        argsSchema:
   873→          args !== null &&
   874→          typeof args === "object" &&
   875→          "safeParse" in args &&
   876→          typeof (args as ZodTypeAny).safeParse === "function"
   877→            ? (args as ZodTypeAny)
   878→            : undefined,
   879→        risk: risk === "low" || risk === "medium" || risk === "high" ? risk : undefined,
   880→      };
   881→    }
   882→  }
   883→  return undefined;
   884→}

   886→/** Write `value` at a `layer`/`layers[i]`-style path inside a params object. */
   887→function applyParamPath(
   888→  params: Record<string, unknown>,
   889→  path: string,
   890→  value: string | number,
   891→): void {
   892→  const match = /^([A-Za-z0-9_]+)(?:\[(\d+)\])?$/.exec(path);
   893→  if (match === null) return;
   894→  const key = match[1];
   895→  if (key === undefined) return;
   896→  const indexText = match[2];
   897→  if (indexText === undefined) {
   898→    params[key] = value;
   899→    return;
   900→  }
   901→  const list = params[key];
   902→  if (Array.isArray(list)) {
   903→    const index = Number.parseInt(indexText, 10);
   904→    if (index >= 0 && index < list.length) list[index] = value;
   905→  }
   906→}

   908→function isCompileInput(value: unknown): value is CompilePlanInput {
   909→  return isPlainObject(value) && "plan" in value && isPlainObject(value.plan);
   910→}

   912→// ---------------------------------------------------------------------------
   913→// Main entry (SEAMS: typed, never throws)
   914→// ---------------------------------------------------------------------------
   915→
   916→function compilePlanInner(plan: unknown, deps: CompileDeps): CompilePlanResult {
   917→  const compiler = new Compiler(deps.registry);
   918→
   919→  // -- 0. pre-flight: the plan must serialize (cycle-hostile) and be sane.
   920→  let serializedPlan: string;
   921→  try {
   922→    serializedPlan = JSON.stringify(plan) ?? "null";
   923→  } catch {
   924→    compiler.fatal("plan_too_large", "the plan is not serializable JSON");
   925→    return { ok: false, issues: compiler.fatals };
   926→  }
   927→  if (serializedPlan.length > MAX_PLAN_BYTES) {
   928→    compiler.fatal(
   929→      "plan_too_large",
   930→      `the plan is too large to compile (${serializedPlan.length} bytes)`,
   931→    );
   932→    return { ok: false, issues: compiler.fatals };
   933→  }
   934→
   935→  // -- 1. defense-in-depth re-validation against the §8.6 schema (the runner
   936→  // validates first; hostile direct calls stop here).
   937→  const parsed = EDIT_PLAN_SCHEMA.safeParse(plan);
   938→  if (!parsed.success) {
   939→    const messages = parsed.error.issues.slice(0, 5).map((issue) => issue.message);
   940→    compiler.fatal(
   941→      "plan_schema",
   942→      `the plan failed its schema: ${messages.join("; ") || "invalid shape"}`,
   943→    );
   944→    return { ok: false, issues: compiler.fatals };
   945→  }
   946→  const editPlan = parsed.data;
   947→  compiler.planLayerNames = editPlan.affectedLayers
   948→    .map((layer) => (typeof layer.name === "string" && layer.name.length > 0 ? layer.name : ""))
   949→    .filter((name) => name !== "");
   950→
   951→  // -- 2. per-step: allowlist the tool, validate args, translate, validate
   952→  // the proposed batch (warns only — the translation wins, law #5).
   953→  for (const [index, planStep] of editPlan.steps.entries()) {
   954→    const position = index + 1;
   955→    if (typeof planStep.tool !== "string" || planStep.tool.length === 0) {
   956→      compiler.fatal("unknown_tool", `step ${position} has no tool`, { step: position });
   957→      continue;
   958→    }
   959→    const tool = lookupWriteTool(deps, planStep.tool);
   960→    if (tool === undefined) {
   961→      compiler.fatal(
   962→        "unknown_tool",
   963→        `tool \`${clip(planStep.tool, 64)}\` is not one of the 8 write tools`,
   964→        {
   965→          step: position,
   966→          tool: planStep.tool,
   967→        },
   968→      );
   969→      continue;
   970→    }
   971→
   972→    if (tool.argsSchema !== undefined) {
   973→      let argsOk = true;
   974→      let argsMessage = "";
   975→      try {
   976→        const safe = tool.argsSchema.safeParse(planStep.args);
   977→        if (!safe.success) {
   978→          argsOk = false;
   979→          argsMessage = safe.error.issues
   980→            .slice(0, 5)
   981→            .map((issue) => issue.message)
   982→            .join("; ");
   983→        }
   984→      } catch {
   985→        argsOk = false;
   986→        argsMessage = "args could not be validated";
   987→      }
   988→      if (!argsOk) {
   989→        compiler.fatal(
   990→          "invalid_args",
   991→          `args for \`${tool.bareId}\` failed validation: ${argsMessage}`,
   992→          {
   993→            step: position,
   994→            tool: tool.bareId,
   995→          },
   996→        );
   997→        continue;
   998→      }
   999→    }
  1000→
  1001→    const translation = compiler.translate(position, tool.bareId, planStep.args);
  1002→    compiler.validateProposed(position, tool.bareId, planStep.batch.steps, translation.steps);
  1003→    compiler.merge(position, translation, tool.risk);
  1004→  }
  1005→  if (compiler.fatals.length > 0) {
  1006→    return { ok: false, issues: compiler.fatals };
  1007→  }
  1008→
  1009→  // -- 3. bounds + registry gate on the COMPILED batch (violations are fatal).
  1010→  const steps: BatchStep[] = [...compiler.steps()];
  1011→  if (steps.length === 0) {
  1012→    compiler.fatal(
  1013→      "empty_batch",
  1014→      "the plan compiles to an empty command batch (only annotation steps)",
  1015→    );
  1016→    return { ok: false, issues: compiler.fatals };
  1017→  }
  1018→  if (steps.length > MAX_BATCH_STEPS) {
  1019→    compiler.fatal(
  1020→      "too_many_commands",
  1021→      `the compiled batch has ${steps.length} commands (max ${MAX_BATCH_STEPS})`,
  1022→    );
  1023→    return { ok: false, issues: compiler.fatals };
  1024→  }
  1025→  for (const issue of checkBatchCommands({ steps }, deps.registry)) {
  1026→    if (issue.reason === "unknown_command") {
  1027→      compiler.fatal(
  1028→        "unknown_command",
  1029→        `compiled command \`${issue.command}\` is not in the agent registry`,
  1030→        {
  1031→          batchStep: issue.step,
  1032→          command: issue.command,
  1033→        },
  1034→      );
  1035→    } else {
  1036→      compiler.fatal(
  1037→        "missing_required_param",
  1038→        `\`${issue.command}\` is missing required param \`${issue.param}\``,
  1039→        {
  1040→          batchStep: issue.step,
  1041→          command: issue.command,
  1042→          param: issue.param,
  1043→        },
  1044→      );
  1045→    }
  1046→  }
  1047→  // The compiled batch is name-based by construction — refs are validated as
  1048→  // defense-in-depth anyway (§8.5 backwards-only $N.key).
  1049→  for (const issue of validateBatchRefs(steps)) {
  1050→    compiler.fatal(
  1051→      "unresolvable_ref",
  1052→      `compiled batch has $-reference \`${issue.ref.raw}\` (${issue.reason})`,
  1053→      {
  1054→        batchStep: issue.step,
  1055→      },
  1056→    );
  1057→  }
  1058→  if (compiler.fatals.length > 0) {
  1059→    return { ok: false, issues: compiler.fatals };
  1060→  }
  1061→
  1062→  // -- 4. layer-name resolution: requests + placeholders (names stay names in
  1063→  // the batch; the client substitutes ids via the bridge — §8.7).
  1064→  const declaredNames: string[] = [];
  1065→  const disambiguation = new Map<string, string[]>();
  1066→  for (const layer of editPlan.affectedLayers) {
  1067→    if (typeof layer.name !== "string" || layer.name.length === 0) continue;
  1068→    if (!declaredNames.includes(layer.name)) declaredNames.push(layer.name);
  1069→    const notes = disambiguation.get(layer.name) ?? [];
  1070→    if (!notes.includes(layer.change)) notes.push(layer.change);
  1071→    disambiguation.set(layer.name, notes);
  1072→  }
  1073→  const uniqueNames = [...declaredNames];
  1074→  for (const site of compiler.layerSites()) {
  1075→    if (!uniqueNames.includes(site.layerName)) uniqueNames.push(site.layerName);
  1076→  }
  1077→  if (uniqueNames.length > MAX_PLAN_AFFECTED_LAYERS) {
  1078→    compiler.fatal(
  1079→      "too_many_layer_names",
  1080→      `the plan references ${uniqueNames.length} distinct layer names (max ${MAX_PLAN_AFFECTED_LAYERS})`,
  1081→    );
  1082→    return { ok: false, issues: compiler.fatals };
  1083→  }
  1084→
  1085→  const warnedUndeclared = new Set<string>();
  1086→  for (const site of compiler.layerSites()) {
  1087→    if (declaredNames.includes(site.layerName) || warnedUndeclared.has(site.layerName)) continue;
  1088→    warnedUndeclared.add(site.layerName);
  1089→    compiler.warn(
  1090→      "undeclared_layer",
  1091→      `layer \`${clip(site.layerName, 64)}\` is used by a step but not declared in the plan`,
  1092→      {
  1093→        step: site.planStep,
  1094→      },
  1095→    );
  1096→  }
  1097→
  1098→  const resolved = new Map<string, string | number>();
  1099→  if (deps.layerResolver !== undefined) {
  1100→    for (const name of uniqueNames) {
  1101→      try {
  1102→        const id = deps.layerResolver(name);
  1103→        if (id !== undefined) resolved.set(name, id);
  1104→      } catch {
  1105→        compiler.warn(
  1106→          "resolver_error",
  1107→          "the layer resolver failed; the layer name stays in the batch",
  1108→        );
  1109→      }
  1110→    }
  1111→  }
  1112→
  1113→  const requests: LayerResolutionRequest[] = [];
  1114→  const requestIdByName = new Map<string, string>();
  1115→  let requestIndex = 0;
  1116→  for (const name of uniqueNames) {
  1117→    if (resolved.has(name)) continue; // substituted server-side — no request
  1118→    requestIndex += 1;
  1119→    const requestId = `lr-${requestIndex}`;
  1120→    requestIdByName.set(name, requestId);
  1121→    requests.push({ requestId, layerName: name, disambiguation: disambiguation.get(name) ?? [] });
  1122→  }
  1123→
  1124→  const placeholders: BatchPlaceholder[] = [];
  1125→  for (const site of compiler.layerSites()) {
  1126→    const resolvedId = resolved.get(site.layerName);
  1127→    if (resolvedId !== undefined) {
  1128→      // Substitute the authoritative id (resolver-provided, never fabricated).
  1129→      const step = steps[site.globalStep];
  1130→      if (step !== undefined) applyParamPath(step.params, site.param, resolvedId);
  1131→      placeholders.push({
  1132→        step: site.globalStep + 1,
  1133→        param: site.param,
  1134→        layerName: site.layerName,
  1135→        resolvedId,
  1136→      });
  1137→      continue;
  1138→    }
  1139→    placeholders.push({
  1140→      step: site.globalStep + 1,
  1141→      param: site.param,
  1142→      layerName: site.layerName,
  1143→      requestId: requestIdByName.get(site.layerName),
  1144→    });
  1145→  }
  1146→
  1147→  // -- 5. assemble the ONE atomic batch (§8.4/§8.5: "Agent: <summary>").
  1148→  const label = `${AGENT_LABEL_PREFIX}${clip(editPlan.summary, MAX_BATCH_LABEL_LENGTH - AGENT_LABEL_PREFIX.length)}`;
  1149→  const batch: AgentBatch = { label, atomic: true, steps };
  1150→
  1151→  const serializedBatch = JSON.stringify(batch);
  1152→  if (serializedBatch !== undefined && serializedBatch.length > MAX_BATCH_BYTES) {
  1153→    compiler.fatal(
  1154→      "batch_too_large",
  1155→      `the compiled batch exceeds the ${MAX_BATCH_BYTES}-byte limit (${serializedBatch.length} bytes)`,
  1156→    );
  1157→    return { ok: false, issues: compiler.fatals };
  1158→  }
  1159→
  1160→  // -- 6. risk + credits recomputed server-side (never the model's numbers).
  1161→  // G-4: the recomputed risk is a floor-safeguarded value -- the model's
  1162→  // declared risk is never LOWERED (the card must never understate risk),
  1163→  // but the command-based floors still raise it.
  1164→  const risk = riskMax(recomputePlanRisk(steps, compiler.toolRiskFloor()), editPlan.risk);
  1165→  const estimatedCredits = estimateCredits({ risk, steps: editPlan.steps });
  1166→
  1167→  // -- 7. dedupe advisory issues (unknown/param warns can double-report).
  1168→  const seen = new Set<string>();
  1169→  const warnings = compiler.warnings.filter((issue) => {
  1170→    const key = `${issue.code}|${issue.step ?? ""}|${issue.batchStep ?? ""}|${issue.command ?? ""}|${issue.param ?? ""}`;
  1171→    if (seen.has(key)) return false;
  1172→    seen.add(key);
  1173→    return true;
  1174→  });
  1175→
  1176→  const base: CompilePlanResult = {
  1177→    ok: true,
  1178→    batch,
  1179→    risk,
  1180→    estimatedCredits,
  1181→    resolutionRequests: requests,
  1182→    placeholders,
  1183→    warnings,
  1184→  };
  1185→  const annotation = compiler.annotation();
  1186→  return annotation !== undefined ? { ...base, saveVersion: annotation } : base;
  1187→}
  1188→
  1189→/**
  1190→ * Compile an EditPlan into ONE atomic AgentBatch (§8.5 bounds enforced), with
  1191→ * layer-name resolution requests, recomputed risk + credits, and typed issues.
  1192→ * Accepts a bare EditPlan or a SEAMS CompilePlanInput. NEVER throws (law #1).
  1193→ */
  1194→export function compilePlan(
  1195→  planOrInput: EditPlan | CompilePlanInput,
  1196→  deps: CompileDeps,
  1197→): CompilePlanResult {
  1198→  try {
  1199→    const plan = isCompileInput(planOrInput) ? planOrInput.plan : planOrInput;
  1200→    return compilePlanInner(plan, deps);
  1201→  } catch {
  1202→    // Absolute backstop: no stack traces, no details (law #1).
  1203→    return {
  1204→      ok: false,
  1205→      issues: [
  1206→        { severity: "fatal", code: "internal", message: "plan compilation failed unexpectedly" },
  1207→      ],
  1208→    };
  1209→  }
  1210→}
  1211→
  1212→/**
  1213→ * Live deps from the M5-A/M5-J contract surfaces, read at CALL time (ESM live
  1214→ * bindings) so peer stubs landing mid-wave are picked up. The registry is the
  1215→ * REAL curated command allowlist (M5-J); the write tools are the REAL M5-A
  1216→ * catalog projected into the WriteToolDef shape (F-S2): each entry carries its
  1217→ * args schema AND its risk badge, so lookupWriteTool's per-tool risk floor
  1218→ * works on the live wire path (defaultCompileDeps previously passed the bare
  1219→ * WRITE_TOOL_ARGS_SCHEMAS and the catalog badges never reached the recomputed
  1220→ * risk). Built once — the catalog is frozen.
  1221→ */
  1222→const CATALOG_WRITE_TOOLS: WriteToolsDep = Object.fromEntries(
  1223→  WRITE_TOOLS.map((tool) => [tool.id, { args: tool.inputSchema, risk: tool.risk }]),
  1224→);
  1225→
  1226→export function defaultCompileDeps(): CompileDeps {
  1227→  return { registry: AGENT_COMMAND_REGISTRY, writeTools: CATALOG_WRITE_TOOLS };
  1228→}
