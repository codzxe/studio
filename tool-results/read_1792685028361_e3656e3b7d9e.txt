   186→   186→function defaultSleep(ms: number): Promise<void> {
   187→   187→  if (ms <= 0) return Promise.resolve();
   188→   188→  return new Promise((resolve) => {
   189→   189→    setTimeout(resolve, ms);
   190→   190→  });
   191→   191→}

   193→   193→/** Serialized query string; `undefined`/`null` values are skipped. */
   194→   194→export function buildQuery(
   195→   195→  params: Record<string, string | number | boolean | null | undefined>,
   196→   196→): string {
   197→   197→  const search = new URLSearchParams();
   198→   198→  for (const [key, value] of Object.entries(params)) {
   199→   199→    if (value === undefined || value === null) continue;
   200→   200→    search.set(key, String(value));
   201→   201→  }
   202→   202→  const text = search.toString();
   203→   203→  return text === "" ? "" : `?${text}`;
   204→   204→}

   221→   221→interface ClientInternals {

   225→   225→  readonly newKey: IdempotencyKeyFactory;

   233→   233→
   234→   234→class ApiClientImpl {
   235→   235→  constructor(readonly core: ClientInternals) {}
   236→   236→
   237→   237→  /**
   238→   238→   * One HTTP call: envelope unwrap, response validation, GET retry with
   239→   239→   * backoff. Never surfaces raw response text — errors are typed ApiErrors.
   240→   240→   */
   241→   241→  async request<T>(plan: RequestPlan<T>): Promise<T> {
   242→   242→    const url = `${this.core.baseUrl}${plan.path}${plan.query ? buildQuery(plan.query) : ""}`;
   243→   243→
   244→   244→    let attempt = 0;
   245→   245→    for (;;) {
   246→   246→      let status = 0;
   247→   247→      let statusText = "";
   248→   248→      let text = "";
   249→   249→      try {
   250→   250→        const init = this.buildInit(plan);
   251→   251→        const response = await this.core.fetchImpl(url, init);
   252→   252→        status = response.status;
   253→   253→        statusText = response.statusText;
   254→   254→        text = await response.text();
   255→   255→      } catch (error) {
   256→   256→        if (isAbortError(error) || plan.signal?.aborted) throw error;
   257→   257→        const netError = networkError(error);
   258→   258→        if (!this.shouldRetry(plan, netError, attempt)) throw netError;
   259→   259→        await this.backoff(plan, attempt);
   260→   260→        attempt += 1;
   261→   261→        continue;
   262→   262→      }
   263→   263→
   264→   264→      if (status < 200 || status > 299) {
   265→   265→        const apiError = await fromResponse({
   266→   266→          status,
   267→   267→          statusText,
   268→   268→          text: async () => text,
   269→   269→        });
   270→   270→        if (!this.shouldRetry(plan, apiError, attempt)) throw apiError;
   271→   271→        await this.backoff(plan, attempt);
   272→   272→        attempt += 1;
   273→   273→        continue;
   274→   274→      }
   275→   275→
   276→   276→      return this.unwrapOk<T>(status, text, plan);
   277→   277→    }
   278→   278→  }
   279→   279→
   280→   280→  private buildInit(plan: RequestPlan<unknown>): RequestInit {
   281→   281→    const headers = new Headers(this.core.extraHeaders);
   282→   282→    headers.set("Accept", "application/json");
   283→   283→    if (plan.body !== undefined) headers.set("Content-Type", "application/json");
   284→   284→    if (plan.idempotencyKey !== undefined) headers.set("Idempotency-Key", plan.idempotencyKey);
   285→   285→    const init: RequestInit = {
   286→   286→      method: plan.method,
   287→   287→      headers,
   288→   288→      signal: plan.signal,
   289→   289→    };
   290→   290→    if (plan.body !== undefined) init.body = JSON.stringify(plan.body);
   291→   291→    return init;
   292→   292→  }
   293→   293→
   294→   294→  /** Only idempotent-safe GETs retry, and only when the failure is retryable. */
   295→   295→  private shouldRetry(plan: RequestPlan<unknown>, error: ApiError, attempt: number): boolean {
   296→   296→    if (plan.method !== "GET") return false;
   297→   297→    if (attempt >= this.core.retryDelays.length) return false;
   298→   298→    if (error.status === 0) return true; // network error before any response
   299→   299→    if (error.status === 408 || error.status === 429 || error.status >= 500) {
   300→   300→      return error.retryable;
   301→   301→    }
   302→   302→    return false;
   303→   303→  }
   304→   304→
   305→   305→  private backoff(plan: RequestPlan<unknown>, attempt: number): Promise<void> {
   306→   306→    const delay = this.core.retryDelays[attempt] ?? 0;
   307→   307→    return sleepWithAbort(this.core.sleep, delay, plan.signal);
   308→   308→  }
   309→   309→
   310→   310→  /** 2xx handling: empty body → `null`; otherwise a §7.3 Ok envelope is required. */
   311→   311→  private unwrapOk<T>(status: number, text: string, plan: RequestPlan<T>): T {
   312→   312→    if (text === "") {
   313→   313→      if (plan.responseSchema === undefined) return null as T;
   314→   314→      throw malformedResponseError(status);
   315→   315→    }
   316→   316→    let json: unknown;
   317→   317→    try {
   318→   318→      json = JSON.parse(text);
   319→   319→    } catch {
   320→   320→      throw malformedResponseError(status);
   321→   321→    }
   322→   322→    if (isErr(json)) {
   323→   323→      const error: Err["error"] = json.error;
   324→   324→      // A 2xx status with an Err body is still the §7.3 failure arm.
   325→   325→      throw new ApiError({
   326→   326→        code: error.code,
   327→   327→        message: error.message,
   328→   328→        hint: error.hint,
   329→   329→        retryable: error.retryable,
   330→   330→        requestId: error.requestId,
   331→   331→        status,
   332→   332→        cause: json,
   333→   333→      });
   334→   334→    }
   335→   335→    if (!isOk(json)) throw malformedResponseError(status);
   336→   336→    const data = json.data;
   337→   337→    if (plan.responseSchema === undefined) return data as T;
   338→   338→    try {
   339→   339→      return plan.responseSchema.parse(data);
   340→   340→    } catch (error) {
   341→   341→      throw malformedResponseError(status, error);
   342→   342→    }
   343→   343→  }
   344→   344→
   345→   345→  /** Client-side §7.2 request validation — fails fast, never sends. */
   346→   346→  validateBody<T>(schema: z.ZodType<T>, body: unknown, path: string): T {
   347→   347→    try {
   348→   348→      return schema.parse(body);
   349→   349→    } catch (error) {
   350→   350→      if (isApiError(error)) throw error;
   351→   351→      throw invalidRequestError(
   352→   352→        Object.assign(new Error(`invalid body for ${path}`), { cause: error }),
   353→   353→      );
   354→   354→    }
   355→   355→  }
   356→   356→
   357→   357→  autoKey(): string {
   358→   358→    return this.core.newKey();
   359→   359→  }
   360→   360→
   361→   361→  // -- GETs don't send Idempotency-Key; mutations always do (auto when absent).
   362→   362→  async get<T>(
   363→   363→    path: string,
   364→   364→    query: Record<string, string | number | boolean | undefined> | undefined,
   365→   365→    schema: z.ZodType<T>,
   366→   366→    signal?: AbortSignal,
   367→   367→  ): Promise<T> {
   368→   368→    return this.request<T>({ method: "GET", path, query, responseSchema: schema, signal });
   369→   369→  }
   370→   370→
   371→   371→  async mutate<T>(
   372→   372→    method: "POST" | "PATCH" | "DELETE",
   373→   373→    path: string,
   374→   374→    plan: {
   375→   375→      body?: unknown;
   376→   376→      bodySchema?: z.ZodType<unknown>;
   377→   377→      responseSchema?: z.ZodType<T>;
   378→   378→      idempotencyKey?: string;
   379→   379→      signal?: AbortSignal;
   380→   380→    },
   381→   381→  ): Promise<T> {
   382→   382→    const body =
   383→   383→      plan.bodySchema !== undefined
   384→   384→        ? this.validateBody(plan.bodySchema, plan.body, path)
   385→   385→        : plan.body;
   386→   386→    return this.request<T>({
   387→   387→      method,
   388→   388→      path,
   389→   389→      body,
   390→   390→      responseSchema: plan.responseSchema,
   391→   391→      idempotencyKey: plan.idempotencyKey ?? this.autoKey(),
   392→   392→      signal: plan.signal,
   393→   393→    });
   394→   394→  }
   395→   395→}

   841→   841→/** Two-phase version upload (§9.2/§9.3): descriptor POST + signed-URL byte PUT. */
   842→   842→async function uploadVersion(
   843→   843→  impl: ApiClientImpl,
   844→   844→  projectId: string,
   845→   845→  input: {
   846→   846→    readonly baseVersionId: string;
   847→   847→    readonly sha256: string;
   848→   848→    readonly label?: string;
   849→   849→    readonly agentRunId?: string;
   850→   850→    readonly bytes: Blob | Uint8Array;
   851→   851→  },
   852→   852→  options: CallOptions | undefined,
   853→   853→): Promise<ProjectVersion> {
   854→   854→  const { bytes } = input;
   855→   855→  if (!(bytes instanceof Blob) && !(bytes instanceof Uint8Array)) {
   856→   856→    throw invalidRequestError(new Error("bytes must be a Blob or Uint8Array"));
   857→   857→  }
   858→   858→  const sizeBytes = bytes instanceof Blob ? bytes.size : bytes.byteLength;
   859→   859→  const descriptor = impl.validateBody(
   860→   860→    UPLOAD_VERSION_SCHEMA,
   861→   861→    {
   862→   862→      baseVersionId: input.baseVersionId,
   863→   863→      sha256: input.sha256,
   864→   864→      label: input.label,
   865→   865→      agentRunId: input.agentRunId,
   866→   866→      sizeBytes,
   867→   867→    },
   868→   868→    "upload version descriptor",
   869→   869→  );
   870→   870→
   871→   871→  const response = await impl.mutate<UploadVersionResponse>(
   872→   872→    "POST",
   873→   873→    `/projects/${encodeId(projectId)}/versions`,
   874→   874→    {
   875→   875→      body: descriptor,
   876→   876→      responseSchema: UPLOAD_VERSION_RESPONSE_SCHEMA,
   877→   877→      idempotencyKey: options?.idempotencyKey ?? impl.autoKey(),
   878→   878→      signal: options?.signal,
   879→   879→    },
   880→   880→  );
   881→   881→
   882→   882→  // Phase two: bytes go straight to object storage via the signed URL — the
   883→   883→  // storage response is NOT a §7.3 envelope, so only the status is checked.
   884→   884→  const { url, headers } = response.upload;
   885→   885→  const put = await impl.core.fetchImpl(url, {
   886→   886→    method: "PUT",
   887→   887→    headers: { ...headers },
   888→   888→    // TS 5.7's DOM BodyInit doesn't accept Uint8Array<ArrayBufferLike>
   889→   889→    // (SharedArrayBuffer edge); runtime fetch (browser + undici) does.
   890→   890→    body: bytes instanceof Blob ? bytes : (bytes as unknown as BodyInit),
   891→   891→    signal: options?.signal,
   892→   892→  });
   893→   893→  if (!put.ok) {
   894→   894→    throw uploadFailedError(put.status);
   895→   895→  }
   896→   896→  return response.version;
   897→   897→}
