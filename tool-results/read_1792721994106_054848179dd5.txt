     7→function header(call: RecordedCall, name: string): string | null {
     8→  if (call.init?.headers === undefined) return null;
     9→  if (call.init.headers instanceof Headers) return call.init.headers.get(name);
    10→  const record = call.init.headers as Record<string, string>;
    11→  return record[name] ?? null;
    12→}

    16→// A §7.2-valid `GET /me` payload — /me responses are schema-validated, so
    17→// tests that await `me.get()` must feed it a real MeResponse (the
    18→// MALFORMED_RESPONSE tests pin the validation itself).
    19→const ME = {
    20→  user: { id: "usr_1", name: null, email: "maya@example.com" },
    21→  memberships: [],
    22→  plan: { plan: "pro", creditsBalance: 3200, includedCredits: 2500, renewsAt: null },

    24→
    25→function makeClient(handler: (input: string, init?: RequestInit) => Promise<Response>) {
    26→  const calls: RecordedCall[] = [];
    27→  const fetch = async (input: string, init?: RequestInit) => {
    28→    calls.push({ input, init });
    29→    return handler(input, init);
    30→  };
    31→  return { calls, fetch, client: createApiClient({ baseUrl: BASE, fetch }) };
    32→}

   467→describe("createApiClient — surface map", () => {
   468→  it("hits the documented §7.2 paths for every namespace", async () => {
   469→    const { client, calls } = makeClient(() => Promise.resolve(jsonOk(null)));
   470→    // Rejections here are schema-validation noise (Ok<null>); paths are what
   471→    // matters, and every call is recorded before its response is unwrapped.
   472→    const ignore = (promise: Promise<unknown>): Promise<void> =>
   473→      promise.then(
   474→        () => undefined,
   475→        () => undefined,
   476→      );
   477→    const recorder = client;
   478→    // A Response body is single-use: every paginated GET needs a FRESH page
   479→    // response (a reused one rejects on the second read as a network error,
   480→    // and the GET retry backoff — real sleeps, no injected seam here — burns
   481→    // seconds per call). okFetch records into the same call log so the
   482→    // paginated paths appear in await order.
   483→    const okFetch = async (input: string, init?: RequestInit): Promise<Response> => {
   484→      calls.push({ input, init });
   485→      return jsonOk({ items: [], nextCursor: null });
   486→    };
   487→    const paginated = createApiClient({ baseUrl: BASE, fetch: okFetch });
   488→
   489→    await ignore(recorder.me.get());
   490→    await ignore(paginated.workspaces.list());
   491→    await ignore(recorder.workspaces.patch("w1", { name: "New" }));
   492→    await ignore(recorder.invites.create("w1", { email: "a@b.example", role: "editor" }));
   493→    await ignore(recorder.invites.accept("tok_123"));
   494→    await ignore(paginated.projects.list());
   495→    await ignore(recorder.projects.get("p1"));
   496→    await ignore(recorder.projects.patch("p1", { description: "d" }));
   497→    await ignore(recorder.projects.delete("p1"));
   498→    await ignore(recorder.projects.manifest("p1"));
   499→    await ignore(paginated.versions.list("p1"));
   500→    await ignore(recorder.versions.restore("p1", "ver_1", { confirmLabel: "v1" }));
   501→    await ignore(recorder.versions.label("p1", "ver_1", { label: "shipped cut" }));
   502→    await ignore(recorder.versions.diff("p1", "ver_1"));
   503→    await ignore(recorder.leases.acquire("p1", { sessionId: "sess_12345678" }));
   504→    await ignore(recorder.leases.release("p1"));
   505→    await ignore(paginated.assets.list());
   506→    await ignore(
   507→      recorder.assets.createDescriptor({
   508→        name: "a.png",
   509→        mime: "image/png",
   510→        sizeBytes: 10,
   511→        sha256: "c".repeat(64),
   512→        width: 10,
   513→        height: 10,
   514→      }),
