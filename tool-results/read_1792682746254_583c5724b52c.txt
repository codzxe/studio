    17→// OPFS file names can't hold "/": keys are percent-encoded.
    18→const enc = (k) => encodeURIComponent(k);
    19→const dec = (n) => decodeURIComponent(n);

    21→async function opfs() {
    22→  if (!navigator.storage || !navigator.storage.getDirectory) throw new Error("no OPFS");
    23→  const root = await navigator.storage.getDirectory();
    24→  const dir = await root.getDirectoryHandle(DIR, { create: true });
    25→  // Probe a write: some browsers expose OPFS but only write it from workers.
    26→  const probe = await dir.getFileHandle(".probe", { create: true });
    27→  if (!probe.createWritable) throw new Error("OPFS is read-only on this thread");
    28→  const w = await probe.createWritable();
    29→  await w.write(new Uint8Array([1]));
    30→  await w.close();
    31→  await dir.removeEntry(".probe");
    32→  return {
    33→    name: "opfs",
    34→    async loadAll() {
    35→      const out = [];
    36→      for await (const [name, h] of dir.entries()) {
    37→        if (h.kind !== "file" || name.startsWith(".") || name.endsWith(".crswap")) continue;
    38→        const f = await h.getFile();
    39→        out.push({ key: dec(name), data: new Uint8Array(await f.arrayBuffer()), modified: f.lastModified });
    40→      }
    41→      return out;
    42→    },
    43→    async put(key, data) {
    44→      // createWritable writes a swap file and replaces the entry on close: atomic.
    45→      const h = await dir.getFileHandle(enc(key), { create: true });
    46→      const w = await h.createWritable();
    47→      await w.write(data);
    48→      await w.close();
    49→    },
    50→    async del(key) {
    51→      try { await dir.removeEntry(enc(key)); } catch (e) { if (e.name !== "NotFoundError") throw e; }
    52→    },
    53→  };
    54→}
    55→
    56→function req(r) {
    57→  return new Promise((res, rej) => { r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
    58→}
    59→
    60→// Run `op` on `store` in a readwrite transaction of its own; resolves once the transaction has
    61→// committed. A request's success doesn't mean the write is stored: the transaction can still
    62→// abort afterwards (quota at commit, the browser closing), rolling it back, and a write
    63→// reported as saved then is never retried (#1242). A failed request aborts the transaction
    64→// with its error; an explicit abort leaves no error, hence the AbortError.
    65→function write(db, store, op) {
    66→  return new Promise((res, rej) => {
    67→    let tx = null;
    68→    try {
    69→      tx = db.transaction(store, "readwrite");
    70→      tx.oncomplete = () => res();
    71→      tx.onabort = () => rej(tx.error ?? new DOMException("the storage transaction was aborted", "AbortError"));
    72→      op(tx.objectStore(store));
    73→    } catch (e) {
    74→      try { tx?.abort(); } catch {}
    75→      rej(e);
    76→    }
    77→  });
    78→}

    80→async function idb() {
    81→  if (!self.indexedDB) throw new Error("no IndexedDB");
    82→  const open = indexedDB.open(DB, 1);
    83→  open.onupgradeneeded = () => open.result.createObjectStore(STORE, { keyPath: "key" });
    84→  const db = await req(open);
    85→  const tx = (mode) => db.transaction(STORE, mode).objectStore(STORE);
    86→  return {
    87→    name: "indexeddb",
    88→    async loadAll() {
    89→      const all = await req(tx("readonly").getAll());
    90→      return all.map((r) => ({ key: r.key, data: new Uint8Array(r.data), modified: r.modified }));
    91→    },
    92→    async put(key, data, modified) {
    93→      await write(db, STORE, (s) => s.put({ key, data: data.slice().buffer, modified }));
    94→    },
    95→    async del(key) {
    96→      await write(db, STORE, (s) => s.delete(key));
    97→    },
    98→  };
    99→}

   101→function memory() {
   102→  const m = new Map();
   103→  return {
   104→    name: "memory",
   105→    async loadAll() { return [...m.values()]; },
   106→    async put(key, data, modified) { m.set(key, { key, data: data.slice(), modified }); },
   107→    async del(key) { m.delete(key); },
   108→  };

   111→/// Open the store: OPFS, else IndexedDB, else memory (nothing persists). Resolves with the
   112→/// backend's name. `prefer` ("opfs" | "indexeddb" | "memory") forces one (tests).
   113→export async function storeOpen(prefer) {
   114→  const order = { opfs: [opfs, idb, memory], indexeddb: [idb, memory], memory: [memory] }[prefer] ?? [opfs, idb, memory];
   115→  for (const make of order) {
   116→    try { backend = await make(); break; } catch (e) { console.info(`storage: ${make.name}: ${e.message ?? e}`); }
   117→  }
   118→  // Ask the browser not to evict the store under storage pressure (granted for installed apps).
   119→  try { if (navigator.storage && navigator.storage.persist) await navigator.storage.persist(); } catch {}
   120→  return backend.name;
   121→}

   149→// One AudioContext for the page. Browsers start it suspended until the user interacts with the
   150→// page, so the first pointer/key event resumes it ("unlock").
   151→let ctx = null;
   152→let node = null;
   153→let kind = null;
   154→// Sample frames handed to the output since `audioStart`, and the context time it started at.
   155→let posted = 0;
   156→let startTime = 0;
   157→let underruns = 0;
   158→// ScriptProcessor fallback: the queue lives on this thread.
   159→let spQueue = [];
   160→let spOffset = 0;
   161→// Blocks pushed while the output node is being created.
   162→let starting = false;

   167→function context() {
   168→  if (!ctx) {
   169→    const AC = self.AudioContext || self.webkitAudioContext;
   170→    if (!AC) return null;
   171→    ctx = new AC({ latencyHint: "interactive" });
   172→  }
   173→  return ctx;

   176→function unlock() {
   177→  const c = context();
   178→  if (c && c.state !== "running") c.resume().catch(() => {});
   179→}

   209→/// Start an output node; resolves when it is ready. Samples pushed before then queue up.
   210→export async function audioStart(workletUrl) {
   211→  audioStop();
   212→  const c = context();
   213→  if (!c) throw new Error("Web Audio is not available");
   214→  unlock();
   215→  posted = 0;
   216→  underruns = 0;
   217→  spQueue = [];
   218→  spOffset = 0;
   219→  starting = true;
   220→  pendingPush = [];
   221→  const gen = ++generation;
   222→  if (c.audioWorklet && workletUrl) {
   223→    try {
   224→      if (!c.__ecWorklet) {
   225→        await c.audioWorklet.addModule(workletUrl);
   226→        c.__ecWorklet = true;
   227→      }
   228→      if (gen !== generation) return;
   229→      node = new AudioWorkletNode(c, "effectcraft-output", { numberOfInputs: 0, outputChannelCount: [2] });
   230→      node.port.onmessage = (e) => { if (e.data && e.data.underrun) underruns += e.data.underrun; };
   231→      kind = "worklet";
   232→    } catch (e) {
   233→      console.warn("audio: AudioWorklet unavailable, using ScriptProcessorNode", e);
   234→      node = null;
   235→    }
   236→  }
   237→  if (!node) {
   238→    node = c.createScriptProcessor(2048, 0, 2);
   239→    node.onaudioprocess = (e) => {
   240→      const l = e.outputBuffer.getChannelData(0);
   241→      const r = e.outputBuffer.getChannelData(1);
   242→      let i = 0;
   243→      while (i < l.length && spQueue.length) {
   244→        const b = spQueue[0];
   245→        const n = Math.min(l.length - i, (b.length - spOffset) >> 1);
   246→        for (let k = 0; k < n; k++) { l[i + k] = b[spOffset + 2 * k]; r[i + k] = b[spOffset + 2 * k + 1]; }
   247→        i += n; spOffset += 2 * n;
   248→        if (spOffset >= b.length) { spQueue.shift(); spOffset = 0; }
   249→      }
   250→      if (i < l.length) { underruns += l.length - i; l.fill(0, i); r.fill(0, i); }
   251→    };
   252→    kind = "scriptProcessor";
   253→  }
   254→  starting = false;
   255→  if (!node) return;
   256→  node.connect(c.destination);
   257→  startTime = c.currentTime;
   258→  const p = pendingPush;
   259→  pendingPush = [];
   260→  for (const b of p) send(b);
   261→}
   262→
   263→function send(samples) {
   264→  if (kind === "worklet") node.port.postMessage(samples, [samples.buffer]);
   265→  else spQueue.push(samples);
   266→}
   267→
   268→/// Queue interleaved stereo samples (Float32Array) for output.
   269→export function audioPush(samples) {
   270→  posted += samples.length >> 1;
   271→  if (node && !starting) send(samples);
   272→  else if (starting) pendingPush.push(samples);
   273→}
   274→
   275→/// Sample frames played since `audioStart`, by the context clock (AudioContext.currentTime).
   276→export function audioPlayed() {
   277→  if (!ctx || !node || starting) return 0;
   278→  return Math.max(0, Math.round((ctx.currentTime - startTime) * ctx.sampleRate));
   279→}
   280→
   281→/// Output latency in sample frames (what's handed over but not audible yet comes on top).
   282→export function audioOutputLatency() {
   283→  if (!ctx) return 0;
   284→  return Math.round(((ctx.outputLatency || 0) + (ctx.baseLatency || 0)) * ctx.sampleRate);
   285→}

   287→export function audioStop() {
   288→  generation++;
   289→  starting = false;
   290→  pendingPush = [];
   291→  if (node) {
   292→    try { node.disconnect(); } catch {}
   293→    if (kind === "worklet") node.port.postMessage("stop");
   294→  }
   295→  node = null;
   296→  spQueue = [];
   297→}

   301→// Idle workers are reused; each keeps the files it was sent (path → version) so a file goes to a

   305→// Job workers open a WebGPU device of their own (unless `?nogpuworkers`): their jobs' frames
   306→// render on it. `jobGpu`: what the last started worker reported; `jobStats`: the last jobs'
   307→// {kind, gpu, passes, readbacks, ms}.
   308→let jobWorkerGpu = true;

   312→export function setJobWorkerGpu(on) {

   316→function spawn(base) {
   317→  const worker = new Worker(new URL("worker.js", base), { type: "module", name: "effectcraft-worker" });
   318→  const w = { worker, files: new Map(), ready: null, gpu: null };
   319→  w.ready = new Promise((res, rej) => {
   320→    worker.onmessage = (e) => {
   321→      if (e.data && e.data.type === "gpu") {
   322→        w.gpu = e.data.gpu ? { adapter: e.data.gpu } : { cpu: e.data.error || "no GPU" };
   323→        jobGpu = w.gpu;
   324→      }
   325→      if (e.data && e.data.type === "ready") res();
   326→    };
   327→    worker.onerror = (e) => rej(new Error(e.message || "worker failed to start"));
   328→  });
   329→  worker.postMessage({ type: "init", module: self.effectcraftModule ?? null, base, gpu: jobWorkerGpu });
   330→  return w;
   331→}
   332→
   333→/// Run a job in a worker. `files`: [[path, Uint8Array, version]] the job reads (`version`
   334→/// changes with every write of the file); `onMessage(type, json,
   335→/// path, bytes)` receives "reply" (json), "file" (path, bytes: a rendered file to download),
   336→/// "store" (path, bytes: a file to keep in browser storage) and "error" (json = message).
   337→export function workerRun(id, json, files, base, onMessage) {
   338→  let w;
   339→  try {
   340→    w = idle.pop() ?? spawn(base);
   341→  } catch (e) {
   342→    // `new Worker` throws at once when the page's origin is opaque (a sandboxed iframe): the job
   343→    // fails like one whose worker didn't start.
   344→    setTimeout(() => onMessage("error", String((e && e.message) || e), null, null));
   345→    return;
   346→  }
   347→  running.set(id, w);
   348→  w.ready.then(() => {
   349→    w.worker.onmessage = (e) => {
   350→      const m = e.data;
   351→      if (m.type === "reply") {
   352→        onMessage("reply", m.json, null, null);
   353→        if (m.done) {
   354→          running.delete(id);
   355→          idle.push(w);
   356→        }
   357→      } else if (m.type === "file") {
   358→        onMessage("file", null, m.path, new Uint8Array(m.bytes));
   359→      } else if (m.type === "store") {
   360→        onMessage("store", null, m.path, new Uint8Array(m.bytes));
   361→      } else if (m.type === "stats") {
   362→        jobStats.push({ kind: m.kind, gpu: m.gpu, passes: m.passes, readbacks: m.readbacks, ms: m.ms });
   363→        if (jobStats.length > 16) jobStats.shift();
   364→      }
   365→    };
   366→    w.worker.onerror = (e) => {
   367→      running.delete(id);
   368→      onMessage("error", String(e.message || "worker error"), null, null);
   369→      w.worker.terminate();
   370→    };
   371→    for (const [path, bytes, version] of files) {
   372→      const tag = version ?? bytes.length;
   373→      if (w.files.get(path) === tag) continue;
   374→      w.files.set(path, tag);
   375→      w.worker.postMessage({ type: "file", path, bytes });
   376→    }
   377→    w.worker.postMessage({ type: "job", json });
   378→  }, (e) => {
   379→    // The worker didn't start (#216): the job fails and the worker goes.
   380→    running.delete(id);
   381→    w.worker.terminate();
   382→    onMessage("error", String(e.message || e), null, null);
   383→  });
   384→}

   386→/// Stop a job: its worker is terminated (it can't be interrupted mid-job).
   387→export function workerCancel(id) {
   388→  const w = running.get(id);
   389→  if (!w) return;
   390→  running.delete(id);
   391→  w.worker.terminate();
   392→}

   401→// Long-lived workers that render viewer frames from a project replica (src/frames.rs). Unlike job

   403→const frameWorkers = [];

   405→/// Start a frame worker; returns its index. `onMessage(index, json, bytes, error)` receives each
   406→/// `frameReply` (JSON, pixels as a Uint8Array) or, if the worker dies, the error. `gpu`: the
   407→/// worker opens its own WebGPU device (when the browser has WebGPU in workers) and renders
   408→/// GPU effects on it.
   409→export function frameWorkerStart(base, onMessage, gpu) {
   410→  const index = frameWorkers.length;
   411→  let worker;
   412→  try {
   413→    worker = new Worker(new URL("worker.js", base), { type: "module", name: `effectcraft-frames-${index}` });
   414→  } catch (e) {
   415→    // The constructor throws at once when the page's origin is opaque (a sandboxed iframe). The
   416→    // slot stays, and the failure is reported like a worker that died, after the caller has
   417→    // recorded the index: frames then render on the page's thread (as with `?noworkers`).
   418→    frameWorkers.push(null);
   419→    setTimeout(() => onMessage(index, undefined, undefined, String((e && e.message) || e || "frame worker failed")));
   420→    return index;
   421→  }
   422→  worker.onmessage = (e) => {
   423→    const m = e.data;
   424→    if (m && m.type === "frameReply") onMessage(index, m.json, m.bytes ? new Uint8Array(m.bytes) : null, undefined);
   425→  };
   426→  worker.onerror = (e) => {
   427→    onMessage(index, undefined, undefined, String(e.message || "frame worker failed"));
   428→    worker.terminate();
   429→  };
   430→  worker.postMessage({ type: "init", module: self.effectcraftModule ?? null, base, frames: true, gpu: !!gpu });
   431→  frameWorkers.push(worker);
   432→  return index;
   433→}

   435→// ------------------------------------------------------------------ disk cache (OPFS)
   436→
   437→// The disk cache (src/diskcache.rs): one file per frame under
   438→// effectcraft-cache/v1/frames/<32 hex>.ecc and one per layer buffer under
   439→// effectcraft-cache/v1/layers/<32 hex>.ecc in the Origin Private File System (entry names with a
   440→// `layers/` prefix are layer buffers). Frame workers write them with synchronous access handles
   441→// (a temporary file moved into place, so a reader never sees a partial entry; every entry also
   442→// ends in a checksum) and read the layer buffers they prefetch; the page lists, reads and
   443→// deletes.
   444→const CACHE_ROOT = "effectcraft-cache";
   445→const CACHE_KINDS = ["frames", "layers", "mattes"];
   446→const cacheDirHandles = {};
   447→
   448→async function cacheDir(kind = "frames") {
   449→  if (cacheDirHandles[kind]) return cacheDirHandles[kind];
   450→  if (!navigator.storage || !navigator.storage.getDirectory) throw new Error("no Origin Private File System");
   451→  let d = await navigator.storage.getDirectory();
   452→  for (const n of [CACHE_ROOT, "v1", kind]) d = await d.getDirectoryHandle(n, { create: true });
   453→  cacheDirHandles[kind] = d;
   454→  return d;
   455→}
   456→
   457→/// The folder and file name of an entry name (`layers/<hex>.ecc` or `<hex>.ecc`).
   458→async function entryDir(name) {
   459→  if (name.startsWith("layers/")) return [await cacheDir("layers"), name.slice(7)];
   460→  if (name.startsWith("mattes/")) return [await cacheDir("mattes"), name.slice(7)];
   461→  return [await cacheDir("frames"), name];
   462→}
   463→
   464→/// Every cached entry: [{name, size, modified}] (layer buffers named `layers/…`). Temporaries
   465→/// older than ten minutes (a writer that died) are removed.
   466→export async function cacheList() {
   467→  const out = [];
   468→  for (const kind of CACHE_KINDS) {
   469→    const dir = await cacheDir(kind);
   470→    const prefix = kind === "frames" ? "" : `${kind}/`;
   471→    const stale = [];
   472→    for await (const [name, h] of dir.entries()) {
   473→      if (h.kind !== "file") continue;
   474→      const f = await h.getFile();
   475→      if (name.includes(".tmp")) {
   476→        if (Date.now() - f.lastModified > 600000) stale.push(name);
   477→        continue;
   478→      }
   479→      out.push({ name: prefix + name, size: f.size, modified: f.lastModified });
   480→    }
   481→    for (const n of stale) { try { await dir.removeEntry(n); } catch {} }
   482→  }
   483→  return out;
   484→}

   486→/// An entry's bytes.
   487→export async function cacheRead(name) {
   488→  const [dir, file] = await entryDir(name);
   489→  const f = await (await dir.getFileHandle(file)).getFile();
   490→  return new Uint8Array(await f.arrayBuffer());
   491→}

   493→/// Write an entry (in a worker: a synchronous access handle; elsewhere a writable stream);
   494→/// resolves with its size.
   495→export async function cacheWrite(fullName, bytes) {
   496→  const [dir, name] = await entryDir(fullName);
   497→  const canMove = typeof FileSystemHandle !== "undefined" && "move" in FileSystemHandle.prototype;
   498→  const tmp = canMove ? `${name}.tmp.${Math.random().toString(36).slice(2)}` : name;
   499→  const fh = await dir.getFileHandle(tmp, { create: true });
   500→  if (fh.createSyncAccessHandle) {
   501→    const h = await fh.createSyncAccessHandle();
   502→    try {
   503→      h.truncate(0);
   504→      h.write(bytes, { at: 0 });
   505→      h.flush();
   506→    } finally {
   507→      h.close();
   508→    }
   509→  } else {
   510→    const w = await fh.createWritable();
   511→    await w.write(bytes);
   512→    await w.close();
   513→  }
   514→  if (canMove) {
   515→    try { await fh.move(dir, name); } catch (e) { try { await dir.removeEntry(tmp); } catch {} throw e; }
   516→  }
   517→  return bytes.length;
   518→}

   554→// Folders opened with the File System Access API (src/browse.rs): name → {handle, files: Map of
   555→// "name/sub/file" → FileSystemFileHandle}. Their handles are remembered in IndexedDB so the
   556→// folders come back on the next visit (once the browser grants access again).
   557→const folders = new Map();
   558→const HANDLES_DB = "effectcraft-handles";
   559→const MAX_ENTRIES = 20000;
   560→
   561→async function handlesDb() {
   562→  const open = indexedDB.open(HANDLES_DB, 1);
   563→  open.onupgradeneeded = () => open.result.createObjectStore("folders", { keyPath: "name" });
   564→  return req(open);
   565→}
   566→
   567→async function enumerate(dir, prefix, out, files, depth) {
   568→  for await (const [name, h] of dir.entries()) {
   569→    if (name.startsWith(".") || out.length >= MAX_ENTRIES) continue;
   570→    const rel = `${prefix}/${name}`;
   571→    if (h.kind === "directory") {
   572→      out.push({ path: rel, dir: true, size: 0, modified: 0 });
   573→      if (depth < 8) await enumerate(h, rel, out, files, depth + 1);
   574→    } else {
   575→      const f = await h.getFile();
   576→      out.push({ path: rel, dir: false, size: f.size, modified: Math.floor(f.lastModified / 1000) });
   577→      files.set(rel, h);
   578→    }
   579→  }
   580→}
   581→
   582→async function openFolder(handle, remember) {
   583→  const name = handle.name;
   584→  const entries = [];
   585→  const files = new Map();
   586→  await enumerate(handle, name, entries, files, 0);
   587→  folders.set(name, { handle, files });
   588→  if (remember) {
   589→    try {
   590→      const db = await handlesDb();
   591→      await write(db, "folders", (s) => s.put({ name, handle }));
   592→    } catch (e) { console.info("media browser: cannot remember the folder", e); }
   593→  }
   594→  return { name, entries };
   595→}

   608→/// Folders opened on earlier visits: [{name, granted, entries?}] (entries when access is granted).
   609→export async function browseRestore() {
   610→  if (!self.indexedDB || !browseCanPickFolder()) return [];
   611→  let all = [];
   612→  try {
   613→    const db = await handlesDb();
   614→    all = await req(db.transaction("folders", "readonly").objectStore("folders").getAll());
   615→  } catch { return []; }
   616→  const out = [];
   617→  for (const { name, handle } of all) {
   618→    let granted = false;
   619→    try { granted = (await handle.queryPermission({ mode: "read" })) === "granted"; } catch {}
   620→    if (granted) {
   621→      try { out.push({ granted, ...(await openFolder(handle, false)) }); continue; } catch {}
   622→    }
   623→    out.push({ name, granted: false });
   624→  }
   625→  return out;
   626→}

   628→/// Ask for access to a remembered folder again (needs a user gesture); resolves like
   629→/// browsePickFolder.
   630→export async function browseReconnect(name) {
   631→  const db = await handlesDb();
   632→  const rec = await req(db.transaction("folders", "readonly").objectStore("folders").get(name));
   633→  if (!rec) throw new Error(`${name} is not remembered`);
   634→  if ((await rec.handle.requestPermission({ mode: "read" })) !== "granted") throw new Error(`access to ${name} was not granted`);
   635→  return openFolder(rec.handle, false);
   636→}

   638→/// Forget a remembered folder.
   639→export async function browseForget(name) {
   640→  folders.delete(name);
   641→  try {
   642→    const db = await handlesDb();
   643→    await write(db, "folders", (s) => s.delete(name));
   644→  } catch {}
   645→}

   647→/// A file of an opened folder ("/Folders/name/sub/file"): its bytes.
   648→export async function browseRead(path) {
   649→  const rel = path.replace(/^\/Folders\//, "");
   650→  const folder = folders.get(rel.split("/")[0]);
   651→  const h = folder && folder.files.get(rel);
   652→  if (!h) throw new Error(`${path}: its folder is not open (reconnect it)`);
   653→  return new Uint8Array(await (await h.getFile()).arrayBuffer());
   654→}

   656→/// The fallback where folders can't be opened: pick files (or a whole folder, `directory`) with a
   657→/// plain file input; resolves with [{path, bytes}] (path relative to the picked folder). A file
   658→/// that can't be read is listed as {path, error} instead (#226); cancelling resolves with [].
   659→export function browsePickFiles(directory) {
   660→  return new Promise((resolve) => {
   661→    const input = document.createElement("input");
   662→    input.type = "file";
   663→    input.multiple = true;
   664→    if (directory) input.webkitdirectory = true;
   665→    input.oncancel = () => resolve([]);
   666→    input.onchange = async () => {
   667→      const out = [];
   668→      for (const f of input.files) {
   669→        const path = (directory && f.webkitRelativePath) || f.name;
   670→        try {
   671→          out.push({ path, bytes: new Uint8Array(await f.arrayBuffer()) });
   672→        } catch (e) {
   673→          out.push({ path, error: String((e && (e.message || e.name)) || e) });
   674→        }
   675→      }
   676→      resolve(out);
   677→    };
   678→    input.click();
   679→  });
   680→}

   682→/// The page's WebGPU device was lost (#225): the canvas can't show anything new while the project
   683→/// stays open and editable. Say so over the stale picture, and offer to save the project (a
   684→/// download) and to reload; the reload first writes pending changes to browser storage, from
   685→/// which the session comes back when it is `kept` there.
   686→export function showDeviceLost(message, kept) {
   687→  if (document.getElementById("device-lost")) return;
   688→  const box = document.createElement("div");
   689→  box.id = "device-lost";
   690→  box.setAttribute("role", "alertdialog");
   691→  const title = document.createElement("strong");
   692→  title.textContent = "The graphics device was lost";
   693→  const text = document.createElement("p");
   694→  text.textContent =
   695→    "Studio can't update the display any more. Your project is still open. " +
   696→    (kept ? "Reload to continue: the project and its unsaved changes come back." : "This session isn't kept in browser storage: save the project before you reload.");
   697→  const detail = document.createElement("small");
   698→  detail.textContent = message;
   699→  const save = document.createElement("button");
   700→  save.textContent = "Save Project";
   701→  save.onclick = () =>
   702→    self.effectcraft
   703→      .execute("file.save", {})
   704→      .catch(() => self.effectcraft.execute("file.saveAs", { path: "/Untitled Project.ecproj" }))
   705→      .catch((e) => (text.textContent = `Saving failed: ${(e && e.message) || e}`));
   706→  const reload = document.createElement("button");
   707→  reload.textContent = "Reload";
   708→  reload.onclick = async () => {
   709→    reload.disabled = true;
   710→    try {
   711→      await self.effectcraft.flush();
   712→    } catch (e) {
   713→      text.textContent = `Writing to browser storage failed (${(e && e.message) || e}). Save the project, or reload anyway and lose the changes.`;
   714→      reload.textContent = "Reload Anyway";
   715→      reload.disabled = false;
   716→      reload.onclick = () => location.reload();
   717→      return;
   718→    }
   719→    location.reload();
   720→  };
   721→  const buttons = document.createElement("div");
   722→  buttons.append(save, reload);
   723→  box.append(title, text, detail, buttons);
   724→  document.body.append(box);
   725→}
