   264→/**
   265→ * Runtime-narrowed subscription view of an event object. StripeEventLike is
   266→ * deliberately minimal (the seam's structural contract), so the period/price
   267→ * fields are validated HERE before use (law #2: webhook input is external).
   268→ */
   269→interface SubscriptionEventView {
   270→  readonly customer: string | null;
   271→  readonly status: string;
   272→  readonly trialEnd: number | null;
   273→  readonly cancelAtPeriodEnd: boolean;
   274→  readonly currentPeriodEnd: number | null;
   275→  readonly priceId: string | null;
   276→}

   278→function priceIdOf(value: unknown): string | null {
   279→  if (typeof value === "string") return value;
   280→  if (isRecord(value)) return nonEmptyString(value.id);
   281→  return null;
   282→}
   283→
   284→function subscriptionViewOf(
   285→  object: StripeEventLike["data"]["object"],
   286→): SubscriptionEventView | null {
   287→  const record = object as unknown as Record<string, unknown>;
   288→  if (!isRecord(record)) return null;
   289→  const status = nonEmptyString(record.status);
   290→  if (status === null) return null;
   291→  let currentPeriodEnd: number | null = null;
   292→  let priceId: string | null = null;
   293→  const items = record.items;
   294→  if (isRecord(items) && Array.isArray(items.data) && items.data.length > 0) {
   295→    const item = items.data[0];
   296→    if (isRecord(item)) {
   297→      const periodEnd = item.current_period_end;
   298→      if (typeof periodEnd === "number" && Number.isFinite(periodEnd)) {
   299→        currentPeriodEnd = periodEnd;
   300→      }
   301→      priceId = priceIdOf(item.price);
   302→    }
   303→  }
   304→  const trialEnd = record.trial_end;
   305→  return {
   306→    customer: nonEmptyString(record.customer),
   307→    status,
   308→    trialEnd: typeof trialEnd === "number" && Number.isFinite(trialEnd) ? trialEnd : null,
   309→    cancelAtPeriodEnd: record.cancel_at_period_end === true,
   310→    currentPeriodEnd,
   311→    priceId,
   312→  };
   313→}

   341→/** Maps a typed seam failure ({ok:false, code, reason}) to the nearest wire

   352→        "BILLING_NOT_CONFIGURED",
   353→        "Billing is not configured in this environment.",
   354→        { status: 400, ...options },

   356→    case "not_found":
   357→      return new GatewayError("NOT_FOUND", "That billing object was not found.", {
   358→        status: 404,

   362→      return new GatewayError("VALIDATION_FAILED", "The billing provider rejected that request.", {

   364→        ...options,
   365→      });

   367→      return new GatewayError(
   368→        "SERVICE_UNAVAILABLE",

   379→export function billingRoutes(deps: GatewayDeps): Hono<AppBindings> {
   380→  const app = new Hono<AppBindings>();

   382→

   393→  function requireStripe(c: AppContext): NonNullable<GatewayDeps["stripe"]> {
   394→    if (deps.stripe === null || deps.env.stripe.secretKey === undefined) {

   396→        hint: "billing not configured in this environment",
   397→        requestId: c.get("requestId"),
   398→      });
   399→    }

   401→  }
   402→
   403→  function requireBillingRole(c: AppContext): WorkspaceContext {
   404→    const workspace = c.get("workspace");
   405→    if (!can(workspace.role, "manageBilling")) {
   406→      throw new StudioError("FORBIDDEN", { requestId: c.get("requestId") });
   407→    }
   408→    return workspace;

   410→
   411→  function appBase(): string {
   412→    return deps.env.app.url.replace(/\/+$/, "");

   414→
   415→  /**
   416→   * Resolves the workspace's Stripe customer id (the durable repo row first,
   417→   * legacy org-metadata adoption, else creates the customer). The repo row is
   418→   * the source of truth for customer → workspace billing state.
   419→   */
   420→  async function ensureWorkspaceCustomer(
   421→    c: AppContext,
   422→    workspace: WorkspaceContext,
   423→    stripe: StripeSeam,
   424→  ): Promise<string> {
   425→    if (subscriptions !== null) {
   426→      const existing = await subscriptions.findStripeCustomer(workspace.id);
   427→      if (existing !== null && existing.stripeCustomerId !== "") {
   428→        return existing.stripeCustomerId;
   429→      }
   430→    }
   431→    const legacy = parseOrgMetadata(workspace.organization.metadata).stripeCustomerId;
   432→    if (typeof legacy === "string" && legacy !== "") {
   433→      if (subscriptions !== null) {
   434→        await subscriptions.ensureStripeCustomer(workspace.id, { stripeCustomerId: legacy });
   435→      }
   436→      return legacy;
   437→    }
   438→    const customer = await stripe.createCustomer({
   439→      email: c.get("auth").user.email,
   440→      metadata: { organizationId: workspace.id },
   441→    });
   442→    if (subscriptions !== null) {
   443→      await subscriptions.ensureStripeCustomer(workspace.id, { stripeCustomerId: customer.id });
   444→    }
   445→    return customer.id;
   446→  }

   448→  /**
   449→   * Price-id lookup through the M6-A helper (typed problem vocabulary).
   450→   * StudioEnv parses STRIPE_PRICE_IDS into a record at boot; the helper
   451→   * consumes the raw JSON value, so the parsed map is re-serialized here —
   452→   * values are env-validated strings, the round-trip is exact.
   453→   */
   454→  function planPriceId(planId: string): StripePriceIdLookup {

   458→  function requirePlanPriceId(c: AppContext, planId: string): string {
   459→    const lookup = planPriceId(planId);
   460→    if (lookup.ok) return lookup.priceId;
   461→    switch (lookup.problem.kind) {
   462→      case "plan-without-price":
   463→        throw new GatewayError(
   464→          "VALIDATION_FAILED",
   465→          "That plan has no price configured in this environment.",
   466→          {
   467→            status: 400,
   468→            hint: "The STRIPE_PRICE_IDS map has no price for that plan.",
   469→            requestId: c.get("requestId"),
   470→          },
   471→        );
   472→      case "unknown-plan":
   473→      case "plan-not-purchasable":
   474→        // Unreachable behind CHECKOUT_SCHEMA (D-4: trial is never a checkout
   475→        // target) — kept fail-closed anyway.
   476→        throw new GatewayError("VALIDATION_FAILED", "That plan is not available for checkout.", {
   477→          status: 400,
   478→          requestId: c.get("requestId"),
   479→        });
   480→      default:
   481→        throw new GatewayError(
   482→          "BILLING_NOT_CONFIGURED",
   483→          "Billing is not configured in this environment.",
   484→          { status: 400, requestId: c.get("requestId"), cause: lookup.problem },
   485→        );
   486→    }
   487→  }

   489→  // ── POST /v1/billing/checkout — plan subscription / plan change ──────────
   490→
   491→  app.post(
   492→    "/v1/billing/checkout",
   493→    requireWorkspace("admin"),
   494→    async (c: AppContext): Promise<Response> => {
   495→      const workspace = requireBillingRole(c);
   496→      const stripe = requireStripe(c);
   497→      const parsed = CHECKOUT_SCHEMA.safeParse(await c.req.json().catch(() => null));
   498→      if (!parsed.success) {
   499→        throw new StudioError("VALIDATION_FAILED", { requestId: c.get("requestId") });
   500→      }
   501→      const planId = parsed.data.planId;
   502→
   503→      // Server-side entitlement pre-checks (§M6 task 1): the target plan must
   504→      // fit the CURRENT workspace shape — checked here, never trusted to the
   505→      // client (the seats gate on invites is the sibling enforcement).
   506→      const entitlements = entitlementsFor(planId);
   507→      if (workspace.members.length > entitlements.seats) {
   508→        throw new GatewayError(
   509→          "VALIDATION_FAILED",
   510→          "This workspace has more members than that plan allows.",
   511→          {
   512→            status: 400,
   513→            hint: "Remove members before switching to a smaller plan.",
   514→            requestId: c.get("requestId"),
   515→          },
   516→        );
   517→      }
   518→
   519→      const priceId = requirePlanPriceId(c, planId);
   520→      const row =
   521→        subscriptions !== null ? await subscriptions.getByOrganization(workspace.id) : null;
   522→
   523→      if (row !== null && row.status !== "canceled") {
   524→        // A live subscription: Checkout sessions always create NEW
   525→        // subscriptions, so a plan change goes through subscriptions.update.
   526→        // DOWNGRADES ARE NEVER INSTANT (§M6 task 2): a smaller plan is
   527→        // scheduled at period end from the customer portal (Stripe's native
   528→        // semantics) — current entitlements stay until the period ends.
   529→        if (planRank(planId) < planRank(row.planId)) {
   530→          throw new GatewayError(
   531→            "VALIDATION_FAILED",
   532→            "Downgrades take effect at the end of the current billing period.",
   533→            {
   534→              status: 400,
   535→              hint: "Use the billing portal to schedule a downgrade; your current features stay until the period ends.",
   536→              requestId: c.get("requestId"),
   537→            },
   538→          );

   894→  /**

   896→   * UPDATE re-checks `status IN from` — the racing/stale delivery loses with
   897→   * a typed CONFLICT, reported here as "guarded", never an error: a stale
   898→   * webhook is still a DELIVERED webhook (Stripe must see 2xx, not a retry loop).

   900→  async function guardedTransition(
   901→    organizationId: string,
   902→    input: SubscriptionTransitionInput,
   903→  ): Promise<boolean> {
   904→    if (subscriptions === null) return false;
   905→    try {
   906→      await subscriptions.transitionStatus(organizationId, input);
   907→      return true;
   908→    } catch (error) {
   909→      if (isDbError(error) && (error.code === "CONFLICT" || error.code === "NOT_FOUND")) {
   910→        return false;
   911→      }
   912→      throw error;
   913→    }
   914→  }

   916→  /** Ledger grant for a top-up from external µc metadata (D-12 + int32 ceiling). */
   917→  async function grantTopUp(
   918→    organizationId: string,
   919→    event: StripeEventLike,
   920→    metadata: Record<string, string> | null | undefined,
   921→    idempotencyKey: string,
   922→  ): Promise<boolean> {
   923→    const raw = metadata?.microcredits;
   924→    if (typeof raw !== "string" || raw === "") return false;
   925→    const microcredits = Number.parseInt(raw, 10);
   926→    if (!Number.isInteger(microcredits) || microcredits <= 0 || microcredits > MAX_LEDGER_AMOUNT) {
   927→      // Hostile/overflowing amounts are audited-ignored — never a poison retry.
   928→      return false;
   929→    }
   930→    await deps.repos.ledger.grant(organizationId, {
   931→      amount: microcredits,
   932→      idempotencyKey,
   933→      metadata: {
   934→        stripeEventId: event.id,
   935→        kind: "topup",
   936→        packId: metadata?.packId ?? null,
   937→      },
   938→    });
   939→    return true;
   940→  }

   973→  async function handleCheckoutCompleted(
   974→    stripe: StripeSeam,
   975→    event: StripeEventLike,
   976→  ): Promise<WebhookOutcome> {
   977→    const object = event.data.object;
   978→    const organizationId = organizationIdOf(object);
   979→    if (organizationId === null) return { ignored: "unattributed" };
   980→
   981→    // Customer + payment-method bookkeeping (§M6 task 5): the repo is the
   982→    // durable customer mapping; cardFingerprint metadata is the fake's card
   983→    // model (real Stripe attaches the fingerprint with the payment method).
   984→    await recordCustomer(organizationId, object);
   985→
   986→    const kind = object.metadata?.kind ?? null;
   987→    if (kind === "topup") {
   988→      const granted = await grantTopUp(
   989→        organizationId,
   990→        event,
   991→        object.metadata,
   992→        `stripe:${event.id}`,
   993→      );
   994→      return granted ? { handled: "topup.granted" } : { ignored: "topup.amount-invalid" };
   995→    }
   996→    if (kind !== "subscription") {
   997→      return { ignored: "unknown-kind" };
   998→    }
   999→
  1000→    // The session object carries no period data — resync the canonical
  1001→    // snapshot from the provider (D-4: the trial status rides on the sub).
  1002→    const customer = nonEmptyString(object.customer);
  1003→    if (customer === null) return { ignored: "no-customer" };
  1004→    const snapshot = await stripe.subscription(customer);
  1005→    if (!snapshot.ok) return { ignored: "subscription-unavailable" };
  1006→    const planId = planIdFromMetadata(object) ?? planIdFromPrice(deps.env, snapshot.data.priceId);
  1007→    if (planId === null) return { ignored: "unknown-plan" };
  1008→    if (subscriptions !== null) {
  1009→      // First row (or resubscription): upsert is race-proof on the unique
  1010→      // organization_id — concurrent completions converge on one row.
  1011→      await subscriptions.upsertState(organizationId, {
  1012→        planId,
  1013→        status: mapStripeStatus(snapshot.data.status),
  1014→        stripeSubscriptionId: snapshot.data.subscriptionId,
  1015→        currentPeriodEnd: toDate(snapshot.data.currentPeriodEnd),
  1016→        trialEnd: toDate(snapshot.data.trialEnd ?? null),
  1017→        cancelAtPeriodEnd: snapshot.data.cancelAtPeriodEnd,
  1018→      });
  1019→    }
  1020→    return { handled: "subscription.synced" };
  1021→  }

  1023→  /**
  1024→   * Shared guarded-transition driver for subscription lifecycle events
  1025→   * (customer.subscription.updated/deleted). The event's object is the
  1026→   * HISTORICAL truth (webhooks replay history); minimal objects resync from
  1027→   * the provider.
  1028→   */
  1029→  async function applySubscriptionTransition(
  1030→    stripe: StripeSeam,
  1031→    event: StripeEventLike,
  1032→    forcedTarget?: SubscriptionStatus,
  1033→  ): Promise<WebhookOutcome> {
  1034→    const object = event.data.object;
  1035→    const organizationId = organizationIdOf(object);
  1036→    if (organizationId === null) return { ignored: "unattributed" };
  1037→    if (subscriptions === null) return { ignored: "no-persistence" };
  1038→
  1039→    let view = subscriptionViewOf(object);
  1040→    if (view === null || view.currentPeriodEnd === null) {
  1041→      const customer = view?.customer ?? nonEmptyString(object.customer);
  1042→      if (customer !== null) {
  1043→        view = await resyncView(stripe, customer);
  1044→      }
  1045→    }
  1046→    if (view === null) return { ignored: "unparsable" };
  1047→    const target = forcedTarget ?? mapStripeStatus(view.status);
  1048→
  1049→    const row = await subscriptions.getByOrganization(organizationId);
  1050→    if (row === null) {
  1051→      // A subscription event for a workspace we never tracked: attribution
  1052→      // exists (metadata) but there is no state to transition — audited.
  1053→      return { ignored: "no-subscription" };
  1054→    }
  1055→
  1056→    // From-lists (the guarded state machine; heals land via invoice.paid):
  1057→    //  active   ← trialing | active | incomplete   (trial conversion, renewal)
  1058→    //  grace    ← active | grace | past_due | incomplete (past-due dunning)
  1059→    //  canceled ← any live state (BP §M6.2: status only — NO work deletion)
  1060→    //  trialing ← trialing | incomplete (idempotent re-pin)
  1061→    //  incomplete ← trialing | active | incomplete
  1062→    //  past_due ← grace (legacy rows only; Stripe past_due maps to grace)
  1063→    const FROM_LISTS: Record<SubscriptionStatus, readonly SubscriptionStatus[]> = {
  1064→      active: ["trialing", "active", "incomplete"],
  1065→      grace: ["active", "grace", "past_due", "incomplete"],
  1066→      canceled: ["trialing", "active", "grace", "past_due", "incomplete"],
  1067→      trialing: ["trialing", "incomplete"],
  1068→      incomplete: ["trialing", "active", "incomplete"],
  1069→      past_due: ["grace"],
  1070→    };
  1071→    const from = FROM_LISTS[target];
  1072→    if (!from.includes(row.status)) {
  1073→      return { ignored: "transition-guarded" };
  1074→    }
  1075→
  1076→    const periodEndMs = view.currentPeriodEnd !== null ? view.currentPeriodEnd * 1000 : null;
  1077→    // Never move the period BACKWARDS (an out-of-order renewal must not
  1078→    // regress the row), and never write a null over a known period.
  1079→    const periodAdvances =
  1080→      periodEndMs !== null &&
  1081→      (row.currentPeriodEnd === null || periodEndMs >= row.currentPeriodEnd.getTime());
  1082→
  1083→    // Plan pinning (§M6 task 2 "never instant feature loss"): the event's new
  1084→    // price only becomes the row's plan when it is an UPGRADE (features grow)
  1085→    // or the period actually advanced (the change took effect). A downgrade
  1086→    // scheduled at period end keeps the row's plan until then.
  1087→    const targetPlan = planIdFromPrice(deps.env, view.priceId) ?? planIdFromMetadata(object);
  1088→    const planId =
  1089→      targetPlan !== null &&
  1090→      target !== "canceled" &&
  1091→      (planRank(targetPlan) > planRank(row.planId) || periodAdvances)
  1092→        ? targetPlan
  1093→        : null;
  1094→
  1095→    const graceEnd =
  1096→      target === "grace"
  1097→        ? periodEndMs !== null
  1098→          ? new Date(periodEndMs + GRACE_PERIOD_DAYS * DAY_MS)
  1099→          : undefined
  1100→        : target === "active" || target === "canceled"
  1101→          ? null
  1102→          : undefined;
  1103→    const applied = await guardedTransition(organizationId, {
  1104→      from,
  1105→      to: target,
  1106→      ...(planId !== null ? { planId } : {}),
  1107→      ...(periodAdvances ? { currentPeriodEnd: new Date(periodEndMs) } : {}),
  1108→      ...(view.trialEnd !== null ? { trialEnd: new Date(view.trialEnd * 1000) } : {}),
  1109→      ...(graceEnd !== undefined ? { graceEnd } : {}),
  1110→      cancelAtPeriodEnd: view.cancelAtPeriodEnd,
  1111→    });
  1112→    return applied ? { handled: `subscription.${target}` } : { ignored: "transition-guarded" };
  1113→  }

  1116→    stripe: StripeSeam,
  1117→    event: StripeEventLike,
  1118→  ): Promise<WebhookOutcome> {

  1123→    stripe: StripeSeam,
  1124→    event: StripeEventLike,
  1125→  ): Promise<WebhookOutcome> {
  1126→    // BP §M6.2: cancellation writes STATUS ONLY. Projects, versions and

  1131→  async function handleInvoice(event: StripeEventLike): Promise<WebhookOutcome> {
  1132→    const object = event.data.object;
  1133→    const organizationId = organizationIdOf(object);
  1134→    if (organizationId === null) return { ignored: "unattributed" };
  1135→    const invoices = invoicesRepo(deps.repos);
  1136→    if (invoices === null) return { ignored: "no-persistence" };
  1137→
  1138→    const record = object as unknown as Record<string, unknown>;
  1139→    const invoiceId = nonEmptyString(record.id);
  1140→    if (invoiceId === null) return { ignored: "invoice-unparsable" };
  1141→    const amountRaw = record.amount_paid ?? record.amount_due;
  1142→    const amountCents =
  1143→      typeof amountRaw === "number" && Number.isFinite(amountRaw) && amountRaw >= 0
  1144→        ? Math.round(amountRaw)
  1145→        : 0;
  1146→    const currencyRaw = nonEmptyString(record.currency);
  1147→    const periodStart = toDate(
  1148→      typeof record.period_start === "number" ? record.period_start : null,
  1149→    );
  1150→    const periodEnd = toDate(typeof record.period_end === "number" ? record.period_end : null);
  1151→
  1152→    const status = event.type === "invoice.paid" ? ("paid" as const) : ("void" as const);
  1153→    try {
  1154→      await invoices.record(organizationId, {
  1155→        stripeInvoiceId: invoiceId,
  1156→        amountCents,
  1157→        currency: currencyRaw ?? "usd",
  1158→        status,
  1159→        periodStart,
  1160→        periodEnd,
  1161→        hostedUrl: nonEmptyString(record.hosted_invoice_url),
  1162→        pdfUrl: nonEmptyString(record.invoice_pdf),
  1163→      });
  1164→    } catch (error) {
  1165→      if (isDbError(error) && (error.code === "VALIDATION_FAILED" || error.code === "CONFLICT")) {
  1166→        // A foreign/malformed invoice is audited-ignored — never a retry loop.
  1167→        return { ignored: "invoice-invalid" };
  1168→      }
  1169→      throw error;
  1170→    }
  1171→
  1172→    if (status !== "paid") {
  1173→      return { handled: "invoice.voided" };
  1174→    }
  1175→
  1176→    // Top-up grants invoiced outside Checkout (§M6 task 2): keyed on the
  1177→    // INVOICE id, so re-deliveries under fresh event ids cannot double-grant.
  1178→    let handled = "invoice.paid";
  1179→    if (object.metadata?.kind === "topup") {
  1180→      const granted = await grantTopUp(
  1181→        organizationId,
  1182→        event,
  1183→        object.metadata,
  1184→        `stripe:invoice:${invoiceId}:topup`,
  1185→      );
  1186→      if (granted) handled = "invoice.topup-granted";
  1187→    }
  1188→
  1189→    // Dunning heal (§M6 task 2 trial/renewal lifecycle): a PAID invoice while
  1190→    // the row sits in the grace window returns the subscription to active.
  1191→    if (subscriptions !== null) {
  1192→      const row = await subscriptions.getByOrganization(organizationId);
  1193→      if (row !== null && (row.status === "grace" || row.status === "past_due")) {
  1194→        const healed = await guardedTransition(organizationId, {
  1195→          from: ["grace", "past_due"],
  1196→          to: "active",
  1197→          ...(periodEnd !== null ? { currentPeriodEnd: periodEnd } : {}),
  1198→          graceEnd: null,
  1199→        });
  1200→        if (healed) handled = "invoice.paid-healed";
  1201→      }
  1202→    }
  1203→    return { handled };
  1204→  }
  1205→
  1206→  async function handleStripeEvent(
  1207→    stripe: StripeSeam,
  1208→    event: StripeEventLike,
  1209→  ): Promise<WebhookOutcome> {
  1210→    switch (event.type) {
  1211→      case "checkout.session.completed":
  1212→        return handleCheckoutCompleted(stripe, event);
  1213→      case "customer.subscription.updated":
  1214→        return handleSubscriptionUpdated(stripe, event);
  1215→      case "customer.subscription.deleted":
  1216→        return handleSubscriptionDeleted(stripe, event);
  1217→      case "invoice.paid":
  1218→      case "invoice.voided":
  1219→        return handleInvoice(event);
  1220→      default:
  1221→        // Unknown event types are 200 + audited-ignored: Stripe retries on
  1222→        // non-2xx, so an unhandled type must NEVER be a 500.
  1223→        return { ignored: "unhandled-type" };
  1224→    }
  1225→  }
  1226→
  1227→  return app;
  1228→}
