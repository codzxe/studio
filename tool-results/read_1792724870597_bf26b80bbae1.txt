     3→/**
     4→ * `CommentsPanel` — the M7-F review surface (BUILD_PLAN §M7 task 1; UI
     5→ * design language per EDITOR_CHAT_UX §10 + UI_DESIGN_SYSTEM §8.7/§9: the
     6→ * panel sits ON the dark editor, tokens only, no blur/glass).
     7→ *
     8→ * Composition (anchored threads over the flat M7-A wire):
     9→ * - HEADER: title + project name + close. The project is the panel's scope
    10→ *   (the wire requires a project), so the PROJECT chip lives here — rows
    11→ *   carry the comp/layer/time anchor chips.
    12→ * - FILTERS: resolved filter (Open | Resolved | All → the wire's server-side
    13→ *   `resolved` param) + comp filter (Menu, options from the live project —
    14→ *   D-15: `compId` filters server-side, so a filter never splits a thread).
    15→ * - LIST: threaded rows (tree.ts), cursor pagination ("Load more" appends
    16→ *   pages; threads rebuild over the accumulated window).
    17→ * - CREATE BOX: prefilled anchor from the CURRENT editor context via the
    18→ *   bridge `ui.inspect` seam (anchor.ts — the chat collector's recipe,
    19→ *   reused), refreshed on demand.
    20→ *
    21→ * LEASE LAW (the D-23 pin): when M7-C's lease-state says another member
    22→ * holds the edit lease, the panel shows the note and comment creation STAYS
    23→ * ENABLED — review is the point of the read-only state; the lease gates
    24→ * project WRITES, never comments. Nothing in this panel is disabled by the
    25→ * lease.
    26→ *
    27→ * DATA LAW (PRODUCT_ARCHITECTURE §8.3 + the M4-N corpus): every body,
    28→ * comp/layer name and guest name is UNTRUSTED — rendered as React text,
    29→ * truncated with CSS (`truncate`), never interpolated into markup, commands
    30→ * or copy. Anchor time renders as `mm:ss` (engine ticks).
    31→ *
    32→ * Author labels never leak ids (law #1): "You" (viewer id match), "Member"
    33→ * (any other member — no member-list endpoint exists on this surface), or
    34→ * the guest's recorded name for guest rows.
    35→ */

   129→/** Truncation bound for anchor names (CSS `truncate` still guards layout). */
   130→const NAME_TRUNCATE = 32;
   131→
   132→/** Comp/layer display: the live name, else the raw anchor id as data. */
   133→function anchorName(name: string | null | undefined, id: string): string {
   134→  if (typeof name === "string" && name.length > 0) return name;
   135→  return id.length > NAME_TRUNCATE ? `${id.slice(0, NAME_TRUNCATE)}…` : id;
   136→}
   137→
   138→/** Row author label — ids never surface (law #1). */
   139→function authorLabel(comment: Comment, viewerId: string | null): string {
   140→  if (comment.authorId === null) {
   141→    return comment.guestName !== null && comment.guestName !== "" ? comment.guestName : "Guest";
   142→  }
   143→  if (viewerId !== null && comment.authorId === viewerId) return "You";
   144→  return "Member";
   145→}
   146→
   147→/** Deterministic short stamp ("Oct 26, 14:32") — UTC so tests never see a TZ. */
   148→const WHEN_FORMAT = new Intl.DateTimeFormat("en-US", {
   149→  month: "short",
   150→  day: "numeric",
   151→  hour: "2-digit",
   152→  minute: "2-digit",
   153→  timeZone: "UTC",
   154→});
   155→
   156→function formatWhen(iso: string): string {
   157→  const date = new Date(iso);
   158→  return Number.isNaN(date.getTime()) ? "" : WHEN_FORMAT.format(date);
   159→}
   160→
   161→/** A safe human message for any rejection (api-client messages are safe). */
   162→function failureMessage(error: unknown): string {
   163→  return error instanceof Error && error.message !== "" ? error.message : GENERIC_ERROR;
   164→}

   166→/** One anchor chip (comp / layer / time) — hostile names render as text. */
   167→function AnchorChip(props: {
   168→  readonly testId: string;
   169→  readonly icon: "comp" | "layer" | "time";
   170→  readonly label: string;
   171→}): ReactElement {
   172→  const title =
   173→    props.icon === "comp" ? "Composition" : props.icon === "layer" ? "Layer" : "Playhead";
   174→  return (
   175→    <span
   176→      data-testid={props.testId}
   177→      title={`${title}: ${props.label}`}
   178→      className="inline-flex max-w-40 items-center rounded-xs border border-border bg-input px-1.5 py-0.5 font-sans text-hint whitespace-nowrap text-secondary"
   179→    >
   180→      <span className="truncate">{props.label}</span>
   181→    </span>
   182→  );
   183→}

   188→
   189→export function CommentsPanel({
   190→  client,
   191→  projectId,
   192→  projectName,
   193→  bridge,
   194→  viewerId = null,
   195→  lease = null,
   196→  open,
   197→  onClose,
   198→}: CommentsPanelProps): ReactElement | null {
   199→  // -- list state -------------------------------------------------------------
   200→  const [rows, setRows] = useState<readonly Comment[]>([]);
   201→  const [nextCursor, setNextCursor] = useState<string | null>(null);
   202→  const [loading, setLoading] = useState(false);
   203→  const [loadingMore, setLoadingMore] = useState(false);
   204→  const [listError, setListError] = useState<string | null>(null);
   205→  const [loadedOnce, setLoadedOnce] = useState(false);
   206→
   207→  // -- filters ----------------------------------------------------------------
   208→  const [resolvedFilter, setResolvedFilter] = useState<ResolvedFilter>("open");
   209→  const [compFilter, setCompFilter] = useState<string | null>(null);
   210→
   211→  // -- editor-context state (bridge-derived) ------------------------------------
   212→  const [anchor, setAnchor] = useState<CommentAnchorPrefill | null>(null);
   213→  const [anchorReady, setAnchorReady] = useState(false);
   214→  const [names, setNames] = useState<AnchorNames | null>(null);
   215→  const [compOptions, setCompOptions] = useState<readonly CompOption[]>([]);
   216→
   217→  // -- composer state -----------------------------------------------------------
   218→  const [draft, setDraft] = useState("");
   219→  const [sending, setSending] = useState(false);
   220→  const [composerError, setComposerError] = useState<string | null>(null);
   221→  const [postedNotice, setPostedNotice] = useState<string | null>(null);
   222→
   223→  // -- reply state --------------------------------------------------------------
   224→  const [replyTargetId, setReplyTargetId] = useState<string | null>(null);
   225→  const [replyDraft, setReplyDraft] = useState("");
   226→  const [replySending, setReplySending] = useState(false);
   227→  const [replyError, setReplyError] = useState<string | null>(null);
   228→
   229→  // Stale-response guard: a generation counter per list fetch.
   230→  const generationRef = useRef(0);
   231→
   232→  const resolvedQueryValue = useCallback(
   233→    (): boolean | undefined =>
   234→      resolvedFilter === "open" ? false : resolvedFilter === "resolved" ? true : undefined,
   235→    [resolvedFilter],
   236→  );
   237→
   238→  // -- list fetching ------------------------------------------------------------
   239→  const fetchPage = useCallback(
   240→    async (
   241→      cursor: string | null,
   242→    ): Promise<{ items: readonly Comment[]; nextCursor: string | null }> => {
   243→      const query: Record<string, string | number | boolean | undefined> = {
   244→        limit: COMMENTS_PAGE_LIMIT,
   245→        ...(compFilter === null ? {} : { compId: compFilter }),
   246→        ...(resolvedQueryValue() === undefined ? {} : { resolved: resolvedQueryValue() }),
   247→      };
   248→      if (cursor !== null) query.cursor = cursor;
   249→      const page = await client.comments.listThreadedComments(projectId, query);
   250→      return { items: page.items, nextCursor: page.nextCursor };
   251→    },
   252→    [client, compFilter, projectId, resolvedQueryValue],
   253→  );
   254→
   255→  const refresh = useCallback(async (): Promise<void> => {
   256→    const generation = ++generationRef.current;
   257→    setLoading(true);
   258→    setListError(null);
   259→    try {
   260→      const page = await fetchPage(null);
   261→      if (generationRef.current !== generation) return;
   262→      setRows(page.items);
   263→      setNextCursor(page.nextCursor);
   264→    } catch (error) {
   265→      if (generationRef.current !== generation) return;
   266→      setRows([]);
   267→      setNextCursor(null);
   268→      setListError(failureMessage(error));
   269→    } finally {
   270→      if (generationRef.current === generation) {
   271→        setLoading(false);
   272→        setLoadedOnce(true);
   273→      }
   274→    }
   275→  }, [fetchPage]);
   276→
   277→  const loadMore = useCallback(async (): Promise<void> => {
   278→    if (loadingMore || nextCursor === null) return;
   279→    const generation = ++generationRef.current;
   280→    setLoadingMore(true);
   281→    setListError(null);
   282→    try {
   283→      const page = await fetchPage(nextCursor);
   284→      if (generationRef.current !== generation) return;
   285→      setRows((previous) => {
   286→        const seen = new Set(previous.map((row) => row.id));
   287→        return [...previous, ...page.items.filter((row) => !seen.has(row.id))];
   288→      });
   289→      setNextCursor(page.nextCursor);
   290→    } catch (error) {
   291→      if (generationRef.current !== generation) return;
   292→      setListError(failureMessage(error));
   293→    } finally {
   294→      if (generationRef.current === generation) setLoadingMore(false);
   295→    }
   296→  }, [fetchPage, loadingMore, nextCursor]);
   297→
   298→  // Open → load the first page. `refresh` identity changes with the FILTERS
   299→  // (fetchPage deps), so this one effect covers open + every filter change
   300→  // with exactly one fetch — no separate filter effect, no double fetch.
   301→  useEffect(() => {
   302→    if (!open) return;
   303→    void refresh();
   304→  }, [open, refresh]);
   305→
   306→  // Editor context per OPEN: anchor prefill + comp/layer names + comp options.
   307→  // No bridge (region mounted outside a live editor) → degraded immediately.
   308→  useEffect(() => {
   309→    if (!open) return;
   310→    if (bridge === null || bridge === undefined) {
   311→      setAnchorReady(true);
   312→      return;
   313→    }
   314→    let cancelled = false;
   315→    void (async (): Promise<void> => {
   316→      const [collected, anchorNames, options] = await Promise.all([
   317→        collectCommentAnchor(bridge),
   318→        collectAnchorNames(bridge),
   319→        listCompOptions(bridge),
   320→      ]);
   321→      if (cancelled) return;
   322→      setAnchor(collected);
   323→      setAnchorReady(true);
   324→      setNames(anchorNames);
   325→      setCompOptions(options);
   326→      setCompFilter((current) =>
   327→        current !== null && options.every((option) => option.id !== current) ? null : current,
   328→      );
   329→    })();
   330→    return () => {
   331→      cancelled = true;
   332→    };
   333→  }, [open, bridge]);
   334→
   335→  // -- mutations ----------------------------------------------------------------
   336→
   337→  const trimmedDraft = draft.trim();
   338→  const draftTooLong = trimmedDraft.length > MAX_BODY_CHARS;
   339→  const canCreate = anchor !== null && !sending && !draftTooLong && trimmedDraft.length > 0;
   340→
   341→  const submitCreate = useCallback(async (): Promise<void> => {
   342→    if (anchor === null || trimmedDraft.length === 0 || draftTooLong) return;
   343→    setSending(true);
   344→    setComposerError(null);
   345→    setPostedNotice(null);
   346→    try {
   347→      const created = await client.comments.create(projectId, {
   348→        compId: anchor.compId,
   349→        ...(anchor.layerId === null ? {} : { layerId: anchor.layerId }),
   350→        ...(anchor.timeTicks === null ? {} : { timeTicks: anchor.timeTicks }),
   351→        body: trimmedDraft,
   352→      });
   353→      setDraft("");
   354→      // Newest-first wire: prepend when the row is visible under the current
   355→      // filters, else keep the draft clear + say where it went.
   356→      const visibleUnderFilters =
   357→        (resolvedFilter === "open" || resolvedFilter === "all") &&
   358→        (compFilter === null || compFilter === created.compId);
   359→      if (visibleUnderFilters) {
   360→        setRows((previous) =>
   361→          previous.some((row) => row.id === created.id) ? previous : [created, ...previous],
   362→        );
   363→      } else {
   364→        setPostedNotice("Comment posted. Switch filters to see it.");
   365→      }
   366→    } catch (error) {
   367→      setComposerError(failureMessage(error));
   368→    } finally {
   369→      setSending(false);
   370→    }
   371→  }, [anchor, client, compFilter, draftTooLong, projectId, resolvedFilter, trimmedDraft]);
   372→
   373→  const submitReply = useCallback(
   374→    async (targetId: string): Promise<void> => {
   375→      const body = replyDraft.trim();
   376→      if (body.length === 0 || body.length > MAX_BODY_CHARS) return;
   377→      setReplySending(true);
   378→      setReplyError(null);
   379→      try {
   380→        const reply = await client.comments.replyComment(targetId, { body });
   381→        setReplyDraft("");
   382→        setReplyTargetId(null);
   383→        // Replies render reading-order (oldest first), so APPEND — the
   384→        // rebuilt tree puts the newest reply last in its thread.
   385→        setRows((previous) =>
   386→          previous.some((row) => row.id === reply.id) ? previous : [...previous, reply],
   387→        );
   388→      } catch (error) {
   389→        setReplyError(failureMessage(error));
   390→      } finally {
   391→        setReplySending(false);
   392→      }
   393→    },
   394→    [client, replyDraft],
   395→  );
   396→
   397→  const toggleResolve = useCallback(
   398→    async (comment: Comment): Promise<void> => {
   399→      try {
   400→        const updated =
   401→          comment.resolvedAt === null
   402→            ? await client.comments.resolveComment(comment.id)
   403→            : await client.comments.resolveComment(comment.id, { resolved: false });
   404→        const threadRootId = rootIdOfRow(updated.id, rows);
   405→        // Thread-level filter semantics: a resolved root takes its replies
   406→        // out of "Open"; a reopened root leaves "Resolved".
   407→        const threadFailsFilter =
   408→          (updated.resolvedAt !== null && resolvedFilter === "open") ||
   409→          (updated.resolvedAt === null && resolvedFilter === "resolved");
   410→        setRows((previous) => {
   411→          const replaced = previous.map((row) => (row.id === updated.id ? updated : row));
   412→          if (!threadFailsFilter || threadRootId === null) return replaced;
   413→          return replaced.filter((row) => rootIdOfRow(row.id, previous) !== threadRootId);
   414→        });
   415→      } catch (error) {
   416→        setListError(failureMessage(error));
   417→      }
   418→    },
   419→    [client, resolvedFilter, rows],
   420→  );
   421→
   422→  // -- derived render state -----------------------------------------------------
   423→
   424→  const threads = useMemo(() => buildCommentThreads(rows), [rows]);
   425→  const resolvedCount = useMemo(
   426→    () => rows.filter((row) => row.resolvedAt !== null && row.parentCommentId === null).length,
   427→    [rows],
   428→  );
   429→  const leaseHeldByOther = lease !== null && lease.holderName !== null && lease.youAre !== "holder";
   430→  const compName = compFilter === null ? null : compOptions.find((c) => c.id === compFilter);
   431→
   432→  const onKeyDown = (event: ReactKeyboardEvent<HTMLElement>): void => {
   433→    if (event.key === "Escape" && !event.defaultPrevented) {
   434→      event.stopPropagation();
   435→      onClose();
   436→    }
   437→  };
   438→
   439→  if (!open) return null;
   440→
   441→  return (
   442→    <aside
   443→      aria-label="Comments"
   444→      data-testid="comments-panel"
   445→      tabIndex={-1}
   446→      onKeyDown={onKeyDown}
   447→      className="pointer-events-auto absolute top-0 right-0 bottom-0 z-[35] flex w-[380px] max-w-full flex-col overflow-hidden border-l border-border bg-panel shadow-modal"
   448→    >
   449→      {/* Header — the project IS the panel scope, so the project chip lives here. */}
   450→      <header
   451→        data-testid="comments-header"
   452→        className="flex h-12 shrink-0 items-center gap-2 border-b border-border bg-panel px-3"
   453→      >
   454→        <MessageSquare
   455→          size={15}
   456→          strokeWidth={1.75}
   457→          aria-hidden="true"
   458→          className="shrink-0 text-secondary"
   459→        />
   460→        <span data-testid="comments-title" className="shrink-0 font-medium text-primary">
   461→          Comments
   462→        </span>
   463→        {projectName !== null && projectName !== undefined && projectName !== "" ? (
   464→          <span
   465→            data-testid="comments-project"
   466→            title={projectName}
   467→            className="min-w-0 flex-1 truncate text-caption text-muted"
   468→          >
   469→            {projectName}
   470→          </span>
   471→        ) : (
   472→          <div aria-hidden="true" className="min-w-0 flex-1" />
   473→        )}
   474→        <Button
   475→          variant="ghost"
   476→          size="sm"
   477→          className="h-7 w-7 shrink-0 p-0 text-secondary hover:text-primary"
   478→          aria-label="Close comments"
   479→          data-testid="comments-close"
   480→          onClick={onClose}
   481→        >
   482→          <X size={15} strokeWidth={1.75} aria-hidden="true" />
   483→        </Button>
   484→      </header>
   485→
   486→      {/* Filters: resolved tri-state (server-side `resolved`) + comp (D-15). */}
   487→      <div
   488→        data-testid="comments-filters"
   489→        className="flex h-10 shrink-0 items-center gap-1 border-b border-border px-2"
   490→      >
   491→        <div className="flex items-center gap-0.5" aria-label="Filter by resolved state">
   492→          <ResolvedFilterButton
   493→            filter="open"
   494→            active={resolvedFilter === "open"}
   495→            label="Open"
   496→            onSelect={() => setResolvedFilter("open")}
   497→          />
   498→          <ResolvedFilterButton
   499→            filter="resolved"
   500→            active={resolvedFilter === "resolved"}
   501→            label={resolvedCount > 0 ? `Resolved · ${resolvedCount}` : "Resolved"}
   502→            onSelect={() => setResolvedFilter("resolved")}
   503→          />
   504→          <ResolvedFilterButton
   505→            filter="all"
   506→            active={resolvedFilter === "all"}
   507→            label="All"
   508→            onSelect={() => setResolvedFilter("all")}
   509→          />
   510→        </div>
   511→        <div aria-hidden="true" className="min-w-0 flex-1" />
   512→        {compOptions.length > 0 ? (
   513→          <Menu>
   514→            <MenuTrigger asChild>
   515→              <Button
   516→                variant="ghost"
   517→                size="sm"
   518→                className="h-6 max-w-36 shrink-0 gap-1 px-2 text-caption text-secondary hover:text-primary"
   519→                aria-label="Filter by composition"
   520→                data-testid="comments-comp-filter"
   521→              >
   522→                <span className="truncate">{compName?.name ?? "All comps"}</span>
   523→                <ChevronDown size={12} strokeWidth={1.75} aria-hidden="true" />
   524→              </Button>
   525→            </MenuTrigger>
   526→            <MenuContent align="end" data-testid="comments-comp-menu">
   527→              <MenuRadioGroup
   528→                value={compFilter ?? "all"}
   529→                onValueChange={(value) => setCompFilter(value === "all" ? null : value)}
   530→              >
   531→                <MenuRadioItem value="all" data-testid="comments-comp-option-all">
   532→                  All comps
   533→                </MenuRadioItem>
   534→                {compOptions.map((option) => (
   535→                  <MenuRadioItem
   536→                    key={option.id}
   537→                    value={option.id}
   538→                    data-testid={`comments-comp-option-${option.id}`}
   539→                  >
   540→                    <span className="max-w-56 truncate">{option.name}</span>
   541→                  </MenuRadioItem>
   542→                ))}
   543→              </MenuRadioGroup>
   544→            </MenuContent>
   545→          </Menu>
   546→        ) : null}
   547→      </div>
   548→
   549→      {/* D-23 pin: another member holds the edit lease — review still works. */}
   550→      {leaseHeldByOther && lease !== null ? (
   551→        <div
   552→          data-testid="comments-lease-note"
   553→          data-holder={lease.holderName}
   554→          className="shrink-0 border-b border-border bg-elevated px-3 py-2 text-caption text-secondary"
   555→        >
   556→          {lease.holderName} is editing — you can still read and comment.
   557→        </div>
   558→      ) : null}
   559→
   560→      {/* Thread list */}
   561→      <div data-testid="comments-list" className="min-h-0 flex-1 overflow-y-auto px-2 py-2">
   562→        {loading && !loadedOnce ? (

   564→            <Skeleton className="h-16 w-full" />
   565→            <Skeleton className="h-16 w-full" />
   566→            <Skeleton className="h-16 w-full" />
   567→          </div>

   569→          <div
   570→            data-testid="comments-error"
   571→            className="flex flex-col items-start gap-2 px-1 py-3 text-caption text-danger"
   572→          >
   573→            <p className="text-body-s text-primary">{listError}</p>
   574→            <Button

   576→              size="sm"
   577→              data-testid="comments-retry"
   578→              onClick={() => void refresh()}
   579→            >

   582→          </div>
   583→        ) : threads.length === 0 ? (
   584→          <div data-testid="comments-empty" className="px-1">
   585→            <EmptyState
   586→              withIllustration={false}
   587→              className="py-10"
   588→              title={
   589→                resolvedFilter === "open"
   590→                  ? compFilter === null
   591→                    ? "No open comments. Leave the first note."
   592→                    : "No open comments on this comp."
   593→                  : resolvedFilter === "resolved"
   594→                    ? "Nothing resolved yet."
   595→                    : "No comments yet."
   596→              }
   597→            />
   598→          </div>
   599→        ) : (

   601→            {threads.map((thread) => (

   606→              >
   607→                {thread.rows.map((row) => (
   608→                  <CommentRow

   610→                    row={row}
   611→                    viewerId={viewerId}
   612→                    names={names}
   613→                    canReply={row.depth < COMMENT_MAX_VISUAL_DEPTH}
   614→                    replyOpen={replyTargetId === row.comment.id}
   615→                    replyDraft={replyDraft}
   616→                    replySending={replySending}
   617→                    replyError={replyError}
   618→                    onReplyOpen={() => {
   619→                      setReplyTargetId(row.comment.id);
   620→                      setReplyDraft("");
   621→                      setReplyError(null);
   622→                    }}
   623→                    onReplyDraftChange={setReplyDraft}
   624→                    onReplyCancel={() => setReplyTargetId(null)}
   625→                    onReplySubmit={() => void submitReply(row.comment.id)}
   626→                    onToggleResolve={() => void toggleResolve(row.comment)}
   627→                  />
   628→                ))}
   629→              </section>
   630→            ))}
   631→            {nextCursor !== null ? (
   632→              <div className="flex justify-center pb-2">
   633→                <Button
   634→                  variant="secondary"
   635→                  size="sm"
   636→                  data-testid="comments-load-more"
   637→                  disabled={loadingMore}
   638→                  onClick={() => void loadMore()}
   639→                >
   640→                  {loadingMore ? "Loading…" : "Load more"}
   641→                </Button>
   642→              </div>
   643→            ) : null}
   644→          </>
   645→        )}
   646→      </div>
   647→
   648→      {/* Create box — prefilled anchor from the current editor context. */}
   649→      <div
   650→        data-testid="comments-composer"
   651→        className="shrink-0 border-t border-border bg-panel px-3 py-2"
   652→      >
   653→        {anchor !== null ? (

   674→                label={formatTimeTicks(anchor.timeTicks) ?? "0:00"}

   679→              variant="ghost"
   680→              size="iconSm"
   681→              className="h-6 w-6 p-0 text-secondary hover:text-primary"
   682→              aria-label="Refresh anchor from the editor"
   683→              title="Refresh anchor from the editor"
   684→              data-testid="comments-composer-refresh"
   685→              disabled={bridge === null || bridge === undefined}
   686→              onClick={() => {
   687→                if (bridge === null || bridge === undefined) return;
   688→                void collectCommentAnchor(bridge).then(setAnchor);
   689→              }}
   690→            >
   691→              <RotateCw size={12} strokeWidth={1.75} aria-hidden="true" />
   692→            </Button>
   693→          </div>
   694→        ) : (
   695→          <p data-testid="comments-composer-hint" className="mb-1.5 text-caption text-muted">
   696→            {anchorReady ? NO_ANCHOR_HINT : "Reading the editor context…"}
   697→          </p>
   698→        )}
   699→
   700→        <div className="relative w-full">
   701→          <Textarea
   702→            rows={2}
   703→            aria-label="Write a comment"
   704→            placeholder="Leave a note for the team…"
   705→            value={draft}
   706→            disabled={anchor === null || sending}
   707→            onChange={(event) => setDraft(event.target.value)}
   708→            onKeyDown={(event) => {
   709→              if (event.key !== "Enter") return;
   710→              if (event.nativeEvent.isComposing || event.nativeEvent.keyCode === 229) return;
   711→              const withModifier = event.metaKey || event.ctrlKey;
   712→              if (event.shiftKey && !withModifier) return;
   713→              event.preventDefault();
   714→              void submitCreate();
   715→            }}
   716→            className="pr-11 text-body-s"
   717→          />
   718→          <Button
   719→            variant="primary"
   720→            size="iconSm"
   721→            className="absolute right-1.5 bottom-1.5"
   722→            aria-label="Post comment"
   723→            data-testid="comments-composer-send"
   724→            disabled={!canCreate}
   725→            onClick={() => void submitCreate()}
   726→          >
   727→            <MessageSquare size={13} strokeWidth={2} aria-hidden="true" />
   728→          </Button>
   729→        </div>
   730→
   731→        <div className="mt-1 flex w-full items-start justify-between gap-2 text-caption text-muted">
   732→          {composerError !== null ? (
   733→            <output data-testid="comments-composer-error" className="text-danger">
   734→              {composerError}
   735→            </output>
   736→          ) : draftTooLong ? (
   737→            <output className="text-danger">{TOO_LONG_HINT}</output>
   738→          ) : postedNotice !== null ? (
   739→            <output data-testid="comments-composer-posted">{postedNotice}</output>
   740→          ) : (
   741→            <span>Enter to post · Shift+Enter for a new line</span>
   742→          )}
   743→        </div>
   744→      </div>
   745→    </aside>
   746→  );

   753→function CommentRow(props: {
   754→  readonly row: CommentRowView;
   755→  readonly viewerId: string | null;
   756→  readonly names: AnchorNames | null;
   757→  readonly canReply: boolean;
   758→  readonly replyOpen: boolean;
   759→  readonly replyDraft: string;
   760→  readonly replySending: boolean;
   761→  readonly replyError: string | null;
   762→  readonly onReplyOpen: () => void;
   763→  readonly onReplyDraftChange: (value: string) => void;
   764→  readonly onReplyCancel: () => void;
   765→  readonly onReplySubmit: () => void;
   766→  readonly onToggleResolve: () => void;
   767→}): ReactElement {
   768→  const { comment, depth, promoted } = props.row;
   769→  const indent = (Math.min(depth, COMMENT_MAX_VISUAL_DEPTH) - 1) * 14;
   770→  const isRoot = comment.parentCommentId === null;
   771→  const resolved = comment.resolvedAt !== null;
   772→  const compName = props.names?.comps.get(comment.compId) ?? null;
   773→  const layerName =
   774→    comment.layerId === null ? null : (props.names?.layers.get(comment.layerId) ?? null);
   775→  const resolvedByYou =
   776→    comment.resolvedBy !== null && props.viewerId !== null && comment.resolvedBy === props.viewerId;
   777→  // Depth 3 is the last visual level (server caps replies at depth 3), and a
   778→  // resolved row closes its reply affordance — reopen to continue a thread.
   779→  const canReply = props.canReply && !resolved;
   780→
   781→  return (
   782→    <article
   783→      data-testid={`comment-row-${comment.id}`}
   784→      data-depth={depth}
   785→      data-promoted={promoted || undefined}
   786→      className={cn(
   787→        "flex flex-col gap-1 rounded-sm",
   788→        isRoot ? "bg-transparent" : "bg-elevated/50",
   789→        resolved && "opacity-80",
   790→      )}
   791→      style={indent > 0 ? { paddingLeft: indent } : undefined}
   792→    >
   793→      <div className="flex min-w-0 items-center gap-1.5">
   794→        <span
   795→          data-testid={`comment-author-${comment.id}`}
   796→          className="truncate font-medium text-primary"
   797→        >
   798→          {authorLabel(comment, props.viewerId)}
   799→        </span>
   800→        <span data-testid={`comment-when-${comment.id}`} className="shrink-0 text-hint text-muted">
   801→          {formatWhen(comment.createdAt)}
   802→        </span>
   803→        {resolved ? (
   804→          <Badge variant="success" data-testid={`comment-resolved-${comment.id}`} className="gap-1">
   805→            <Check size={10} strokeWidth={2} aria-hidden="true" />
   806→            Resolved{resolvedByYou ? " by you" : ""}
   807→          </Badge>
   808→        ) : null}
   809→      </div>
   810→
   811→      <p
   812→        data-testid={`comment-body-${comment.id}`}
   813→        className="text-body-s whitespace-pre-wrap break-words text-secondary"
   814→      >
   815→        {comment.body}
   816→      </p>
   817→
   818→      <div className="flex flex-wrap items-center gap-1">
   819→        <AnchorChip
   820→          testId={`comment-anchor-comp-${comment.id}`}
   821→          icon="comp"
   822→          label={anchorName(compName, comment.compId)}
   823→        />
   824→        {comment.layerId !== null ? (
   825→          <AnchorChip
   826→            testId={`comment-anchor-layer-${comment.id}`}
   827→            icon="layer"
   828→            label={anchorName(layerName, comment.layerId)}
   829→          />
   830→        ) : null}
   831→        {comment.timeTicks !== null ? (
   832→          <AnchorChip
   833→            testId={`comment-anchor-time-${comment.id}`}
   834→            icon="time"
   835→            label={formatTimeTicks(comment.timeTicks) ?? "0:00"}
   836→          />
   837→        ) : null}
   838→      </div>
   839→
   840→      <div className="flex items-center gap-1.5">
   841→        {canReply ? (
   842→          <Button
   843→            variant="ghost"
   844→            size="sm"
   845→            className="h-6 px-2 text-caption text-secondary hover:text-primary"
   846→            data-testid={`comment-reply-open-${comment.id}`}
   847→            onClick={props.onReplyOpen}
   848→          >
   849→            Reply
   850→          </Button>
   851→        ) : null}
   852→        {isRoot ? (
   853→          resolved ? (
   854→            <Button
   855→              variant="ghost"
   856→              size="sm"
   857→              className="h-6 px-2 text-caption text-secondary hover:text-primary"
   858→              data-testid={`comment-reopen-${comment.id}`}
   859→              onClick={props.onToggleResolve}
   860→            >
   861→              Reopen
   862→            </Button>
   863→          ) : (
   864→            <Button
   865→              variant="ghost"
   866→              size="sm"
   867→              className="h-6 px-2 text-caption text-secondary hover:text-primary"
   868→              data-testid={`comment-resolve-${comment.id}`}
   869→              onClick={props.onToggleResolve}
   870→            >
   871→              Resolve
   872→            </Button>
   873→          )
   874→        ) : null}
   875→      </div>
   876→
   877→      {props.replyOpen ? (
   878→        <div data-testid={`comment-reply-box-${comment.id}`} className="flex flex-col gap-1 pt-0.5">
   879→          <Textarea
   880→            rows={2}
   881→            aria-label="Write a reply"
   882→            placeholder="Reply…"
   883→            value={props.replyDraft}
   884→            disabled={props.replySending}
   885→            onChange={(event) => props.onReplyDraftChange(event.target.value)}
   886→            onKeyDown={(event) => {
   887→              if (event.key !== "Enter") return;
   888→              if (event.nativeEvent.isComposing || event.nativeEvent.keyCode === 229) return;
   889→              const withModifier = event.metaKey || event.ctrlKey;
   890→              if (event.shiftKey && !withModifier) return;
   891→              event.preventDefault();
   892→              props.onReplySubmit();
   893→            }}
   894→            className="text-body-s"
   895→          />
   896→          {props.replyError !== null ? (
   897→            <p
   898→              className="text-caption text-danger"
   899→              data-testid={`comment-reply-error-${comment.id}`}
   900→            >
   901→              {props.replyError}
   902→            </p>
   903→          ) : null}
   904→          <div className="flex items-center justify-end gap-1.5">
   905→            <Button
   906→              variant="ghost"
   907→              size="sm"
   908→              className="h-6 px-2 text-caption"
   909→              data-testid={`comment-reply-cancel-${comment.id}`}
   910→              onClick={props.onReplyCancel}
   911→            >
   912→              Cancel
   913→            </Button>
   914→            <Button
   915→              variant="primary"
   916→              size="sm"
   917→              className="h-6 px-2 text-caption"
   918→              aria-label="Post reply"
   919→              data-testid={`comment-reply-send-${comment.id}`}
   920→              disabled={props.replySending || props.replyDraft.trim().length === 0}
   921→              onClick={props.onReplySubmit}
   922→            >
   923→              Reply
   924→            </Button>
   925→          </div>
   926→        </div>
   927→      ) : null}
   928→    </article>
   929→  );
   930→}

   936→/** The thread root id of one row inside the CURRENT accumulated window. */
   937→function rootIdOfRow(commentId: string, rows: readonly Comment[]): string | null {
   938→  const byId = new Map<string, Comment>();
   939→  for (const row of rows) {
   940→    if (!byId.has(row.id)) byId.set(row.id, row);
   941→  }
   942→  const seen = new Set<string>();
   943→  let cursor: Comment | undefined = byId.get(commentId);
   944→  for (;;) {
   945→    if (cursor === undefined) return null;
   946→    if (cursor.parentCommentId === null) return cursor.id;
   947→    if (seen.has(cursor.id) || seen.has(cursor.parentCommentId)) return cursor.id;
   948→    seen.add(cursor.id);
   949→    cursor = byId.get(cursor.parentCommentId);
   950→  }
   951→}
