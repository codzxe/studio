   118→function parseInstant(iso: string): Date | null {
   119→  if (typeof iso !== "string" || iso.length === 0) return null;
   120→  const date = new Date(iso.length > 100 ? iso.slice(0, 100) : iso);
   121→  return Number.isNaN(date.getTime()) ? null : date;
   122→}

   124→export type RenderExpiryState = "none" | "until" | "expired" | "unknown";
   125→
   126→export interface RenderExpiryLabel {
   127→  readonly state: RenderExpiryState;
   128→  /** Ready-to-render copy; null for the none/unknown shapes. */
   129→  readonly text: string | null;
   130→}
   131→
   132→/**
   133→ * Expiry copy against the RSC-stamped `nowIso` (deterministic per page load,
   134→ * the activity-feed house pattern): "Available until Nov 9, 2026" while the
   135→ * signed download window is open, "Expired — the download window has
   136→ * closed." once it is past. Null/invalid expiresAt degrade honestly.
   137→ */
   138→export function renderExpiryLabel(expiresAt: string | null, nowIso: string): RenderExpiryLabel {
   139→  if (expiresAt === null) return { state: "none", text: null };
   140→  const expires = parseInstant(expiresAt);
   141→  const now = parseInstant(nowIso);
   142→  if (expires === null || now === null) return { state: "unknown", text: null };
   143→  if (expires.getTime() <= now.getTime()) {
   144→    return { state: "expired", text: "Expired — the download window has closed." };
   145→  }
   146→  return { state: "until", text: `Available until ${DAY_FORMAT.format(expires)}` };
   147→}

   149→/**
   150→ * Friendly explanation for the stable machine `errorCode` on failed rows
   151→ * (the §7.3-safe human `error` message renders beside it; the code is
   152→ * copy-mapped, never echoed as UI copy). Unknown codes → null (the server
   153→ * message alone stays).
   154→ */
   155→export function renderErrorCodeCopy(code: string | null): string | null {
   156→  switch (code) {
   157→    case "RENDER_TIMEOUT":
   158→      return "The render ran past its time budget and was stopped.";
   159→    case "RENDER_ASSET_MISSING":
   160→      return "An asset this composition needs could not be found.";
   161→    case "RENDER_CLI_FAILED":
   162→    case "RENDER_CLI_MISSING":
   163→      return "The render engine could not run this job.";
   164→    case "RENDER_CANCELLED":
   165→      return "The render was cancelled before it finished.";
   166→    case "RENDER_QUOTA_EXCEEDED":
   167→      return "This render is past the plan's monthly render minutes.";
   168→    case "RENDER_INTERNAL":
   169→      return "Something went wrong on our side while rendering.";
   170→    default:
   171→      return null;
   172→  }

   259→// ---------------------------------------------------------------------------
   260→// useRendersPages — race-guarded cursor pagination + the poller merge
   261→// ---------------------------------------------------------------------------
   262→
   263→export type RendersPageFetcher = RendersHubPort["listRenders"];
   264→
   265→export interface RendersPagesState {
   266→  readonly entries: readonly RenderJob[];
   267→  readonly nextCursor: string | null;
   268→  readonly loading: boolean;
   269→  readonly loadingMore: boolean;
   270→  readonly error: string | null;
   271→  readonly loadMore: () => void;
   272→  readonly retry: () => void;
   273→  readonly reload: () => void;
   274→  /** Replaces one loaded row by id (cancel responses; no refetch needed). */
   275→  readonly patchEntry: (next: RenderJob) => void;
   276→  /**
   277→   * Polling refresh: re-fetches the FIRST page and merges by id (rows in
   278→   * place, unseen rows prepended, loaded pages preserved). Silent on
   279→   * failure — the last list stays rendered.
   280→   */
   281→  readonly mergeFirstPage: () => Promise<void>;
   282→}
   283→
   284→export function useRendersPages(
   285→  fetchPage: RendersPageFetcher,
   286→  options?: { readonly pageSize?: number },
   287→): RendersPagesState {
   288→  const pageSize = options?.pageSize ?? 25;
   289→
   290→  const [entries, setEntries] = useState<readonly RenderJob[]>([]);
   291→  const [nextCursor, setNextCursor] = useState<string | null>(null);
   292→  const [loading, setLoading] = useState(true);
   293→  const [loadingMore, setLoadingMore] = useState(false);
   294→  const [error, setError] = useState<string | null>(null);
   295→
   296→  const seq = useRef(0);
   297→  const inFlight = useRef(false);
   298→  const mergeInFlight = useRef(false);
   299→  const fetchRef = useRef(fetchPage);
   300→  fetchRef.current = fetchPage;
   301→
   302→  const fetchFirst = useCallback(async (): Promise<void> => {
   303→    seq.current += 1;
   304→    const mySeq = seq.current;
   305→    setLoading(true);
   306→    setLoadingMore(false);
   307→    setError(null);
   308→    try {
   309→      const result = await fetchRef.current({ limit: pageSize });
   310→      if (seq.current !== mySeq) return;
   311→      setEntries(result.items);
   312→      setNextCursor(result.nextCursor);
   313→    } catch (cause) {
   314→      if (seq.current !== mySeq) return;
   315→      setError(fetchErrorMessage(cause));
   316→    } finally {
   317→      if (seq.current === mySeq) setLoading(false);
   318→    }
   319→  }, [pageSize]);
   320→
   321→  const loadMore = useCallback((): void => {
   322→    if (inFlight.current) return;
   323→    if (loading || loadingMore) return;
   324→    const cursor = nextCursor;
   325→    if (cursor === null) return;
   326→    inFlight.current = true;
   327→    seq.current += 1;
   328→    const mySeq = seq.current;
   329→    setLoadingMore(true);
   330→    setError(null);
   331→    void (async () => {
   332→      try {
   333→        const result = await fetchRef.current({ cursor, limit: pageSize });
   334→        if (seq.current !== mySeq) return;
   335→        setEntries((current) => [...current, ...result.items]);
   336→        setNextCursor(result.nextCursor);
   337→      } catch (cause) {
   338→        if (seq.current !== mySeq) return;
   339→        setError(fetchErrorMessage(cause));
   340→      } finally {
   341→        inFlight.current = false;
   342→        if (seq.current === mySeq) setLoadingMore(false);
   343→      }
   344→    })();
   345→  }, [loading, loadingMore, nextCursor, pageSize]);
   346→
   347→  const retry = useCallback((): void => {
   348→    if (entries.length === 0) void fetchFirst();
   349→    else loadMore();
   350→  }, [entries.length, fetchFirst, loadMore]);
   351→
   352→  const patchEntry = useCallback((next: RenderJob): void => {
   353→    setEntries((current) => current.map((row) => (row.id === next.id ? next : row)));
   354→  }, []);
   355→
   356→  const mergeFirstPage = useCallback(async (): Promise<void> => {
   357→    if (mergeInFlight.current) return;
   358→    mergeInFlight.current = true;
   359→    try {
   360→      const result = await fetchRef.current({ limit: pageSize });
   361→      setEntries((current) => {
   362→        const next = [...current];
   363→        const indexById = new Map(next.map((row, index) => [row.id, index] as const));
   364→        const unseen: RenderJob[] = [];
   365→        let changed = false;
   366→        for (const row of result.items) {
   367→          const at = indexById.get(row.id);
   368→          if (at === undefined) {
   369→            unseen.push(row);
   370→          } else if (next[at] !== row) {
   371→            next[at] = row;
   372→            changed = true;
   373→          }
   374→        }
   375→        if (unseen.length === 0) return changed ? next : current;
   376→        return [...unseen, ...next];
   377→      });
   378→    } catch {
   379→      /* Poll failures are silent (the presence poller posture): the last
   380→       * list stays rendered and the next tick retries. */
   381→    } finally {
   382→      mergeInFlight.current = false;
   383→    }
   384→  }, [pageSize]);
   385→
   386→  useEffect(() => {
   387→    void fetchFirst();
   388→  }, [fetchFirst]);
   389→
   390→  return {
   391→    entries,
   392→    nextCursor,
   393→    loading,
   394→    loadingMore,
   395→    error,
   396→    loadMore,
   397→    retry,
   398→    reload: fetchFirst,
   399→    patchEntry,
   400→    mergeFirstPage,
   401→  };
   402→}

   404→// ---------------------------------------------------------------------------
   405→// useRendersPolling — the active-row refresh (presence-cadence recipe)
   406→// ---------------------------------------------------------------------------
   407→
   408→/** UI refresh cadence while renders are active (not the worker's poll). */
   409→export const RENDERS_POLL_INTERVAL_MS = 5_000;
   410→
   411→/**
   412→ * Re-fetches the first page on an interval WHILE `enabled` (any loaded row
   413→ * queued|running for this project) and the tab is visible. Mirrors the M7-B
   414→ * presence cadence recipe: `visibilitychange` → hidden PAUSES the timer;
   415→ * → visible ticks IMMEDIATELY, then resumes; an in-flight guard collapses
   416→ * overlapping ticks; unmount cancels. Never global — the enabled flag is
   417→ * per-project and the Radix TabsContent unmounts inactive panels.
   418→ */
   419→export function useRendersPolling(options: {
   420→  readonly enabled: boolean;
   421→  readonly intervalMs?: number;
   422→  readonly refresh: () => Promise<void>;
   423→}): void {
   424→  const enabled = options.enabled;
   425→  const intervalMs = options.intervalMs ?? RENDERS_POLL_INTERVAL_MS;
   426→  const refreshRef = useRef(options.refresh);
   427→  refreshRef.current = options.refresh;
   428→
   429→  useEffect(() => {
   430→    if (!enabled) return;
   431→    let cancelled = false;
   432→    let inFlight = false;
   433→    let timer: ReturnType<typeof setInterval> | null = null;
   434→
   435→    const tick = async (): Promise<void> => {
   436→      if (cancelled || inFlight) return;
   437→      inFlight = true;
   438→      try {
   439→        await refreshRef.current();
   440→      } finally {
   441→        inFlight = false;
   442→      }
   443→    };
   444→
   445→    const start = (): void => {
   446→      if (timer !== null) return; // already ticking
   447→      void tick();
   448→      timer = setInterval(() => void tick(), intervalMs);
   449→    };
   450→    const stop = (): void => {
   451→      if (timer !== null) {
   452→        clearInterval(timer);
   453→        timer = null;
   454→      }
   455→    };
   456→
   457→    if (document.visibilityState === "visible") start();
   458→
   459→    const onVisibility = (): void => {
   460→      if (document.visibilityState === "hidden") stop();
   461→      else start();
   462→    };
   463→    document.addEventListener("visibilitychange", onVisibility);
   464→    return () => {
   465→      cancelled = true;
   466→      stop();
   467→      document.removeEventListener("visibilitychange", onVisibility);
   468→    };
   469→  }, [enabled, intervalMs]);
   470→}

   556→              style={{ width: `${Math.min(100, Math.max(0, job.progress.percent))}%` }}
   557→            />
   558→          </div>
   559→          <p className="tabular-nums text-caption text-muted">
   560→            {job.progress.framesTotal === 0
   561→              ? job.status === "queued"
   562→                ? "Waiting in the queue…"
   563→                : "Preparing frames…"
   564→              : `${job.progress.framesDone} of ${job.progress.framesTotal} frames · ${job.progress.percent}%`}
   565→          </p>
   566→        </div>
   567→      ) : null}
   568→
   569→      {job.status === "failed" ? (
   570→        <p
   571→          role="alert"
   572→          data-testid={`renders-hub-row-${job.id}-error`}
   573→          className="flex flex-col gap-0.5 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   574→        >
   575→          {job.error !== null && job.error.length > 0 ? (
   576→            <span>{truncateTimelineString(job.error, MAX_RENDER_TEXT)}</span>
   577→          ) : null}
   578→          {codeCopy !== null ? <span className="text-muted">{codeCopy}</span> : null}
   579→          {job.error === null && codeCopy === null ? (
   580→            <span className="text-muted">
   581→              This render failed. Queue it again when you're ready.
   582→            </span>
   583→          ) : null}
   584→        </p>
   585→      ) : null}
   586→
   587→      <div className="flex flex-wrap items-center gap-x-3 gap-y-1">
   588→        <span className="text-caption text-muted" data-testid={`renders-hub-row-${job.id}-version`}>
   589→          {versionLabel}
   590→        </span>
   591→        {expiry.text !== null ? (
   592→          <span
   593→            className="text-caption text-muted"
   594→            data-state={expiry.state}
   595→            data-testid={`renders-hub-row-${job.id}-expires`}
   596→          >
   597→            {expiry.text}
   598→          </span>
   599→        ) : null}
   600→        {job.sizeBytes !== null ? (
   601→          <span
   602→            className="tabular-nums text-caption text-muted"
   603→            data-testid={`renders-hub-row-${job.id}-size`}
   604→          >
   605→            {formatVersionBytes(job.sizeBytes)}
   606→          </span>
   607→        ) : null}
   608→        {job.cancelRequested && active ? (
   609→          <span
   610→            className="text-caption text-muted"
   611→            data-testid={`renders-hub-row-${job.id}-cancelling`}
   612→          >
   613→            Cancelling…
   614→          </span>
   615→        ) : null}
   616→      </div>
   617→
   618→      <div className="flex flex-wrap items-center gap-2">
   619→        {downloadable ? (
   620→          <Button
   621→            variant="secondary"
   622→            size="sm"
   623→            onClick={() => onDownload(job)}
   624→            data-testid={`renders-hub-row-${job.id}-download`}
   625→          >
   626→            <Download size={14} strokeWidth={1.75} aria-hidden="true" />
   627→            Download
   628→          </Button>
   629→        ) : null}
   630→        {canQueue && active ? (
   631→          <Button
   632→            variant="ghost"
   633→            size="sm"
   634→            loading={cancelBusy}
   635→            disabled={cancelBusy || job.cancelRequested}
   636→            onClick={() => onCancel(job)}
   637→            data-testid={`renders-hub-row-${job.id}-cancel`}
   638→          >
   639→            <XCircle size={14} strokeWidth={1.75} aria-hidden="true" />
   640→            Cancel
   641→          </Button>
   642→        ) : null}
   643→      </div>
   644→    </li>
   645→  );
   646→}

   652→/** One renderable composition for the queue card's picker. */
   653→export interface HubCompOption {
   654→  readonly id: string;
   655→  readonly name: string;
   656→}
   657→
   658→export interface RendersHubPanelProps {
   659→  readonly projectId: string;
   660→  readonly workspaceId: string;
   661→  /** Editor+ roles queue renders (the gateway re-checks; viewer+ reads). */
   662→  readonly canQueue: boolean;
   663→  /** ISO instant stamped by the RSC — pins the expiry math (deterministic). */
   664→  readonly nowIso: string;
   665→  /** Deep link to the editor (empty-state + queue-card fallbacks). */
   666→  readonly editorHref: string;
   667→  /** Host-known compositions (id + display name); history rows add more. */
   668→  readonly comps?: readonly HubCompOption[];
   669→  /** Injectable port (tests). Production leaves it undefined. */
   670→  readonly api?: RendersHubPort;
   671→  /** URL opener seam for completed downloads (tests spy it). */
   672→  readonly openUrl?: (url: string) => void;
   673→  /** Poll cadence override while renders are active (tests). */
   674→  readonly pollIntervalMs?: number;
   675→}
   676→
   677→export function RendersHubPanel({
   678→  projectId,
   679→  workspaceId,
   680→  canQueue,
   681→  nowIso,
   682→  editorHref,
   683→  comps,
   684→  api,
   685→  openUrl,
   686→  pollIntervalMs,
   687→}: RendersHubPanelProps): ReactElement {
   688→  const port = api ?? createRendersHubPort(workspaceId, projectId);
   689→  const open = openUrl ?? ((url: string) => window.open(url, "_blank", "noopener,noreferrer"));
   690→
   691→  const pages = useRendersPages(port.listRenders);
   692→  const [cancelBusyId, setCancelBusyId] = useState<string | null>(null);
   693→  const [actionError, setActionError] = useState<RendersActionError | null>(null);
   694→  const [downloadError, setDownloadError] = useState<string | null>(null);
   695→  const [versionNumbers, setVersionNumbers] = useState<ReadonlyMap<string, number> | null>(null);
   696→
   697→  // Version pin labels: resolve the server-pinned versionId to a human
   698→  // "v{N}" through the §M7.4 timeline; any failure degrades silently to the
   699→  // neutral "Pinned snapshot" copy (labels never block the list).
   700→  useEffect(() => {
   701→    let cancelled = false;
   702→    void (async () => {
   703→      try {
   704→        const page = await port.listVersions({ limit: VERSION_LABEL_LIMIT });
   705→        if (cancelled) return;
   706→        const map = new Map<string, number>();
   707→        for (const entry of page.items) map.set(entry.id, entry.version);
   708→        setVersionNumbers(map);
   709→      } catch {
   710→        /* degrade to the neutral pin copy */
   711→      }
   712→    })();
   713→    return () => {
   714→      cancelled = true;
   715→    };
   716→  }, [port]);
   717→
   718→  // Poll only while THIS project has queued/running rows (never global).
   719→  useRendersPolling({
   720→    enabled: pages.entries.some(isActiveRender),
   721→    intervalMs: pollIntervalMs,
   722→    refresh: pages.mergeFirstPage,
   723→  });
   724→
   725→  const runCancel = useCallback(
   726→    (job: RenderJob): void => {
   727→      setCancelBusyId(job.id);
   728→      setActionError(null);
   729→      void (async () => {
   730→        try {
   731→          const next = await port.cancelRender(job.id);
   732→          pages.patchEntry(next);
   733→        } catch (cause) {
   734→          setActionError(toActionError(cause));
   735→        } finally {
   736→          setCancelBusyId(null);
   737→        }
   738→      })();
   739→    },
   740→    [pages, port],
   741→  );
   742→
   743→  const runDownload = useCallback(
   744→    (job: RenderJob): void => {
   745→      setDownloadError(null);
   746→      const url = job.resultUrl;
   747→      if (typeof url !== "string" || !(url.startsWith("http://") || url.startsWith("https://"))) {
   748→        setDownloadError("The download link could not be opened.");
   749→        return;
   750→      }
   751→      open(url);
   752→    },
   753→    [open],
   754→  );
   755→
   756→  // Queue-card options: host-known comps first, then every comp this project
   757→  // has already rendered (the M7-F `Comp <id>` house fallback — compId is
   758→  // engine data, never a secret, never raw UI copy).
   759→  const compOptions = useMemo(() => {
   760→    const byId = new Map<string, HubCompOption>();
   761→    for (const comp of comps ?? []) byId.set(comp.id, comp);
   762→    for (const job of pages.entries) {
   763→      if (!byId.has(job.compId))
   764→        byId.set(job.compId, { id: job.compId, name: `Comp ${job.compId}` });
   765→    }
   766→    return [...byId.values()];
   767→  }, [comps, pages.entries]);
   768→
   769→  const canLoadMore = pages.nextCursor !== null;
   770→  const atEnd = !pages.loading && pages.nextCursor === null && pages.entries.length > 0;
   771→
   772→  return (
   773→    <section
   774→      data-testid="renders-hub-panel"
   775→      aria-label="Project renders"
   776→      className="flex flex-col gap-4"
   777→    >
   778→      {canQueue ? (
   779→        <QueueRenderCard
   780→          projectId={projectId}
   781→          comps={compOptions}
   782→          editorHref={editorHref}
   783→          api={port}
   784→          onQueued={() => void pages.reload()}
   785→        />
   786→      ) : (
   787→        <p
   788→          data-testid="renders-hub-queue-readonly"
   789→          className="rounded-sm border border-border bg-elevated/40 px-3 py-2 text-caption text-secondary"
   790→        >
   791→          You need edit access to queue renders. Everyone can watch the history below.
   792→        </p>
   793→      )}
   794→
   795→      <p className="text-caption text-muted" data-testid="renders-hub-list-note">
   796→        Every final output queued for this project, newest first.
   797→      </p>
   798→
   799→      {pages.error !== null ? (
   800→        <div
   801→          role="alert"
   802→          data-testid="renders-hub-error"
   803→          className="flex items-center gap-2 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   804→        >
   805→          <CircleAlert
   806→            size={16}
   807→            strokeWidth={1.75}
   808→            aria-hidden="true"
   809→            className="shrink-0 text-danger"
   810→          />
   811→          <p className="min-w-0 flex-1">{pages.error}</p>
   812→          <Button
   813→            variant="secondary"
   814→            size="sm"
   815→            data-testid="renders-hub-error-retry"
   816→            disabled={pages.loading}
   817→            onClick={pages.retry}
   818→          >
   819→            Try again
   820→          </Button>
   821→        </div>
   822→      ) : null}
   823→
   824→      {actionError !== null ? (
   825→        <div
   826→          role="alert"
   827→          data-testid="renders-hub-action-error"
   828→          className="flex items-start gap-2 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   829→        >
   830→          <CircleAlert
   831→            size={16}
   832→            strokeWidth={1.75}
   833→            aria-hidden="true"
   834→            className="mt-0.5 shrink-0 text-danger"
   835→          />
   836→          <span className="min-w-0 flex-1">
   837→            {errorLine(actionError)}
   838→            {actionError.hint === undefined ? null : (
   839→              <span className="block text-muted">
   840→                {truncateTimelineString(actionError.hint, MAX_RENDER_TEXT)}
   841→              </span>
   842→            )}
   843→          </span>
   844→        </div>
   845→      ) : null}
   846→
   847→      {downloadError !== null ? (
   848→        <p
   849→          role="alert"
   850→          data-testid="renders-hub-download-error"
   851→          className="flex items-center gap-2 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   852→        >
   853→          <CircleAlert
   854→            size={16}
   855→            strokeWidth={1.75}
   856→            aria-hidden="true"
   857→            className="shrink-0 text-danger"
   858→          />
   859→          {downloadError}
   860→        </p>
   861→      ) : null}
   862→
   863→      {pages.loading && pages.entries.length === 0 ? (
   864→        <div className="flex flex-col gap-3 py-2" data-testid="renders-hub-loading">
   865→          <Skeleton className="h-4 w-28" />
   866→          <Skeleton className="h-6 w-full max-w-[560px]" />
   867→          <Skeleton className="h-6 w-full max-w-[520px]" />
   868→          <Skeleton className="h-6 w-full max-w-[540px]" />
   869→        </div>
   870→      ) : pages.entries.length === 0 ? (
   871→        <EmptyState
   872→          data-testid="renders-hub-empty"
   873→          title="No renders yet."
   874→          secondaryLink={{ label: "Open the editor", href: editorHref }}
   875→        />
   876→      ) : (
   877→        <ol
   878→          data-testid="renders-hub-list"
   879→          className="divide-y divide-border/60"
   880→          aria-label="Render history"
   881→        >
   882→          {pages.entries.map((job) => (
   883→            <RendersRow
   884→              key={job.id}
   885→              job={job}
   886→              nowIso={nowIso}
   887→              versionLabel={versionLabelFor(versionNumbers, job.versionId)}
   888→              canQueue={canQueue}
   889→              cancelBusy={cancelBusyId === job.id}
   890→              onDownload={runDownload}
   891→              onCancel={runCancel}
   892→            />
   893→          ))}
   894→        </ol>
   895→      )}
   896→
   897→      {canLoadMore || atEnd ? (
   898→        <div className="flex items-center justify-center gap-3">
   899→          {canLoadMore ? (
   900→            <Button
   901→              variant="secondary"
   902→              size="sm"
   903→              data-testid="renders-hub-load-more"
   904→              loading={pages.loadingMore}
   905→              disabled={pages.loading || pages.loadingMore}
   906→              onClick={() => pages.loadMore()}
   907→            >
   908→              Load more
   909→            </Button>
   910→          ) : (
   911→            <p data-testid="renders-hub-end" className="text-caption text-muted">
   912→              That’s every render so far.
   913→            </p>
   914→          )}
   915→        </div>
   916→      ) : null}
   917→    </section>
   918→  );
   919→}

   921→/** "Snapshot v{N}" once the timeline resolves the pin; neutral otherwise. */
   922→function versionLabelFor(
   923→  versionNumbers: ReadonlyMap<string, number> | null,
   924→  versionId: string,
   925→): string {
   926→  const number = versionNumbers?.get(versionId);
   927→  return number === undefined ? "Pinned snapshot" : `Snapshot v${number}`;
   928→}
