   104→function frame(data: unknown): string {
   105→  return `data: ${JSON.stringify(data)}\n\n`;

   107→function textDelta(content: string): string {

   110→function toolCallStart(index: number, id: string, name: string): string {
   111→  return frame({
   112→    choices: [

   137→class FakeOpenAIServer {
   138→  private readonly queue: ScriptEntry[] = [];
   139→  private server: Server | null = null;
   140→  url = "";
   141→  readonly calls: Array<{ body: ChatCompletionBody; authorization: string }> = [];
   142→
   143→  enqueue(...entries: ScriptEntry[]): void {
   144→    this.queue.push(...entries);
   145→  }
   146→
   147→  reset(): void {
   148→    this.queue.length = 0;
   149→    this.calls.length = 0;
   150→  }
   151→
   152→  async start(): Promise<void> {
   153→    const server = createServer((request, response) => {
   154→      void this.handle(request, response);
   155→    });
   156→    this.server = server;
   157→    await new Promise<void>((resolve) => {
   158→      server.listen(0, "127.0.0.1", resolve);
   159→    });
   160→    const address = server.address() as AddressInfo;
   161→    this.url = `http://127.0.0.1:${address.port}`;
   162→  }
   163→
   164→  async stop(): Promise<void> {
   165→    await new Promise<void>((resolve) => {
   166→      const server = this.server;
   167→      if (server === null) {
   168→        resolve();
   169→        return;
   170→      }
   171→      server.close(() => resolve());
   172→    });
   173→  }
   174→
   175→  private async handle(request: IncomingMessage, response: ServerResponse): Promise<void> {
   176→    response.on("error", () => undefined);
   177→    request.on("error", () => undefined);
   178→    const chunks: Buffer[] = [];
   179→    for await (const chunk of request) {
   180→      chunks.push(chunk as Buffer);
   181→      if (response.destroyed) return;
   182→    }
   183→    let body: ChatCompletionBody = { model: "", messages: [] };
   184→    try {
   185→      body = JSON.parse(Buffer.concat(chunks).toString("utf8")) as ChatCompletionBody;
   186→    } catch {
   187→      body = { model: "", messages: [] };
   188→    }
   189→    this.calls.push({ body, authorization: String(request.headers.authorization ?? "") });
   190→    const next = this.queue.shift();
   191→    if (next === undefined) {
   192→      response.writeHead(500, { "content-type": "application/json" });
   193→      response.end(JSON.stringify({ error: { message: "fake script exhausted" } }));
   194→      return;
   195→    }
   196→    response.writeHead(200, {
   197→      "content-type": "text/event-stream; charset=utf-8",
   198→      "cache-control": "no-store",
   199→    });
   200→    for (const wireFrame of next(body)) {
   201→      if (response.destroyed) return;
   202→      response.write(encoder.encode(wireFrame));
   203→      await new Promise<void>((resolveTick) => {
   204→        setImmediate(resolveTick);
   205→      });
   206→    }
   207→    response.end();
   208→  }
   209→}

   228→// ---------------------------------------------------------------------------
   229→// SSE wire reader (light: §7.4 schema-valid events + contiguous seq + one
   230→// terminal — the BYTE contract is chat-loop scenario 8's turf, not repeated).
   231→// ---------------------------------------------------------------------------
   232→
   233→/** Read every §7.4 event off the SSE wire through clean EOF (never cancels —
   234→ * §7.4 disconnect semantics cancel the live run; the server ends the stream
   235→ * after the terminal event, so a clean EOF read is race-free). */
   236→async function readAllEvents(response: Response): Promise<StudioStreamEvent[]> {
   237→  const body = response.body;
   238→  if (body === null) throw new Error("chat response had no SSE body");
   239→  const reader = body.getReader();
   240→  const decoder = new TextDecoder();
   241→  let buffer = "";
   242→  const events: StudioStreamEvent[] = [];
   243→  const deadline = Date.now() + 15_000;
   244→  for (;;) {
   245→    const boundary = buffer.indexOf("\n\n");
   246→    if (boundary !== -1) {
   247→      const raw = buffer.slice(0, boundary);
   248→      buffer = buffer.slice(boundary + 2);
   249→      const dataLine = raw.split("\n").find((line) => line.startsWith("data: "));
   250→      if (dataLine === undefined) continue;
   251→      const parsed = JSON.parse(dataLine.slice("data: ".length));
   252→      const checked = STUDIO_STREAM_EVENT_SCHEMA.safeParse(parsed);
   253→      if (!checked.success) {
   254→        throw new Error(`wire event failed the §7.4 schema: ${checked.error.message}`);
   255→      }
   256→      events.push(checked.data);
   257→      continue;
   258→    }
   259→    if (Date.now() > deadline) throw new Error("timed out reading the SSE wire");
   260→    const read = await reader.read();
   261→    if (read.done) break;
   262→    buffer += decoder.decode(read.value, { stream: true });
   263→  }
   264→  return events;
   265→}
