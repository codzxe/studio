    19→/**
    20→ * Studio schema (Postgres, drizzle-orm) — PRODUCT_ARCHITECTURE §9.1 (all tables),
    21→ * §9.3 (versions/conflict), §8.8 (credit ledger) + Better Auth 1.7 core/org/api-key/admin
    22→ * plugin tables, verified against `getSchema()` of the installed better-auth.
    23→ *
    24→ * Conventions:
    25→ * - text primary keys (`${prefix}_${uuid}`) generated app-side; timestamps with
    26→ *   time zone (JS Date mode).
    27→ * - Column names are snake_case; TS property names are camelCase.
    28→ *   EXCEPTION: Better Auth tables — property names MUST match the plugin model
    29→ *   field names exactly (the drizzle adapter addresses columns by property name).
    30→ * - Tenant scoping: content tables carry `workspace_id` (or scope through
    31→ *   `project_id` -> projects.workspace_id). Cross-tenant access must read as
    32→ *   "not found" (§12.1) — enforced in `src/repos/*`, not here.
    33→ *
    34→ * Decisions (documented for reviewers):
    35→ * - `project_versions` is NOT unique on (project_id, sha256): re-saving identical
    36→ *   content is a legitimate action and §9.3 forbids rewriting history; the
    37→ *   object store dedupes the blob itself (§9.2 content-addressed). The composite
    38→ *   index exists for fast "is this content already stored" lookups.
    39→ * - `credit_ledger` is append-only by convention; balance is SUM(amount) under a
    40→ *   per-workspace advisory lock (see src/repos/ledger.ts).
    41→ * - `renders`, `comments` and `agent_*` scope through `project_id` per §9.1; the
    42→ *   repos join projects to enforce tenancy.
    43→ * - `projects.current_version_id` and `project_versions.parent_version_id` are
    44→ *   plain text columns WITHOUT FK constraints: they form a circular reference
    45→ *   pair with `project_versions.project_id`, and both are written only inside
    46→ *   repo transactions that hold the project row lock (src/repos/versions.ts),
    47→ *   so integrity is enforced where the conflict rules live (§9.3) without
    48→ *   making schema-level DELETE semantics impossible.
    49→ * - M5 apply/undo (§M5.4/§M5.5): the plan lifecycle (proposed → applied →
    50→ *   undone) lives on `agent_runs` — `applied_at`, `applied_batch_sha256`,
    51→ *   `applied_version_id`, `undone_at`, `undone_by` — because `project_versions`
    52→ *   is IMMUTABLE (§9.3): applying creates ONE version, undoing marks the run
    53→ *   and NEVER deletes or rewrites a version row. Unlike the circular pair
    54→ *   above, `agent_runs.base_version_id` / `applied_version_id` DO carry FKs to
    55→ *   `project_versions.id` (no cycle: `project_versions.agent_run_id` stays a
    56→ *   plain-text back-reference). All three FKs are `ON DELETE SET NULL` so a
    57→ *   cascading project delete never trips over cross-reference ordering
    58→ *   (mirrors `audit_log.actor_id` / `assets.project_id`). Same-project scope is
    59→ *   enforced in repos (a version of another project reads as NOT_FOUND).
    60→ */

   489→// ─────────────────────────────────────────────────────────────────────────────
   490→// Section 7 — Credit ledger (§8.8, §10.2) — append-only
   491→//   grant   +N   purchase / plan allowance / top-up pack
   492→//   adjust  ±N   manual corrections, storage overage, promos (may be negative)
   493→//   reserve −N   hold at run start (pessimistic: balance drops immediately)
   494→//   commit  −M   actual cost at settle time (bounded by the run's hold)
   495→//   refund  +R   release of an unspent hold (full refund on failure/cancel)
   496→// balance = SUM(amount); writers serialize per workspace via pg_advisory_xact_lock.
   497→// ─────────────────────────────────────────────────────────────────────────────
   498→
   499→export type LedgerKind = "grant" | "reserve" | "commit" | "refund" | "adjust";
   500→
   501→export const creditLedger = pgTable(
   502→  "credit_ledger",
   503→  {
   504→    id: text("id")
   505→      .primaryKey()
   506→      .$defaultFn(() => createId("led")),
   507→    workspaceId: text("workspace_id")
   508→      .notNull()
   509→      .references((): AnyPgColumn => organization.id, { onDelete: "cascade" }),
   510→    kind: text("kind").$type<LedgerKind>().notNull(),
   511→    amount: integer("amount").notNull(), // signed
   512→    balanceAfter: integer("balance_after").notNull(),
   513→    runId: text("run_id"),
   514→    idempotencyKey: text("idempotency_key").notNull(),
   515→    metadata: jsonb("metadata"),
   516→    createdAt: ts("created_at").notNull().defaultNow(),
   517→  },
   518→  (t) => [
