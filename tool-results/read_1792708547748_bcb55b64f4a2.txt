   268→function defaultSleep(ms: number): Promise<void> {
   269→  if (ms <= 0) return Promise.resolve();
   270→  return new Promise((resolve) => {
   271→    setTimeout(resolve, ms);
   272→  });
   273→}

   275→/** Serialized query string; `undefined`/`null` values are skipped. */
   276→export function buildQuery(
   277→  params: Record<string, string | number | boolean | null | undefined>,
   278→): string {
   279→  const search = new URLSearchParams();
   280→  for (const [key, value] of Object.entries(params)) {
   281→    if (value === undefined || value === null) continue;
   282→    search.set(key, String(value));
   283→  }
   284→  const text = search.toString();
   285→  return text === "" ? "" : `?${text}`;
   286→}

   288→interface RequestPlan<T> {

   305→interface ClientInternals {
   306→  readonly baseUrl: string;
   307→  readonly fetchImpl: FetchLike;
   308→  readonly extraHeaders: Record<string, string>;
   309→  readonly newKey: IdempotencyKeyFactory;
   310→  readonly sleep: (ms: number) => Promise<void>;
   311→  readonly retryDelays: readonly number[];
   312→}

   314→function encodeId(id: string): string {
   315→  return encodeURIComponent(id);
   316→}
   317→
   318→class ApiClientImpl {
   319→  constructor(readonly core: ClientInternals) {}
   320→
   321→  /**
   322→   * One HTTP call: envelope unwrap, response validation, GET retry with
   323→   * backoff. Never surfaces raw response text — errors are typed ApiErrors.
   324→   */
   325→  async request<T>(plan: RequestPlan<T>): Promise<T> {
   326→    const url = `${this.core.baseUrl}${plan.path}${plan.query ? buildQuery(plan.query) : ""}`;
   327→
   328→    let attempt = 0;
   329→    for (;;) {
   330→      let status = 0;
   331→      let statusText = "";
   332→      let text = "";
   333→      try {
   334→        const init = this.buildInit(plan);
   335→        const response = await this.core.fetchImpl(url, init);
   336→        status = response.status;
   337→        statusText = response.statusText;
   338→        text = await response.text();
   339→      } catch (error) {
   340→        if (isAbortError(error) || plan.signal?.aborted) throw error;
   341→        const netError = networkError(error);
   342→        if (!this.shouldRetry(plan, netError, attempt)) throw netError;
   343→        await this.backoff(plan, attempt);
   344→        attempt += 1;
   345→        continue;
   346→      }
   347→
   348→      if (status < 200 || status > 299) {
   349→        const apiError = await fromResponse({
   350→          status,
   351→          statusText,
   352→          text: async () => text,
   353→        });
   354→        if (!this.shouldRetry(plan, apiError, attempt)) throw apiError;
   355→        await this.backoff(plan, attempt);
   356→        attempt += 1;
   357→        continue;
   358→      }
   359→
   360→      return this.unwrapOk<T>(status, text, plan);
   361→    }
   362→  }
   363→
   364→  private buildInit(plan: RequestPlan<unknown>): RequestInit {
   365→    const headers = new Headers(this.core.extraHeaders);
   366→    headers.set("Accept", "application/json");
   367→    if (plan.body !== undefined) headers.set("Content-Type", "application/json");
   368→    if (plan.idempotencyKey !== undefined) headers.set("Idempotency-Key", plan.idempotencyKey);
   369→    if (plan.headers !== undefined) {
   370→      for (const [name, value] of Object.entries(plan.headers)) headers.set(name, value);
   371→    }
   372→    const init: RequestInit = {
   373→      method: plan.method,
   374→      headers,
   375→      signal: plan.signal,
   376→    };
   377→    if (plan.body !== undefined) init.body = JSON.stringify(plan.body);
   378→    return init;
   379→  }
   380→
   381→  /** Only idempotent-safe GETs retry, and only when the failure is retryable. */
   382→  private shouldRetry(plan: RequestPlan<unknown>, error: ApiError, attempt: number): boolean {
   383→    if (plan.method !== "GET") return false;
   384→    if (attempt >= this.core.retryDelays.length) return false;
   385→    if (error.status === 0) return true; // network error before any response
   386→    if (error.status === 408 || error.status === 429 || error.status >= 500) {
   387→      return error.retryable;
   388→    }
   389→    return false;
   390→  }
   391→
   392→  private backoff(plan: RequestPlan<unknown>, attempt: number): Promise<void> {
   393→    const delay = this.core.retryDelays[attempt] ?? 0;
   394→    return sleepWithAbort(this.core.sleep, delay, plan.signal);
   395→  }
   396→
   397→  /** 2xx handling: empty body → `null`; otherwise a §7.3 Ok envelope is required. */
   398→  private unwrapOk<T>(status: number, text: string, plan: RequestPlan<T>): T {
   399→    if (text === "") {
   400→      if (plan.responseSchema === undefined) return null as T;
   401→      throw malformedResponseError(status);
   402→    }
   403→    let json: unknown;
   404→    try {
   405→      json = JSON.parse(text);
   406→    } catch {
   407→      throw malformedResponseError(status);
   408→    }
   409→    if (isErr(json)) {
   410→      const error: Err["error"] = json.error;
   411→      // A 2xx status with an Err body is still the §7.3 failure arm.
   412→      throw new ApiError({
   413→        code: error.code,
   414→        message: error.message,
   415→        hint: error.hint,
   416→        retryable: error.retryable,
   417→        requestId: error.requestId,
   418→        status,
   419→        cause: json,
   420→      });
   421→    }
   422→    if (!isOk(json)) throw malformedResponseError(status);
   423→    const data = json.data;
   424→    if (plan.responseSchema === undefined) return data as T;
   425→    try {
   426→      return plan.responseSchema.parse(data);
   427→    } catch (error) {
   428→      throw malformedResponseError(status, error);
   429→    }
   430→  }
   431→
   432→  /** Client-side §7.2 request validation — fails fast, never sends. */
   433→  validateBody<T>(schema: z.ZodType<T>, body: unknown, path: string): T {
   434→    try {
   435→      return schema.parse(body);
   436→    } catch (error) {
   437→      if (isApiError(error)) throw error;
   438→      throw invalidRequestError(
   439→        Object.assign(new Error(`invalid body for ${path}`), { cause: error }),
   440→      );
   441→    }
   442→  }
   443→
   444→  autoKey(): string {
   445→    return this.core.newKey();
   446→  }
   447→
   448→  // -- GETs don't send Idempotency-Key; mutations always do (auto when absent).
   449→  async get<T>(
   450→    path: string,
   451→    query: Record<string, string | number | boolean | undefined> | undefined,
   452→    schema: z.ZodType<T>,
   453→    signal?: AbortSignal,
   454→    headers?: Record<string, string>,
   455→  ): Promise<T> {
   456→    return this.request<T>({
   457→      method: "GET",
   458→      path,
   459→      query,
   460→      responseSchema: schema,
   461→      signal,
   462→      headers,
   463→    });
   464→  }
   465→
   466→  async mutate<T>(
   467→    method: "POST" | "PATCH" | "DELETE",
   468→    path: string,
   469→    plan: {
   470→      body?: unknown;
   471→      bodySchema?: z.ZodType<unknown>;
   472→      responseSchema?: z.ZodType<T>;
   473→      idempotencyKey?: string;
   474→      signal?: AbortSignal;
   475→      headers?: Record<string, string>;
   476→    },
   477→  ): Promise<T> {
   478→    const body =
   479→      plan.bodySchema !== undefined
   480→        ? this.validateBody(plan.bodySchema, plan.body, path)
   481→        : plan.body;
   482→    return this.request<T>({
   483→      method,
   484→      path,
   485→      body,
   486→      responseSchema: plan.responseSchema,
   487→      idempotencyKey: plan.idempotencyKey ?? this.autoKey(),
   488→      signal: plan.signal,
   489→      headers: plan.headers,
   490→    });
   491→  }
   492→}

   802→/**
   803→ * Builds the typed API client. `baseUrl` includes the version segment
   804→ * (default `"/v1"` for same-origin browser use). All mutation methods send
   805→ * `Idempotency-Key` (§7.2) — pass your own per logical operation, or let the
   806→ * client derive one.
   807→ */
   808→export function createApiClient(options: ApiClientOptions = {}): ApiClient {
   809→  const core: ClientInternals = {
   810→    baseUrl: (options.baseUrl ?? DEFAULT_BASE_URL).replace(/\/+$/, ""),
   811→    fetchImpl: options.fetch ?? ((input, init) => fetch(input, init)),
   812→    extraHeaders: { ...(options.headers ?? {}) },
   813→    newKey: options.idempotencyKey ?? (() => crypto.randomUUID()),
   814→    sleep: options.sleep ?? defaultSleep,

   816→  };
   817→  const impl = new ApiClientImpl(core);
   818→
   819→  const mutationKey = (options: CallOptions | undefined): string | undefined => {
   820→    const explicit = options?.idempotencyKey;
   821→    if (explicit !== undefined) {
   822→      // §7.2: client-generated, ≥8 chars (contracts IDEMPOTENCY_KEY_SCHEMA).
   823→      return impl.validateBody(IDEMPOTENCY_KEY_SCHEMA, explicit, "Idempotency-Key");
   824→    }
   825→    return undefined;
   826→  };
   827→
   828→  const queryOf = (
   829→    schema: z.ZodType<unknown>,
   830→    query: unknown,
   831→  ): Record<string, string | number | boolean | undefined> => {
   832→    const parsed = impl.validateBody(schema, query ?? {}, "list query");
   833→    const out: Record<string, string | number | boolean | undefined> = {};
   834→    for (const [k, v] of Object.entries(parsed as Record<string, unknown>)) {
   835→      out[k] = v as string | number | boolean | undefined;
   836→    }
   837→    return out;

  1306→/** Two-phase version upload (§9.2/§9.3): descriptor POST + signed-URL byte PUT. */
  1307→async function uploadVersion(
  1308→  impl: ApiClientImpl,
  1309→  projectId: string,
  1310→  input: {
  1311→    readonly baseVersionId: string;
  1312→    readonly sha256: string;
  1313→    readonly label?: string;
  1314→    readonly agentRunId?: string;
  1315→    readonly bytes: Blob | Uint8Array;
  1316→  },
  1317→  options: CallOptions | undefined,
  1318→): Promise<ProjectVersion> {
  1319→  const { bytes } = input;
  1320→  if (!(bytes instanceof Blob) && !(bytes instanceof Uint8Array)) {
  1321→    throw invalidRequestError(new Error("bytes must be a Blob or Uint8Array"));
  1322→  }
  1323→  const sizeBytes = bytes instanceof Blob ? bytes.size : bytes.byteLength;
  1324→  const descriptor = impl.validateBody(
  1325→    UPLOAD_VERSION_SCHEMA,
  1326→    {
  1327→      baseVersionId: input.baseVersionId,
  1328→      sha256: input.sha256,
  1329→      label: input.label,
  1330→      agentRunId: input.agentRunId,
  1331→      sizeBytes,
  1332→    },
  1333→    "upload version descriptor",
  1334→  );
  1335→
  1336→  const response = await impl.mutate<UploadVersionResponse>(
  1337→    "POST",
  1338→    `/projects/${encodeId(projectId)}/versions`,
  1339→    {
  1340→      body: descriptor,
  1341→      responseSchema: UPLOAD_VERSION_RESPONSE_SCHEMA,
  1342→      idempotencyKey: options?.idempotencyKey ?? impl.autoKey(),
  1343→      signal: options?.signal,
  1344→    },
  1345→  );
  1346→
  1347→  // Phase two: bytes go straight to object storage via the signed URL — the
  1348→  // storage response is NOT a §7.3 envelope, so only the status is checked.
  1349→  const { url, headers } = response.upload;
  1350→  const put = await impl.core.fetchImpl(url, {
  1351→    method: "PUT",
  1352→    headers: { ...headers },
  1353→    // TS 5.7's DOM BodyInit doesn't accept Uint8Array<ArrayBufferLike>
  1354→    // (SharedArrayBuffer edge); runtime fetch (browser + undici) does.
  1355→    body: bytes instanceof Blob ? bytes : (bytes as unknown as BodyInit),
  1356→    signal: options?.signal,
  1357→  });
  1358→  if (!put.ok) {
  1359→    throw uploadFailedError(put.status);
  1360→  }
  1361→  return response.version;
  1362→}
