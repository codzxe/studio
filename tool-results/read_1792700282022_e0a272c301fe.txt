    33→// FINDINGS this corpus surfaces (report-only — law #12: nothing outside the

    35→// tool-growth rule). The golden-prompts.test.ts suite PINS today's actual
    36→// behavior, so the orchestrator's fixes flip these fixtures to full goldens
    37→// without touching fixture data.
    38→//
    39→// G-1 (HIGH, seam) — write-tool ARGS-SHAPE DIVERGENCE between the curated
    40→//   args schemas (M5-A, packages/contracts/src/write-tools.ts) and the
    41→//   compiler's per-tool translation (M5-B, services/gateway/src/plan-compiler.ts
    42→//   translate()):
    43→//     set_properties: schema `{layers:[{name}], properties:[{path,value}]}`  vs
    44→//                     compiler `{updates:[{layer,path,value}]}`
    45→//     set_keyframes:  schema `{layers:[{name}], propertyPath, keys:[{timeSeconds,
    46→//                     value, interpolation?}]}` vs compiler `{layer, propertyPath,
    47→//                     keyframes:[{time, value, easing?}]}`
    48→//     add_effect:     schema `{layers:[{name}], effect:{name}, params?}` vs
    49→//                     compiler `{layer, effect: <string>, params?}`
    50→//     apply_preset:   schema `{preset:{name}|{builtinId}, layers?, args?}` vs
    51→//                     compiler `{preset: <string>, targets:[<string>]}`
    52→//   Consequence (verified against the real pipeline): schema-valid model plans
    53→//   for these four tools translate to ZERO commands ⇒ the compile answers a
    54→//   typed `empty_batch` fatal ⇒ the plan event goes out UN-enriched (no
    55→//   compiledBatch ⇒ Apply is gated) and write-tool proposals answer
    56→//   PROPOSAL_INVALID. The failure is FAIL-CLOSED (never a partial or wrong
    57→//   batch for single-tool plans) but it blocks 6 of the 10 required golden
    58→//   scenarios today. Existing suites never caught this: plan-compiler.test.ts
    59→//   injects fixture args schemas shaped like the compiler, litellm-runner
    60→//   tests inject a stub compiler. Fix (one file, src/plan-compiler.ts):
    61→//   translate() must read the write-tools.ts arg names (layers/properties,
    62→//   layers/keys, layers+effect.name, preset.name|preset.builtinId+layers) —
    63→//   including the set_properties cross-product (N layers × M properties).
    64→//
    65→// G-2 (MEDIUM, silent drop) — create_text typography: CREATE_TEXT_ARGS_SCHEMA
    66→//   accepts font/style/size/fill/stroke/strokeWidth/tracking/leading/box, but
    67→//   translate() reads `fontSize`/`fontFamily`/`color`/`compId` — none of which
    68→//   the STRICT schema allows. Validated typography params are therefore
    69→//   SILENTLY DROPPED from the compiled layer.newText (gp-06 pins the exact
    70→//   reduced params: only text/name/position survive). Also comp targeting
    71→//   (compId) is unexpressible. Fix belongs in translate() (read size/font/fill
    72→//   + box/style/stroke* /tracking/leading passthrough per the engine's
    73→//   layer.newText params).
    74→//
    75→// G-3 (HIGH, partial drop) — MIXED plans compile PARTIALLY: plan steps whose
    76→//   tool args diverge (G-1) translate to zero commands with only WARN
    77→//   severity, so a multi-step plan mixing an aligned tool and a divergent tool
    78→//   returns ok:true with a batch that SILENTLY MISSING the divergent steps
    79→//   (gp-11: "create text + animate + save" compiles to create-text only).
    80→//   The plan card shows all 3 steps; the applied batch performs 1. This
    81→//   violates the spirit of "a failed step never leaves a partial edit" —
    82→//   recommend: zero-command steps in an otherwise-compilable plan should be
    83→//   FATAL (or the plan should be rejected with a typed issue), not warn.
    84→//
    85→// G-4 (LOW, risk floors) — defaultCompileDeps() passes WRITE_TOOL_ARGS_SCHEMAS
    86→//   (bare Zod objects, no risk metadata), so the write-tool catalog's risk
    87→//   badges (medium for create_composition/add_effect/apply_preset/undo_run)
    88→//   never reach the compiled risk — only the command-side rules apply
    89→//   (delete-ish/render-ish/edit.undo|redo ⇒ medium). gp-07 pins it: the model
    90→//   says medium, the wire carries low. Fix: defaultCompileDeps should project
    91→//   WRITE_TOOLS as {args: inputSchema, risk} WriteToolDefs.
    92→//
    93→// G-5 (INFO, by design) — save_version is compile-time ANNOTATION ONLY; a
    94→//   save_version-only plan is a pinned empty_batch fatal (apply needs a
    95→//   batch), and the studio.save_version write-tool proposal likewise answers
    96→//   PROPOSAL_INVALID — so the agent cannot produce a labeled snapshot without
    97→//   also editing something. Every apply creates a version anyway (apply
    98→//   route requires the uploaded version); save_version only labels it.
    99→//   Documented as a product seam: "save a version of what I have now" needs
   100→//   either a client-side action or a ruling.
   101→//
   102→// G-6 (MEDIUM, selection targeting) — APPLY_PRESET_ARGS_SCHEMA documents
   103→//   "Omit layers to use the current selection", but the compiler's
   104→//   selection-capture policy refuses to compile unnamed targets (gp-08: the
   105→//   §4.4 quick action "Make it pop" answers empty_batch). The tool description
   106→//   promises semantics the compiler deliberately refuses — needs an
   107→//   orchestrator ruling (client resolves selection→names before compile, or a
   108→//   typed "needs-selection" request rides the plan event).
   109→//
   110→// G-7 (MEDIUM, inexpressible edit) — RENAME: the engine has layer.rename (in
   111→//   AGENT_COMMAND_IDS), but none of the 8 write tools maps to it; a realistic
   112→//   "rename these layers" prompt (gp-05) can only be approximated by

   114→//   whose support for path "name" is unverified). Recorded per the mission:

   151→/** §5 context snapshot the client sends with the message (§5.1/§5.2). */

   154→  readonly activeComp: string;

   165→export interface GoldenExpectation {
   166→  /**
   167→   * Today's REAL-pipeline outcome. "compiled" ⇒ the assertions below hold on
   168→   * the compiled result now. "blocked" ⇒ the compile answers a TYPED FATAL
   169→   * (fail-closed: no batch is produced at all) because of FINDING G-1/G-6 —
   170→   * `commands` then carries the pinned POST-FIX golden for the fix.
   171→   */
   172→  readonly compile: "compiled" | "blocked";
   173→  /** The exact fatal issue code asserted when `compile` is "blocked". */
   174→  readonly fatalCode?: "empty_batch";
   175→  /**
   176→   * Registry command ids of the compiled batch, in order (the §M5.2 golden).
   177→   * For blocked fixtures: the post-reconciliation golden per the compiler's
   178→   * own translation table — every id is in AGENT_COMMAND_IDS (self-checked).
   179→   */
   180→  readonly commands: readonly string[];
   181→  /** Inclusive [min, max] range on the compiled batch step count. */
   182→  readonly batchStepRange: readonly [number, number];
   183→  /** Server-RECOMPUTED risk carried on the plan event (never the model's). */
   184→  readonly risk: PlanRisk;
   185→  /** estimateCredits({ risk, steps: plan.steps }) — never the model's number. */
   186→  readonly estimatedCredits: number;
   187→  /** Layer NAMES needing client-side id resolution, in request order (§8.7). */
   188→  readonly resolutionRequests: readonly string[];
   189→  /** Expected WARN issue codes on a compiled result (subset match). */
   190→  readonly warningCodes: readonly string[];
   191→  /** Present when the plan carries a save_version annotation. */
   192→  readonly saveVersionLabel?: string;
   193→  /** Finding references explaining a non-golden expectation. */
   194→  readonly findings: readonly string[];
   195→}

   213→function baseContext(overrides: Partial<ContextSelection> = {}): ContextSelection {
   214→  return {
   215→    projectSummary: true,
   216→    activeComp: true,
   217→    selection: false,
   218→    playhead: true,
   219→    focusedPanel: true,
   220→    selectedProperties: false,
   221→    currentFrame: false,
   222→    recentCommands: false,
   223→    comments: false,
   224→    ...overrides,
   225→  };
   226→}
   227→
   228→function selectionContext(): ContextSelection {
   229→  return baseContext({ selection: true, selectedProperties: true });
   230→}

   232→/** A §8.6 plan step: curated write tool id + args + the model's ADVISORY
   233→ * batch proposal (pseudo command ids = the bare tool id, per the M5-C write
   234→ * tool carrier convention — never a real registry id, so the wire proofs can
   235→ * assert the compiled registry ids NEVER enter the model conversation). */
   236→function step(
   237→  title: string,
   238→  tool: string,
   239→  args: unknown,
   240→  proposedParams: Record<string, unknown>,
   241→): EditPlan["steps"][number] {
   242→  return {
   243→    title,
   244→    tool,
   245→    args,
   246→    batch: {
   247→      label: title,
   248→      steps: [{ command: tool.replace(/^studio\./, ""), params: proposedParams }],
   249→    },
   250→  };
   251→}
