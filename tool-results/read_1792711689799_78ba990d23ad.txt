    72→/** Hard string bound for anything rendered from wire data. */
    73→export function truncateTimelineString(value: string, max: number): string {
    74→  const text = value.trim();
    75→  if (text.length <= max) return text;
    76→  return `${text.slice(0, max - 1)}…`;
    77→}
    78→
    79→/**
    80→ * §14 file sizes: `12.4 MB`, `20 GB`, `1 TB` (base 1024, one decimal above
    81→ * bytes). Hostile numbers (negative, non-finite) render as "—".
    82→ */
    83→export function formatVersionBytes(bytes: number): string {
    84→  if (typeof bytes !== "number" || !Number.isFinite(bytes) || bytes < 0) return "—";
    85→  if (bytes < 1024) return `${Math.round(bytes)} B`;
    86→  const units = ["KB", "MB", "GB", "TB"] as const;
    87→  let value = bytes;
    88→  let unit = -1;
    89→  do {
    90→    value /= 1024;
    91→    unit += 1;
    92→  } while (value >= 1024 && unit < units.length - 1);
    93→  const rounded = Math.round(value * 10) / 10;
    94→  return `${rounded.toLocaleString("en-US", { maximumFractionDigits: 1 })} ${units[unit]}`;
    95→}
    96→
    97→/** Signed size delta for the diff panel: "+1.2 KB" / "-240 B" / "no change". */
    98→export function formatSizeDeltaBytes(deltaBytes: number): string {
    99→  if (typeof deltaBytes !== "number" || !Number.isFinite(deltaBytes)) return "—";
   100→  if (deltaBytes === 0) return "no change";
   101→  const body = formatVersionBytes(Math.abs(deltaBytes));
   102→  return `${deltaBytes > 0 ? "+" : "-"}${body}`;
   103→}

   105→/**
   106→ * Timeline timestamps: "Oct 21, 2026, 14:32" (UI_DS §14 — absolute dates on a
   107→ * history surface; relative time stays on the activity feed). Pinned to UTC
   108→ * so the display is deterministic everywhere; hostile ISO degrades to "—".
   109→ */
   110→const TIMELINE_DATE_FORMAT = new Intl.DateTimeFormat("en-US", {
   111→  timeZone: "UTC",
   112→  month: "short",
   113→  day: "numeric",
   114→  year: "numeric",
   115→  hour: "2-digit",
   116→  minute: "2-digit",
   117→  hour12: false,
   118→});
   119→
   120→export function formatTimelineTimestamp(iso: string): string {
   121→  if (typeof iso !== "string" || iso.length === 0) return "—";
   122→  const date = new Date(iso.length > 100 ? iso.slice(0, 100) : iso);
   123→  if (Number.isNaN(date.getTime())) return "—";
   124→  return TIMELINE_DATE_FORMAT.format(date);
   125→}

   127→/** Creation gap between a version and its parent, as short copy. */
   128→export function formatCreatedGap(gapMs: number): string {
   129→  if (typeof gapMs !== "number" || !Number.isFinite(gapMs) || gapMs < 0) return "—";
   130→  if (gapMs < 60_000) return "moments later";
   131→  if (gapMs < 3_600_000) return `${Math.round(gapMs / 60_000)}m later`;
   132→  if (gapMs < 86_400_000) {
   133→    const hours = Math.floor(gapMs / 3_600_000);
   134→    const minutes = Math.round((gapMs % 3_600_000) / 60_000);
   135→    return minutes === 0 ? `${hours}h later` : `${hours}h ${minutes}m later`;
   136→  }
   137→  const days = Math.floor(gapMs / 86_400_000);
   138→  if (days < 7) return `${days}d later`;
   139→  return "more than a week later";
   140→}

   142→/**
   143→ * The typed confirmation phrase for a restore (§M7 task 4 confirmation
   144→ * semantics): the version's exact label, or its `v{N}` designation when it has
   145→ * none. The user types THIS, exactly, before the POST is allowed to fire.
   146→ */
   147→export function restoreConfirmPhrase(entry: VersionTimelineEntry): string {
   148→  const label = entry.label;
   149→  if (typeof label === "string" && label.trim().length > 0) return label.trim();
   150→  return `v${entry.version}`;

   183→function toActionError(cause: unknown): TimelineActionError {
   184→  if (isApiError(cause)) {
   185→    return { message: cause.message, hint: cause.hint, requestId: cause.requestId };
   186→  }

   192→  return error.requestId === undefined ? message : `${message} (${error.requestId})`;

   195→// ---------------------------------------------------------------------------

   199→export interface VersionPagesState {
   200→  readonly entries: readonly VersionTimelineEntry[];
   201→  readonly nextCursor: string | null;
   202→  readonly loading: boolean;
   203→  readonly loadingMore: boolean;
   204→  readonly error: string | null;
   205→  readonly loadMore: () => void;
   206→  readonly retry: () => void;
   207→  readonly reload: () => void;
   208→  /** Replaces one loaded row by id (inline label saves; no refetch needed). */
   209→  readonly patchEntry: (next: VersionTimelineEntry) => void;
   210→}
   211→
   212→function fetchErrorMessage(cause: unknown): string {
   213→  if (isApiError(cause)) return truncateTimelineString(cause.message, MAX_META_DISPLAY);
   214→  return "The version timeline could not be loaded.";
   215→}
   216→
   217→export function useVersionPages(
   218→  fetchPage: VersionPageFetcher,
   219→  options?: { readonly pageSize?: number },
   220→): VersionPagesState {
   221→  const pageSize = options?.pageSize ?? 25;
   222→
   223→  const [entries, setEntries] = useState<readonly VersionTimelineEntry[]>([]);
   224→  const [nextCursor, setNextCursor] = useState<string | null>(null);
   225→  const [loading, setLoading] = useState(true);
   226→  const [loadingMore, setLoadingMore] = useState(false);
   227→  const [error, setError] = useState<string | null>(null);
   228→
   229→  const seq = useRef(0);
   230→  // Synchronous in-flight guard for load-more: two calls before a re-render
   231→  // (double click) can never double-fetch a page. fetchFirst (reload) never
   232→  // blocks on it — a reload SUPERSEDES an in-flight load-more (seq guard).
   233→  const inFlight = useRef(false);
   234→  const fetchRef = useRef(fetchPage);
   235→  fetchRef.current = fetchPage;
   236→
   237→  const fetchFirst = useCallback(async (): Promise<void> => {
   238→    seq.current += 1;
   239→    const mySeq = seq.current;
   240→    setLoading(true);
   241→    setLoadingMore(false);
   242→    setError(null);
   243→    try {
   244→      const result = await fetchRef.current({ limit: pageSize });
   245→      if (seq.current !== mySeq) return;
   246→      setEntries(result.items);
   247→      setNextCursor(result.nextCursor);
   248→    } catch (cause) {
   249→      if (seq.current !== mySeq) return;
   250→      setError(fetchErrorMessage(cause));
   251→    } finally {
   252→      if (seq.current === mySeq) setLoading(false);
   253→    }
   254→  }, [pageSize]);
   255→
   256→  const loadMore = useCallback((): void => {
   257→    if (inFlight.current) return;
   258→    if (loading || loadingMore) return;
   259→    const cursor = nextCursor;
   260→    if (cursor === null) return;
   261→    inFlight.current = true;
   262→    seq.current += 1;
   263→    const mySeq = seq.current;
   264→    setLoadingMore(true);
   265→    setError(null);
   266→    void (async () => {
   267→      try {
   268→        const result = await fetchRef.current({ cursor, limit: pageSize });
   269→        if (seq.current !== mySeq) return;
   270→        setEntries((current) => [...current, ...result.items]);
   271→        setNextCursor(result.nextCursor);
   272→      } catch (cause) {
   273→        if (seq.current !== mySeq) return;
   274→        setError(fetchErrorMessage(cause));
   275→      } finally {
   276→        inFlight.current = false;
   277→        if (seq.current === mySeq) setLoadingMore(false);
   278→      }
   279→    })();
   280→  }, [loading, loadingMore, nextCursor, pageSize]);
   281→
   282→  const retry = useCallback((): void => {
   283→    if (entries.length === 0) void fetchFirst();
   284→    else loadMore();
   285→  }, [entries.length, fetchFirst, loadMore]);
   286→
   287→  const patchEntry = useCallback((next: VersionTimelineEntry): void => {
   288→    setEntries((current) => current.map((entry) => (entry.id === next.id ? next : entry)));
   289→  }, []);
   290→
   291→  useEffect(() => {
   292→    void fetchFirst();
   293→  }, [fetchFirst]);
   294→
   295→  return {
   296→    entries,
   297→    nextCursor,
   298→    loading,
   299→    loadingMore,
   300→    error,
   301→    loadMore,
   302→    retry,
   303→    reload: fetchFirst,
   304→    patchEntry,
   305→  };
   306→}

   371→  const [labelEdit, setLabelEdit] = useState<LabelEditorState>(LABEL_EDITOR_IDLE);
   372→
   373→  const label =
   374→    entry.label !== null && entry.label.trim().length > 0
   375→      ? truncateTimelineString(entry.label, MAX_LABEL_DISPLAY)
   376→      : null;
   377→  const author =
   378→    entry.createdByName !== null ? truncateTimelineString(entry.createdByName, 80) : null;
   379→
   380→  const saveLabel = (): void => {
   381→    if (!validTimelineLabel(labelEdit.draft)) return;

   389→        const actionError = toActionError(cause);

   391→          ...state,
   392→          saving: false,

   574→            <div className="flex flex-col gap-2" data-testid="version-diff-loading">
   575→              <Skeleton className="h-4 w-2/3" />
   576→              <Skeleton className="h-4 w-1/2" />
   577→            </div>
   578→          ) : diffState.status === "error" ? (
   579→            <p
   580→              role="alert"
   581→              data-testid="version-diff-error"
   582→              className="flex flex-wrap items-center gap-2 text-caption text-primary"
   583→            >
   584→              <CircleAlert
   585→                size={16}
   586→                strokeWidth={1.75}
   587→                aria-hidden="true"
   588→                className="shrink-0 text-danger"
   589→              />
   590→              <span className="min-w-0 flex-1">{diffState.message}</span>
   591→              <Button
   592→                variant="secondary"
   593→                size="sm"
   594→                data-testid="version-diff-retry"
   595→                onClick={() => onRetryDiff(entry)}
   596→              >
   597→                Try again
   598→              </Button>
   599→            </p>
   600→          ) : diffState.diff.fromParent === null ? (
   601→            <p className="text-body-s text-secondary" data-testid="version-diff-summary">
   602→              {truncateTimelineString(diffState.diff.summary, MAX_META_DISPLAY)}
   603→            </p>
   604→          ) : (
   605→            <>
   606→              <p className="text-body-s text-secondary" data-testid="version-diff-summary">
   607→                {truncateTimelineString(diffState.diff.summary, MAX_META_DISPLAY)}
   608→              </p>
   609→              <dl className="mt-3 grid gap-x-6 gap-y-1 text-caption sm:grid-cols-2">
   610→                <div className="flex justify-between gap-4 sm:block">
   611→                  <dt className="text-muted">Compared with</dt>
   612→                  <dd className="text-secondary" data-testid="version-diff-parent">
   613→                    v{diffState.diff.fromParent.parentVersion}
   614→                  </dd>
   615→                </div>
   616→                <div className="flex justify-between gap-4 sm:block">
   617→                  <dt className="text-muted">Size</dt>
   618→                  <dd className="text-secondary tabular-nums" data-testid="version-diff-size">
   619→                    {formatSizeDeltaBytes(diffState.diff.fromParent.sizeDeltaBytes)}
   620→                  </dd>
   621→                </div>
   622→                <div className="flex justify-between gap-4 sm:block">
   623→                  <dt className="text-muted">Saved</dt>
   624→                  <dd className="text-secondary" data-testid="version-diff-gap">
   625→                    {formatCreatedGap(diffState.diff.fromParent.createdGapMs)}
   626→                  </dd>
   627→                </div>
   628→                <div className="flex justify-between gap-4 sm:block">
   629→                  <dt className="text-muted">Author</dt>
   630→                  <dd className="text-secondary" data-testid="version-diff-author">
   631→                    {diffState.diff.fromParent.authorChanged
   632→                      ? "Changed since the parent"
   633→                      : "Same as the parent"}
   634→                  </dd>
   635→                </div>
   636→              </dl>
   637→            </>
   638→          )}
   639→        </div>
   640→      )}
   641→    </li>
   642→  );
   643→}

   645→// ---------------------------------------------------------------------------
   646→// The restore confirmation dialog (§9.3 + §7.5 type-to-confirm)
   647→// ---------------------------------------------------------------------------
   648→
   649→function RestoreDialog({
   650→  target,
   651→  busy,
   652→  error,
   653→  onOpenChange,
   654→  onConfirm,
   655→}: {
   656→  readonly target: VersionTimelineEntry | null;
   657→  readonly busy: boolean;
   658→  readonly error: string | null;
   659→  readonly onOpenChange: (open: boolean) => void;
   660→  readonly onConfirm: (confirmLabel: string) => void;
   661→}): ReactElement | null {
   662→  const [typed, setTyped] = useState("");
   663→  const expected = target === null ? "" : restoreConfirmPhrase(target);
   664→  // EXACT-label gate: the typed confirmation must match the version's label
   665→  // (or its `v{N}` designation) verbatim — trimmed, case-sensitive. A wrong
   666→  // label is rejected here, BEFORE the POST fires (§M7 task 4 confirmation).
   667→  const matches = target !== null && typed.trim() === expected && expected.length > 0;
   668→  const withinBounds = typed.trim().length <= TIMELINE_LABEL_MAX_CHARS;
   669→
   670→  // Fresh input whenever a new target opens the dialog.
   671→  useEffect(() => {
   672→    if (target !== null) setTyped("");
   673→  }, [target]);
   674→
   675→  if (target === null) return null;
   676→
   677→  const label =
   678→    target.label !== null && target.label.trim().length > 0
   679→      ? truncateTimelineString(target.label, MAX_LABEL_DISPLAY)
   680→      : null;
   681→
   682→  return (
   683→    <Dialog open onOpenChange={onOpenChange}>
   684→      <DialogContent aria-describedby={undefined}>
   685→        <DialogTitle data-testid="restore-dialog-title">Restore v{target.version}?</DialogTitle>
   686→        <DialogDescription asChild>
   687→          <div data-testid="restore-dialog-description">
   688→            <p>
   689→              {label === null ? (
   690→                "This version has no label. "
   691→              ) : (
   692→                <>
   693→                  Labeled <span className="text-primary">{label}</span>.{" "}
   694→                </>
   695→              )}
   696→              Restoring saves this snapshot as a new version — the version history is never
   697→              rewritten, and nothing is lost.
   698→            </p>
   699→            <p className="mt-2">
   700→              Type <span className="text-primary">{expected}</span> to confirm.
   701→            </p>
   702→          </div>
   703→        </DialogDescription>
   704→        <div className="mt-4">
   705→          <Input
   706→            label={`Type "${expected}" to confirm`}
   707→            autoComplete="off"
   708→            spellCheck={false}
   709→            value={typed}
   710→            onChange={(event) => setTyped(event.target.value)}
   711→            aria-invalid={typed.length > 0 && !matches ? true : undefined}
   712→            data-testid="restore-input"
   713→          />
   714→        </div>
   715→        {error !== null ? (
   716→          <p
   717→            role="alert"
   718→            data-testid="restore-error"
   719→            className="mt-3 flex items-start gap-2 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   720→          >
   721→            <CircleAlert
   722→              size={16}
   723→              strokeWidth={1.75}
   724→              aria-hidden="true"
   725→              className="mt-0.5 shrink-0 text-danger"
   726→            />
   727→            <span className="min-w-0 flex-1">{error}</span>
   728→          </p>
   729→        ) : null}
   730→        <DialogActions>
   731→          <DialogClose asChild>
   732→            <Button variant="secondary" disabled={busy} data-testid="restore-cancel">
   733→              Cancel
   734→            </Button>
   735→          </DialogClose>
   736→          <Button
   737→            disabled={!matches || !withinBounds}
   738→            loading={busy}
   739→            onClick={() => onConfirm(typed.trim())}
   740→            data-testid="restore-confirm"
   741→          >
   742→            Restore version
   743→          </Button>
   744→        </DialogActions>
   745→      </DialogContent>
   746→    </Dialog>
   747→  );
   748→}

   754→export interface VersionTimelineProps {
   755→  /** The version row the project currently points at (the "Current" marker). */
   756→  readonly currentVersionId: string | null;
   757→  /** Structurally `client.versions.list(projectId, query)` — timeline pages. */
   758→  readonly fetchPage: VersionPageFetcher;
   759→  /** Structurally `client.versions.diff(projectId, versionId)`. */
   760→  readonly fetchDiff: FetchDiffCaller;
   761→  /**
   762→   * Structurally `client.versions.restore(projectId, versionId, body)`.
   763→   * Absent (or the caller not authorized) → the Restore action is hidden.
   764→   */
   765→  readonly restore?: RestoreVersionCaller;
   766→  /**
   767→   * Structurally `client.versions.label(projectId, versionId, body)`.
   768→   * Absent (or the caller not authorized) → inline label editing is hidden.
   769→   */
   770→  readonly setLabel?: SetVersionLabelCaller;
   771→  /**
   772→   * Download hook. The wire only exposes snapshot bytes for the CURRENT
   773→   * version (the manifest's short-lived signed URL, §9.2) — the host passes
   774→   * this only when that surface exists; the row renders it for the current
   775→   * version only.
   776→   */
   777→  readonly onDownload?: (versionId: string) => void;
   778→  /** Notifies the host that a restore made a new current version (§9.3). */
   779→  readonly onCurrentVersionChange?: (versionId: string) => void;
   780→  /** Deep link to the editor for the empty state's secondary action. */
   781→  readonly editorHref?: string;
   782→  readonly className?: string;
   783→}
   784→
   785→export function VersionTimeline({
   786→  currentVersionId,
   787→  fetchPage,
   788→  fetchDiff,
   789→  restore,
   790→  setLabel,
   791→  onDownload,
   792→  onCurrentVersionChange,
   793→  editorHref,
   794→  className,
   795→}: VersionTimelineProps): ReactElement {
   796→  const pages = useVersionPages(fetchPage);
   797→  const [restoreTarget, setRestoreTarget] = useState<VersionTimelineEntry | null>(null);
   798→  const [restoreBusy, setRestoreBusy] = useState(false);
   799→  const [restoreError, setRestoreError] = useState<string | null>(null);
   800→  const [diffFor, setDiffFor] = useState<string | null>(null);
   801→  const [diffStates, setDiffStates] = useState<Record<string, DiffState>>({});
   802→
   803→  const byId = useMemo(() => {
   804→    const map = new Map<string, VersionTimelineEntry>();
   805→    for (const entry of pages.entries) map.set(entry.id, entry);
   806→    return map;
   807→  }, [pages.entries]);
   808→
   809→  const fetchDiffFor = useCallback(
   810→    (entry: VersionTimelineEntry): void => {
   811→      setDiffStates((current) => ({ ...current, [entry.id]: { status: "loading" } }));
   812→      void (async () => {
   813→        try {
   814→          const diff = await fetchDiff(entry.id);
   815→          setDiffStates((current) => ({ ...current, [entry.id]: { status: "ready", diff } }));
   816→        } catch (cause) {
   817→          setDiffStates((current) => ({
   818→            ...current,
   819→            [entry.id]: { status: "error", message: boundedErrorText(toActionError(cause)) },
   820→          }));
   821→        }
   822→      })();
   823→    },
   824→    [fetchDiff],
   825→  );
   826→
   827→  const toggleDiff = useCallback(
   828→    (entry: VersionTimelineEntry): void => {
   829→      if (diffFor === entry.id) {
   830→        setDiffFor(null);
   831→        return;
   832→      }
   833→      setDiffFor(entry.id);
   834→      if (diffStates[entry.id] === undefined) fetchDiffFor(entry);
   835→    },
   836→    [diffFor, diffStates, fetchDiffFor],
   837→  );
   838→
   839→  const retryDiff = useCallback(
   840→    (entry: VersionTimelineEntry): void => {
   841→      fetchDiffFor(entry);
   842→    },
   843→    [fetchDiffFor],
   844→  );
   845→
   846→  const openRestore = useCallback((entry: VersionTimelineEntry): void => {
   847→    setRestoreTarget(entry);
   848→    setRestoreError(null);
   849→  }, []);
   850→
   851→  const closeRestore = useCallback(
   852→    (open: boolean): void => {
   853→      // A restore in flight is never dismissible — the POST must settle.
   854→      if (!open && !restoreBusy) {
   855→        setRestoreTarget(null);
   856→        setRestoreError(null);
   857→      }
   858→    },
   859→    [restoreBusy],
   860→  );
   861→
   862→  const confirmRestore = useCallback(
   863→    (confirmLabel: string): void => {
   864→      if (restoreTarget === null || restore === undefined) return;
   865→      const target = restoreTarget;
   866→      setRestoreBusy(true);
   867→      setRestoreError(null);
   868→      void (async () => {
   869→        try {
   870→          // §9.3: the response IS the newly created version; it becomes the
   871→          // project's current version. History is never rewritten.
   872→          const created = await restore(target.id, { confirmLabel });
   873→          setRestoreBusy(false);
   874→          setRestoreTarget(null);
   875→          setRestoreError(null);
   876→          onCurrentVersionChange?.(created.id);
   877→          void pages.reload();
   878→        } catch (cause) {
   879→          setRestoreBusy(false);
   880→          setRestoreError(boundedErrorText(toActionError(cause)));
   881→        }
   882→      })();
   883→    },
   884→    [onCurrentVersionChange, pages, restore, restoreTarget],
   885→  );
   886→
   887→  const saveLabel = useCallback(
   888→    (next: VersionTimelineEntry): Promise<unknown> => {
   889→      if (setLabel === undefined) return Promise.resolve();
   890→      return setLabel(next.id, { label: next.label ?? "" }).then((saved) => {
   891→        // The wire returns the updated row (D-24: only the label changes);
   892→        // keep the timeline's display fields, adopt the saved label.
   893→        pages.patchEntry({ ...next, label: saved.label });
   894→        return saved;
   895→      });
   896→    },
   897→    [pages, setLabel],
   898→  );
   899→
   900→  const handleDownload = useCallback(
   901→    (entry: VersionTimelineEntry): void => {
   902→      onDownload?.(entry.id);
   903→    },
   904→    [onDownload],
   905→  );
   906→
   907→  const canLoadMore = pages.nextCursor !== null;
   908→  const atEnd = !pages.loading && pages.nextCursor === null && pages.entries.length > 0;
   909→
   910→  return (
   911→    <section data-testid="version-timeline" className={className} aria-label="Version timeline">
   912→      {pages.error !== null ? (
   913→        <div
   914→          role="alert"
   915→          data-testid="timeline-error"
   916→          className="mb-4 flex items-center gap-2 rounded-sm border border-danger/40 bg-danger/10 px-3 py-2 text-caption text-primary"
   917→        >
   918→          <CircleAlert
   919→            size={16}
   920→            strokeWidth={1.75}
   921→            aria-hidden="true"
   922→            className="shrink-0 text-danger"
   923→          />
   924→          <p className="min-w-0 flex-1">{pages.error}</p>
   925→          <Button
   926→            variant="secondary"
   927→            size="sm"
   928→            data-testid="timeline-error-retry"
   929→            disabled={pages.loading}
   930→            onClick={pages.retry}
   931→          >
   932→            Try again
   933→          </Button>
   934→        </div>
   935→      ) : null}
   936→
   937→      {pages.loading && pages.entries.length === 0 ? (
   938→        <div className="flex flex-col gap-6 py-2 pl-5" data-testid="timeline-loading">
   939→          <Skeleton className="h-16 w-full max-w-[720px]" />
   940→          <Skeleton className="h-16 w-full max-w-[680px]" />
   941→          <Skeleton className="h-16 w-full max-w-[700px]" />
   942→        </div>
   943→      ) : pages.entries.length === 0 ? (
   944→        <EmptyState
   945→          data-testid="timeline-empty"
   946→          title="No versions yet."
   947→          secondaryLink={
   948→            editorHref === undefined ? undefined : { label: "Open the editor", href: editorHref }
   949→          }
   950→        />
   951→      ) : (
   952→        <ol data-testid="timeline-list" className="ml-2">
   953→          {pages.entries.map((entry) => (
   954→            <VersionRow
   955→              key={entry.id}
   956→              entry={entry}
   957→              parent={
   958→                entry.parentVersionId === null ? null : (byId.get(entry.parentVersionId) ?? null)
   959→              }
   960→              isCurrent={entry.id === currentVersionId}
   961→              canRestore={restore !== undefined}
   962→              canEditLabel={setLabel !== undefined}
   963→              canDownload={onDownload !== undefined}
   964→              diffState={diffFor === entry.id ? (diffStates[entry.id] ?? null) : null}
   965→              onToggleDiff={toggleDiff}
   966→              onRetryDiff={retryDiff}
   967→              onRestore={openRestore}
   968→              onDownload={handleDownload}
   969→              onSaveLabel={saveLabel}
   970→            />
   971→          ))}
   972→        </ol>
   973→      )}
   974→
   975→      {canLoadMore || atEnd ? (
   976→        <div className="mt-2 flex items-center justify-center gap-3">
   977→          {canLoadMore ? (
   978→            <Button
   979→              variant="secondary"
   980→              size="sm"
   981→              data-testid="timeline-load-more"
   982→              loading={pages.loadingMore}
   983→              disabled={pages.loading || pages.loadingMore}
   984→              onClick={() => pages.loadMore()}
   985→            >
   986→              Load more
   987→            </Button>
   988→          ) : (
   989→            <p data-testid="timeline-end" className="text-caption text-muted">
   990→              That’s the whole history.
   991→            </p>
   992→          )}
   993→        </div>
   994→      ) : null}
   995→
   996→      <RestoreDialog
   997→        target={restoreTarget}
   998→        busy={restoreBusy}
   999→        error={restoreError}
  1000→        onOpenChange={closeRestore}
  1001→        onConfirm={confirmRestore}
  1002→      />
  1003→    </section>
  1004→  );
  1005→}
