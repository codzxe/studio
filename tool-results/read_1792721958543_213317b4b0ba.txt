   271→function defaultSleep(ms: number): Promise<void> {
   272→  if (ms <= 0) return Promise.resolve();
   273→  return new Promise((resolve) => {
   274→    setTimeout(resolve, ms);
   275→  });
   276→}

   278→/** Serialized query string; `undefined`/`null` values are skipped. */
   279→export function buildQuery(
   280→  params: Record<string, string | number | boolean | null | undefined>,
   281→): string {
   282→  const search = new URLSearchParams();
   283→  for (const [key, value] of Object.entries(params)) {
   284→    if (value === undefined || value === null) continue;
   285→    search.set(key, String(value));
   286→  }
   287→  const text = search.toString();
   288→  return text === "" ? "" : `?${text}`;
   289→}

   308→interface ClientInternals {
   309→  readonly baseUrl: string;
   310→  readonly fetchImpl: FetchLike;
   311→  readonly extraHeaders: Record<string, string>;
   312→  readonly newKey: IdempotencyKeyFactory;
   313→  readonly sleep: (ms: number) => Promise<void>;
   314→  readonly retryDelays: readonly number[];
   315→}
   316→
   317→function encodeId(id: string): string {
   318→  return encodeURIComponent(id);
   319→}
   320→
   321→class ApiClientImpl {
   322→  constructor(readonly core: ClientInternals) {}
   323→
   324→  /**
   325→   * One HTTP call: envelope unwrap, response validation, GET retry with
   326→   * backoff. Never surfaces raw response text — errors are typed ApiErrors.
   327→   */
   328→  async request<T>(plan: RequestPlan<T>): Promise<T> {
   329→    const url = `${this.core.baseUrl}${plan.path}${plan.query ? buildQuery(plan.query) : ""}`;
   330→
   331→    let attempt = 0;
   332→    for (;;) {
   333→      let status = 0;
   334→      let statusText = "";
   335→      let text = "";
   336→      try {
   337→        const init = this.buildInit(plan);
   338→        const response = await this.core.fetchImpl(url, init);
   339→        status = response.status;
   340→        statusText = response.statusText;
   341→        text = await response.text();
   342→      } catch (error) {
   343→        if (isAbortError(error) || plan.signal?.aborted) throw error;
   344→        const netError = networkError(error);
   345→        if (!this.shouldRetry(plan, netError, attempt)) throw netError;
   346→        await this.backoff(plan, attempt);
   347→        attempt += 1;
   348→        continue;
   349→      }
   350→
   351→      if (status < 200 || status > 299) {
   352→        const apiError = await fromResponse({
   353→          status,
   354→          statusText,
   355→          text: async () => text,
   356→        });
   357→        if (!this.shouldRetry(plan, apiError, attempt)) throw apiError;
   358→        await this.backoff(plan, attempt);
   359→        attempt += 1;
   360→        continue;
   361→      }
   362→
   363→      return this.unwrapOk<T>(status, text, plan);
   364→    }
   365→  }
   366→
   367→  private buildInit(plan: RequestPlan<unknown>): RequestInit {
   368→    const headers = new Headers(this.core.extraHeaders);
   369→    headers.set("Accept", "application/json");
   370→    if (plan.body !== undefined) headers.set("Content-Type", "application/json");
   371→    if (plan.idempotencyKey !== undefined) headers.set("Idempotency-Key", plan.idempotencyKey);
   372→    if (plan.headers !== undefined) {
   373→      for (const [name, value] of Object.entries(plan.headers)) headers.set(name, value);
   374→    }
   375→    const init: RequestInit = {
   376→      method: plan.method,
   377→      headers,
   378→      signal: plan.signal,
   379→    };
   380→    if (plan.body !== undefined) init.body = JSON.stringify(plan.body);
   381→    return init;
   382→  }
   383→
   384→  /** Only idempotent-safe GETs retry, and only when the failure is retryable. */
   385→  private shouldRetry(plan: RequestPlan<unknown>, error: ApiError, attempt: number): boolean {
   386→    if (plan.method !== "GET") return false;
   387→    if (attempt >= this.core.retryDelays.length) return false;
   388→    if (error.status === 0) return true; // network error before any response
   389→    if (error.status === 408 || error.status === 429 || error.status >= 500) {
   390→      return error.retryable;
   391→    }
   392→    return false;
   393→  }
   394→
   395→  private backoff(plan: RequestPlan<unknown>, attempt: number): Promise<void> {
   396→    const delay = this.core.retryDelays[attempt] ?? 0;
   397→    return sleepWithAbort(this.core.sleep, delay, plan.signal);
   398→  }
   399→
   400→  /** 2xx handling: empty body → `null`; otherwise a §7.3 Ok envelope is required. */
   401→  private unwrapOk<T>(status: number, text: string, plan: RequestPlan<T>): T {
   402→    if (text === "") {
   403→      if (plan.responseSchema === undefined) return null as T;
   404→      throw malformedResponseError(status);
   405→    }
   406→    let json: unknown;
   407→    try {
   408→      json = JSON.parse(text);
   409→    } catch {
   410→      throw malformedResponseError(status);
   411→    }
   412→    if (isErr(json)) {
   413→      const error: Err["error"] = json.error;
   414→      // A 2xx status with an Err body is still the §7.3 failure arm.
   415→      throw new ApiError({
   416→        code: error.code,
   417→        message: error.message,
   418→        hint: error.hint,
   419→        retryable: error.retryable,
   420→        requestId: error.requestId,
   421→        status,
   422→        cause: json,
   423→      });
   424→    }
   425→    if (!isOk(json)) throw malformedResponseError(status);
   426→    const data = json.data;
   427→    if (plan.responseSchema === undefined) return data as T;
   428→    try {
   429→      return plan.responseSchema.parse(data);
   430→    } catch (error) {
   431→      throw malformedResponseError(status, error);
   432→    }
   433→  }
   434→
   435→  /** Client-side §7.2 request validation — fails fast, never sends. */
   436→  validateBody<T>(schema: z.ZodType<T>, body: unknown, path: string): T {
   437→    try {
   438→      return schema.parse(body);
   439→    } catch (error) {
   440→      if (isApiError(error)) throw error;
   441→      throw invalidRequestError(
   442→        Object.assign(new Error(`invalid body for ${path}`), { cause: error }),
   443→      );
   444→    }
   445→  }
   446→
   447→  autoKey(): string {
   448→    return this.core.newKey();
   449→  }
   450→
   451→  // -- GETs don't send Idempotency-Key; mutations always do (auto when absent).
   452→  async get<T>(
   453→    path: string,
   454→    query: Record<string, string | number | boolean | undefined> | undefined,
   455→    schema: z.ZodType<T>,
   456→    signal?: AbortSignal,
   457→    headers?: Record<string, string>,
   458→  ): Promise<T> {
   459→    return this.request<T>({
   460→      method: "GET",
   461→      path,
   462→      query,
   463→      responseSchema: schema,
   464→      signal,
   465→      headers,
   466→    });
   467→  }
   468→
   469→  async mutate<T>(
   470→    method: "POST" | "PATCH" | "DELETE",
   471→    path: string,
   472→    plan: {
   473→      body?: unknown;
   474→      bodySchema?: z.ZodType<unknown>;
   475→      responseSchema?: z.ZodType<T>;
   476→      idempotencyKey?: string;
   477→      signal?: AbortSignal;
   478→      headers?: Record<string, string>;
   479→    },
   480→  ): Promise<T> {
   481→    const body =
   482→      plan.bodySchema !== undefined
   483→        ? this.validateBody(plan.bodySchema, plan.body, path)
   484→        : plan.body;
   485→    return this.request<T>({
   486→      method,
   487→      path,
   488→      body,
   489→      responseSchema: plan.responseSchema,
   490→      idempotencyKey: plan.idempotencyKey ?? this.autoKey(),
   491→      signal: plan.signal,
   492→      headers: plan.headers,
   493→    });
   494→  }
   495→}

   810→ * Builds the typed API client. `baseUrl` includes the version segment
   811→ * (default `"/v1"` for same-origin browser use). All mutation methods send
   812→ * `Idempotency-Key` (§7.2) — pass your own per logical operation, or let the
   813→ * client derive one.
   814→ */

   816→  const core: ClientInternals = {
   817→    baseUrl: (options.baseUrl ?? DEFAULT_BASE_URL).replace(/\/+$/, ""),
   818→    fetchImpl: options.fetch ?? ((input, init) => fetch(input, init)),
   819→    extraHeaders: { ...(options.headers ?? {}) },

   827→    const explicit = options?.idempotencyKey;

   829→      // §7.2: client-generated, ≥8 chars (contracts IDEMPOTENCY_KEY_SCHEMA).
   830→      return impl.validateBody(IDEMPOTENCY_KEY_SCHEMA, explicit, "Idempotency-Key");
   831→    }
   832→    return undefined;
   833→  };

   837→    query: unknown,

   842→      out[k] = v as string | number | boolean | undefined;

  1320→/** Two-phase version upload (§9.2/§9.3): descriptor POST + signed-URL byte PUT. */
  1321→async function uploadVersion(
  1322→  impl: ApiClientImpl,
  1323→  projectId: string,
  1324→  input: {
  1325→    readonly baseVersionId: string;
  1326→    readonly sha256: string;
  1327→    readonly label?: string;
  1328→    readonly agentRunId?: string;
  1329→    readonly bytes: Blob | Uint8Array;
  1330→  },
  1331→  options: CallOptions | undefined,
  1332→): Promise<ProjectVersion> {
  1333→  const { bytes } = input;
  1334→  if (!(bytes instanceof Blob) && !(bytes instanceof Uint8Array)) {
  1335→    throw invalidRequestError(new Error("bytes must be a Blob or Uint8Array"));
  1336→  }
  1337→  const sizeBytes = bytes instanceof Blob ? bytes.size : bytes.byteLength;
  1338→  const descriptor = impl.validateBody(
  1339→    UPLOAD_VERSION_SCHEMA,
  1340→    {
  1341→      baseVersionId: input.baseVersionId,
  1342→      sha256: input.sha256,
  1343→      label: input.label,
  1344→      agentRunId: input.agentRunId,
  1345→      sizeBytes,
  1346→    },
  1347→    "upload version descriptor",
  1348→  );
  1349→
  1350→  const response = await impl.mutate<UploadVersionResponse>(
  1351→    "POST",
  1352→    `/projects/${encodeId(projectId)}/versions`,
  1353→    {
  1354→      body: descriptor,
  1355→      responseSchema: UPLOAD_VERSION_RESPONSE_SCHEMA,
  1356→      idempotencyKey: options?.idempotencyKey ?? impl.autoKey(),
  1357→      signal: options?.signal,
  1358→    },
  1359→  );
  1360→
  1361→  // Phase two: bytes go straight to object storage via the signed URL — the
  1362→  // storage response is NOT a §7.3 envelope, so only the status is checked.
  1363→  const { url, headers } = response.upload;
  1364→  const put = await impl.core.fetchImpl(url, {
  1365→    method: "PUT",
  1366→    headers: { ...headers },
  1367→    // TS 5.7's DOM BodyInit doesn't accept Uint8Array<ArrayBufferLike>
  1368→    // (SharedArrayBuffer edge); runtime fetch (browser + undici) does.
  1369→    body: bytes instanceof Blob ? bytes : (bytes as unknown as BodyInit),
  1370→    signal: options?.signal,
  1371→  });
  1372→  if (!put.ok) {
  1373→    throw uploadFailedError(put.status);
  1374→  }
  1375→  return response.version;
  1376→}
